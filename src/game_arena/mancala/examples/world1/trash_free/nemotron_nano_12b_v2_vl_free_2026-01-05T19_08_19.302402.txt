def policy(you: list[int], opponent: list[int]) -> int:
    """
    Selects the optimal move by simulating each possible move and evaluating based on captures, extra moves, and store gain.
    
    Args:
        you (list[int]): Your game state with 6 houses and a store.
        opponent (list[int]): Opponent's game state with 6 houses and a store.
        
    Returns:
        int: The index of the best move (0-5).
    """
    best_move = -1
    best_score = -1
    best_captured = 0
    best_extra = False
    
    for move in range(6):
        if you[move] == 0:
            continue  # Skip empty houses

        new_you, new_opponent, captured, extra_move = simulate_move(you, opponent, move)
        
        # Calculate score based on captures, extra move, and store gain
        score = captured * 100 + 100 * int(extra_move) + (new_you[6] - you[6]) * 2
        
        # Update best move if current move is better
        if score > best_score:
            best_score = score
            best_move = move
            best_captured = captured
            best_extra = extra_move
    
    return best_move

def simulate_move(you: list[int], opponent: list[int], move: int) -> tuple:

    Args:

        opponent (list[int]): Opponent's game state.

    new_opponent = opponent.copy()

    new_opponent = list(opponent.copy())  # Create a deep copy

        return new_you, new_opponent, 0, False  # No valid move

    current_pos = (move + 1) % 12  # Start distributing from the next house
    
    stored_seeds = 0

        elif current_pos == 6:  # Your store
            new_you[6] += 1
        else:  # Opponent's houses
            opponent_idx = current_pos - 7
            new_opponent[opponent_idx] += 1
        
        last_pos = current_pos
        current_pos = (current_pos + 1) % 12  # Move to next position
    
    # Check for capture after simulation
    captured = 0
    if last_pos < 6:  # Last seed in your own house
        if new_you[last_pos] == 1:

            if new_opponent[opp_house] > 0:

                new_opponent[opp_house] = 0

    extra_move = (last_pos == 6)  # Extra move if last seed in store

**Explanation:**

1. **Simulation of Each Move:**
   - The `policy` function iterates through each possible move (houses 0-5) that contains seeds.
   - For each move, it simulates the distribution of seeds using `simulate_move`, which accurately models Kalah's rules.

2. **Capture Logic:**
   - After distributing all seeds, the function checks if the last seed landed in your own house (0-5) and if that house was empty before the move. If so, it captures the corresponding opponent's house seeds.

3. **Scoring Heuristic:**
   - **Captured Seeds:** Prioritizes moves that capture the most opponent seeds.
   - **Extra Move:** Moves ending in your store grant an extra turn, highly valued.
   - **Store Gain:** Points awarded for seeds added to your store, encouraging strategic storage.

4. **Efficiency:**
   - The heuristic scores moves by weighting captured seeds highest, followed by extra moves and store gains, ensuring quick evaluation while retaining strategic depth.
