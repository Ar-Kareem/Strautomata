Minimax with alpha-beta pruning.

def policy(you, opponent):
    def simulate(you, opp, move):
        y, o = you[:], opp[:]
        seeds = y[move]
        y[move] = 0
        pos = move + 1
        side = 0  # 0=your side, 1=opp side
        while seeds:
            if side == 0:
                if pos <= 5:
                    y[pos] += 1; seeds -= 1; pos += 1
                elif pos == 6:
                    y[6] += 1; seeds -= 1; pos = 0; side = 1
                else:
                    pos = 0; side = 1
            else:
                if pos <= 5:
                    o[pos] += 1; seeds -= 1; pos += 1
                else:
                    pos = 0; side = 0
        extra = False
        if side == 0 and pos == 7:
            extra = True
        if side == 0 and 1 <= pos <= 6:
            last = pos - 1
            if y[last] == 1 and 0 <= last <= 5 and o[5 - last] > 0:
                y[6] += y[last] + o[5 - last]
                y[last] = 0; o[5 - last] = 0
        return y, o, extra

    def evaluate(y, o):
        return y[6] - o[6]

    def minimax(y, o, depth, alpha, beta, maximizing):
        y_empty = all(s == 0 for s in y[:6])
        o_empty = all(s == 0 for s in o[:6])
        if y_empty or o_empty:
            sy = y[6] + sum(y[:6])
            so = o[6] + sum(o[:6])
            return sy - so
        if depth == 0:
            return evaluate(y, o)
        if maximizing:
            val = -999
            for m in range(6):
                if y[m] == 0: continue
                ny, no, extra = simulate(y, o, m)
                if extra:
                    v = minimax(ny, no, depth, alpha, beta, True)
                else:
                    v = minimax(no, ny, depth - 1, -beta, -alpha, True)
                    v = -v
                if v > val: val = v
                if val > alpha: alpha = val
                if alpha >= beta: break
            return val
        return 0

    best_move = -1; best_val = -9999
    for m in range(6):
        if you[m] == 0: continue
        ny, no, extra = simulate(you, opponent, m)
        if extra:
            v = minimax(ny, no, 8, -9999, 9999, True)
        else:
            v = -minimax(no, ny, 8, -9999, 9999, True)
        if v > best_val:
            best_val = v; best_move = m
    return best_move
