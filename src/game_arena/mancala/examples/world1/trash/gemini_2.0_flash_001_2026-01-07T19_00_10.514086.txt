def policy(you: list[int], opponent: list[int]) -> int:
    """
    Selects a move in the game of Kalah.

    Args:
        you: A list of integers representing your side of the board.
             you[0..5] are your houses, you[6] is your store.
        opponent: A list of integers representing the opponent's side of the board.
                  opponent[0..5] are their houses, opponent[6] is their store.

    Returns:
        An integer between 0 and 5 (inclusive) representing the house you choose to move from.
    """

    def make_move(you, opponent, move):
        """
        Simulates a move and returns the resulting board state and whether an extra turn is granted.

        Modifies the you and opponent lists in place and then returns copies.

        """
        you = you[:]
        opponent = opponent[:]
        seeds = you[move]
        you[move] = 0
        current_index = move + 1
        extra_turn = False

        while seeds > 0:
            if current_index == 6:
                you[6] += 1
                seeds -= 1
                if seeds == 0:
                    extra_turn = True
                current_index = 0
            elif current_index == 7:  # opponents store
                current_index = 0
            elif current_index < 6:
                you[current_index] += 1
                seeds -= 1
                current_index += 1

            elif current_index >= 8 and current_index < 14: #opponent houses
                opponent[current_index - 8] += 1
                seeds -= 1
                current_index += 1

            elif current_index >= 14 :
                current_index = 0

        # Capture
        current_index -= 1 # adjust for the last increment
        if 0 <= current_index < 6  and you[current_index] == 1 and opponent[5 - current_index] > 0:
                you[6] += 1 + opponent[5-current_index] #add the captured and your seed to the store
                opponent[5-current_index] = 0
                you[current_index] = 0


        return you, opponent, extra_turn
    
    # Simple policy: prioritize taking another turn if possible, otherwise make a random valid move.
    
    legal_moves = [i for i in range(6) if you[i] > 0]
    
    for move in legal_moves:
        you_copy = you[:]
        opponent_copy = opponent[:]

        new_you, new_opponent, extra_turn = make_move(you_copy, opponent_copy, move)
        if extra_turn:
            return move

    # If no extra turn available, pick the first legal move.
    return legal_moves[0]
