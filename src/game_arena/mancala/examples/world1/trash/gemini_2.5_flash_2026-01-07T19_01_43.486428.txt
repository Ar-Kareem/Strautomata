def minimax(you, opponent, depth, alpha, beta, is_maximizing_player):
    # Check for game over
    my_houses_empty = sum(you[0:6]) == 0
    opp_houses_empty = sum(opponent[0:6]) == 0

    if my_houses_empty or opp_houses_empty:
        # Game over, sweep remaining seeds
        final_you = list(you)
        final_opponent = list(opponent)
        if my_houses_empty:
            final_opponent[6] += sum(final_opponent[0:6])
            for i in range(6): final_opponent[i] = 0
        elif opp_houses_empty:
            final_you[6] += sum(final_you[0:6])
            for i in range(6): final_you[i] = 0
        return evaluate_board(final_you, final_opponent) # Use final scores for evaluation


    if depth == 0:
        return evaluate_board(you, opponent)

    if is_maximizing_player:
        max_eval = -math.inf
        # Iterate over possible moves (your houses 0-5)
        for i in range(6):
            if you[i] == 0:
                continue # Cannot move from an empty house

            new_you, new_opponent, extra_turn, game_over_sim, winner_sim = simulate_move(you, opponent, i)

            if game_over_sim: # If move ends the game, return the final evaluation
                if winner_sim == 'you':
                    return math.inf # Win is infinity
                elif winner_sim == 'opponent':
                    return -math.inf # Loss is -infinity
                else: # Draw
                    return 0 # Or some reasonable neutral score

            if extra_turn:
                # If extra turn, we (maximizing player) move again, so still maximizing, same depth
                eval = minimax(new_you, new_opponent, depth - 1, alpha, beta, True)
            else:
                # If no extra turn, opponent moves, so minimizing player
                # Swap you and opponent for the next call to represent opponent's perspective
                eval = minimax(new_opponent, new_you, depth - 1, alpha, beta, False)

            alpha = max(alpha, eval)

                break # Alpha-beta pruning

                            # So `you` here represents the actual opponent's current state.
                continue

            # and `opponent` here is `current_opponent_perspective_opponent`

                eval = minimax(new_opp_opponent, new_opp_you, depth - 1, alpha, beta, True)

            min_eval = min(min_eval, eval)
            beta = min(beta, eval)
            if beta <= alpha:
                break # Alpha-beta pruning
        return min_eval

def policy(you: list[int], opponent: list[int]) -> int:
    best_move = -1
    max_eval = -math.inf

    # Try a shallower depth first to quickly get some good moves
    # This might be useful for time management, but for simplicity let's stick to MAX_DEPTH as fixed.

            if winner_sim == 'you':
                return i # Winning move, take it immediately!

                current_eval = -math.inf # This move leads to immediate loss
            else: # Draw

            current_eval = minimax(new_opponent, new_you, MAX_DEPTH - 1, -math.inf, math.inf, False)

            max_eval = current_eval
            best_move = i

    # just pick the first legal move. This case should not happen given the problem statement

    if best_move == -1:
        for i in range(6):
            if you[i] > 0:
                return i # Fallback: return the first legal move.

    return best_move

    always from the perspective of the original 'you' (player 0).

    opponent_list = list(board[7:13]) + [board[13]]

            is_current_player_house = True
            _final_landed_house_idx = last_landed_idx
            _opposite_house_on_board_idx = 7 + (5 - last_landed_idx) # P1's corresponding house
        elif not current_player_is_player_0 and 7 <= last_landed_idx <= 12: # P1 landed in their own house
            is_current_player_house = True
            _final_landed_house_idx = last_landed_idx
            _opposite_house_on_board_idx = (5 - (last_landed_idx - 7)) # P0's corresponding house
        
        if is_current_player_house and temp_board[_final_landed_house_idx] == 1 and temp_board[_opposite_house_on_board_idx] > 0:
            # Capture!
            if current_player_is_player_0:
                temp_board[6] += temp_board[_final_landed_house_idx] # Capture own seed
                temp_board[_final_landed_house_idx] = 0
                temp_board[6] += temp_board[_opposite_house_on_board_idx] # Capture opponent's seeds
                temp_board[_opposite_house_on_board_idx] = 0
            else: # Player 1 (opponent) performing capture
                temp_board[13] += temp_board[_final_landed_house_idx] # Capture own seed
                temp_board[_final_landed_house_idx] = 0
                temp_board[13] += temp_board[_opposite_house_on_board_idx] # Capture opponent's (P0's) seeds
                temp_board[_opposite_house_on_board_idx] = 0

    # Check for game over
    game_over = False
    winner_internal_str = None

    player_0_houses_empty = sum(temp_board[0:6]) == 0
    player_1_houses_empty = sum(temp_board[7:13]) == 0

    if player_0_houses_empty:
        # Player 0's houses are empty. Player 1 sweeps their remaining seeds.
        temp_board[13] += sum(temp_board[7:13])
        for i in range(7, 13): temp_board[i] = 0
        game_over = True
    elif player_1_houses_empty:
        # Player 1's houses are empty. Player 0 sweeps their remaining seeds.
        temp_board[6] += sum(temp_board[0:6])
        for i in range(6): temp_board[i] = 0
        game_over = True
    
    if game_over:
        if temp_board[6] > temp_board[13]:
            winner_internal_str = 'player_0'
        elif temp_board[13] > temp_board[6]:
            winner_internal_str = 'player_1'
        else:
            winner_internal_str = 'draw'

    return temp_board, extra_turn, game_over, winner_internal_str


def simulate_move(player_to_move_state, other_player_state, move_idx, player_to_move_is_original_player: bool):
    """
    Simulates a Kalah move and returns the resulting board state from the ORIGINAL player's perspective.

    Args:
        player_to_move_state (list[int]): The list of houses/store for the player whose turn it is.
        other_player_state (list[int]): The list for the other player.
        move_idx (int): The house index (0-5) the current player chooses.
        player_to_move_is_original_player (bool): True if 'player_to_move_state' corresponds to
                                                  the `you` that initialized the policy call.

    Returns:
        tuple: (new_you, new_opponent, extra_turn, game_over, winner_str)
        `new_you` and `new_opponent` are always from the original `you`'s perspective.
        `winner_str` can be 'you', 'opponent', 'draw', or None.
    """
    # Convert to 14-slot board assuming player_to_move_state is 'you' (slots 0-6)
    # and other_player_state is 'opponent' (slots 7-13) for _simulate_move_internal.
    board_14 = _convert_to_14_slot_board(player_to_move_state, other_player_state)
    
    # Simulate the move
    temp_board_14, extra_turn, game_over, winner_internal = \
        _simulate_move_internal(board_14, move_idx, player_to_move_is_original_player)

    # Convert the resulting 14-slot board back to (you, opponent) from the ORIGINAL player's perspective
    final_my_you, final_my_opponent = _convert_from_14_slot_board(temp_board_14)

    winner_str = None
    if game_over:
        if winner_internal == 'player_0': # Player 0 is the 'you' in _convert_to_14_slot_board
            winner_str = 'you'
        elif winner_internal == 'player_1': # Player 1 is the 'opponent' in _convert_to_14_slot_board
            winner_str = 'opponent'
        else:
            winner_str = 'draw'

    return final_my_you, final_my_opponent, extra_turn, game_over, winner_str


# Heuristic evaluation function - always evaluates from the ORIGINAL player's perspective
def evaluate_board(you_state, opponent_state):
    """
    Evaluates a board state from the perspective of 'you' (the original player).
    Positive scores are good for 'you', negative for 'opponent'.
    """
    # Weight store difference much higher
    score = (you_state[6] - opponent_state[6]) * 20
    
    # Add difference in seeds on respective sides (houses) with less weight
    score += (sum(you_state[0:6]) - sum(opponent_state[0:6])) * 1

    # Small bonus for houses that if moved, land exactly in player's store
    for i in range(6):
        if you_state[i] > 0:
            if you_state[i] == (6 - i): # Exactly lands in store
                score += 5 # This is a good opportunity, a form of extra turn
    
    return score


def minimax(current_player_you_state, current_player_opponent_state, depth, alpha, beta, is_maximizing_player):
    """
    Minimax with Alpha-Beta Pruning.
    All scores are evaluated from the perspective of the initial maximizing player (original 'you').
    
    Args:
        current_player_you_state (list[int]): houses/store for the player whose turn it is in this minimax step.
        current_player_opponent_state (list[int]): houses/store for the other player in this minimax step.
        depth (int): Current search depth.
        alpha (float): Alpha value for alpha-beta pruning.
        beta (float): Beta value for alpha-beta pruning.
        is_maximizing_player (bool): True if this is the maximizing player's turn (original 'you').
                                     False if it's the minimizing player's turn (original 'opponent').
    """

    # Determine real 'you' and 'opponent' for this evaluation/game over check
    evaluating_you = current_player_you_state if is_maximizing_player else current_player_opponent_state
    evaluating_opponent = current_player_opponent_state if is_maximizing_player else current_player_you_state

    # Check for game over (no seeds in houses)
    player_to_move_houses_empty = sum(current_player_you_state[0:6]) == 0
    other_player_houses_empty = sum(current_player_opponent_state[0:6]) == 0

    if player_to_move_houses_empty or other_player_houses_empty:
        # Game over, simulate sweeps for final evaluation
        final_you, final_opponent = list(evaluating_you), list(evaluating_opponent)
        
        # Whichever player had empty houses, the other player sweeps.
        if sum(final_you[0:6]) == 0:
            final_opponent[6] += sum(final_opponent[0:6])
            for i in range(6): final_opponent[i] = 0
        elif sum(final_opponent[0:6]) == 0:
            final_you[6] += sum(final_you[0:6])
            for i in range(6): final_you[i] = 0
        
        return evaluate_board(final_you, final_opponent)

    if depth == 0:
        return evaluate_board(evaluating_you, evaluating_opponent)

    if is_maximizing_player:
        max_eval = -math.inf
        for i in range(6):
            if current_player_you_state[i] == 0: # Can't move from empty house
                continue

            new_my_you, new_my_opponent, extra_turn, game_over_sim, winner_sim = \
                simulate_move(current_player_you_state, current_player_opponent_state, i, True)

            current_eval = -math.inf # Initialize for this path

            if game_over_sim:
                if winner_sim == 'you': return math.inf # Immediate win
                elif winner_sim == 'opponent': return -math.inf # Immediate loss
                else: current_eval = 0 # Immediate draw
            elif extra_turn:
                # Still maximizing player, same depth (extra move)
                current_eval = minimax(new_my_you, new_my_opponent, depth, alpha, beta, True)
            else:
                # Opponent's turn. Swap roles.
                # `new_my_opponent` becomes the 'you' for the minimizing player's perspective.
                # `new_my_you` becomes the 'opponent' for the minimizing player's perspective.
                current_eval = minimax(new_my_opponent, new_my_you, depth - 1, alpha, beta, False)

            max_eval = max(max_eval, current_eval)
            alpha = max(alpha, current_eval)
            if beta <= alpha:
                break # Alpha-beta pruning
        return max_eval
    else: # Minimizing player's turn (original 'opponent')
        min_eval = math.inf
        for i in range(6):
            if current_player_you_state[i] == 0: # Opponent cannot move from empty house
                continue

            # Simulate opponent's move. `current_player_you_state` is opponent's 'you'.
            new_my_you, new_my_opponent, extra_turn, game_over_sim, winner_sim = \
                simulate_move(current_player_you_state, current_player_opponent_state, i, False)

            current_eval = math.inf # Initialize for this path

            if game_over_sim:
                if winner_sim == 'you': current_eval = math.inf # Max player wins (bad for min player)
                elif winner_sim == 'opponent': current_eval = -math.inf # Min player wins (good for min player)
                else: current_eval = 0 # Draw
            elif extra_turn:
                # Opponent gets extra turn, still minimizing player, same depth
                current_eval = minimax(new_my_opponent, new_my_you, depth, alpha, beta, False)
            else:
                # Opponent's turn ends, it's original 'you's turn (maximizing player)
                current_eval = minimax(new_my_you, new_my_opponent, depth - 1, alpha, beta, True)

            min_eval = min(min_eval, current_eval)
            beta = min(beta, current_eval)
            if beta <= alpha:
                break # Alpha-beta pruning
        return min_eval


def policy(you: list[int], opponent: list[int]) -> int:
    """
    Selects the best move for the current player using Minimax with Alpha-Beta Pruning.

    Args:
        you (list[int]): Your current board state ([h0..h5, store]).
        opponent (list[int]): Opponent's current board state ([h0..h5, store]).

    Returns:
        int: The index (0-5) of the house to move from.
    """
    best_move = -1
    max_eval = -math.inf

    # Iterate through all possible moves (your houses 0-5)
    for i in range(6):
        if you[i] == 0:
            continue # Cannot move from an empty house

        # Simulate the chosen move
        new_my_you, new_my_opponent, extra_turn, game_over_sim, winner_sim = \
            simulate_move(you, opponent, i, True) # Simulating as the original 'you'

        current_eval = -math.inf # Default for paths leading to immediate loss or very bad states

        if game_over_sim:
            if winner_sim == 'you':
                return i # Found an immediate winning move, take it!
            elif winner_sim == 'opponent':
                current_eval = -math.inf # This move leads to an immediate loss
            else: # Draw
                current_eval = 0 # This move leads to an immediate draw
        elif extra_turn:
            # If I get an extra turn, I continue to maximize from the new state, same depth
            current_eval = minimax(new_my_you, new_my_opponent, MAX_DEPTH, -math.inf, math.inf, True)
        else:
            # My turn ends, it's opponent's turn. Call minimax from opponent's perspective.
            # `new_my_opponent` becomes the 'you' for minimax, `new_my_you` becomes the 'opponent'.
            current_eval = minimax(new_my_opponent, new_my_you, MAX_DEPTH - 1, -math.inf, math.inf, False)

        if current_eval > max_eval:
            max_eval = current_eval
            best_move = i
        
        # Early exit if possible to win immediately
        if max_eval == math.inf:
            return best_move

    # Fallback: if somehow no best move is determined (e.g., all moves lead to negative infinity
    # and max_eval remains -inf, though theoretically a non-losing move should exist).
    # The problem statement guarantees a legal move exists.
    if best_move == -1:
        # This branch should ideally not be reached if evaluation and game rules are correct
        # and a legal move is always guaranteed. Pick the first legal move as a safeguard.
        for i in range(6):
            if you[i] > 0:
                return i

    return best_move
