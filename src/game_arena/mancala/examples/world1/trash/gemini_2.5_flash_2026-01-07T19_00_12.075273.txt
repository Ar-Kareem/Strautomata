def policy(you: list[int], opponent: list[int]) -> int:
    """
    Implements a move-selection policy for the Kalah game.
    It evaluates all possible legal moves and selects the best one based on a greedy strategy:
    
    1. Prioritize moves that grant an immediate extra turn.
    2. Among moves that don't grant an extra turn, prioritize those that result in a capture,
       favoring captures that yield more seeds.
    3. For all other moves, or to break ties, choose the move that maximizes the
       (my_final_score - opponent_final_score) differential.
       The final scores are calculated considering the game-ending conditions.
    
    Args:
        you: A list representing your side of the board [h0, h1, h2, h3, h4, h5, store].
        opponent: A list representing the opponent's side [h0, h1, h2, h3, h4, h5, store].
    
    Returns:
        An integer (0-5) representing the index of the house from which to make a move.
    """

    best_move = -1
    # Initialize criteria tuple: (has_extra_move, captured_seeds_count, net_score_differential)
    # Using negative infinity for net_score to ensure any valid move is considered better than the initial state.
    best_criteria_value = (False, 0, -sys.maxsize) 

    # Iterate through each of your houses (indices 0-5) to find legal moves
    for move_idx in range(6): 
        if you[move_idx] == 0:
            continue # Skip empty houses, as they are not legal moves

        # Simulate the chosen move
        simulation_result = _simulate_move(you, opponent, move_idx)
        
        # Calculate the potential final scores for both players after this simulated move,
        # taking into account game-ending conditions.
        my_final_score, opponent_final_score = _calculate_final_scores(
            simulation_result['you'], simulation_result['opponent']
        )
        
        # Calculate the net score differential (my score - opponent's score)
        net_score_differential = my_final_score - opponent_final_score
        
        # Combine the criteria into a tuple for easy comparison
        current_criteria_value = (
            simulation_result['extra_move'],
            simulation_result['captured_seeds'],
            net_score_differential
        )

        # Compare the current move's criteria with the best found so far.
        # Python's tuple comparison works lexicographically (left-to-right),
        # which perfectly matches our priority order: extra move > captures > net score.
        if current_criteria_value > best_criteria_value:
            best_criteria_value = current_criteria_value
            best_move = move_idx
            
    # The problem statement guarantees that policy is never called when no legal move exists,
    # so `best_move` will always be updated from its initial -1 to a valid move index.
    return best_move
