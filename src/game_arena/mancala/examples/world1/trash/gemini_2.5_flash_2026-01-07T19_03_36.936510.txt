class KalahAI:
    MAX_DEPTH = 6  # Adjust this based on performance

    def policy(self, you: list[int], opponent: list[int]) -> int:
        best_move = -1
        max_eval = float('-inf')

        # Find all legal moves
        possible_moves = [i for i in range(6) if you[i] > 0]
        
        # If no legal moves, the game should have ended.
        # The problem statement guarantees "policy is never called when you have no legal move."
        # So this case should ideally not be hit, but defensive check
        if not possible_moves:
             return -1 # Should not happen

        for move_idx in possible_moves:
            # Simulate the move for 'you'
            next_you, next_opponent, extra_turn_granted = self._distribute_seeds(you[:], opponent[:], move_idx)
            
            # Check if the game ended after this simulated move
            final_you_if_game_over, final_opponent_if_game_over, game_over = self._check_game_over(next_you[:], next_opponent[:])

            current_eval = 0
            if game_over:
                current_eval = self._evaluate(final_you_if_game_over, final_opponent_if_game_over)
            elif extra_turn_granted:
                # If extra turn, we remain the maximizing player
                current_eval = self._minimax(next_you, next_opponent, self.MAX_DEPTH - 0, float('-inf'), float('inf'), True)
            else:
                # Otherwise, it's opponent's turn, we are minimizing
                current_eval = self._minimax(next_you, next_opponent, self.MAX_DEPTH - 1, float('-inf'), float('inf'), False)

            if current_eval > max_eval:
                max_eval = current_eval
                best_move = move_idx
            
        # If best_move is still -1, it means no valid moves were found,
        # which again, should not happen based on problem statement.
        # As a fallback, just pick the first legal move available.
        if best_move == -1 and possible_moves:
            return possible_moves[0]

        return best_move

            else: # Currently on opponent's side houses

                    last_landed_on_player_side = False
                else: # Skip opponent's store (index 6), cycle back to player's board
                    board_iterator = 0 # Switch to player's side
                    current_pos = -1 # Reset index for player's houses
                    continue # Do not decrement seeds_to_distribute as no seed was placed

            seeds_to_distribute -= 1

        extra_turn_granted = False
        # Check for special conditions (extra move or capture)
        if last_landed_on_player_side and last_landed_house_idx == 6: # Last seed landed in player's store
            extra_turn_granted = True
        elif last_landed_on_player_side and 0 <= last_landed_house_idx < 6: # Last seed landed in player's empty house
            # Note: current_player_board[last_landed_house_idx] should be 1 if it was empty before the last seed
            if current_player_board[last_landed_house_idx] == 1 and current_opponent_board[5 - last_landed_house_idx] > 0:
                # Capture: move seeds from opponent's opposite house and the one seed from player's house to player's store
                current_player_board[6] += (current_opponent_board[5 - last_landed_house_idx] + current_player_board[last_landed_house_idx])
                current_opponent_board[5 - last_landed_house_idx] = 0 # Empty opponent's house
                current_player_board[last_landed_house_idx] = 0 # Empty player's house

        return current_player_board, current_opponent_board, extra_turn_granted

    def _check_game_over(self, player_board: list[int], opponent_board: list[int]) -> tuple[list[int], list[int], bool]:
        """
        Checks if the game has ended and finalizes scores if so.
        Returns modified boards and a boolean indicating if game is over.
        """
        game_over = False
        temp_player_board = list(player_board)
        temp_opponent_board = list(opponent_board)

        player_houses_empty = sum(temp_player_board[0:6]) == 0
        opponent_houses_empty = sum(temp_opponent_board[0:6]) == 0

        if player_houses_empty or opponent_houses_empty:
            game_over = True
            # Move all remaining seeds to respective stores
            for i in range(6):
                temp_player_board[6] += temp_player_board[i]
                temp_player_board[i] = 0
                temp_opponent_board[6] += temp_opponent_board[i]
                temp_opponent_board[i] = 0
                
        return temp_player_board, temp_opponent_board, game_over

    def _evaluate(self, you_board: list[int], opponent_board: list[int]) -> int:
        """
        Simple evaluation function: difference in score.
        """
        return you_board[6] - opponent_board[6]

    def _minimax(self, you: list[int], opponent: list[int], depth: int, alpha: float, beta: float, maximizing_player: bool) -> int:
        """
        Minimax algorithm with alpha-beta pruning.
        `you` and `opponent` are the current board states from the perspective of the maximizing_player.
        """
        
        # Check for terminal state or depth limit
        final_you, final_opponent, game_over = self._check_game_over(you[:], opponent[:])
        if depth == 0 or game_over:
            return self._evaluate(final_you, final_opponent)

        if maximizing_player:
            max_eval = float('-inf')
            for i in range(6):
                if you[i] == 0:
                    continue # Skip empty houses

                # Simulate my move
                next_you, next_opponent, extra_turn = self._distribute_seeds(you[:], opponent[:], i)
                
                if extra_turn:
                    # If I get an extra turn, I am still the maximizing player, depth doesn't decrease
                    eval = self._minimax(next_you, next_opponent, depth - 0, alpha, beta, True)
                else:
                    # My turn ends, opponent (minimizing player) moves next
                    eval = self._minimax(next_you, next_opponent, depth - 1, alpha, beta, False)
                
                max_eval = max(max_eval, eval)
                alpha = max(alpha, eval)
                if beta <= alpha: # Alpha-beta pruning
                    break
            return max_eval
        else: # Minimizing player (opponent's turn)
            min_eval = float('inf')
            for i in range(6):
                if opponent[i] == 0:
                    continue # Skip empty houses for opponent

                # Simulate opponent's move.
                # `_distribute_seeds` expects (player_board, opponent_board).
                # So we pass current_opponent as 'player_board' and current_you as 'opponent_board'.
                next_opponent, next_you, extra_turn = self._distribute_seeds(opponent[:], you[:], i)
                
                if extra_turn:
                    # Opponent gets an extra turn, they are still the minimizing player, depth doesn't decrease
                    eval = self._minimax(next_you, next_opponent, depth - 0, alpha, beta, False)
                else:
                    # Opponent's turn ends, it's my turn (maximizing player) next
                    eval = self._minimax(next_you, next_opponent, depth - 1, alpha, beta, True)
                
                min_eval = min(min_eval, eval)
                beta = min(beta, eval)
                if beta <= alpha: # Alpha-beta pruning
                    break
            return min_eval

When calling `_minimax`, I need to maintain the convention that `you` is always *my* perspective and `opponent` is the *other* player's perspective.

In `policy` and `_minimax` for `maximizing_player`:
`_distribute_seeds(you[:], opponent[:], i)` -> `(next_you, next_opponent, extra_turn)`
This is already consistent since `you` is `player_board` and `opponent` is `opponent_board`.

In `_minimax` for `not maximizing_player`:
`_distribute_seeds(opponent[:], you[:], i)` -> `(next_opponent_from_their_perspective, next_my_board_from_their_perspective, extra_turn)`
This means `next_opponent_from_their_perspective` is what `opponent` would be FROM MY PERSPECTIVE after their move.
And `next_my_board_from_their_perspective` is what `you` would be FROM MY PERSPECTIVE after their move.

`eval = self._minimax(next_my_board_from_their_perspective, next_opponent_from_their_perspective, ...)`

class KalahAI:
    MAX_DEPTH = 6  # Maximum search depth for the minimax algorithm

    def policy(self, you: list[int], opponent: list[int]) -> int:
        """
        Determines the best move for the current player using a minimax algorithm
        with alpha-beta pruning.
        
        Args:
            you: A list representing the current player's board state (houses 0-5, store 6).
            opponent: A list representing the opponent's board state (houses 0-5, store 6).
            
        Returns:
            An integer (0-5) representing the index of the house to move from.
        """
        best_move = -1
        max_eval = float('-inf')

        # Identify all legal moves (houses with seeds)
        # The problem statement guarantees this function is never called when you have no legal move.
        possible_moves = [i for i in range(6) if you[i] > 0]
        
        for move_idx in possible_moves:
            # Simulate the chosen move
            # _distribute_seeds returns the board states AFTER the move and whether an extra turn was granted
            next_you, next_opponent, extra_turn_granted = self._distribute_seeds(list(you), list(opponent), move_idx)
            
            # After a move, check if the game has ended for evaluation
            final_you_if_game_over, final_opponent_if_game_over, game_over = self._check_game_over(list(next_you), list(next_opponent))

            current_eval = 0
            if game_over:
                # If the game ends, simply evaluate the final score
                current_eval = self._evaluate(final_you_if_game_over, final_opponent_if_game_over)
            elif extra_turn_granted:
                # If an extra turn is granted, it's still our turn (maximizing player).
                # Depth does not decrease as it's part of the same player's sequence of moves.
                current_eval = self._minimax(next_you, next_opponent, self.MAX_DEPTH - 0, float('-inf'), float('inf'), True)
            else:
                # If no extra turn, it's the opponent's turn (minimizing player).
                # Depth decreases by one
                current_eval = self._minimax(next_you, next_opponent, self.MAX_DEPTH - 1, float('-inf'), float('inf'), False)

            # Update best move if current move leads to a better outcome
            if current_eval > max_eval:
                max_eval = current_eval
                best_move = move_idx
        
        # If no best move was found (e.g., all moves lead to -inf), or as a fallback
        # This branch should rarely/never be hit if `possible_moves` is guaranteed to be non-empty.
        if best_move == -1 and possible_moves:
            return possible_moves[0] # Return the first legal move as a failsafe

        return best_move

    def _distribute_seeds(self, player_board: list[int], opponent_board: list[int], start_house_idx: int) -> tuple[list[int], list[int], bool]:
        """
        Simulates the distribution of seeds from a given house.
        Handles wrapping around the board, landing in stores, captures, and extra turns.
        
        Args:
            player_board: The board state of the player making the move.
            opponent_board: The board state of the other player.
            start_house_idx: The index (0-5) of the house from which seeds are picked.
            
        Returns:
            A tuple containing:
            - The new state of player_board.
            - The new state of opponent_board.
            - A boolean indicating whether an extra turn was granted.
        """
        
        # Make copies of the boards to simulate without affecting the original state
        current_player_board = list(player_board)
        current_opponent_board = list(opponent_board)

        seeds_to_distribute = current_player_board[start_house_idx]
        current_player_board[start_house_idx] = 0 # Empty the starting house

        last_landed_house_idx = -1
        last_landed_on_player_side = False

        board_iterator_state = 0 # 0 for current player's side, 1 for opponent's side
        current_pos_in_bank = start_house_idx # Position within the current 'bank' (0-6 for player, 0-5 for opponent)

        while seeds_to_distribute > 0:
            if board_iterator_state == 0: # Current player's side (houses or store)
                current_pos_in_bank += 1
                if current_pos_in_bank < 6: # Player's houses (0-5)
                    current_player_board[current_pos_in_bank] += 1
                    last_landed_house_idx = current_pos_in_bank
                    last_landed_on_player_side = True
                elif current_pos_in_bank == 6: # Player's store (6)
                    current_player_board[current_pos_in_bank] += 1
                    last_landed_house_idx = current_pos_in_bank # Mark as landed in store (index 6)
                    last_landed_on_player_side = True
                    board_iterator_state = 1 # Switch to opponent's side for next seeds
                    current_pos_in_bank = -1 # Reset index for next iteration to start at opponent's house 0
                # No other `current_pos_in_bank` values are valid here; control flows to seeds_to_distribute -= 1
            else: # Opponent's side (houses only, skipping their store)
                current_pos_in_bank += 1
                if current_pos_in_bank < 6: # Opponent's houses (0-5)
                    current_opponent_board[current_pos_in_bank] += 1
                    last_landed_house_idx = current_pos_in_bank
                    last_landed_on_player_side = False
                else: # Pass opponent's houses, skip their store, cycle back to player's board
                    board_iterator_state = 0 # Switch back to player's side
                    current_pos_in_bank = -1 # Reset index for next iteration to start at player's house 0
                    seeds_to_distribute -= 1 # Crucially, a seed was NOT placed, but iteration happens. So decrement here.
                    continue # Skip general seed decrement at bottom of loop as no seed was placed

            seeds_to_distribute -= 1 # Decrement for each seed placed

        extra_turn_granted = False

        if last_landed_on_player_side and last_landed_house_idx == 6:
            extra_turn_granted = True
        elif last_landed_on_player_side and 0 <= last_landed_house_idx < 6:
            # Capture condition: last seed landed in current player's house, and it was empty before this seed
            # (which means it now contains exactly 1 seed), and the opponent's opposite house has seeds.
            if current_player_board[last_landed_house_idx] == 1 and current_opponent_board[5 - last_landed_house_idx] > 0:
                # Move seeds from opponent's opposite house and the one seed from player's house to player's store
                current_player_board[6] += (current_opponent_board[5 - last_landed_house_idx] + current_player_board[last_landed_house_idx])
                current_opponent_board[5 - last_landed_house_idx] = 0 # Empty opponent's house
                current_player_board[last_landed_house_idx] = 0 # Empty player's house

        return current_player_board, current_opponent_board, extra_turn_granted

    def _check_game_over(self, player_board: list[int], opponent_board: list[int]) -> tuple[list[int], list[int], bool]:
        """
        Checks if the game has ended based on either player having no seeds in their houses.
        If the game is over, remaining seeds are moved to the respective stores.
        
        Args:
            player_board: The current player's board state.
            opponent_board: The opponent's board state.
            
        Returns:
            A tuple containing:
            - The finalized player_board.
            - The finalized opponent_board.
            - A boolean indicating if the game is over.

        temp_player_board = list(player_board) # Work on copies
        temp_opponent_board = list(opponent_board)

        player_houses_empty = sum(temp_player_board[0:6]) == 0
        opponent_houses_empty = sum(temp_opponent_board[0:6]) == 0

        if player_houses_empty or opponent_houses_empty:

            # Move all remaining seeds to respective stores
            for i in range(6):
                temp_player_board[6] += temp_player_board[i]
                temp_player_board[i] = 0
                temp_opponent_board[6] += temp_opponent_board[i]
                temp_opponent_board[i] = 0
                
        return temp_player_board, temp_opponent_board, game_over

    def _evaluate(self, you_board: list[int], opponent_board: list[int]) -> int:
        """
        Evaluation function: simply returns the difference in seeds in stores.
        Positive values favor 'you', negative values favor 'opponent'.
        """
        return you_board[6] - opponent_board[6]

    def _minimax(self, you: list[int], opponent: list[int], depth: int, alpha: float, beta: float, maximizing_player: bool) -> int:
        """
        Minimax algorithm with alpha-beta pruning.
        
        Args:
            you: The 'you' player's board state.
            opponent: The 'opponent' player's board state.
            depth: The current remaining search depth.
            alpha: The alpha value for alpha-beta pruning.
            beta: The beta value for alpha-beta pruning.
            maximizing_player: True if it's the current player's turn to maximize score, False for opponent.
            
        Returns:
            The evaluated score for the current state.
        """
        
        # Check for terminal state or depth limit
        final_you, final_opponent, game_over = self._check_game_over(list(you), list(opponent))
        if depth == 0 or game_over:
            return self._evaluate(final_you, final_opponent)

        if maximizing_player:
            max_eval = float('-inf')

            for i in range(6):

                    continue # Skip empty houses

                next_you, next_opponent, extra_turn = self._distribute_seeds(list(you), list(opponent), i)

                    # Depth does not decrease as it's part of the same player's turn
                    eval = self._minimax(next_you, next_opponent, depth - 0, alpha, beta, True)
                else:
                    # If turn passes, recursively call minimax for the opponent (minimizing)
                    eval = self._minimax(next_you, next_opponent, depth - 1, alpha, beta, False)
                
                max_eval = max(max_eval, eval)
                alpha = max(alpha, eval)
                if beta <= alpha: # Alpha-beta pruning
                    break # Prune this branch
            return max_eval
        else: # Minimizing player (opponent's turn)
            min_eval = float('inf')
            # Iterate through all possible moves for the minimizing player
            for i in range(6):
                if opponent[i] == 0:
                    continue # Skip empty houses for opponent

                # Simulate opponent's move.
                # _distribute_seeds expects `player_board` first, so we pass opponent's board as `player_board`
                # and our board (`you`) as `opponent_board`.
                next_opponent_board, next_my_board, extra_turn = self._distribute_seeds(list(opponent), list(you), i)

                    # Opponent gets an extra turn, so they are still the minimizing player
                    eval = self._minimax(next_my_board, next_opponent_board, depth - 0, alpha, beta, False)
                else:
                    # Opponent's turn ends, it's our turn (maximizing player) next
                    eval = self._minimax(next_my_board, next_opponent_board, depth - 1, alpha, beta, True)
                
                min_eval = min(min_eval, eval)
                beta = min(beta, eval)
                if beta <= alpha: # Alpha-beta pruning
                    break # Prune this branch
            return min_eval
