def policy(you: list[int], opponent: list[int]) -> int:
    def simulate_move(you_state, opp_state, i):
        you_copy = you_state.copy()
        opp_copy = opp_state.copy()
        seeds = you_copy[i]
        if seeds == 0:
            return None
        you_copy[i] = 0
        start_index = i + 1
        for seed_num in range(seeds):
            pos_in_loop = (start_index + seed_num) % 13
            if pos_in_loop < 6:
                you_copy[pos_in_loop] += 1
            elif pos_in_loop == 6:
                you_copy[6] += 1
            else:
                opp_index = pos_in_loop - 7
                opp_copy[opp_index] += 1
        last_pos = (start_index + seeds - 1) % 13
        extra_move = False
        is_capture = False
        capture_amount = 0
        if last_pos == 6:
            extra_move = True
        elif 0 <= last_pos <= 5:
            prev_house_value = you_copy[last_pos] - 1
            if prev_house_value == 0:
                opp_house_index = 5 - last_pos
                if opp_copy[opp_house_index] > 0:
                    is_capture = True
                    capture_amount = you_copy[last_pos] + opp_copy[opp_house_index]
                    you_copy[6] += capture_amount
                    you_copy[last_pos] = 0
                    opp_copy[opp_house_index] = 0
        return (you_copy, opp_copy, extra_move, is_capture, capture_amount)
    
    best_move = -1
    max_priority = -1
    max_score = -float('inf')
    for i in range(6):
        if you[i] == 0:
            continue
        sim_results = simulate_move(you, opponent, i)
        if sim_results is None:
            continue
        you_new, opp_new, extra_move, is_capture, cap_amt = sim_results
        store_diff = you_new[6] - opp_new[6]
        if is_capture:
            priority = 2
            move_score = cap_amt
        elif extra_move:
            priority = 1
            move_score = store_diff + 10  # Bonus for extra move
        else:
            priority = 0
            move_score = store_diff
        if (priority > max_priority) or (priority == max_priority and move_score > max_score):
            max_priority = priority
            max_score = move_score
            best_move = i
    return best_move
