#!/usr/bin/env python3
"""
Simple heuristic policy for Kalah (6 houses per side + 1 store).

The policy evaluates every legal move, simulates its effect,
and selects the move with the highest priority:
    1. Extra turn (bool)
    2. Increase in our store (int)
    3. Number of seeds taken from the chosen house (int)

The function signature required by the arena is:

    def policy(you: list[int], opponent: list[int]) -> int:
        ...

Both `you` and `opponent` are lists of length 7:
    indices 0..5 -> houses, index 6 -> store.
"""

from __future__ import annotations
from typing import List, Tuple


def _advance(side: int, idx: int) -> Tuple[int, int]:
    """
    Return the next pit in the sowing order, skipping opponent's store.
    side == 0 -> our side, side == 1 -> opponent's side.
    """
    if side == 0:            # our side
        if idx < 5:          # house -> next house
            return 0, idx + 1
        elif idx == 5:       # last house -> our store
            return 0, 6
        else:                # our store -> opponent house 0
            return 1, 0
    else:                    # opponent's side (store is never visited)
        if idx < 5:          # house -> next house
            return 1, idx + 1
        else:                # last opponent house -> our house 0
            return 0, 0


def _simulate_move(
    move_idx: int,
    you: List[int],
    opponent: List[int]
) -> Tuple[List[int], List[int], bool, int]:
    """
    Simulate sowing from house `move_idx` (0..5) on our side.

    Returns:
        new_you, new_opponent   copies of the boards after the move
        extra_turn              True if the last seed lands in our store
        store_gain              Number of seeds added to our store by this move
                                (captures and the seed(s) placed in the store)
    """
    # copy boards to avoid mutating the originals
 new_you = you.copy()
    new_opp = opponent.copy()

    seeds = new_you[move_idx]
    new_you[move_idx] = 0               # pick up all seeds

    side, idx = 0, move_idx            # start from the emptied house
    last_side = last_idx = -1
    while seeds:
        side, idx = _advance(side, idx)
        # skip opponent's store (side==1 and idx==6) – never generated by _advance
        if side == 0:
            new_you[idx] += 1
        else:
            new_opp[idx] += 1
        seeds -= 1
        last_side, last_idx = side, idx

    extra_turn = (last_side == 0 and last_idx == 6)

    store_gain = 0
    # capture rule
    if not extra_turn and last_side == 0 and 0 <= last_idx <= 5:
        # check if the pit was empty before the last seed landed
        if new_you[last_idx] == 1:  # it became 1 now, meaning it was empty
            opposite = 5 - last_idx
            if new_opp[opposite] > 0:
                captured = new_opp[opposite] + 1  # opponent seeds + our last seed
                new_you[last_idx] = 0
                new_opp[opposite] = 0
                new_you[6] += captured
                store_gain = captured

    # seeds placed into our store during sowing (including the last one if it lands there)
    if extra_turn:
        # count how many seeds we dropped into the store this turn
        # we can compute it as the difference in store size
        store_gain = new_you[6] - you[6]
    else:
        # we may have already added store_gain via capture; also include any seeds
        # that landed in the store during the sowing (if any)
        store_gain = new_you[6] - you[6]

    return new_you, new_opp, extra_turn, store_gain


def policy(you: List[int], opponent: List[int]) -> int:
    """
    Choose a legal move (house index 0..5) based on a simple heuristic.
    """
    best_move = None
    best_score = None  # will store a tuple for lexicographic comparison

    for i in range(6):
        if you[i] == 0:
            continue  # illegal move

        new_you, new_opp, extra, gain = _simulate_move(i, you, opponent)

        # Priority tuple:
        #   extra turn (bool, True > False)
        #   store gain (int, larger is better)
        #   seeds taken from the house (int, larger is better)
        score = (extra, gain, you[i])

        if best_score is None or score > best_score:
            best_score = score
            best_move = i

    # According to the problem statement, there is always at least one legal move.
    return best_move  # type: ignore

# The module exports only the required function.
__all__ = ["policy"]
