def simulate_move(you_initial: list[int], opponent_initial: list[int], move_idx: int) -> tuple[list[int], list[int], bool, bool]:
    """
    Simulates a single move for the current player without modifying the original states.
    Returns the resulting board state (temp_you, temp_opponent), whether an extra move was granted,
    and whether a capture occurred.
    """
    temp_you = list(you_initial)
    temp_opponent = list(opponent_initial)

    seeds_to_sow = temp_you[move_idx]
    temp_you[move_idx] = 0  # Remove seeds from chosen house

    current_sowing_idx = move_idx  # This is the house index from which sowing started. The first seed goes to current_sowing_idx + 1.
    last_landed_side = None
    last_landed_house_idx = -1  # For house indices 0-5, or 6 for your store

            last_landed_house_idx = current_sowing_idx
        elif current_sowing_idx == 6:  # Your store
            temp_you[current_sowing_idx] += 1
            last_landed_side = 'you'
            last_landed_house_idx = current_sowing_idx
        elif current_sowing_idx < 13:  # Opponent's houses (0-5, corresponding to original indices as 7->0, 8->1 etc.)

            temp_opponent[opponent_house_idx] += 1
            last_landed_side = 'opponent'
            last_landed_house_idx = opponent_house_idx
