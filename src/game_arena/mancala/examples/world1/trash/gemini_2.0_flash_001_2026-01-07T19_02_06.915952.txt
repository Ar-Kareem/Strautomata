```python
import random

def policy(you: list[int], opponent: list[int]) -> int:
    """
    Selects a legal move for the Kalah game.

    Args:
        you: Your board state (list of 7 integers).
        opponent: The opponent's board state (list of 7 integers).

    Returns:
        An integer representing the chosen house (0-5).
    """
    legal_moves = [i for i in range(6) if you[i] > 0]

    # If there's only one legal move, take it
    if len(legal_moves) == 1:
        return legal_moves[0]

    # Try for a capture
    for move in legal_moves:
        seeds = you[move]
        index = move
        temp_you = you[:]
        temp_opponent = opponent[:]
        temp_you[move] = 0
        
        seeds_to_distribute  = seeds
        current_index = move
        current_side = 0 # 0 for you, 1 for opponent

        while seeds_to_distribute > 0:
          current_index = (current_index + 1) % 14 # total num spots 6+6+1+1
          seeds_to_distribute -= 1

          
          if current_index < 6:
              temp_you[current_index] += 1
          elif current_index == 6:
              temp_you[6] += 1
          elif current_index < 12:
              temp_opponent[current_index - 7] +=1
          elif current_index == 12:
            pass # skip opponent store
          else:
              temp_you[current_index - 13] += 1
        
        final_index = current_index
        adjusted_final_index = final_index if final_index < 7 else final_index - 7

        if final_index < 6: # you side
            if temp_you[adjusted_final_index] == 1 and you[adjusted_final_index] == 0 and temp_opponent[5 - adjusted_final_index] > 0: #capture
                    return move

    # Try for an extra turn
    for move in legal_moves:
        seeds = you[move]
        index = move
        temp_you = you[:]
        temp_opponent = opponent[:]
        temp_you[move] = 0
        
        seeds_to_distribute  = seeds
        current_index = move
        current_side = 0 # 0 for you, 1 for opponent

        while seeds_to_distribute > 0:
          current_index = (current_index + 1) % 14 # total num spots 6+6+1+1
          seeds_to_distribute -= 1

          
          if current_index < 6:
              temp_you[current_index] += 1
          elif current_index == 6:
              temp_you[6] += 1
          elif current_index < 12:
              temp_opponent[current_index - 7] +=1
          elif current_index == 12:
            pass # skip opponent store
          else:
              temp_you[current_index - 13] += 1
        
        final_index = current_index
        
        if final_index == 6:
            return move


    # Otherwise, choose a random legal move
    return random.choice(legal_moves)
```
