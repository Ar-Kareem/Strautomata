```python
# Policy function
def policy(you, opponent):
    legal_moves = [i for i in range(6) if you[i] > 0]

    best_move_score = -float('inf')
    best_move = -1

    # Store moves that grant an extra turn separately
    extra_turn_moves = []

    for move in legal_moves:
        original_you_store = you[6]
        original_opponent_store = opponent[6] # Not strictly needed for immediate calculation, but good to have

        # Simulate the move
        temp_you, temp_opp, extra_move, captured_seeds = simulate_move(you, opponent, move)

        current_move_score = temp_you[6] - original_you_store

        # Strong preference for extra moves
        if extra_move:
            # If an extra move is possible, this is usually the best.
            # We want to pick the EXTRA_MOVE that also maximizes own store.
            extra_turn_moves.append((current_move_score, move))
            continue # Don't consider other criteria if this gives an extra move

        # If not an extra move, consider captures
        if captured_seeds > 0:
            # Captures are also very good. Give a large bonus.
            # The captured_seeds are already included in temp_you[6], so just add a capture bonus.
            # Maybe a fixed bonus to make it higher priority than just store gain.
            # The captured_seeds value *is* the total collected at that step in the simulate.
            # No, the `current_move_score` already includes the captured seeds as a gain to `temp_you[6]`.
            # So `captured_seeds` itself just tells us that a capture *happened*.
            # The value from the capture is already reflected in `current_move_score`.
            # To prioritize captures over mere store gains, we need an explicit bonus.
            # Let's say a capture itself earns a value like '100' (much higher than typical store gain)
            # This makes sure captures are prioritized over just landing in store.
            current_move_score += 100 # Arbitrary high bonus for a capture.

        if current_move_score > best_move_score:
            best_move_score = current_move_score
            best_move = move

    # Decision logic:
    if extra_turn_moves:
        # If there are moves that give an extra turn, choose the one maximizing store gain.
        # Sort by score (first element of tuple) in descending order.
        extra_turn_moves.sort(key=lambda x: x[0], reverse=True)
        return extra_turn_moves[0][1] # Return the move index

    # If no extra turn moves, return the best_move found based on score (which includes capture bonus)
    return best_move
```

This greedy approach prioritizes:
1.  Extra turns (and within those, maximizing immediate store gain).
2.  Captures (and within those, maximizing immediate store gain, due to the bonus).
3.  Maximizing immediate store gain.

import collections

def simulate_move(you_state: list[int], opp_state: list[int], move_idx: int) -> tuple[list[int], list[int], bool, int]:
    """
    Simulates a single move in Kalah and returns the resulting board state,
    whether an extra move occurred, and how many seeds were captured.

    Args:
        you_state: Your current board state [h0, h1, h2, h3, h4, h5, store].
        opp_state: Opponent's current board state [h0, h1, h2, h3, h4, h5, store].
        move_idx: The index of the house (0-5) from which to make the move.

    Returns:
        A tuple: (modified_you_state, modified_opp_state, extra_move_flag, captured_seeds_count)
    """
    temp_you = list(you_state)
    temp_opp = list(opp_state)

    seeds_to_sow = temp_you[move_idx]
    temp_you[move_idx] = 0  # Pick up seeds from the chosen house

    current_pit_absolute_idx = move_idx 
    
    last_sown_on_your_side = False
    last_sown_pit_idx_on_side = -1 # 0-5 for houses, 6 for store (on _your_ side)

    # Conceptual board for sowing:
    # 0-5: your houses
    # 6: your store
    # 7-12: opponent's houses (map to opp_state[0] to opp_state[5])
    # Opponent's store (13) is skipped. Total 13 soweable pits.

    for _ in range(seeds_to_sow):
        current_pit_absolute_idx += 1
        
        # Wrap around the board, skipping opponent's store
        if current_pit_absolute_idx == 13: 
            current_pit_absolute_idx = 0 # Wrap around to your h0

        if current_pit_absolute_idx < 6: # Your houses 0-5
            temp_you[current_pit_absolute_idx] += 1
            last_sown_on_your_side = True
            last_sown_pit_idx_on_side = current_pit_absolute_idx
        elif current_pit_absolute_idx == 6: # Your store (index 6)
            temp_you[6] += 1
            last_sown_on_your_side = True
            last_sown_pit_idx_on_side = 6
        else: # Opponent's houses 7-12 (which map to opp_state[0-5])
            # Map absolute index (7-12) to opponent's house index (0-5)
            opposite_house_idx = current_pit_absolute_idx - 7 
            temp_opp[opposite_house_idx] += 1
            last_sown_on_your_side = False
            last_sown_pit_idx_on_side = opposite_house_idx # This is the 0-5 index for opponent's houses

    # Evaluate consequences of the last seed drop
    extra_move = False
    captured_seeds = 0

    if last_sown_on_your_side:
        if last_sown_pit_idx_on_side == 6: # Last seed landed in your store
            extra_move = True
        elif last_sown_pit_idx_on_side < 6: # Last seed landed in one of your houses (0-5)
            # Check for capture:
            # 1. The house must have been empty before this last seed was dropped (now contains 1 seed)
            # 2. The opposite house on opponent's side must contain seeds
            opposite_opp_house_idx = 5 - last_sown_pit_idx_on_side
            if temp_you[last_sown_pit_idx_on_side] == 1 and temp_opp[opposite_opp_house_idx] > 0:
                # Capture occurs!
                captured_seeds = temp_you[last_sown_pit_idx_on_side] + temp_opp[opposite_opp_house_idx]
                temp_you[6] += captured_seeds # Add captured seeds to your store
                temp_you[last_sown_pit_idx_on_side] = 0 # Empty your house
                temp_opp[opposite_opp_house_idx] = 0 # Empty opponent's house
                
    return temp_you, temp_opp, extra_move, captured_seeds


def policy(you: list[int], opponent: list[int]) -> int:
    """
    Implements a move-selection policy for the game Kalah.
    The policy prioritizes moves that give an extra turn, then captures,
    and finally maximizes the immediate score difference.

    Args:
        you: Your current board state [h0, h1, h2, h3, h4, h5, store].
        opponent: Opponent's current board state [h0, h1, h2, h3, h4, h5, store].

    Returns:
        The index (0-5) of the house from which to make the desired move.
    """

    legal_moves = [i for i in range(6) if you[i] > 0]

    best_move_score = -float('inf')
    best_move = legal_moves[0] if legal_moves else -1 # Fallback, though problem guarantees legal move exists

    # Stores moves that result in an extra turn.
    # Each entry is (store_gain_from_move, move_idx)
    extra_turn_moves = []

    for move_idx in legal_moves:
        original_you_store = you[6]

        # Simulate the move to get the outcome
        temp_you, _, extra_move, captured_seeds = simulate_move(you, opponent, move_idx)

        # Calculate the immediate gain to your store
        current_move_store_gain = temp_you[6] - original_you_store
        
        # Assign a score to the current move
        move_score = current_move_store_gain

        # Prioritize moves that grant an extra turn
        if extra_move:
            extra_turn_moves.append((move_score, move_idx))
            continue # Skip further scoring for now, extra turns are handled separately

        # Prioritize captures if no extra turn
        if captured_seeds > 0:
            # Add a significant bonus to differentiate captures from simple store gains.
            # This ensures captures are preferred over non-capture, non-extra-turn moves.
            move_score += 1000 # Large arbitrary bonus to elevate capture priority

        # Update best move if this one is better based on combined score
        if move_score > best_move_score:
            best_move_score = move_score
            best_move = move_idx

    # Decision logic:
    if extra_turn_moves:
        # If there are moves that give an extra turn, pick the one that also maximizes
        # the immediate store gain during that extra turn.
        # Sort by store gain (first element of tuple) in descending order.
        extra_turn_moves.sort(key=lambda x: x[0], reverse=True)
        return extra_turn_moves[0][1] # Return the move index of the best extra-turn move
    else:
        # If no extra turn moves, return the best move found
        # based on score (which incorporates capture bonus).
        return best_move
