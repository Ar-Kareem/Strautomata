def policy(you: list[int], opponent: list[int]) -> int:
    def simulate_move(you_state, opponent_state, i):
        new_you = you_state.copy()
        new_opponent = opponent_state.copy()
        m = new_you[i]
        new_you[i] = 0
        current_side = 'you'
        current_index_in_side = i + 1
        last_seed_extra_info = None
        
        for seed in range(m):
            if current_side == 'you':
                prev_val = new_you[current_index_in_side]
                new_you[current_index_in_side] += 1
                if seed == m - 1:
                    last_seed_extra_info = ('you', current_index_in_side, prev_val == 0)
                # Move to next index
                if current_index_in_side < 5:
                    current_index_in_side += 1
                else:
                    current_side = 'store'
                    current_index_in_side = 0  # dummy index
            elif current_side == 'store':
                new_you[6] += 1
                if seed == m - 1:
                    last_seed_extra_info = ('store', 6, None)
                current_side = 'opponent'
                current_index_in_side = 0
            elif current_side == 'opponent':
                prev_val = new_opponent[current_index_in_side]
                new_opponent[current_index_in_side] += 1
                if seed == m - 1:
                    last_seed_extra_info = ('opponent', current_index_in_side, None)
                # Move to next index
                if current_index_in_side < 5:
                    current_index_in_side += 1
                else:
                    current_side = 'you'
                    current_index_in_side = 0
        
        extra_move_flag = False
        if last_seed_extra_info and last_seed_extra_info[0] == 'store':
            extra_move_flag = True
        elif last_seed_extra_info and last_seed_extra_info[0] == 'you':
            house_index = last_seed_extra_info[1]
            was_empty = last_seed_extra_info[2]
            if was_empty:
                opposite_house_index = 5 - house_index
                if new_opponent[opposite_house_index] > 0:
                    captured_seeds = new_you[house_index] + new_opponent[opposite_house_index]
                    new_you[6] += captured_seeds
                    new_you[house_index] = 0
                    new_opponent[opposite_house_index] = 0
        return new_you, new_opponent, extra_move_flag

    best_score = -float('inf')
    best_move = 0
    for i in range(6):
        if you[i] == 0:
            continue
        new_you, new_opponent, extra_move = simulate_move(you, opponent, i)
        score = new_you[6] - new_opponent[6]
        if extra_move:
            score += 10  # Bonus for extra move
        # Update best move if current score is better
        if score > best_score:
            best_score = score
            best_move = i
        elif score == best_score:
            # Prefer the move with the smallest index in case of a tie
            if i < best_move:
                best_move = i
    return best_move
