    you = list(you_orig)
    opponent = list(opponent_orig)

    seeds_to_sow = you[move_idx]
    you[move_idx] = 0 # Pick up seeds

    # Variables to track last landed position
    last_landed_idx = -1
    last_landed_is_you_side = False

    current_idx_on_your_side = move_idx + 1 # Start sowing from the next house
    current_idx_on_opponent_side = 0

    while seeds_to_sow > 0:
        # Sowing on your side
        while seeds_to_sow > 0 and current_idx_on_your_side < 7: # 0-5 houses, 6 store
            you[current_idx_on_your_side] += 1
            seeds_to_sow -= 1
            last_landed_idx = current_idx_on_your_side
            last_landed_is_you_side = True
            current_idx_on_your_side += 1

        # Reset for opponent's side, unless it's the end of a round
        current_idx_on_your_side = 0 # Wrap around if more seeds

        # Sowing on opponent's side

            opponent[current_idx_on_opponent_side] += 1
            seeds_to_sow -= 1

import math

def policy(you: list[int], opponent: list[int]) -> int:
    """
    Implements a move-selection policy for the game Kalah.
    The policy uses a 1-ply search with a heuristic that prioritizes:
    1. Getting an extra move.
    2. Maximizing the difference between its own store and the opponent's store.
    3. Captures are implicitly handled as they increase the store difference.

    Args:
        you: A list of 7 integers representing your houses (0-5) and your store (6).
        opponent: A list of 7 integers representing the opponent's houses (0-5) and their store (6).

    Returns:
        An integer (0-5) representing the chosen house to make a move from.
    """

    def simulate_move(you_orig: list[int], opponent_orig: list[int], move_idx: int) -> tuple[list[int], list[int], bool]:
        """
        Simulates a single move on a Kalah board.

        Args:
            you_orig: The original state of your side.
            opponent_orig: The original state of the opponent's side.
            move_idx: The index of the house (0-5) from which seeds are taken.

        Returns:
            A tuple containing:
            - new_you: The state of your side after the move.
            - new_opponent: The state of the opponent's side after the move.
            - got_extra_move: True if the last seed landed in your store, False otherwise.
        """
        # Make deep copies to avoid modifying original game state
        you_sim = list(you_orig)
        opponent_sim = list(opponent_orig)

        seeds_to_sow = you_sim[move_idx]
        you_sim[move_idx] = 0  # Pick up seeds from the chosen house

        # Variables to track the position where the last seed landed
        last_landed_idx = -1
        last_landed_is_you_side = False

        # Start sowing from the house immediately after the chosen one
        current_idx_on_your_side = move_idx + 1
        current_idx_on_opponent_side = 0 # Reset for opponent's side when it's their turn

        while seeds_to_sow > 0:
            # Sowing on your side
            # Iterate through your houses (0-5) and your store (6)
            while seeds_to_sow > 0 and current_idx_on_your_side <= 6:
                you_sim[current_idx_on_your_side] += 1
                seeds_to_sow -= 1
                last_landed_idx = current_idx_on_your_side
                last_landed_is_you_side = True
                current_idx_on_your_side += 1
            if seeds_to_sow == 0:
                break  # All seeds sown

            # If seeds remain, transition to opponent's side
            current_idx_on_your_side = 0 # Reset for next round on your side

            # Sowing on opponent's side
            # Iterate through opponent's houses (0-5), skipping their store (6)
            while seeds_to_sow > 0 and current_idx_on_opponent_side <= 5:
                opponent_sim[current_idx_on_opponent_side] += 1
                seeds_to_sow -= 1
                last_landed_idx = current_idx_on_opponent_side
                last_landed_is_you_side = False
                current_idx_on_opponent_side += 1
            if seeds_to_sow == 0:
                break  # All seeds sown

            # If seeds remain, transition back to your side, restarting from your house 0
            current_idx_on_opponent_side = 0 # Reset for next round on opponent's side


        # --- Check for special effects after all seeds are sown ---

        got_extra_move = False
        # If the last seed landed in your store, you get an extra move
        if last_landed_is_you_side and last_landed_idx == 6:
            got_extra_move = True

        # Capture logic:
        # 1. Last seed landed in one of your houses (0-5).
        # 2. That house was empty BEFORE the entire move (you_orig[last_landed_idx] == 0).
        # 3. That house currently has exactly 1 seed (because the last seed landed there).
        # 4. The opposite opponent's house has seeds.
        if last_landed_is_you_side and 0 <= last_landed_idx <= 5:
            if you_orig[last_landed_idx] == 0 and you_sim[last_landed_idx] == 1:
                opponent_opposite_house_idx = 5 - last_landed_idx
                if opponent_sim[opponent_opposite_house_idx] > 0:
                    # Perform capture
                    you_sim[6] += you_sim[last_landed_idx]  # Capture the single seed from your house
                    you_sim[last_landed_idx] = 0
                    you_sim[6] += opponent_sim[opponent_opposite_house_idx]  # Capture opponent's seeds
                    opponent_sim[opponent_opposite_house_idx] = 0

        return you_sim, opponent_sim, got_extra_move

    # Initialize best score to negative infinity to ensure any legal move is better
    best_score = -math.inf
    best_move = -1

    # A significant bonus for getting an extra move, making it highly preferred.
    # This value should be much larger than any possible store difference from a single move.
    # Max possible seeds on board: 12 houses * average_seeds/house (e.g., 4) = 48 seeds.
    # So a bonus of 1000 is safely above any difference.
    EXTRA_MOVE_BONUS = 1000

    # Iterate through all your houses (0 to 5) to find legal moves
    for move_idx in range(6):
        if you[move_idx] > 0:  # Only consider moves from non-empty houses
            temp_you, temp_opponent, got_extra_move = simulate_move(you, opponent, move_idx)

            # Calculate the primary heuristic: difference in seeds in stores
            current_score = temp_you[6] - temp_opponent[6]

            # Apply a large bonus if an extra move was gained
            if got_extra_move:
                current_score += EXTRA_MOVE_BONUS

            # Update best_move if current move's score is better
            if current_score > best_score:
                best_score = current_score
                best_move = move_idx
            # If scores are equal, we can keep the first found (smallest index) or add
            # other tie-breaking heuristics if desired. For now, simply keep the first.

    return best_move
