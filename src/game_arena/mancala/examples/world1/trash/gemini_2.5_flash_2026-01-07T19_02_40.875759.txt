    best_move = -1

            # To avoid infinite recursion, we limit the depth or use direct evaluation.

                move_score = temp_you[6] - temp_opponent[6]

                        best_extra_turn_sub_score = current_et_score
                        best_extra_turn_sub_move = extra_move_idx # Not really needed, just for value
                
                # The score of my initial move is the score after my best extra move
                move_score = best_extra_turn_sub_score
            
            # Additional bonus for getting an extra turn. Might need adjustment.

        else: # No extra turn, opponent's turn next
            # Evaluate the current state after my move

            # Opponent wants to maximize (opponent's_store - my_store).

                    if score_after_opp_move < opponent_best_response_score_for_me:

        # Update best overall move
        if move_score > best_overall_score:
            best_overall_score = move_score
            best_move = my_move
            
    return best_move

**Refining `simulate_move` (renamed from `make_move`):**

The current `simulate_move` has a slight issue with `you_state[last_landed_index] == 0` for capture. This `you_state` is the *initial* state passed. If sowing passes through a house and puts seeds in it *before* the last seed lands, that house is no longer empty according to `you_state`.
The rule is "it was empty before the drop", which means `you[i]` in the *current* iteration of `simulate_move`, *before* any seeds were sowed into it in the current turn.
My existing `temp_next_you` gets filled as seeds are sowed. So `temp_next_you[last_landed_index]` will be `1` at the point of capture check. To truly check "was it empty before this turn's sowing?", I need the initial state.

Let's maintain a copy of the state *before any seeds are sowed from the chosen house*.

# Helper function to simulate a single move
def simulate_move(you_state: list[int], opp_state: list[int], move_idx: int) -> tuple[list[int], list[int], bool]:
    # Deep copy for the working state
    current_you = list(you_state)
    current_opponent = list(opp_state)

    # Store initial state of the current player's houses for capture check
    initial_house_you_for_capture_check = list(you_state[0:6])

    seeds_to_sow = current_you[move_idx]
    current_you[move_idx] = 0 # Empty the chosen house

    # Path using logical indices (0-5 for houses, 6 for store; then 7-12 for opponent's houses)
    # Total effective holes in the circuit (excluding opponent's store) = 6 (my houses) + 1 (my store) + 6 (opponent's houses) = 13
    
    current_logical_idx = move_idx # Start after emptied house
    
    # Track the actual last landed position
    last_landed_target_list = None
    last_landed_target_idx = -1

    while seeds_to_sow > 0:
        current_logical_idx += 1
        
        target_list = None
        target_idx = -1

        if current_logical_idx < 6: # My houses 0..5
            target_list = current_you
            target_idx = current_logical_idx
        elif current_logical_idx == 6: # My store 6
            target_list = current_you
            target_idx = current_logical_idx
        elif current_logical_idx < 13: # Opponent's houses 0..5 (maps from logical index 7-12)
            target_list = current_opponent
            target_idx = current_logical_idx - 7
        else: # Wrap around to my side, starting from my first house
            current_logical_idx = 0 
            target_list = current_you
            target_idx = current_logical_idx
            
        target_list[target_idx] += 1
        seeds_to_sow -= 1
        
        last_landed_target_list = target_list
        last_landed_target_idx = target_idx

    extra_turn = False
    
    # Check for extra turn
    if last_landed_target_list is current_you and last_landed_target_idx == 6:
        extra_turn = True
            
    # Check for capture
    # Conditions:
    # 1. Not an extra turn.
    # 2. Last seed landed in one of your houses (not store, 0-5).
    # 3. That house was empty *before* this move (original state, not `current_you` after partial sowing).
    # 4. Opponent's opposite house has seeds.
    if (not extra_turn and
        last_landed_target_list is current_you and

        for i in range(6):
            current_you[6] += current_you[i]
            current_you[i] = 0

def policy(you: list[int], opponent: list[int]) -> int:

    best_move = -1

    legal_moves = [i for i, seeds in enumerate(you[0:6]) if seeds > 0]
    
    # Shuffle legal moves to introduce some randomness for tie-breaking.
    # This can prevent predictable play patterns if multiple moves have the same score.
    random.shuffle(legal_moves) 

    for my_move in legal_moves:
        # Simulate my move
        temp_you, temp_opponent, extra_turn = simulate_move(you, opponent, my_move)

        move_score = 0

        if extra_turn:
            # If I get an extra turn, I evaluate the best outcome of that extra turn.
            # This is a one-level lookahead for the extra turn, prioritizing immediate store gain.
            
            best_extra_turn_sub_score = -math.inf
            legal_extra_moves = [i for i, seeds in enumerate(temp_you[0:6]) if seeds > 0]

            # the game ends. The simulate_move already handled collecting seeds.
            # So, `temp_you[6] - temp_opponent[6]` is the final score.
            if not legal_extra_moves:
                best_extra_turn_sub_score = temp_you[6] - temp_opponent[6]
            else:
                # Simulate each possible extra move
                for extra_move_idx in legal_extra_moves:
                    et_you, et_opponent, et_extra_turn = simulate_move(temp_you, temp_opponent, extra_move_idx)
                    
                    # Score for this extra turn path, primarily based on store difference
                    current_et_score = et_you[6] - et_opponent[6]
                    if et_extra_turn:
                        # Give a bonus for potentially chaining extra turns further.
                        # This is a heuristic to encourage extra turns.
                        current_et_score += EXTRA_TURN_BONUS 
                    
                    if current_et_score > best_extra_turn_sub_score:
                        best_extra_turn_sub_score = current_et_score
                
            move_score = best_extra_turn_sub_score

        else: # No extra turn, opponent's turn next
            # Evaluate the current state immediately after my move (before opponent's move)
            # my_immediate_score = temp_you[6] - temp_opponent[6] # This is just a baseline for my current position without opponent's counterplay.
            
            # Simulate opponent's best response (from their perspective, which is my worst outcome).
            # Opponent wants to maximize (opponent's_store - my_store).
            # This is equivalent to minimizing (my_store - opponent's_store).
            
            opponent_worst_outcome_for_me = math.inf # Opponent tries to make my score as low as possible
            
            opponent_legal_moves = [i for i, seeds in enumerate(temp_opponent[0:6]) if seeds > 0]
            
            if not opponent_legal_moves: # Opponent has no moves, game ends immediately after my turn.
                # My score is simply the current state after my move, as no further moves from opponent.
                opponent_worst_outcome_for_me = temp_you[6] - temp_opponent[6] 
            else:
                # For each of opponent's possible moves
                for opp_move in opponent_legal_moves:
                    # Simulate opponent's move (swap temporary you and opponent for simulate_move)
                    # The simulate_move function expects the current player's state first, then opponent's.
                    # When simulating opponent, THEIR state is `_temp_opponent`, and MY state is `_temp_you`.
                    op_final_opp, op_final_you, opp_extra_turn = simulate_move(temp_opponent, temp_you, opp_move)
                    
                    # Score from my perspective (my_store - opponent_store) after opponent's move
                    score_after_opp_move = op_final_you[6] - op_final_opp[6]

                    if opp_extra_turn:
                        # Opponent gets an extra turn. This is generally bad for me.
                        # Applying a penalty to my score.
                        score_after_opp_move -= EXTRA_TURN_BONUS
                        
                        # More ideal would be to simulate opponent's extra turn as well (minimax with depth 2 etc.)
                        # But for 1-ply efficiency and meeting time limits, a penalty is a heuristic simplification.
                        
                    if score_after_opp_move < opponent_worst_outcome_for_me:
                        opponent_worst_outcome_for_me = score_after_opp_move
            
            # My current move's score is the worst possible outcome for me after the opponent's best counterplay.
            move_score = opponent_worst_outcome_for_me 

        # Update the best overall move found so far.
        if move_score > best_overall_score:
            best_overall_score = move_score
            best_move = my_move
