An effective Go policy must balance opening principles, tactical awareness (life, death, liberties, and atari), and territorial influence while guaranteeing move legality and fast execution within the 1-second limit.

### Strategy

1. **Board Representation & Pre-computation**:
   - The board is modeled as a 19x19 grid.
   - Connected chains (groups) and their liberties are computed for both friendly and opponent stones.
   - Legality is verified strictly: avoiding occupied squares and suicide (moves with zero liberties that capture no opponent stones).

2. **Tactical Hierarchy (Urgent Moves First)**:
   - **Capture Opponent in Atari**: If any opponent chain has 1 liberty, capturing it is given top priority (proportional to group size and strategic value).
   - **Save Friendly Stones in Atari**: If a friendly chain has 1 liberty, identify escape moves (playing the liberty if it yields $\ge 2$ liberties, or counter-capturing the adjacent attacker).
   - **Atari Threat (Offense)**: Look for moves that reduce opponent chains to 1 liberty, provided the move does not walk into a self-atari trap.
   - **Self-Atari Pruning**: Penalize moves that leave the played group with only 1 liberty unless they successfully capture enemy stones.

3. **Positional and Shape Heuristics**:
   - **Opening**: Corner star points (4, 4), 3-4 points (komoku), and side star points are prioritized in the early phase.
   - **Lines of Influence**: Stones on the 3rd line (territory) and 4th line (influence) are favored over the 1st line (death/endgame) and 2nd line during early and midgame.
   - **Cutting and Connecting**: Moves that connect friendly stones or split opponent stones receive substantial weight.
   - **Eye Defense & Attack**: Vital points that complete or destroy eyes are evaluated.

4. **Fallback & Robustness**:
   - Scores every candidate legal move using the weighted heuristic.
   - If no moves are beneficial or the board is completely resolved, it gracefully passes with `(0, 0)`.

import random
import time

BOARD_SIZE = 19

def in_bounds(r: int, c: int) -> bool:
    return 1 <= r <= 19 and 1 <= c <= 19

def get_neighbors(r: int, c: int):
    for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
        nr, nc = r + dr, c + dc
        if in_bounds(nr, nc):
            yield (nr, nc)

def get_diagonals(r: int, c: int):
    for dr, dc in [(-1, -1), (-1, 1), (1, -1), (1, 1)]:
        nr, nc = r + dr, c + dc
        if in_bounds(nr, nc):
            yield (nr, nc)

def build_board(me: list[tuple[int, int]], opponent: list[tuple[int, int]]):
    board = {}
    for r in range(1, BOARD_SIZE + 1):
        for c in range(1, BOARD_SIZE + 1):
            board[(r, c)] = 0
    for r, c in me:
        board[(r, c)] = 1
    for r, c in opponent:
        board[(r, c)] = 2
    return board

def find_chains(board):
    visited = set()
    chains = []  # list of dicts: {'color': int, 'stones': set, 'liberties': set}
    stone_to_chain = {}

    for r in range(1, BOARD_SIZE + 1):
        for c in range(1, BOARD_SIZE + 1):
            color = board[(r, c)]
            if color != 0 and (r, c) not in visited:
                chain_stones = set()
                liberties = set()
                queue = [(r, c)]
                visited.add((r, c))

                while queue:
                    curr = queue.pop()
                    chain_stones.add(curr)
                    for nxt in get_neighbors(curr[0], curr[1]):
                        ncol = board[nxt]
                        if ncol == 0:
                            liberties.add(nxt)
                        elif ncol == color and nxt not in visited:
                            visited.add(nxt)
                            queue.append(nxt)

                chain_info = {
                    'color': color,
                    'stones': chain_stones,
                    'liberties': liberties
                }
                chains.append(chain_info)
                for s in chain_stones:
                    stone_to_chain[s] = chain_info

    return chains, stone_to_chain

def is_legal_move(r: int, c: int, color: int, board, stone_to_chain) -> bool:
    if board.get((r, c), -1) != 0:
        return False

    opp_color = 2 if color == 1 else 1

    # If adjacent to an empty spot, it immediately has at least 1 liberty
    for nxt in get_neighbors(r, c):
        if board[nxt] == 0:
            return True

    # If adjacent to opponent stone, check if any opponent group would be captured
    for nxt in get_neighbors(r, c):
        if board[nxt] == opp_color:
            chain = stone_to_chain.get(nxt)
            if chain and len(chain['liberties']) == 1 and (r, c) in chain['liberties']:
                return True

    # If adjacent to friendly stone, check if it connects to a group with > 1 liberties
    for nxt in get_neighbors(r, c):
        if board[nxt] == color:
            chain = stone_to_chain.get(nxt)
            if chain and len(chain['liberties']) > 1:
                return True

    return False

def count_post_move_liberties(r: int, c: int, color: int, board, stone_to_chain) -> int:
    """Estimates the liberties of the chain after placing a stone at (r, c)."""
    opp_color = 2 if color == 1 else 1
    new_liberties = set()
    connected_chains = set()

    for nxt in get_neighbors(r, c):
        ncol = board[nxt]
        if ncol == 0:
            new_liberties.add(nxt)
        elif ncol == color:
            chain = stone_to_chain.get(nxt)
            if chain:
                connected_chains.add(id(chain))
                for lib in chain['liberties']:
                    if lib != (r, c):
                        new_liberties.add(lib)
        elif ncol == opp_color:
            # Check if this opponent chain is captured
            chain = stone_to_chain.get(nxt)
            if chain and len(chain['liberties']) == 1 and (r, c) in chain['liberties']:
                # The captured stones become liberties!
                for s in chain['stones']:
                    new_liberties.add(s)

    return len(new_liberties)

def policy(me: list[tuple[int, int]], opponent: list[tuple[int, int]]) -> tuple[int, int]:
    start_time = time.time()
    board = build_board(me, opponent)
    chains, stone_to_chain = find_chains(board)

    total_stones = len(me) + len(opponent)

    # 1. Opening book / classic corner and side priorities
    opening_points = [
        (4, 4), (4, 16), (16, 4), (16, 16),  # 4-4 Star points
        (4, 10), (16, 10), (10, 4), (10, 16), # Side star points
        (3, 4), (4, 3), (3, 16), (4, 17),    # 3-4 points (Komoku)
        (16, 3), (17, 4), (16, 17), (17, 16),
        (3, 3), (3, 17), (17, 3), (17, 17),  # 3-3 San-san
        (10, 10)                             # Tengen
    ]

    if total_stones < 6:
        for pt in opening_points:
            if board[pt] == 0:
                # Don't place right adjacent to an enemy star point if empty board
                return pt

    # Tactical analysis:
    # 2. Urgent: Capture opponent groups in atari
    best_capture = None
    max_captured_size = 0
    for chain in chains:
        if chain['color'] == 2 and len(chain['liberties']) == 1:
            lib = next(iter(chain['liberties']))
            if is_legal_move(lib[0], lib[1], 1, board, stone_to_chain):
                if len(chain['stones']) > max_captured_size:
                    max_captured_size = len(chain['stones'])
                    best_capture = lib

    if best_capture is not None and max_captured_size >= 1:
        return best_capture

    # 3. Urgent: Save friendly groups in atari
    # Try to save the largest friendly group in atari
    save_move = None
    max_saved_size = 0
    for chain in chains:
        if chain['color'] == 1 and len(chain['liberties']) == 1:
            lib = next(iter(chain['liberties']))
            if is_legal_move(lib[0], lib[1], 1, board, stone_to_chain):
                # Verify that playing this liberty actually gives >= 2 liberties
                libs_after = count_post_move_liberties(lib[0], lib[1], 1, board, stone_to_chain)
                if libs_after >= 2:
                    if len(chain['stones']) > max_saved_size:
                        max_saved_size = len(chain['stones'])
                        save_move = lib

    if save_move is not None:
        return save_move

    # 4. Generate and score all candidate moves
    # To stay fast and effective, prioritize candidates near existing stones or key points
    candidate_moves = set()
    if total_stones < 20:
        for pt in opening_points:
            if board[pt] == 0:
                candidate_moves.add(pt)

    for r in range(1, BOARD_SIZE + 1):
        for c in range(1, BOARD_SIZE + 1):
            if board[(r, c)] != 0:
                for nxt in get_neighbors(r, c):
                    if board[nxt] == 0:
                        candidate_moves.add(nxt)
                for diag in get_diagonals(r, c):
                    if board[diag] == 0:
                        candidate_moves.add(diag)

    if not candidate_moves:
        # Fallback to any empty spot
        for r in range(1, BOARD_SIZE + 1):
            for c in range(1, BOARD_SIZE + 1):
                if board[(r, c)] == 0:
                    candidate_moves.add((r, c))

    best_move = None
    best_score = -1e9

    for r, c in candidate_moves:
        if time.time() - start_time > 0.85:
            break

        if not is_legal_move(r, c, 1, board, stone_to_chain):
            continue

        libs_after = count_post_move_liberties(r, c, 1, board, stone_to_chain)

        # Avoid self-atari
        if libs_after <= 1:
            continue

        score = 0.0

        # Base liberties reward
        score += min(libs_after, 5) * 6.0

        # Check line on board (distance to edge)
        d_edge = min(r, 20 - r, c, 20 - c)
        if d_edge == 1:
            score -= 25.0  # Avoid 1st line unless endgame / capture
        elif d_edge == 2:
            score -= 5.0
        elif d_edge in (3, 4):
            score += 10.0  # 3rd line (territory) & 4th line (influence)

        # Neighbor interaction
        adj_me = 0
        adj_opp = 0
        for nxt in get_neighbors(r, c):
            ncol = board[nxt]
            if ncol == 1:
                adj_me += 1
            elif ncol == 2:
                adj_opp += 1
                chain = stone_to_chain.get(nxt)
                if chain:
                    # Reducing opponent's liberties
                    if len(chain['liberties']) == 2:
                        score += 35.0  # Put opponent in atari!
                    elif len(chain['liberties']) == 3:
                        score += 15.0

        # Connect friendly groups
        if adj_me >= 2:
            score += 18.0
        elif adj_me == 1:
            score += 8.0

        # Cutting opponent
        if adj_opp >= 2:
            score += 20.0

        # Prefer center-oriented moves early on
        if total_stones < 40:
            center_dist = abs(r - 10) + abs(c - 10)
            score -= center_dist * 0.5

        # Small tie-breaking noise to avoid deterministic cycling
        score += random.uniform(0.0, 0.5)

        if score > best_score:
            best_score = score
            best_move = (r, c)

    if best_move is not None:
        return best_move

    # Absolute fallback: find ANY legal move
    for r in range(1, BOARD_SIZE + 1):
        for c in range(1, BOARD_SIZE + 1):
            if is_legal_move(r, c, 1, board, stone_to_chain):
                return (r, c)

    return (0, 0)
