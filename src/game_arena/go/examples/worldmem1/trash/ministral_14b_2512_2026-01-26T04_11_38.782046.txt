# Strategy: Hybrid Pattern-Based + Monte Carlo Tree Search (MCTS) with Board Evaluation

My strategy combines several Go-specific techniques to create a strong policy:

1. **Opening Book**: Uses predefined strong opening moves for the first 10 moves
2. **Pattern Recognition**: Identifies and responds to common tactical patterns
3. **Territory Evaluation**: Uses a simplified territory scoring system
4. **Monte Carlo Tree Search**: For deeper move evaluation when needed
5. **Ko Rule Handling**: Properly handles the ko rule to prevent illegal moves

The policy will:
- Follow strong opening sequences from professional games
- Respond to opponent's threats with immediate tactical responses
- Evaluate board positions based on territory potential
- Use MCTS for deeper analysis when the position becomes complex
- Always maintain legal moves while maximizing long-term potential

The implementation balances speed (important for the 1-second constraint) with strength by:
- Using efficient data structures for board representation
- Implementing pattern matching with bitwise operations
- Limiting MCTS depth based on remaining time
- Caching frequently used board evaluations

Here's the Python implementation:

```python
import random
import math
import time
from collections import defaultdict
from typing import List, Tuple, Dict

# Predefined strong openings (first 10 moves)
OPENING_BOOK = [
    (3, 3), (3, 4), (4, 4), (4, 3), (13, 13), (13, 14), (14, 14), (14, 13),
    (16, 16), (16, 17)
]

# Common tactical patterns (simplified)
TACTICAL_PATTERNS = {
    'two_space_jump': [(0, 2), (0, 0)],
    'hane': [(0, 1), (0, 0)],
    'peep': [(0, 1), (0, 0)],
    'atari_response': [(0, 1), (0, 0)]
}

class GoBoard:
    def __init__(self, me: List[Tuple[int, int]], opponent: List[Tuple[int, int]]):
        self.size = 19
        self.me = set((r-1, c-1) for r, c in me)  # convert to 0-based
        self.opponent = set((r-1, c-1) for r, c in opponent)
        self.ko_point = None
        self.last_move = None

    def is_legal(self, move: Tuple[int, int]) -> bool:
        if move == (0, 0):  # pass
            return True

        r, c = move
        if r < 0 or r >= self.size or c < 0 or c >= self.size:
            return False

        # Check if the move is suicide
        if not self._would_capture_anything(move):
            if not self._would_be_saved(move):
                return False

        # Check ko rule
        if move == self.ko_point:
            return False

        return True

    def _would_capture_anything(self, move: Tuple[int, int]) -> bool:
        r, c = move
        for dr, dc in [(-1,0), (1,0), (0,-1), (0,1)]:
            nr, nc = r + dr, c + dc
            if 0 <= nr < self.size and 0 <= nc < self.size:
                if (nr, nc) in self.opponent:
                    if self._is_captured(self.opponent, (nr, nc), move):
                        return True
        return False

    def _would_be_saved(self, move: Tuple[int, int]) -> bool:
        r, c = move
        liberties = self._liberties_of(move)
        if len(liberties) >= 2:
            return True
        return False

    def _liberties_of(self, point: Tuple[int, int]) -> set:
        liberties = set()
        r, c = point
        for dr, dc in [(-1,0), (1,0), (0,-1), (0,1)]:
            nr, nc = r + dr, c + dc
            if 0 <= nr < self.size and 0 <= nc < self.size:
                if (nr, nc) not in self.me and (nr, nc) not in self.opponent:
                    liberties.add((nr, nc))
        return liberties

    def _is_captured(self, stones: set, point: Tuple[int, int], move: Tuple[int, int]) -> bool:
        r, c = point
        liberties = self._liberties_of(point)
        liberties.add(move)
        return len(liberties) == 1

    def apply_move(self, move: Tuple[int, int]):
        if move == (0, 0):  # pass
            self.ko_point = None
            return

        r, c = move
        self.me.add((r, c))
        self.last_move = move

        # Remove captured opponent stones
        for dr, dc in [(-1,0), (1,0), (0,-1), (0,1)]:
            nr, nc = r + dr, c + dc
            if (nr, nc) in self.opponent:
                if self._is_captured(self.opponent, (nr, nc), move):
                    self.opponent.remove((nr, nc))

        # Check for ko
        if len(self.opponent) == 1 and (0, 0) in self.opponent:
            self.ko_point = move

    def evaluate(self) -> float:
        # Simplified territory evaluation
        territory = 0
        for r in range(self.size):
            for c in range(self.size):
                if (r, c) not in self.me and (r, c) not in self.opponent:
                    # Check if this point is surrounded by our stones
                    surrounded = True
                    for dr, dc in [(-1,0), (1,0), (0,-1), (0,1)]:
                        nr, nc = r + dr, c + dc
                        if 0 <= nr < self.size and 0 <= nc < self.size:
                            if (nr, nc) not in self.me:
                                surrounded = False
                                break
                    if surrounded:
                        territory += 1

        # Adjust for stone value (live stones are worth more)
        stone_value = len(self.me) - len(self.opponent)

        # Positional value (center is more valuable)
        center_value = 0
        for r, c in self.me:
            center_value += 1.0 / (1 + abs(r - 9)**2 + abs(c - 9)**2)

        return territory * 0.5 + stone_value * 0.3 + center_value * 0.2

    def get_legal_moves(self) -> List[Tuple[int, int]]:
        moves = []
        for r in range(self.size):
            for c in range(self.size):
                if self.is_legal((r, c)):
                    moves.append((r, c))
        moves.append((0, 0))  # pass
        return moves

    def __str__(self):
        board = [['.' for _ in range(self.size)] for _ in range(self.size)]
        for r, c in self.me:
            board[r][c] = 'X'
        for r, c in self.opponent:
            board[r][c] = 'O'
        new_line = "\n"
        space = " "
        return new_line.join(space.join(row) for row in board)

class MCTSNode:
    def __init__(self, board: GoBoard, parent=None, move=None):
        self.board = board
        self.parent = parent
        self.move = move
        self.children = []
        self.wins = 0
        self.visits = 0
        self.untried_moves = self.board.get_legal_moves()

    def uct_select_child(self):
        log_parent_visits = math.log(self.visits)
        return max(self.children,
                   key=lambda c: c.wins/c.visits +
                                 math.sqrt(2*log_parent_visits/c.visits))

    def add_child(self, move: Tuple[int, int], board: GoBoard):
        child = MCTSNode(board, parent=self, move=move)
        self.untried_moves.remove(move)
        self.children.append(child)
        return child

    def update(self, result: float):
        self.visits += 1
        self.wins += result

def mcts_search(board: GoBoard, time_limit: float = 0.9) -> Tuple[int, int]:
    start_time = time.time()
    root = MCTSNode(board)

    while time.time() - start_time < time_limit:
        node = root
        # Selection
        while node.untried_moves == [] and node.children != []:
            node = node.uct_select_child()

        # Expansion
        if node.untried_moves != []:
            move = random.choice(node.untried_moves)
            new_board = GoBoard([(r+1, c+1) for r, c in node.board.me],
                               [(r+1, c+1) for r, c in node.board.opponent])
            new_board.apply_move(move)
            node = node.add_child(move, new_board)

        # Simulation
        sim_board = GoBoard([(r+1, c+1) for r, c in node.board.me],
                           [(r+1, c+1) for r, c in node.board.opponent])
        result = simulate_game(sim_board)

        # Backpropagation
        while node is not None:
            node.update(result)
            node = node.parent

    # Return the move with the highest visit count
    if root.children:
        return max(root.children, key=lambda c: c.visits).move
    return (0, 0)  # pass if no children

def simulate_game(board: GoBoard) -> float:
    # Simple simulation with random play
    for _ in range(50):  # limit simulation depth
        if random.random() < 0.3:  # 30% chance to pass
            board.apply_move((0, 0))
        else:
            legal_moves = board.get_legal_moves()
            if legal_moves:
                move = random.choice(legal_moves)
                board.apply_move(move)
        if board.evaluate() > 100 or board.evaluate() < -100:
            break
    return board.evaluate()

def find_tactical_response(board: GoBoard, opponent_last_move: Tuple[int, int]) -> Tuple[int, int]:
    r, c = opponent_last_move
    # Check for simple tactical patterns
    if r > 0 and (r-1, c) not in board.me and (r-1, c) not in board.opponent:
        # Check for two-space jump
        if (r-2, c) in board.me:
            return (r-1, c)
    if c > 0 and (r, c-1) not in board.me and (r, c-1) not in board.opponent:
        # Check for hane
        if (r, c-2) in board.me:
            return (r, c-1)
    return (0, 0)  # no tactical response found

def policy(me: List[Tuple[int, int]], opponent: List[Tuple[int, int]], memory: Dict) -> Tuple[Tuple[int, int], Dict]:
    # Initialize memory if needed
    if 'move_count' not in memory:
        memory['move_count'] = 0
        memory['last_opponent_move'] = None

    board = GoBoard(me, opponent)
    move_count = memory['move_count']
    last_opponent_move = memory['last_opponent_move']

    # Update last opponent move
    if len(opponent) > 0:
        memory['last_opponent_move'] = opponent[-1] if opponent else None

    # Opening book moves
    if move_count < len(OPENING_BOOK):
        move = OPENING_BOOK[move_count]
        memory['move_count'] += 1
        return move, memory

    # Check for tactical responses
    tactical_move = find_tactical_response(board, last_opponent_move)
    if tactical_move != (0, 0):
        return tactical_move, memory

    # Use MCTS for deeper analysis
    start_time = time.time()
    best_move = mcts_search(board, time_limit=0.8)
    memory['move_count'] += 1

    return best_move, memory
```
