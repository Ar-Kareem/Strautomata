### Strategy for Checkers Policy

My strategy employs a **Depth-Limited Minimax algorithm with Alpha-Beta Pruning** to evaluate and select the best move. This approach balances computational efficiency with strategic depth, allowing the AI to look several moves ahead while respecting the 1-second execution time limit.

1.  **Heuristic Evaluation Function**: The board state is scored based on material advantage (kings are worth more than men), positional advantage (controlling the center and promoting rows), and mobility (number of legal moves).
2.  **Mandatory Captures**: The algorithm strictly adheres to checkers rules by prioritizing capture moves. If capture moves are available, non-capturing moves are ignored.
3.  **Iterative Deepening & Alpha-Beta**: To optimize move ordering and time management, the search uses iterative deepening with alpha-beta pruning. This allows the search to be interrupted if it approaches the time limit, returning the best move found so far.
4.  **Symmetry & Color Handling**: The logic is generalized to handle both Black and White players by dynamically adjusting move directions (Black moves down, White moves up) and promotion rows.
5.  **State Representation**: The board is represented using bitmask integers for efficient state generation and evaluation, allowing complex lookups and move generations within the tight time constraints.

This policy ensures legal move generation, prioritizes winning strategies (promotion and captures), and maintains a robust defense against opponent threats.

def rc_to_idx(r, c):
    if r % 2 == 0: # Even rows (0,2,4,6) have odd cols (1,3,5,7)
        if c % 2 != 1: return -1
        return (r * 4) + (c // 2)
    else: # Odd rows (1,3,5,7) have even cols (0,2,4,6)
        if c % 2 != 0: return -1
        return (r * 4) + (c // 2)

def idx_to_rc(idx):
    r = idx // 4
    c_odd = idx % 4
    if r % 2 == 0:
        c = c_odd * 2 + 1
    else:
        c = c_odd * 2
    return (r, c)

def get_captures(my_men, my_kings, all_pieces, opp_pieces, color):
    captures = []
    #Directions
    if color == 'b': # Black moves down (r-1)
        dirs = [(-1, -1), (-1, 1)]
        k_dirs = [(-1, -1), (-1, 1), (1, -1), (1, 1)]
    else: # White moves up (r+1)
        dirs = [(1, -1), (1, 1)]
        k_dirs = [(1, -1), (1, 1), (-1, -1), (-1, 1)]
        
    # Check Men
    men = my_men
    while men:
        p = men & -men
        idx = (p.bit_length() - 1)
        r, c = idx_to_rc(idx)
        men &= ~p
        
        directions = dirs if (my_kings & p) == 0 else k_dirs
        # If it's a man, use dirs. If king, use k_dirs.
        # Wait, my_kings & p checks if this piece is a king. 
        # But we are iterating 'men' which should only contain men. 
        # However, strictly speaking, we separated lists. Let's check membership in my_men just in case.
        is_king = False
        if (1 << idx) & my_kings:
            is_king = True
            directions = k_dirs
        else:
            directions = dirs

        for dr, dc in directions:
            mid_r, mid_c = r + dr, c + dc
            jump_r, jump_c = r + 2*dr, c + 2*dc
            
            mid_idx = rc_to_idx(mid_r, mid_c)
            jump_idx = rc_to_idx(jump_r, jump_c)
            
            if mid_idx != -1 and jump_idx != -1:
                # Check if mid is opponent and jump is empty
                if ((1 << mid_idx) & opp_pieces) and not ((1 << jump_idx) & all_pieces):
                    # Found a capture
                    captures.append(((r, c), (jump_r, jump_c)))
    
    # Check Kings (my_kings)
    kings = my_kings
    while kings:
        p = kings & -kings
        idx = (p.bit_length() - 1)
        r, c = idx_to_rc(idx)
        kings &= ~p
        
        for dr, dc in k_dirs:
            mid_r, mid_c = r + dr, c + dc
            jump_r, jump_c = r + 2*dr, c + 2*dc
            
            mid_idx = rc_to_idx(mid_r, mid_c)
            jump_idx = rc_to_idx(jump_r, jump_c)
            
            if mid_idx != -1 and jump_idx != -1:
                if ((1 << mid_idx) & opp_pieces) and not ((1 << jump_idx) & all_pieces):
                    captures.append(((r, c), (jump_r, jump_c)))

    return captures

def apply_move(move, my_men, my_kings, opp_men, opp_kings, turn, my_color):
    (r1, c1), (r2, c2) = move
    idx1 = rc_to_idx(r1, c1)
    idx2 = rc_to_idx(r2, c2)
    
    # Determine who owns the moving piece based on turn
    # If turn is 'me', it's my piece. If 'opp', it's opp piece.
    # However, in recursion, we maintain 'my_men' as the player whose turn it is.
    # To simplify, let's keep 'my' as the current player in the recursion state.
    
    res = {
        'my_men': my_men, 'my_kings': my_kings,
        'opp_men': opp_men, 'opp_kings': opp_kings
    }
    
    is_capture = abs(r1 - r2) == 2
    
    # Identify source
    src_is_king = False
    if (1 << idx1) & my_kings:
        src_is_king = True
        res['my_kings'] &= ~(1 << idx1)
    else:
        res['my_men'] &= ~(1 << idx1)
        
    # Check Promotion
    promo = False
    if not src_is_king:
        # Determine promotion row based on whose turn it is
        # If turn == 'me', color is my_color. 
        # If turn == 'opp', color is opp_color (opp of my_color).
        current_color = my_color if turn == 'me' else ('w' if my_color == 'b' else 'b')
        
        if current_color == 'b' and r2 == 0: promo = True
        if current_color == 'w' and r2 == 7: promo = True
    
    if src_is_king or not promo:
        res['my_kings'] |= (1 << idx2)
    else:
        res['my_men'] |= (1 << idx2)
        
    if is_capture:
        mid_r, mid_c = (r1 + r2) // 2, (c1 + c2) // 2
        mid_idx = rc_to_idx(mid_r, mid_c)
        
        if (1 << mid_idx) & res['opp_men']:
            res['opp_men'] &= ~(1 << mid_idx)
        else:
            res['opp_kings'] &= ~(1 << mid_idx)
            
    res['all_pieces'] = res['my_men'] | res['my_kings'] | res['opp_men'] | res['opp_kings']
    return res

def minimax_recurse(my_men, my_kings, opp_men, opp_kings, all_pieces, depth, alpha, beta, maximizing, my_color):

    # Wait, `my_men`, `my_kings` always represent the current player in the recursion.

    # Let's fix the convention:
    # In recursion, 'my' always refers to the player whose turn it is.
    # `my_color` refers to the ORIGINAL player (Root).
    # So if it's opponent's turn, `my_color` passed to evaluate is still Root's color?

    # If maximizing is False, we want to minimize score for `my_color`.

    if depth == 0:

    # Determine current color
    # If `maximizing` is True, current player is Root. Color is `my_color`.

    caps = get_captures(my_men, my_kings, all_pieces, opp_men | opp_kings, current_color)

def rc_to_idx(r, c):
    if r % 2 == 0:
        return (r * 4) + (c // 2) if c % 2 != 0 else -1
    else:
        return (r * 4) + (c // 2) if c % 2 != 1 else -1

def idx_to_rc(idx):
    r = idx // 4
    c = (idx % 4) * 2 + (1 if r % 2 == 0 else 0)
    return (r, c)

def policy(my_men, my_kings, opp_men, opp_kings, color):
    global START_TIME
    START_TIME = time.time()
    
    # Bitboards
    mm = 0; mk = 0; om = 0; ok = 0
    for r, c in my_men: mm |= 1 << rc_to_idx(r, c)
    for r, c in my_kings: mk |= 1 << rc_to_idx(r, c)
    for r, c in opp_men: om |= 1 << rc_to_idx(r, c)
    for r, c in opp_kings: ok |= 1 << rc_to_idx(r, c)
    
    all_p = mm | mk | om | ok
    
    # Get Moves
    moves = get_moves_state(mm, mk, all_p, color, True) # True = captures only if exist
    
    if not moves:
        moves = get_moves_state(mm, mk, all_p, color, False) # Normal moves
        if not moves: return ((0,0), (0,0)) # Stuck

    best_move = moves[0]
    
    # Iterative Deepening
    depth = 1
    while time.time() - START_TIME < 0.4:
        best_score = -float('inf')
        local_best = moves[0]
        
        # Sort moves (Promotions first)
        moves.sort(key=lambda m: 10 if (color == 'b' and m[1][0] == 0) or (color == 'w' and m[1][0] == 7) else 0, reverse=True)
        
        for m in moves:
            s = simulate_and_search(m, mm, mk, om, ok, all_p, depth, -float('inf'), float('inf'), True, color)
            if s > best_score:
                best_score = s
                local_best = m
            if time.time() - START_TIME > 0.8: break
        
        best_move = local_best
        depth += 1
        if depth > 8: break
        
    return best_move

def get_moves_state(men, kings, all_p, color, force_captures):
    # Check captures
    caps = []
    dirs = [(-1, -1), (-1, 1)] if color == 'b' else [(1, -1), (1, 1)]
    k_dirs = [(-1, -1), (-1, 1), (1, -1), (1, 1)]
    opp = (all_p ^ (men | kings)) # Opponent pieces
    
    # Scan Men
    temp_men = men
    while temp_men:
        b = temp_men & -temp_men
        idx = b.bit_length() - 1
        r, c = idx_to_rc(idx)
        temp_men ^= b
        
        is_king = (kings & b) != 0
        cur_dirs = k_dirs if is_king else dirs
        
        for dr, dc in cur_dirs:
            mr, mc = r+dr, c+dc; jr, jc = r+2*dr, c+2*dc
            mi = rc_to_idx(mr, mc); ji = rc_to_idx(jr, jc)
            if mi != -1 and ji != -1:
                if ((1 << mi) & opp) and not ((1 << ji) & all_p):
                    caps.append(((r, c), (jr, jc)))
    
    if caps: return caps if force_captures else caps
    
    if force_captures: return []
    
    # Normal Moves
    norm = []
    temp_men = men
    while temp_men:
        b = temp_men & -temp_men
        idx = b.bit_length() - 1
        r, c = idx_to_rc(idx)
        temp_men ^= b
        is_king = (kings & b) != 0
        cur_dirs = k_dirs if is_king else dirs
        for dr, dc in cur_dirs:
            nr, nc = r+dr, c+dc
            ni = rc_to_idx(nr, nc)
            if ni != -1 and not ((1 << ni) & all_p):
                norm.append(((r, c), (nr, nc)))
    return norm
