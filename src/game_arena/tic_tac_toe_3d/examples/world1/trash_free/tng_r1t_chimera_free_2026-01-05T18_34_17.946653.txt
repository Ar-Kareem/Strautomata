def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    # Generate all possible winning lines
    lines = []
    
    # Rows (x-axis)
    for y in range(3):
        for z in range(3):
            lines.append([(x, y, z) for x in range(3)])
    
    # Columns (y-axis)
    for x in range(3):
        for z in range(3):
            lines.append([(x, y, z) for y in range(3)])
    
    # Pillars (z-axis)
    for x in range(3):
        for y in range(3):
            lines.append([(x, y, z) for z in range(3)])
    
    # Face diagonals (x-y plane)
    for z in range(3):
        lines.append([(0, 0, z), (1, 1, z), (2, 2, z)])
        lines.append([(0, 2, z), (1, 1, z), (2, 0, z)])
    
    # Face diagonals (x-z plane)
    for y in range(3):
        lines.append([(0, y, 0), (1, y, 1), (2, y, 2)])
        lines.append([(0, y, 2), (1, y, 1), (2, y, 0)])
    
    # Face diagonals (y-z plane)
    for x in range(3):
        lines.append([(x, 0, 0), (x, 1, 1), (x, 2, 2)])
        lines.append([(x, 0, 2), (x, 1, 1), (x, 2, 0)])
    
    # Space diagonals
    lines.append([(0, 0, 0), (1, 1, 1), (2, 2, 2)])
    lines.append([(0, 0, 2), (1, 1, 1), (2, 2, 0)])
    lines.append([(0, 2, 0), (1, 1, 1), (2, 0, 2)])
    lines.append([(0, 2, 2), (1, 1, 1), (2, 0, 0)])
    
    # Check for winning moves
    for line in lines:
        values = [board[x][y][z] for (x, y, z) in line]
        if sum(values) == 2:
            for (x, y, z) in line:
                if board[x][y][z] == 0:
                    return (x, y, z)
    
    # Check for blocking opponent's wins
    for line in lines:
        values = [board[x][y][z] for (x, y, z) in line]
        if sum(values) == -2:
            for (x, y, z) in line:
                if board[x][y][z] == 0:
                    return (x, y, z)
    
    # Take center if available
    if board[1][1][1] == 0:
        return (1, 1, 1)
    
    # Take any corner
    corners = [
        (0, 0, 0), (0, 0, 2), (0, 2, 0), (0, 2, 2),
        (2, 0, 0), (2, 0, 2), (2, 2, 0), (2, 2, 2)
    ]
    for (x, y, z) in corners:
        if board[x][y][z] == 0:
            return (x, y, z)
    
    # Take first available cell
    for x in range(3):
        for y in range(3):
            for z in range(3):
                if board[x][y][z] == 0:
                    return (x, y, z)
    
    # Fallback (shouldn't occur in valid game)
    return (0, 0, 0)
