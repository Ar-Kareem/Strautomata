def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    # Generate all possible lines in 3x3x3
    all_lines = []
    
    # x-axis lines (fixed y, z)
    for y in range(3):
        for z in range(3):
            line = [(x, y, z) for x in range(3)]
            all_lines.append(line)
            
    # y-axis lines (fixed x, z)
    for x in range(3):
        for z in range(3):
            line = [(x, y, z) for y in range(3)]
            all_lines.append(line)
            
    # z-axis lines (fixed x, y)
    for x in range(3):
        for y in range(3):
            line = [(x, y, z) for z in range(3)]
            all_lines.append(line)
            
    # xy diagonals (fixed z)
    for z in range(3):
        line1 = [(0,0,z), (1,1,z), (2,2,z)]
        line2 = [(0,2,z), (1,1,z), (2,0,z)]
        all_lines.append(line1)
        all_lines.append(line2)
        
    # xz diagonals (fixed y)
    for y in range(3):
        line1 = [(0,y,0), (1,y,1), (2,y,2)]
        line2 = [(0,y,2), (1,y,1), (2,y,0)]
        all_lines.append(line1)
        all_lines.append(line2)
        
    # yz diagonals (fixed x)
    for x in range(3):
        line1 = [(x,0,0), (x,1,1), (x,2,2)]
        line2 = [(x,0,2), (x,1,1), (x,2,0)]
        all_lines.append(line1)
        all_lines.append(line2)
        
    # space diagonals
    all_lines.append([(0,0,0), (1,1,1), (2,2,2)])
    all_lines.append([(0,0,2), (1,1,1), (2,2,0)])
    all_lines.append([(0,2,0), (1,1,1), (2,0,2)])
    all_lines.append([(0,2,2), (1,1,1), (2,0,0)])
    
    # Check for winning moves (player's two 1s and one 0)
    for line in all_lines:
        values = [board[x][y][z] for (x,y,z) in line]
        if values.count(1) == 2 and values.count(0) == 1:
            for idx, (x,y,z) in enumerate(line):
                if board[x][y][z] == 0:
                    return (x, y, z)
                    
    # Check for blocking moves (opponent's two -1s and one 0)
    for line in all_lines:
        values = [board[x][y][z] for (x,y,z) in line]
        if values.count(-1) == 2 and values.count(0) == 1:
            for idx, (x,y,z) in enumerate(line):
                if board[x][y][z] == 0:
                    return (x, y, z)
                    
    # If no winning or blocking move, pick the first empty cell
    for x in range(3):
        for y in range(3):
            for z in range(3):
                if board[x][y][z] == 0:
                    return (x, y, z)
