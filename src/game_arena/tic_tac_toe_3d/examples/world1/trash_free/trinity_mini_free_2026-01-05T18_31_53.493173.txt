def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    # Precompute all winning lines in 3D Tic Tac Toe
    lines = []
    
    # Rows (fixed x,z, varying y)
    for x in range(3):
        for z in range(3):
            lines.append([(x, y, z) for y in range(3)])
    
    # Columns (fixed y,z, varying x)
    for y in range(3):
        for z in range(3):
            lines.append([(x, y, z) for x in range(3)])
    
    # Diagonals in each layer (fixed z, varying x,y)
    for z in range(3):
        lines.append([(0, 0, z), (1, 1, z), (2, 2, z)])
        lines.append([(0, 2, z), (1, 1, z), (2, 0, z)])
    
    # Pillars (fixed x,y, varying z)
    for x in range(3):
        for y in range(3):
            lines.append([(x, y, z) for z in range(3)])
    
    # Space diagonals (4 lines through the cube)
    lines.append([(0, 0, 0), (1, 1, 1), (2, 2, 2)])
    lines.append([(0, 0, 2), (1, 1, 1), (2, 2, 0)])
    lines.append([(0, 2, 0), (1, 1, 1), (2, 0, 2)])
    lines.append([(0, 2, 2), (1, 1, 1), (2, 0, 0)])
    
    # Count how many winning lines each cell is part of
    cell_lines = [[[0]*3 for _ in range(3)] for _ in range(3)]
    for line in lines:
        for (x, y, z) in line:
            cell_lines[x][y][z] += 1
    
    # Find the best move: prioritize cells with most winning lines
    best_count = -1
    best_move = (0, 0, 0)
    for x in range(3):
        for y in range(3):
            for z in range(3):
                if board[z][y][x] == 0:  # Empty cell
                    if cell_lines[x][y][z] > best_count:
                        best_count = cell_lines[x][y][z]
                        best_move = (x, y, z)
    return best_move
