import numpy as np
from typing import List, Tuple

def policy(board: List[List[List[int]]]) -> Tuple[int, int, int]:
    """
    Determines the best move for a 3x3x3 3D Tic Tac Toe game using minimax with alpha-beta pruning.

    Args:
        board: A 3x3x3 list of lists representing the game board (0=empty, 1=player, -1=opponent)

    Returns:
        A tuple (x, y, z) representing the coordinates of the best move (0-2 for each dimension)
    """
    # Convert to numpy array for easier manipulation
    board_np = np.array(board)

    # Find all empty positions
    empty_positions = np.argwhere(board_np == 0)

        return (1, 1, 1)

    best_move = empty_positions[0]

    for pos in empty_positions:
        x, y, z = pos
        # Make a copy of the board

        new_board[x, y, z] = 1

        # Evaluate the move
        score = minimax(new_board, 0, -float('inf'), float('inf'), False)

        # Update best move if this is better
        if score > best_score:
            best_score = score
            best_move = pos

    return tuple(best_move)

def minimax(board: np.ndarray, depth: int, alpha: float, beta: float, is_maximizing: bool) -> float:
    """
    Minimax algorithm with alpha-beta pruning for 3D Tic Tac Toe.

    Args:
        board: Current game state
        depth: Current depth in the search tree
        alpha: Best value that the maximizer can guarantee
        beta: Best value that the minimizer can guarantee
        is_maximizing: Whether current player is maximizing (True) or minimizing (False)

    Returns:
        The evaluated score of the current board state
    """
    # Check for terminal state (win or draw)
    score = evaluate_board(board)

    if score != 0:
        return score

    if is_maximizing:
        best_score = -float('inf')
        for x in range(3):
            for y in range(3):
                for z in range(3):
                    if board[x, y, z] == 0:
                        new_board = board.copy()
                        new_board[x, y, z] = 1
                        score = minimax(new_board, depth + 1, alpha, beta, False)
                        best_score = max(score, best_score)
                        alpha = max(alpha, best_score)
                        if beta <= alpha:
                            break
                if beta <= alpha:
                    break
        return best_score
    else:
        best_score = float('inf')
        for x in range(3):
            for y in range(3):
                for z in range(3):
                    if board[x, y, z] == 0:
                        new_board = board.copy()
                        new_board[x, y, z] = -1
                        score = minimax(new_board, depth + 1, alpha, beta, True)
                        best_score = min(score, best_score)
                        beta = min(beta, best_score)
                        if beta <= alpha:
                            break
                if beta <= alpha:
                    break
        return best_score

def evaluate_board(board: np.ndarray) -> float:
    """
    Evaluates the current board state and returns:
    - Positive score if player (1) is winning
    - Negative score if opponent (-1) is winning
    - 0 if game is not over
    """
    # Check all possible lines (rows, columns, pillars, and 3D diagonals)
    lines = []

    # 2D lines (rows, columns, pillars)
    for i in range(3):
        # Rows in each layer
        for j in range(3):
            lines.append(board[i, j, :])
        # Columns in each layer
        for j in range(3):
            lines.append(board[:, i, j])
        # Pillars (through z-axis)
        for j in range(3):
            lines.append(board[:, :, i])

    # 3D diagonals
    # Main space diagonal
    lines.append(np.diag(board))
    # Anti-diagonal
    lines.append(np.diag(np.fliplr(board)))

    # Check each line for a win
    for line in lines:
        if np.all(line == 1):
            return 10 - depth  # Player wins
        if np.all(line == -1):
            return depth - 10  # Opponent wins

    # Check for draw
    if np.all(board != 0):
        return 0

    return 0
