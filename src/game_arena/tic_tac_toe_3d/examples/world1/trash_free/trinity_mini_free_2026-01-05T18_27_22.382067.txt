def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    # Precompute all winning lines in 3x3x3 Tic Tac Toe
    winning_lines = []
    
    # Lines along x-axis (fixed y, z)
    for y in range(3):
        for z in range(3):
            winning_lines.append([(0, y, z), (1, y, z), (2, y, z)])
    
    # Lines along y-axis (fixed x, z)
    for x in range(3):
        for z in range(3):
            winning_lines.append([(x, 0, z), (x, 1, z), (x, 2, z)])
    
    # Lines along z-axis (fixed x, y)
    for x in range(3):
        for y in range(3):
            winning_lines.append([(x, y, 0), (x, y, 1), (x, y, 2)])
    
    # Diagonals in xy-plane (fixed z)
    for z in range(3):
        for x in range(3):
            for y in range(3):
                # Main diagonal (x=y)
                if x + 2 < 3 and y + 2 < 3:
                    winning_lines.append([(x, y, z), (x+1, y+1, z), (x+2, y+2, z)])
                # Anti-diagonal (x+y=2)
                if x + 2 < 3 and y - 2 >= 0:
                    winning_lines.append([(x, y, z), (x+1, y-1, z), (x+2, y-2, z)])
    
    # Diagonals in xz-plane (fixed y)
    for y in range(3):
        for x in range(3):
            for z in range(3):
                # Main diagonal (x=z)
                if x + 2 < 3 and z + 2 < 3:
                    winning_lines.append([(x, y, z), (x+1, y, z+1), (x+2, y, z+2)])
                # Anti-diagonal (x+z=2)
                if x + 2 < 3 and z - 2 >= 0:
                    winning_lines.append([(x, y, z), (x+1, y, z-1), (x+2, y, z-2)])
    
    # Diagonals in yz-plane (fixed x)
    for x in range(3):
        for y in range(3):
            for z in range(3):
                # Main diagonal (y=z)
                if y + 2 < 3 and z + 2 < 3:
                    winning_lines.append([(x, y, z), (x, y+1, z+1), (x, y+2, z+2)])
                # Anti-diagonal (y+z=2)
                if y + 2 < 3 and z - 2 >= 0:
                    winning_lines.append([(x, y, z), (x, y+1, z-1), (x, y+2, z-2)])
    
    # Space diagonals
    for x in range(1, 3):
        for y in range(1, 3):
            for z in range(1, 3):
                # Main space diagonal (x=y=z)
                winning_lines.append([(x, y, z), (x+1, y+1, z+1), (x+2, y+2, z+2)])
                # Other space diagonals
                winning_lines.append([(x, y, z), (x+1, y+1, z-1), (x+2, y+2, z-2)])
                winning_lines.append([(x, y, z), (x+1, y-1, z+1), (x+2, y-2, z+2)])
                winning_lines.append([(x, y, z), (x+1, y-1, z-1), (x+2, y-2, z-2)])
    
    # Check for winning moves
    for i in range(3):
        for j in range(3):
            for k in range(3):
                if board[i][j][k] == 0:
                    # Check if placing 1 here wins
                    for line in winning_lines:
                        if i in line and j in line and k in line:
                            count = 0
                            for (x, y, z) in line:
                                if (x, y, z) != (i, j, k) and board[x][y][z] == 1:
                                    count += 1
                            if count == 2:
                                return (i, j, k)
                    # Check if placing 1 here blocks opponent's win
                    for line in winning_lines:
                        if i in line and j in line and k in line:
                            count = 0
                            for (x, y, z) in line:
                                if (x, y, z) != (i, j, k) and board[x][y][z] == -1:
                                    count += 1
                            if count == 2:
                                return (i, j, k)
    
    # If no winning or blocking move, choose a cell that maximizes potential
    best_score = -1
    best_move = None
    for i in range(3):
        for j in range(3):
            for k in range(3):
                if board[i][j][k] == 0:
                    score = 0
                    for line in winning_lines:
                        if i in line and j in line and k in line:
                            count1 = 0
                            count2 = 0
                            for (x, y, z) in line:
                                if (x, y, z) == (i, j, k):
                                    continue
                                if board[x][y][z] == 1:
                                    count1 += 1
                                elif board[x][y][z] == -1:
                                    count2 += 1
                            if count1 == 2:
                                score += 1
                            if count2 == 2:
                                score -= 1
                    if score > best_score:
                        best_score = score
                        best_move = (i, j, k)
    return best_move
