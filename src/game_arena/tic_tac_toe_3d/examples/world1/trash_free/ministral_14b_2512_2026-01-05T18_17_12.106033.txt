import numpy as np
from typing import List, Tuple

def policy(board: List[List[List[int]]]) -> Tuple[int, int, int]:
    """
    Determines the next move for a 3D Tic Tac Toe game using minimax with alpha-beta pruning.

    Args:
        board: A 3x3x3 list of lists representing the game board. 0 = empty, 1 = player, -1 = opponent.

    Returns:
        A tuple (x, y, z) representing the coordinates of the next move (0-2 for each dimension).

    # Convert the board to a numpy array for easier manipulation
    board_np = np.array(board, dtype=int)

    empty_cells = np.argwhere(board_np == 0)

    if len(empty_cells) == 0:

        return (1, 1, 1)

    best_move = empty_cells[0]

        x, y, z = move

        new_board[x, y, z] = 1  # Player's move (1)

        score = minimax(new_board, 0, -float('inf'), float('inf'), False)

        # Update best move if current score is better
        if score > best_score:
            best_score = score
            best_move = move

    return tuple(best_move)

def minimax(board: np.ndarray, depth: int, alpha: float, beta: float, is_maximizing: bool) -> int:
    """
    Minimax algorithm with alpha-beta pruning for 3D Tic Tac Toe.

    Args:
        board: Current game board.
        depth: Current depth in the game tree.
        alpha: Alpha value for alpha-beta pruning.
        beta: Beta value for alpha-beta pruning.
        is_maximizing: Whether the current player is maximizing (player) or minimizing (opponent).

    Returns:
        The score of the current board state from the perspective of the maximizing player.
    """

    if check_win(board, 1):
        return 10 - depth  # Player wins
    elif check_win(board, -1):
        return depth - 10  # Opponent wins
    elif np.all(board != 0):
        return 0  # Draw

    if is_maximizing:
        best_score = -float('inf')
        for x in range(3):
            for y in range(3):
                for z in range(3):
                    if board[x, y, z] == 0:
                        board[x, y, z] = 1
                        score = minimax(board, depth + 1, alpha, beta, False)
                        board[x, y, z] = 0
                        best_score = max(score, best_score)
                        alpha = max(alpha, score)
                        if beta <= alpha:
                            break
                if beta <= alpha:
                    break
        return best_score
    else:
        best_score = float('inf')
        for x in range(3):
            for y in range(3):

                    if board[x, y, z] == 0:

                        score = minimax(board, depth + 1, alpha, beta, True)
                        board[x, y, z] = 0
                        best_score = min(score, best_score)
                        beta = min(beta, score)
                        if beta <= alpha:
                            break
                if beta <= alpha:
                    break
        return best_score

def check_win(board: np.ndarray, player: int) -> bool:

    if all(board[layer, 2 - layer, layer] == player for layer in range(3)):

    for z in range(3):

            return True
        if all(board[i, 2 - i, z] == player for i in range(3)):
            return True

### Explanation:
1. **Policy Function**:
   - The `policy` function uses the minimax algorithm with alpha-beta pruning to determine the best move.
   - It first checks if there are any empty cells. If it's the first move, it chooses the center (1, 1, 1) for symmetry.
   - For other moves, it evaluates all possible moves using the minimax algorithm and selects the one with the highest score.

2. **Minimax with Alpha-Beta Pruning**:
   - The `minimax` function recursively evaluates all possible moves, alternating between maximizing (player) and minimizing (opponent) players.
   - It uses alpha-beta pruning to eliminate branches that cannot possibly influence the final decision, improving efficiency.

3. **Win Check**:
   - The `check_win` function checks all possible winning conditions in 3D Tic Tac Toe, including rows, columns, diagonals, and 3D diagonals across all layers and planes.

This implementation ensures that the AI makes optimal moves by considering all possible future game states and choosing the one that maximizes its chances of winning.
