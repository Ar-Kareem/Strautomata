def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    # Generate all possible winning lines in the 3D board
    all_lines = []
    
    # Straight lines along each axis
    for y in range(3):
        for z in range(3):
            all_lines.append([(x, y, z) for x in range(3)])
    for x in range(3):
        for z in range(3):
            all_lines.append([(x, y, z) for y in range(3)])
    for x in range(3):
        for y in range(3):
            all_lines.append([(x, y, z) for z in range(3)])
    
    # Face diagonals
    for z in range(3):
        all_lines.append([(x, x, z) for x in range(3)])
        all_lines.append([(x, 2-x, z) for x in range(3)])
    for y in range(3):
        all_lines.append([(x, y, x) for x in range(3)])
        all_lines.append([(x, y, 2-x) for x in range(3)])
    for x in range(3):
        all_lines.append([(x, y, y) for y in range(3)])
        all_lines.append([(x, y, 2-y) for y in range(3)])
    
    # Space diagonals
    all_lines.extend([
        [(0, 0, 0), (1, 1, 1), (2, 2, 2)],
        [(0, 0, 2), (1, 1, 1), (2, 2, 0)],
        [(0, 2, 0), (1, 1, 1), (2, 0, 2)],
        [(0, 2, 2), (1, 1, 1), (2, 0, 0)],
    ])
    
    # Check for immediate winning moves
    winning_moves = []
    for line in all_lines:
        vals = [board[x][y][z] for (x, y, z) in line]
        if sum(vals) == 2:  # Two 1's and one 0
            for i in range(3):
                if vals[i] == 0:
                    winning_moves.append(line[i])
    
    if winning_moves:
        # Prioritize center and corners
        winning_moves.sort(key=lambda pos: ((pos[0]==1 and pos[1]==1 and pos[2]==1) and 0) or 
                                         (sum(c in {0,2} for c in pos) ==3 and 1) or 2)
        return winning_moves[0]
    
    # Check for opponent's immediate threats
    blocking_moves = []
    for line in all_lines:
        vals = [board[x][y][z] for (x, y, z) in line]
        if sum(vals) == -2:  # Two -1's and one 0
            for i in range(3):
                if vals[i] == 0:
                    blocking_moves.append(line[i])
    
    if blocking_moves:
        # Prioritize center and corners
        blocking_moves.sort(key=lambda pos: ((pos[0]==1 and pos[1]==1 and pos[2]==1) and 0) or 
                                          (sum(c in {0,2} for c in pos) ==3 and 1) or 2)
        return blocking_moves[0]
    
    # Collect all empty cells
    empty_cells = [(x, y, z) for x in range(3) for y in range(3) for z in range(3) if board[x][y][z] == 0]
    
    # Prioritize center, then corners, then edges
    empty_cells.sort(key=lambda pos: (
        (pos[0] != 1 or pos[1] != 1 or pos[2] != 1),  # Center first
        -sum(c in {0,2} for c in pos),  # More extremes first (corners)
        sum(c == 1 for c in pos)        # Then edges (one middle coordinate)
    ))
    
    return empty_cells[0] if empty_cells else (0, 0, 0)
