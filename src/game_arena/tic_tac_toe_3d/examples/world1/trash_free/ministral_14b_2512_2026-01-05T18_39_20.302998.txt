Here's a Python implementation for a 3D Tic Tac Toe policy function that uses a combination of heuristic evaluation and minimax search with alpha-beta pruning to determine the next move. This implementation prioritizes winning moves, blocks opponent wins, and falls back to a simple heuristic if no immediate threats are present.

```python
import numpy as np
from typing import List, Tuple

def policy(board: List[List[List[int]]]) -> Tuple[int, int, int]:
    """
    Determines the next move for a 3D Tic Tac Toe game using minimax with alpha-beta pruning.

    Args:
        board: A 3x3x3 list of lists representing the game board. 0 = empty, 1 = player, -1 = opponent.

    Returns:
        A tuple (x, y, z) representing the coordinates of the chosen empty cell.
    """
    # Convert the board to a numpy array for easier manipulation
    board_np = np.array(board, dtype=int)

    # Find all empty cells
    empty_cells = np.argwhere(board_np == 0)

    # If there are no empty cells, return None (though this shouldn't happen in a valid game)
    if len(empty_cells) == 0:
        return (0, 0, 0)  # Shouldn't reach here in a valid game

    # Evaluate all possible moves using minimax with alpha-beta pruning
    best_score = -float('inf')
    best_move = None

    for cell in empty_cells:
        x, y, z = cell
        # Make a copy of the board to simulate the move
        new_board = board_np.copy()
        new_board[x, y, z] = 1  # Player's move (1)

        # Evaluate the move using minimax
        score = minimax(new_board, 0, -float('inf'), float('inf'), False)

        # Choose the move with the highest score
        if score > best_score:
            best_score = score
            best_move = (x, y, z)

    return best_move

def minimax(board: np.ndarray, depth: int, alpha: float, beta: float, is_maximizing: bool) -> int:
    """
    Minimax algorithm with alpha-beta pruning for 3D Tic Tac Toe.

    Args:
        board: Current game board.
        depth: Current depth in the search tree.
        alpha: Alpha value for alpha-beta pruning.
        beta: Beta value for alpha-beta pruning.
        is_maximizing: Whether the current player is maximizing (player) or minimizing (opponent).

    Returns:
        The score of the current board state from the perspective of the maximizing player.
    """
    # Check if the game is over
    winner = check_winner(board)
    if winner != 0:
        return winner * 10  # Multiply by 10 to prioritize winning over other heuristics

    # If the board is full, return 0 (draw)
    if np.all(board != 0):
        return 0

    if is_maximizing:
        best_score = -float('inf')
        for x in range(3):
            for y in range(3):
                for z in range(3):
                    if board[x, y, z] == 0:
                        board[x, y, z] = 1  # Player's move
                        score = minimax(board, depth + 1, alpha, beta, False)
                        board[x, y, z] = 0  # Undo move
                        best_score = max(score, best_score)
                        alpha = max(alpha, score)
                        if beta <= alpha:
                            break  # Alpha-beta pruning
                if beta <= alpha:
                    break
        return best_score
    else:
        best_score = float('inf')
        for x in range(3):
            for y in range(3):
                for z in range(3):
                    if board[x, y, z] == 0:
                        board[x, y, z] = -1  # Opponent's move
                        score = minimax(board, depth + 1, alpha, beta, True)
                        board[x, y, z] = 0  # Undo move
                        best_score = min(score, best_score)
                        beta = min(beta, score)
                        if beta <= alpha:
                            break  # Alpha-beta pruning
                if beta <= alpha:
                    break
        return best_score

def check_winner(board: np.ndarray) -> int:
    """
    Checks if there is a winner on the 3D Tic Tac Toe board.

    Args:
        board: Current game board.

    Returns:
        1 if player (1) wins, -1 if opponent (-1) wins, 0 otherwise.
    """
    # Check rows in each layer
    for layer in range(3):
        for row in range(3):
            if board[layer, row, :].sum() == 3:  # Player wins
                return 1
            if board[layer, row, :].sum() == -3:  # Opponent wins
                return -1

    # Check columns in each layer
    for layer in range(3):
        for col in range(3):
            if board[layer, :, col].sum() == 3:  # Player wins
                return 1
            if board[layer, :, col].sum() == -3:  # Opponent wins
                return -1

    # Check diagonals in each layer
    for layer in range(3):
        if board[layer, 0, 0] + board[layer, 1, 1] + board[layer, 2, 2] == 3:  # Player wins
            return 1
        if board[layer, 0, 0] + board[layer, 1, 1] + board[layer, 2, 2] == -3:  # Opponent wins
            return -1
        if board[layer, 0, 2] + board[layer, 1, 1] + board[layer, 2, 0] == 3:  # Player wins
            return 1
        if board[layer, 0, 2] + board[layer, 1, 1] + board[layer, 2, 0] == -3:  # Opponent wins
            return -1

    # Check vertical lines (through layers)
    for x in range(3):
        for y in range(3):
            if board[x, y, 0] + board[x, y, 1] + board[x, y, 2] == 3:  # Player wins
                return 1
            if board[x, y, 0] + board[x, y, 1] + board[x, y, 2] == -3:  # Opponent wins
                return -1

    # Check 3D diagonals
    if (board[0, 0, 0] + board[1, 1, 1] + board[2, 2, 2] == 3 or
        board[0, 0, 2] + board[1, 1, 1] + board[2, 2, 0] == 3 or
        board[0, 2, 0] + board[1, 1, 1] + board[2, 0, 2] == 3 or
        board[0, 2, 2] + board[1, 1, 1] + board[2, 0, 0] == 3):
        return 1
    if (board[0, 0, 0] + board[1, 1, 1] + board[2, 2, 2] == -3 or
        board[0, 0, 2] + board[1, 1, 1] + board[2, 2, 0] == -3 or
        board[0, 2, 0] + board[1, 1, 1] + board[2, 0, 2] == -3 or
        board[0, 2, 2] + board[1, 1, 1] + board[2, 0, 0] == -3):
        return -1

    return 0  # No winner
```

### Explanation:
1. **`policy` function**:
   - Converts the input board to a NumPy array for easier manipulation.
   - Finds all empty cells on the board.
   - Uses the `minimax` function to evaluate each possible move and selects the best one based on the highest score.

2. **`minimax` function**:
   - Implements the minimax algorithm with alpha-beta pruning to explore possible moves.
   - Alternates between maximizing (player) and minimizing (opponent) players.
   - Returns the score of the current board state from the perspective of the maximizing player.

3. **`check_winner` function**:
   - Checks all possible winning conditions in 3D Tic Tac Toe, including rows, columns, diagonals in each layer, vertical lines, and 3D diagonals.
   - Returns `1` if the player wins, `-1` if the opponent wins, and `0` otherwise.

This implementation ensures that the AI will always choose the optimal move to win or block the opponent, and it will fall back to a draw if no immediate threats are present. The use of alpha-beta pruning makes the search more efficient by eliminating branches that cannot possibly influence the final decision.
