Here's a Python implementation for a 3D Tic Tac Toe (3x3x3) policy function that uses a simple heuristic-based approach to determine the next move. This solution prioritizes winning moves, then blocks opponent winning moves, and finally chooses the center cell if available. If no strategic moves are found, it picks an empty cell randomly.

```python
import numpy as np
from typing import List, Tuple

def policy(board: List[List[List[int]]]) -> Tuple[int, int, int]:
    """
    Determines the next move for a 3D Tic Tac Toe game using a heuristic-based approach.

    Args:
        board: A 3x3x3 board where 0 is empty, 1 is player, and -1 is opponent.

    Returns:
        A tuple (x, y, z) representing the chosen empty cell in 3D coordinates.
    """
    # Convert to numpy array for easier manipulation
    board_np = np.array(board)

    # Check all possible winning moves for the player (1)
    for x in range(3):
        for y in range(3):
            for z in range(3):
                if board_np[x, y, z] == 0:
                    # Check all possible lines (rows, columns, pillars, and space diagonals)
                    # Rows in x-dimension
                    if np.all(board_np[x, y, :] == 1) or np.all(board_np[x, :, z] == 1) or np.all(board_np[x, :, y] == 1):
                        return (x, y, z)

                    # Columns in y-dimension
                    if np.all(board_np[x, :, z] == 1) or np.all(board_np[:, y, z] == 1) or np.all(board_np[x, :, y] == 1):
                        return (x, y, z)

                    # Pillars in z-dimension
                    if np.all(board_np[x, y, :] == 1) or np.all(board_np[:, :, z] == 1) or np.all(board_np[:, y, z] == 1):
                        return (x, y, z)

                    # Space diagonals (all 4 possible)
                    if x == y == z or x == y and z == 2 - x or x == z and y == 2 - x or y == z and x == 2 - y:
                        if np.all(board_np[np.eye(3, dtype=bool)[x, y, z], :] == 1):
                            return (x, y, z)

                    # Check all possible lines where placing a 1 would complete a win
                    # Rows in x-dimension
                    if np.sum(board_np[x, y, :] == 1) == 2 and np.sum(board_np[x, y, :] == 0) == 1:
                        return (x, y, z)
                    if np.sum(board_np[x, :, z] == 1) == 2 and np.sum(board_np[x, :, z] == 0) == 1:
                        return (x, y, z)
                    if np.sum(board_np[x, :, y] == 1) == 2 and np.sum(board_np[x, :, y] == 0) == 1:
                        return (x, y, z)

                    # Columns in y-dimension
                    if np.sum(board_np[x, :, z] == 1) == 2 and np.sum(board_np[x, :, z] == 0) == 1:
                        return (x, y, z)
                    if np.sum(board_np[:, y, z] == 1) == 2 and np.sum(board_np[:, y, z] == 0) == 1:
                        return (x, y, z)
                    if np.sum(board_np[x, :, y] == 1) == 2 and np.sum(board_np[x, :, y] == 0) == 1:
                        return (x, y, z)

                    # Pillars in z-dimension
                    if np.sum(board_np[x, y, :] == 1) == 2 and np.sum(board_np[x, y, :] == 0) == 1:
                        return (x, y, z)
                    if np.sum(board_np[:, :, z] == 1) == 2 and np.sum(board_np[:, :, z] == 0) == 1:
                        return (x, y, z)
                    if np.sum(board_np[:, y, z] == 1) == 2 and np.sum(board_np[:, y, z] == 0) == 1:
                        return (x, y, z)

                    # Space diagonals (all 4 possible)
                    if x == y == z:
                        if np.sum(board_np[np.arange(3), np.arange(3), np.arange(3)] == 1) == 2 and np.sum(board_np[np.arange(3), np.arange(3), np.arange(3)] == 0) == 1:
                            return (x, y, z)
                    if x == y and z == 2 - x:
                        if np.sum(board_np[np.arange(3), np.arange(3), [2 - i for i in np.arange(3)]] == 1) == 2 and np.sum(board_np[np.arange(3), np.arange(3), [2 - i for i in np.arange(3)]] == 0) == 1:
                            return (x, y, z)
                    if x == z and y == 2 - x:
                        if np.sum(board_np[np.arange(3), [2 - i for i in np.arange(3)], np.arange(3)] == 1) == 2 and np.sum(board_np[np.arange(3), [2 - i for i in np.arange(3)], np.arange(3)] == 0) == 1:
                            return (x, y, z)
                    if y == z and x == 2 - y:
                        if np.sum(board_np[[2 - i for i in np.arange(3)], np.arange(3), np.arange(3)] == 1) == 2 and np.sum(board_np[[2 - i for i in np.arange(3)], np.arange(3), np.arange(3)] == 0) == 1:
                            return (x, y, z)

    # Check all possible winning moves for the opponent (-1)
    for x in range(3):
        for y in range(3):
            for z in range(3):
                if board_np[x, y, z] == 0:
                    # Check all possible lines where placing a -1 would complete a win
                    # Rows in x-dimension
                    if np.sum(board_np[x, y, :] == -1) == 2 and np.sum(board_np[x, y, :] == 0) == 1:
                        return (x, y, z)
                    if np.sum(board_np[x, :, z] == -1) == 2 and np.sum(board_np[x, :, z] == 0) == 1:
                        return (x, y, z)
                    if np.sum(board_np[x, :, y] == -1) == 2 and np.sum(board_np[x, :, y] == 0) == 1:
                        return (x, y, z)

                    # Columns in y-dimension
                    if np.sum(board_np[x, :, z] == -1) == 2 and np.sum(board_np[x, :, z] == 0) == 1:
                        return (x, y, z)
                    if np.sum(board_np[:, y, z] == -1) == 2 and np.sum(board_np[:, y, z] == 0) == 1:
                        return (x, y, z)
                    if np.sum(board_np[x, :, y] == -1) == 2 and np.sum(board_np[x, :, y] == 0) == 1:
                        return (x, y, z)

                    # Pillars in z-dimension
                    if np.sum(board_np[x, y, :] == -1) == 2 and np.sum(board_np[x, y, :] == 0) == 1:
                        return (x, y, z)
                    if np.sum(board_np[:, :, z] == -1) == 2 and np.sum(board_np[:, :, z] == 0) == 1:
                        return (x, y, z)
                    if np.sum(board_np[:, y, z] == -1) == 2 and np.sum(board_np[:, y, z] == 0) == 1:
                        return (x, y, z)

                    # Space diagonals (all 4 possible)
                    if x == y == z:
                        if np.sum(board_np[np.arange(3), np.arange(3), np.arange(3)] == -1) == 2 and np.sum(board_np[np.arange(3), np.arange(3), np.arange(3)] == 0) == 1:
                            return (x, y, z)
                    if x == y and z == 2 - x:
                        if np.sum(board_np[np.arange(3), np.arange(3), [2 - i for i in np.arange(3)]] == -1) == 2 and np.sum(board_np[np.arange(3), np.arange(3), [2 - i for i in np.arange(3)]] == 0) == 1:
                            return (x, y, z)
                    if x == z and y == 2 - x:
                        if np.sum(board_np[np.arange(3), [2 - i for i in np.arange(3)], np.arange(3)] == -1) == 2 and np.sum(board_np[np.arange(3), [2 - i for i in np.arange(3)], np.arange(3)] == 0) == 1:
                            return (x, y, z)
                    if y == z and x == 2 - y:
                        if np.sum(board_np[[2 - i for i in np.arange(3)], np.arange(3), np.arange(3)] == -1) == 2 and np.sum(board_np[[2 - i for i in np.arange(3)], np.arange(3), np.arange(3)] == 0) == 1:
                            return (x, y, z)

    # If no immediate winning/blocking move, choose the center cell if available
    if board_np[1, 1, 1] == 0:
        return (1, 1, 1)

    # Otherwise, choose any empty cell randomly
    empty_cells = np.argwhere(board_np == 0)
    if len(empty_cells) > 0:
        return tuple(empty_cells[np.random.randint(len(empty_cells))])

    # If no empty cells (shouldn't happen in normal gameplay)
    raise ValueError("No empty cells available on the board")
```

### Explanation:
1. **Winning Moves Check**: The function first checks if placing a `1` in any empty cell would complete a winning line (row, column, pillar, or space diagonal).
2. **Blocking Opponent Moves**: If no winning move is found, it checks if placing a `1` would block the opponent from completing a winning line (placing a `-1`).
3. **Center Cell Preference**: If no immediate strategic move is found, it prioritizes the center cell `(1, 1, 1)` as it is often the most advantageous position in symmetric games.
4. **Random Move**: If no strategic or center move is available, it randomly selects an empty cell from the remaining options.

This is a basic heuristic approach. For a more advanced AI, you could implement minimax with alpha-beta pruning or other evaluation functions.
