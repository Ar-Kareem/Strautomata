def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    def is_winning_move(board, cell, player):
        x, y, z = cell
        original = board[x][y][z]
        board[x][y][z] = player
        
        # Check along axes
        if all(board[i][y][z] == player for i in range(3)):
            board[x][y][z] = original
            return True
        if all(board[x][j][z] == player for j in range(3)):
            board[x][y][z] = original
            return True
        if all(board[x][y][k] == player for k in range(3)):
            board[x][y][z] = original
            return True
        
        # Check face diagonals
        if x == y and all(board[i][i][z] == player for i in range(3)):
            board[x][y][z] = original
            return True
        if x + y == 2 and all(board[i][2-i][z] == player for i in range(3)):
            board[x][y][z] = original
            return True
        if x == z and all(board[i][y][i] == player for i in range(3)):
            board[x][y][z] = original
            return True
        if x + z == 2 and all(board[i][y][2-i] == player for i in range(3)):
            board[x][y][z] = original
            return True
        if y == z and all(board[x][i][i] == player for i in range(3)):
            board[x][y][z] = original
            return True
        if y + z == 2 and all(board[x][i][2-i] == player for i in range(3)):
            board[x][y][z] = original
            return True
        
        # Check space diagonals
        if x == y == z and all(board[i][i][i] == player for i in range(3)):
            board[x][y][z] = original
            return True
        if x == y and z == 2 - x and all(board[i][i][2-i] == player for i in range(3)):
            board[x][y][z] = original
            return True
        if x == z and y == 2 - x and all(board[i][2-i][i] == player for i in range(3)):
            board[x][y][z] = original
            return True
        if y == z and x == 2 - y and all(board[2-i][i][i] == player for i in range(3)):
            board[x][y][z] = original
            return True
        
        board[x][y][z] = original
        return False
    
    empty = [(x, y, z) for x in range(3) for y in range(3) for z in range(3) if board[x][y][z] == 0]
    
    # Check for winning move
    for cell in empty:
        if is_winning_move(board, cell, 1):
            return cell
    
    # Check for opponent's winning moves to block
    for cell in empty:
        if is_winning_move(board, cell, -1):
            return cell
    
    # Choose center if available
    center = (1, 1, 1)
    if center in empty:
        return center
    
    # Fallback to first available cell
    return empty[0]
