def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    # Precompute all possible winning lines once
    if not hasattr(policy, 'all_lines'):
        all_lines = []
        
        # Axes-aligned lines
        # Rows (fixed z, y)
        for z in range(3):
            for y in range(3):
                all_lines.append([(x, y, z) for x in range(3)])
        # Columns (fixed z, x)
        for z in range(3):
            for x in range(3):
                all_lines.append([(x, y, z) for y in range(3)])
        # Depth (fixed x, y)
        for x in range(3):
            for y in range(3):
                all_lines.append([(x, y, z) for z in range(3)])
        
        # Layer diagonals (fixed z)
        for z in range(3):
            all_lines.append([(x, x, z) for x in range(3)])          # Main diagonal
            all_lines.append([(x, 2 - x, z) for x in range(3)])     # Anti-diagonal
        
        # Space diagonals
        all_lines.append([(x, x, x) for x in range(3)])
        all_lines.append([(x, x, 2 - x) for x in range(3)])
        all_lines.append([(x, 2 - x, x) for x in range(3)])
        all_lines.append([(x, 2 - x, 2 - x) for x in range(3)])
        
        # Face diagonals (x-z planes, fixed y)
        for y in range(3):
            all_lines.append([(x, y, x) for x in range(3)])
            all_lines.append([(x, y, 2 - x) for x in range(3)])
        # Face diagonals (y-z planes, fixed x)
        for x in range(3):
            all_lines.append([(x, y, y) for y in range(3)])
            all_lines.append([(x, y, 2 - y) for y in range(3)])
        
        policy.all_lines = all_lines
    
    all_empty = [(x, y, z) for z in range(3) for y in range(3) for x in range(3) if board[z][y][x] == 0]
    
    # Check for winning move
    for (x, y, z) in all_empty:
        for line in policy.all_lines:
            if (x, y, z) in line:
                others = [cell for cell in line if cell != (x, y, z)]
                if sum(board[cz][cy][cx] for (cx, cy, cz) in others) == 2:
                    return (x, y, z)
    
    # Check for blocking move
    for (x, y, z) in all_empty:
        for line in policy.all_lines:
            if (x, y, z) in line:
                others = [cell for cell in line if cell != (x, y, z)]
                if sum(board[cz][cy][cx] for (cx, cy, cz) in others) == -2:
                    return (x, y, z)
    
    # Choose cell with highest potential (most lines)
    cell_scores = {}
    for (x, y, z) in all_empty:
        count = 0
        for line in policy.all_lines:
            if (x, y, z) in line:
                count += 1
        cell_scores[(x, y, z)] = count
    
    # Prefer center if available, then high-score cells
    if (1, 1, 1) in cell_scores:
        return (1, 1, 1)
    else:
        max_score = max(cell_scores.values())
        best_cells = [cell for cell, score in cell_scores.items() if score == max_score]
        # Among best cells, pick the first one (or could prioritize corners)
        return best_cells[0]
