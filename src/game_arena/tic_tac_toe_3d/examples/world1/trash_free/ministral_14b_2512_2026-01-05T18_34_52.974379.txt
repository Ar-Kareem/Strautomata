import numpy as np
from typing import List, Tuple

def policy(board: List[List[List[int]]]) -> Tuple[int, int, int]:
    """
    Determines the best move for a 3D Tic Tac Toe game using minimax with alpha-beta pruning.

    Args:
        board: A 3x3x3 board where 0 is empty, 1 is current player, -1 is opponent.

    Returns:
        A tuple (x, y, z) representing the coordinates of the best move.
    """
    # Convert to numpy array for easier manipulation
    board_np = np.array(board, dtype=int)

    # Find all empty cells
    empty_cells = np.argwhere(board_np == 0)

    if len(empty_cells) == 0:
        raise ValueError("Board is full")

    # If it's the first move, choose the center
    if len(empty_cells) == 27:
        return (1, 1, 1)

    # Evaluate all possible moves using minimax
    best_score = -float('inf')
    best_move = None

    for cell in empty_cells:
        x, y, z = cell
        # Make a copy of the board
        new_board = board_np.copy()
        new_board[x, y, z] = 1

        # Evaluate the move
        score = minimax(new_board, 0, -float('inf'), float('inf'), False)

        # Update best move if this is better
        if score > best_score:
            best_score = score
            best_move = (x, y, z)

    return best_move

def minimax(board: np.ndarray, depth: int, alpha: float, beta: float, is_maximizing: bool) -> float:
    """
    Minimax algorithm with alpha-beta pruning for 3D Tic Tac Toe.

    Args:
        board: Current board state
        depth: Current depth in the search tree
        alpha: Best value that the maximizer can guarantee so far
        beta: Best value that the minimizer can guarantee so far
        is_maximizing: Whether the current player is maximizing (1) or minimizing (-1)

    Returns:
        The score of the current board state
    """
    # Check if the game is over
    if check_win(board, 1):
        return 10 - depth  # Higher score for immediate wins
    if check_win(board, -1):
        return -10 + depth  # Lower score for opponent wins

    # Check for draw
    if np.all(board != 0):
        return 0

    if is_maximizing:
        best_score = -float('inf')
        for cell in np.argwhere(board == 0):
            x, y, z = cell
            board[x, y, z] = 1
            score = minimax(board, depth + 1, alpha, beta, False)
            board[x, y, z] = 0  # Undo move
            best_score = max(score, best_score)
            alpha = max(alpha, best_score)
            if beta <= alpha:
                break  # Beta cutoff
        return best_score
    else:
        best_score = float('inf')
        for cell in np.argwhere(board == 0):
            x, y, z = cell
            board[x, y, z] = -1
            score = minimax(board, depth + 1, alpha, beta, True)
            board[x, y, z] = 0  # Undo move
            best_score = min(score, best_score)
            beta = min(beta, best_score)
            if beta <= alpha:
                break  # Alpha cutoff
        return best_score

def check_win(board: np.ndarray, player: int) -> bool:
    """
    Check if the given player has won on the 3D board.

    Args:
        board: Current board state
        player: Player to check (1 or -1)

    Returns:
        True if the player has won, False otherwise
    """

    for z in range(3):
        for x in range(3):
            if all(board[x, y, z] == player for y in range(3)):
                return True
        for y in range(3):
            if all(board[x, y, z] == player for x in range(3)):
                return True

    for z in range(3):
        for x in range(3):
            if all(board[x, y, z] == player for y in range(3)):
                return True
        for y in range(3):
            if all(board[x, y, z] == player for x in range(3)):

    for x in range(3):

            if all(board[x, y, z] == player for z in range(3)):
                return True

    # Check horizontal lines through layers
    for x in range(3):
        for z in range(3):
            if all(board[x, y, z] == player for y in range(3)):
                return True

    # Check vertical columns through layers
    for y in range(3):
        for z in range(3):
            if all(board[x, y, z] == player for x in range(3)):
                return True

    # Check space diagonals through layers
    if (board[0, 0, 0] == player and board[1, 1, 1] == player and board[2, 2, 2] == player) or \
       (board[0, 2, 0] == player and board[1, 1, 1] == player and board[2, 0, 2] == player) or \
       (board[0, 0, 2] == player and board[1, 1, 1] == player and board[2, 2, 0] == player) or \

        return True

    # Check face diagonals through layers
    for z in range(3):

            return True

    return False

This implementation:

1. Uses minimax with alpha-beta pruning to evaluate possible moves
2. Checks all possible winning conditions in 3D space (rows, columns, diagonals in all planes)
3. Prioritizes immediate wins and blocks opponent wins
4. Has a special case for the first move (center)
5. Uses numpy for efficient board manipulation

The algorithm will:
- Return the best move that leads to a win if possible
- Block opponent wins if they're about to win
- Otherwise choose moves that lead to the best possible position

Note that for a full 3D Tic Tac Toe game, this implementation might be computationally intensive for deeper searches. In a real game scenario, you might want to add:
- Move ordering (try more promising moves first)
- Depth limiting
- Additional heuristics to evaluate board positions
- Transposition tables to cache previously evaluated positions
