Here's a Python implementation for a 3D Tic Tac Toe (3x3x3) policy function that uses a simple heuristic-based approach to determine the next move. This solution prioritizes winning moves, then blocks opponent winning moves, then evaluates the center, and finally chooses a random empty cell.

```python
import numpy as np
import random

def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    """
    Determines the next move for a 3D Tic Tac Toe game using a heuristic-based approach.

    Args:
        board: A 3x3x3 list of lists where:
            - 0 represents an empty cell
            - 1 represents the player's cell
            - -1 represents the opponent's cell

    Returns:
        A tuple (x, y, z) representing the chosen empty cell in the 3D space.
    """
    # Convert the board to a numpy array for easier manipulation
    board_np = np.array(board)

    # Find all empty cells (0)
    empty_cells = np.argwhere(board_np == 0)

    # Check for winning moves (if any)
    for cell in empty_cells:
        x, y, z = cell
        # Check all possible lines (rows, columns, pillars, and space diagonals)
        # Rows in x dimension
        if np.all(board_np[x, y, :] == 1):
            return (x, y, z)
        # Columns in y dimension
        if np.all(board_np[x, :, z] == 1):
            return (x, y, z)
        # Pillars in z dimension
        if np.all(board_np[:, y, z] == 1):
            return (x, y, z)
        # Space diagonals (4 main diagonals in each 2D layer)
        # Diagonal 1: (0,0,0) to (2,2,2)
        if x == y == z:
            if np.all(board_np[0, 0, 0] == 1) or np.all(board_np[1, 1, 1] == 1) or np.all(board_np[2, 2, 2] == 1):
                return (x, y, z)
        # Diagonal 2: (0,0,2) to (2,2,0)
        if x == y and z == 2 - x:
            if np.all(board_np[0, 0, 2] == 1) or np.all(board_np[1, 1, 1] == 1) or np.all(board_np[2, 2, 0] == 1):
                return (x, y, z)
        # Diagonal 3: (0,2,0) to (2,0,2)
        if x == 2 - y and z == x:
            if np.all(board_np[0, 2, 0] == 1) or np.all(board_np[1, 1, 1] == 1) or np.all(board_np[2, 0, 2] == 1):
                return (x, y, z)
        # Diagonal 4: (0,2,2) to (2,0,0)
        if x == 2 - y and z == 2 - x:
            if np.all(board_np[0, 2, 2] == 1) or np.all(board_np[1, 1, 1] == 1) or np.all(board_np[2, 0, 0] == 1):
                return (x, y, z)

    # Check for opponent's winning moves and block them
    for cell in empty_cells:
        x, y, z = cell
        # Check all possible lines for opponent
        if np.all(board_np[x, y, :] == -1):
            return (x, y, z)
        if np.all(board_np[x, :, z] == -1):
            return (x, y, z)
        if np.all(board_np[:, y, z] == -1):
            return (x, y, z)
        # Space diagonals
        if x == y == z:
            if np.all(board_np[0, 0, 0] == -1) or np.all(board_np[1, 1, 1] == -1) or np.all(board_np[2, 2, 2] == -1):
                return (x, y, z)
        if x == y and z == 2 - x:
            if np.all(board_np[0, 0, 2] == -1) or np.all(board_np[1, 1, 1] == -1) or np.all(board_np[2, 2, 0] == -1):
                return (x, y, z)
        if x == 2 - y and z == x:
            if np.all(board_np[0, 2, 0] == -1) or np.all(board_np[1, 1, 1] == -1) or np.all(board_np[2, 0, 2] == -1):
                return (x, y, z)
        if x == 2 - y and z == 2 - x:
            if np.all(board_np[0, 2, 2] == -1) or np.all(board_np[1, 1, 1] == -1) or np.all(board_np[2, 0, 0] == -1):
                return (x, y, z)

    # If no winning or blocking moves, prioritize the center (1,1,1)
    if board_np[1, 1, 1] == 0:
        return (1, 1, 1)

    # If no immediate winning or blocking moves, choose a random empty cell
    return tuple(random.choice(empty_cells).tolist())

# Helper function to check if a move is valid (not already occupied)
def is_valid_move(board: list[list[list[int]]], move: tuple[int, int, int]) -> bool:
    x, y, z = move
    return board[x][y][z] == 0

# Helper function to check if the board is full (draw)
def is_board_full(board: list[list[list[int]]]) -> bool:
    return np.all(np.array(board) != 0)

# Helper function to check if the game is won by player or opponent
def is_won(board: list[list[list[int]]], player: int) -> bool:
    board_np = np.array(board)
    # Check rows, columns, pillars, and space diagonals
    for x in range(3):
        for y in range(3):
            if np.all(board_np[x, y, :] == player):
                return True
            if np.all(board_np[x, :, y] == player):
                return True
    for y in range(3):
        for z in range(3):
            if np.all(board_np[:, y, z] == player):
                return True
    # Check space diagonals
    if np.all(board_np[0, 0, 0] == player) and np.all(board_np[1, 1, 1] == player) and np.all(board_np[2, 2, 2] == player):
        return True
    if np.all(board_np[0, 0, 2] == player) and np.all(board_np[1, 1, 1] == player) and np.all(board_np[2, 2, 0] == player):
        return True
    if np.all(board_np[0, 2, 0] == player) and np.all(board_np[1, 1, 1] == player) and np.all(board_np[2, 0, 2] == player):
        return True
    if np.all(board_np[0, 2, 2] == player) and np.all(board_np[1, 1, 1] == player) and np.all(board_np[2, 0, 0] == player):
        return True
    return False
```

### Explanation:
1. **Winning Move Check**: The function first checks if placing a `1` in any empty cell would result in a win (3 in a row in any dimension or space diagonal).
2. **Blocking Opponent's Win**: If no winning move is found, it checks if placing a `1` in any empty cell would block the opponent from winning (3 in a row of `-1`).
3. **Center Priority**: If neither winning nor blocking moves are available, it prioritizes the center cell `(1, 1, 1)` as it is often the most strategic position.
4. **Random Move**: If no immediate moves are found, it selects a random empty cell.

### Note:
- The helper functions (`is_valid_move`, `is_board_full`, `is_won`) are included for completeness but are not part of the required `policy` function.
- This is a **heuristic-based** approach and not a perfect minimax solver (which would be computationally expensive for a 3x3x3 game). For a more advanced solution, you could implement minimax with alpha-beta pruning, but that would require more code and potentially longer execution times.
