Here's a Python implementation for a basic 3D Tic Tac Toe policy function that uses a simple heuristic to choose the next move. This version prioritizes winning moves, then blocks opponent winning moves, then chooses the center if available, and finally picks a random empty cell.

```python
import random
from typing import List, Tuple

def policy(board: List[List[List[int]]]) -> Tuple[int, int, int]:
    """
    Returns the next move for a 3x3x3 3D Tic Tac Toe game.
    The move is chosen based on a simple heuristic:
    1. Try to win immediately
    2. Block opponent from winning
    3. Take the center if available
    4. Choose a random empty cell

    Args:
        board: A 3x3x3 board where 0 is empty, 1 is player, -1 is opponent

    Returns:
        A tuple (x, y, z) representing the next move in 0-2 coordinates
    """

    # Check all possible winning moves for player (1)
    for move in find_winning_moves(board, 1):
        return move

    # Check all possible winning moves for opponent (-1) and block them
    for move in find_winning_moves(board, -1):
        return move

    # Try to take the center (position (1,1,1))
    if board[1][1][1] == 0:
        return (1, 1, 1)

    # If no immediate winning/blocking move, choose a random empty cell
    empty_cells = []
    for x in range(3):
        for y in range(3):
            for z in range(3):
                if board[x][y][z] == 0:
                    empty_cells.append((x, y, z))

    return random.choice(empty_cells)

def find_winning_moves(board: List[List[List[int]]], player: int) -> List[Tuple[int, int, int]]:
    """
    Finds all empty cells that would complete a winning line for the given player.

    Args:
        board: The current game board
        player: The player to check for winning moves (1 or -1)

    Returns:
        A list of tuples representing empty cells that would complete a winning line
    """
    winning_moves = []

    # Check all rows in each layer (x fixed)
    for x in range(3):
        for y in range(3):
            for z in [0, 1, 2]:
                if board[x][y][z] == 0:
                    # Check if this cell completes a row
                    if all(board[x][y][i] == player for i in range(3) if i != z):
                        winning_moves.append((x, y, z))

    # Check all columns in each layer (y fixed)
    for x in range(3):
        for y in [0, 1, 2]:
            for z in range(3):
                if board[x][y][z] == 0:
                    # Check if this cell completes a column
                    if all(board[x][i][z] == player for i in range(3) if i != y):
                        winning_moves.append((x, y, z))

    # Check all depth lines in each layer (z fixed)
    for x in range(3):
        for y in range(3):
            for z in [0, 1, 2]:
                if board[x][y][z] == 0:
                    # Check if this cell completes a depth line
                    if all(board[i][y][z] == player for i in range(3) if i != x):
                        winning_moves.append((x, y, z))

    # Check all 3D diagonals (main and anti-diagonals)
    # Main diagonals (x=y=z)
    for i in range(3):
        if board[i][i][i] == 0:
            if all(board[j][j][j] == player for j in range(3) if j != i):
                winning_moves.append((i, i, i))

    # Anti-diagonals (x+y+z=2)
    for i in range(3):
        for j in range(3):
            for k in range(3):
                if i + j + k == 2 and board[i][j][k] == 0:
                    if all(board[a][b][c] == player for a, b, c in
                         [(0,0,2), (0,1,1), (0,2,0), (1,0,1), (1,1,0), (1,2,2), (2,0,0), (2,1,1), (2,2,2)]
                         if (a,b,c) != (i,j,k) and a + b + c == 2):
                        winning_moves.append((i, j, k))

    # Check all 2D diagonals in each layer (x fixed, y=z or x=y, z fixed)
    # Diagonals where y=z (top-left to bottom-right)
    for x in range(3):
        for y in [0, 1, 2]:
            z = y
            if board[x][y][z] == 0:
                if all(board[x][i][i] == player for i in range(3) if i != y):
                    winning_moves.append((x, y, z))

    # Diagonals where x=y (top-left to bottom-right)
    for y in range(3):
        for z in range(3):
            x = y
            if board[x][y][z] == 0:
                if all(board[i][i][z] == player for i in range(3) if i != y):
                    winning_moves.append((x, y, z))

    # Diagonals where y=z (top-right to bottom-left)
    for x in range(3):
        for y in [0, 1, 2]:
            z = 2 - y
            if board[x][y][z] == 0:
                if all(board[x][i][2-i] == player for i in range(3) if i != y):
                    winning_moves.append((x, y, z))

    # Diagonals where x=y (top-right to bottom-left)
    for y in range(3):
        for z in range(3):
            x = y
            if board[x][y][z] == 0:
                if all(board[i][i][2-i] == player for i in range(3) if i != y):
                    winning_moves.append((x, y, z))

    return winning_moves
```

### Explanation:
1. **`policy` function**:
   - First checks if there's any empty cell that would complete a winning line for the player (`1`).
   - Then checks if there's any empty cell that would complete a winning line for the opponent (`-1`) and blocks them.
   - If the center cell `(1,1,1)` is empty, it takes that cell as a strong heuristic.
   - Finally, it picks a random empty cell if no immediate winning/blocking move is available.

2. **`find_winning_moves` function**:
   - Checks all possible lines (rows, columns, depth lines, 3D diagonals, and 2D diagonals in each layer) to see if placing the player in an empty cell would complete a winning line.
   - Returns a list of all such moves.
