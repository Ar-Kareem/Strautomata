def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    # Generate all possible winning lines
    lines = []
    
    # Lines within each layer (same z)
    for z in range(3):
        # Rows (fixed y, z)
        for y in range(3):
            lines.append([(x, y, z) for x in range(3)])
        # Columns (fixed x, z)
        for x in range(3):
            lines.append([(x, y, z) for y in range(3)])
        # Layer diagonals
        lines.append([(i, i, z) for i in range(3)])
        lines.append([(i, 2 - i, z) for i in range(3)])
    
    # Vertical columns (fixed x, y)
    for x in range(3):
        for y in range(3):
            lines.append([(x, y, z) for z in range(3)])
    
    # Space diagonals
    lines.append([(0, 0, 0), (1, 1, 1), (2, 2, 2)])
    lines.append([(0, 0, 2), (1, 1, 1), (2, 2, 0)])
    lines.append([(0, 2, 0), (1, 1, 1), (2, 0, 2)])
    lines.append([(0, 2, 2), (1, 1, 1), (2, 0, 0)])
    
    # Collect empty cells
    empty_cells = []
    for x in range(3):
        for y in range(3):
            for z in range(3):
                if board[z][y][x] == 0:
                    empty_cells.append((x, y, z))
    
    # Check for winning move
    for (x, y, z) in empty_cells:
        for line in lines:
            if (x, y, z) in line:
                total = sum(board[k][j][i] for (i, j, k) in line)
                if total == 2:  # Two 1s and this cell (0) → place 1 to win
                    return (x, y, z)
    
    # Check for blocking opponent's win
    block_candidates = []
    for (x, y, z) in empty_cells:
        for line in lines:
            if (x, y, z) in line:
                total = sum(board[k][j][i] for (i, j, k) in line)
                if total == -2:  # Two -1s and this cell (0) → block
                    block_candidates.append((x, y, z))
                    break  # Only need one threatening line
    if block_candidates:
        return block_candidates[0]
    
    # Take center if available
    if (1, 1, 1) in empty_cells:
        return (1, 1, 1)
    
    # Take any corner
    corners = [(0, 0, 0), (0, 0, 2), (0, 2, 0), (0, 2, 2),
               (2, 0, 0), (2, 0, 2), (2, 2, 0), (2, 2, 2)]
    for corner in corners:
        if corner in empty_cells:
            return corner
    
    # Take first available cell if none of the above
    return empty_cells[0]
