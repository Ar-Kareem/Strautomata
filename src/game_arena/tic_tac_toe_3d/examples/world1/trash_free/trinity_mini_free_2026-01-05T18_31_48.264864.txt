def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    # Generate all possible winning lines in 3x3x3 Tic Tac Toe
    lines = []
    
    # X-axis lines (fixed y, z; vary x)
    for y in range(3):
        for z in range(3):
            lines.append([(x, y, z) for x in range(3)])
    
    # Y-axis lines (fixed x, z; vary y)
    for x in range(3):
        for z in range(3):
            lines.append([(x, y, z) for y in range(3)])
    
    # Z-axis lines (fixed x, y; vary z)
    for x in range(3):
        for y in range(3):
            lines.append([(x, y, z) for z in range(3)])
    
    # Diagonals in xy-plane (fixed z; two diagonals per layer)
    for z in range(3):
        lines.append([(0, 0, z), (1, 1, z), (2, 2, z)])
        lines.append([(0, 2, z), (1, 1, z), (2, 0, z)])
    
    # Diagonals in xz-plane (fixed y; two diagonals per layer)
    for y in range(3):
        lines.append([(0, y, 0), (1, y, 1), (2, y, 2)])
        lines.append([(0, y, 2), (1, y, 1), (2, y, 0)])
    
    # Diagonals in yz-plane (fixed x; two diagonals per layer)
    for x in range(3):
        lines.append([(x, 0, 0), (x, 1, 1), (x, 2, 2)])
        lines.append([(x, 0, 2), (x, 1, 1), (x, 2, 0)])
    
    # Space diagonals (four main diagonals)
    lines.append([(0, 0, 0), (1, 1, 1), (2, 2, 2)])
    lines.append([(0, 0, 2), (1, 1, 1), (2, 2, 0)])
    lines.append([(0, 2, 0), (1, 1, 1), (2, 0, 2)])
    lines.append([(0, 2, 2), (1, 1, 1), (2, 0, 0)])
    
    # Check each line for winning/blocking opportunities
    for line in lines:
        coords = line
        values = [board[x][y][z] for (x, y, z) in coords]
        if values.count(1) == 2 and values.count(0) == 1:
            # Find and return the empty cell in this line
            for i, val in enumerate(values):
                if val == 0:
                    x, y, z = coords[i]
                    return (x, y, z)
        elif values.count(-1) == 2 and values.count(0) == 1:
            # Block opponent's potential win
            for i, val in enumerate(values):
                if val == 0:
                    x, y, z = coords[i]
                    return (x, y, z)
    
    # If no immediate threats, choose the first empty cell in row-major order
    for x in range(3):
        for y in range(3):
            for z in range(3):
                if board[x][y][z] == 0:
                    return (x, y, z)
    
    # Fallback (should not occur in valid gameplay)
    return (0, 0, 0)
