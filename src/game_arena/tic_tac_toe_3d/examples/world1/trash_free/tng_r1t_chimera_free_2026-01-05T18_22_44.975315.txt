def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    # Generate all possible winning lines
    all_lines = []
    
    # Rows (in each z-layer, y-row)
    for z in range(3):
        for y in range(3):
            all_lines.append([(x, y, z) for x in range(3)])
    
    # Columns (in each z-layer, x-column)
    for z in range(3):
        for x in range(3):
            all_lines.append([(x, y, z) for y in range(3)])
    
    # Pillars (vertical columns along z-axis)
    for x in range(3):
        for y in range(3):
            all_lines.append([(x, y, z) for z in range(3)])
    
    # Face diagonals
    for z in [0, 2]:
        all_lines.append([(0, 0, z), (1, 1, z), (2, 2, z)])
        all_lines.append([(0, 2, z), (1, 1, z), (2, 0, z)])
    for x in [0, 2]:
        all_lines.append([(x, 0, 0), (x, 1, 1), (x, 2, 2)])
        all_lines.append([(x, 0, 2), (x, 1, 1), (x, 2, 0)])
    for y in [0, 2]:
        all_lines.append([(0, y, 0), (1, y, 1), (2, y, 2)])
        all_lines.append([(0, y, 2), (1, y, 1), (2, y, 0)])
    
    # Space diagonals
    all_lines.append([(0, 0, 0), (1, 1, 1), (2, 2, 2)])
    all_lines.append([(0, 0, 2), (1, 1, 1), (2, 2, 0)])
    all_lines.append([(0, 2, 0), (1, 1, 1), (2, 0, 2)])
    all_lines.append([(0, 2, 2), (1, 1, 1), (2, 0, 0)])
    
    # Check for winning moves
    for line in all_lines:
        count_me = 0
        empty = None
        for x, y, z in line:
            if board[x][y][z] == 1:
                count_me += 1
            elif board[x][y][z] == 0:
                empty = (x, y, z)
        if count_me == 2 and empty is not None:
            return empty
    
    # Check for opponent's winning moves to block
    for line in all_lines:
        count_opp = 0
        empty = None
        for x, y, z in line:
            if board[x][y][z] == -1:
                count_opp += 1
            elif board[x][y][z] == 0:
                empty = (x, y, z)
        if count_opp == 2 and empty is not None:
            return empty
    
    # Priority cells in strategic order
    priority_cells = [
        (1, 1, 1),  # Center
        (1, 1, 0), (1, 1, 2),  # Front/back centers
        (1, 0, 1), (1, 2, 1),  # Top/bottom centers
        (0, 1, 1), (2, 1, 1),  # Left/right centers
        (0, 0, 1), (0, 2, 1), (2, 0, 1), (2, 2, 1),  # Horizontal edges
        (0, 1, 0), (0, 1, 2), (2, 1, 0), (2, 1, 2),  # Depth edges
        (1, 0, 0), (1, 0, 2), (1, 2, 0), (1, 2, 2),  # Vertical edges
        (0, 0, 0), (0, 0, 2), (0, 2, 0), (0, 2, 2),  # Corners
        (2, 0, 0), (2, 0, 2), (2, 2, 0), (2, 2, 2),  # Corners
    ]
    
    for cell in priority_cells:
        x, y, z = cell
        if board[x][y][z] == 0:
            return cell
    
    # Fallback (shouldn't be needed if game isn't over)
    for x in range(3):
        for y in range(3):
            for z in range(3):
                if board[x][y][z] == 0:
                    return (x, y, z)
    
    return (0, 0, 0)  # Default return (full board)
