from typing import List, Tuple, Optional

                    # Undo the move
                    board_np[x, y, z] = 0

                    # Update best move if this move is better
                    if score > best_score:
                        best_score = score
                        best_move = (x, y, z)

def minimax(board: np.ndarray, depth: int, alpha: float, beta: float, player: int) -> float:
    """
    Minimax algorithm with alpha-beta pruning to evaluate the best move.

    Args:
        board: Current game board.
        depth: Current depth in the search tree.
        alpha: Alpha value for pruning.
        beta: Beta value for pruning.
        player: Current player (1 for maximizing, -1 for minimizing).

    Returns:
        The evaluated score of the current board position.
    """
    # Check if the game is terminal
    if is_terminal(board, player):
        return evaluate(board, player)

    # If it's the opponent's turn, minimize the score
    if player == -1:
        value = float('inf')
        for x in range(3):
            for y in range(3):
                for z in range(3):
                    if board[x, y, z] == 0:
                        board[x, y, z] = -1
                        value = min(value, minimax(board, depth + 1, alpha, beta, 1))
                        board[x, y, z] = 0
                        alpha = min(alpha, value)
                        if beta <= alpha:
                            break
                if beta <= alpha:
                    break
        return value
    # If it's our turn, maximize the score
    else:
        value = -float('inf')
        for x in range(3):
            for y in range(3):
                for z in range(3):
                    if board[x, y, z] == 0:
                        board[x, y, z] = 1
                        value = max(value, minimax(board, depth + 1, alpha, beta, -1))
                        board[x, y, z] = 0
                        beta = max(beta, value)
                        if beta <= alpha:
                            break
                if beta <= alpha:
                    break
        return value

def is_terminal(board: np.ndarray, player: int) -> bool:
    """
    Checks if the game is terminal (win or draw).

    Args:
        board: Current game board.
        player: The player to check for a win.

    Returns:
        True if the game is terminal, False otherwise.
    """

    for z in range(3):
        for x in range(3):
            if all(board[x, y, z] == player for y in range(3)):

            if all(board[x, y, z] == player for x in range(3)):

    for x in range(3):

            if all(board[x, y, z] == player for y in range(3)):

    # Check vertical columns through layers
    for y in range(3):

            if all(board[x, y, z] == player for x in range(3)):

    # Check space diagonals through layers
    if all(board[x, x, x] == player for x in range(3)):

        return True
    if all(board[x, x, 2 - x] == player for x in range(3)):
        return True
    if all(board[x, 2 - x, 2 - x] == player for x in range(3)):
        return True

    # Check if the board is full (draw)
    return all(board[x, y, z] != 0 for x in range(3) for y in range(3) for z in range(3))

def evaluate(board: np.ndarray, player: int) -> float:
    """
    Evaluates the current board position.

    Args:
        board: Current game board.
        player: The player to evaluate for.

    Returns:
        A score representing how good the position is for the player.
    """
    if is_terminal(board, player):
        return 1000  # Win
    elif is_terminal(board, -player):
        return -1000  # Loss
    else:
        return 0  # Draw or ongoing game

def find_empty_cell(board: np.ndarray) -> Tuple[int, int, int]:
    """
    Finds the first empty cell in the board.

    Args:
        board: Current game board.

    Returns:
        A tuple (x, y, z) representing the coordinates of the first empty cell.
    """
    for x in range(3):
        for y in range(3):
            for z in range(3):
                if board[x, y, z] == 0:
                    return (x, y, z)
    return (0, 0, 0)  # Shouldn't reach here if the game is ongoing

### Explanation:

2. **`minimax` function**: This recursive function evaluates the game tree, alternating between maximizing (our player) and minimizing (opponent) the score.
3. **`is_terminal` function**: Checks if the game is over (win, loss, or draw) by examining all possible winning conditions in 3D Tic Tac Toe.
4. **`evaluate` function**: Assigns a score to the current board position (1000 for win, -1000 for loss, 0 otherwise).
5. **`find_empty_cell` function**: Helper function to find the first empty cell (used in edge cases).
