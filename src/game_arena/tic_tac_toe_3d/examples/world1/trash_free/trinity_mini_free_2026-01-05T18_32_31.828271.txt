def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    # Define all 49 lines in a 3x3x3 Tic Tac Toe board
    lines = []
    
    # Lines parallel to x-axis (fixed y, z)
    for y in range(3):
        for z in range(3):
            lines.append([(x, y, z) for x in range(3)])
    
    # Lines parallel to y-axis (fixed x, z)
    for x in range(3):
        for z in range(3):
            lines.append([(x, y, z) for y in range(3)])
    
    # Lines parallel to z-axis (fixed x, y)
    for x in range(3):
        for y in range(3):
            lines.append([(x, y, z) for z in range(3)])
    
    # Face diagonals (fixed z)
    for z in range(3):
        lines.append([(0, 0, z), (1, 1, z), (2, 2, z)])
        lines.append([(0, 2, z), (1, 1, z), (2, 0, z)])
    
    # Face diagonals (fixed y)
    for y in range(3):
        lines.append([(0, y, 0), (1, y, 1), (2, y, 2)])
        lines.append([(0, y, 2), (1, y, 1), (2, y, 0)])
    
    # Face diagonals (fixed x)
    for x in range(3):
        lines.append([(x, 0, 0), (x, 1, 1), (x, 2, 2)])
        lines.append([(x, 0, 2), (x, 1, 1), (x, 2, 0)])
    
    # Space diagonals
    lines.append([(0, 0, 0), (1, 1, 1), (2, 2, 2)])
    lines.append([(0, 0, 2), (1, 1, 1), (2, 2, 0)])
    lines.append([(0, 2, 0), (1, 1, 1), (2, 0, 2)])
    lines.append([(0, 2, 2), (1, 1, 1), (2, 0, 0)])
    
    # Check for winning moves (two 1s and one empty)
    for line in lines:
        count1 = 0
        empty_pos = None
        for coord in line:
            val = board[coord[0]][coord[1]][coord[2]]
            if val == 1:
                count1 += 1
            elif val == 0:
                empty_pos = coord
        if count1 == 2 and empty_pos is not None:
            return empty_pos
    
    # Check for blocking moves (two -1s and one empty)
    for line in lines:
        count_opponent = 0
        empty_pos = None
        for coord in line:
            val = board[coord[0]][coord[1]][coord[2]]
            if val == -1:
                count_opponent += 1
            elif val == 0:
                empty_pos = coord
        if count_opponent == 2 and empty_pos is not None:
            return empty_pos
    
    # If no winning/blocking moves, pick the first empty cell
    for x in range(3):
        for y in range(3):
            for z in range(3):
                if board[x][y][z] == 0:
                    return (x, y, z)
    
    # Fallback: return (0,0,0) if no empty cells (shouldn't happen in valid game)
    return (0, 0, 0)
