import numpy as np
from typing import List, Tuple

def policy(board: List[List[List[int]]]) -> Tuple[int, int, int]:
    """
    Determines the best move for a 3x3x3 3D Tic Tac Toe game using minimax with alpha-beta pruning.

    Args:
        board: A 3x3x3 list of lists representing the game board (0=empty, 1=player, -1=opponent)

    Returns:
        A tuple (x, y, z) representing the coordinates of the best move (0-2 for each dimension)
    """
    # Convert to numpy array for easier manipulation
    board_np = np.array(board, dtype=int)

    # Find all empty cells
    empty_cells = np.argwhere(board_np == 0)

    if len(empty_cells) == 0:
        raise ValueError("No empty cells left on the board")

    # If it's the first move, choose the center (1,1,1)
    if len(empty_cells) == 27:
        return (1, 1, 1)

    # Evaluate all possible moves using minimax with alpha-beta pruning
    best_score = -float('inf')
    best_move = None

    for cell in empty_cells:
        x, y, z = cell
        # Make a copy of the board
        new_board = board_np.copy()
        new_board[x, y, z] = 1  # Player's move

        # Evaluate the move
        score = minimax(new_board, 0, -float('inf'), float('inf'), False)

        # Update best move if this is better
        if score > best_score:
            best_score = score
            best_move = (x, y, z)

    return best_move

def minimax(board: np.ndarray, depth: int, alpha: float, beta: float, is_maximizing: bool) -> float:
    """
    Minimax algorithm with alpha-beta pruning for 3D Tic Tac Toe.

    Args:
        board: Current game board
        depth: Current depth in the search tree
        alpha: Alpha value for pruning
        beta: Beta value for pruning
        is_maximizing: Whether the current player is maximizing (player) or minimizing (opponent)

    Returns:
        The evaluated score of the current board position
    """
    # Check for terminal states
    winner = check_winner(board)
    if winner != 0:
        return winner * (10 - depth)  # Higher weight for earlier wins

    # If no winner and board is full, it's a draw
    if np.all(board != 0):
        return 0

    if is_maximizing:
        value = -float('inf')
        for x in range(3):
            for y in range(3):
                for z in range(3):
                    if board[x, y, z] == 0:
                        board[x, y, z] = 1
                        value = max(value, minimax(board, depth + 1, alpha, beta, False))
                        board[x, y, z] = 0
                        alpha = max(alpha, value)
                        if beta <= alpha:
                            break
                if beta <= alpha:
                    break
        return value
    else:
        value = float('inf')
        for x in range(3):
            for y in range(3):
                for z in range(3):
                    if board[x, y, z] == 0:
                        board[x, y, z] = -1
                        value = min(value, minimax(board, depth + 1, alpha, beta, True))
                        board[x, y, z] = 0
                        beta = min(beta, value)
                        if beta <= alpha:
                            break
                if beta <= alpha:
                    break
        return value

def check_winner(board: np.ndarray) -> int:
    """
    Checks if there's a winner on the 3D Tic Tac Toe board.

    Args:
        board: Current game board

    Returns:
        1 if player wins, -1 if opponent wins, 0 otherwise
    """
    # Check rows in each layer
    for layer in range(3):
        for row in range(3):
            if board[layer, row, :].sum() == 3:  # Player wins
                return 1
            if board[layer, row, :].sum() == -3:  # Opponent wins
                return -1

    # Check columns in each layer
    for layer in range(3):
        for col in range(3):
            if board[layer, :, col].sum() == 3:  # Player wins
                return 1
            if board[layer, :, col].sum() == -3:  # Opponent wins
                return -1

    # Check diagonals in each layer
    for layer in range(3):
        if board[layer, 0, 0] + board[layer, 1, 1] + board[layer, 2, 2] == 3:  # Player wins
            return 1
        if board[layer, 0, 0] + board[layer, 1, 1] + board[layer, 2, 2] == -3:  # Opponent wins
            return -1
        if board[layer, 0, 2] + board[layer, 1, 1] + board[layer, 2, 0] == 3:  # Player wins
            return 1
        if board[layer, 0, 2] + board[layer, 1, 1] + board[layer, 2, 0] == -3:  # Opponent wins
            return -1

    # Check vertical lines (through layers)
    for x in range(3):
        for y in range(3):
            if board[x, y, 0] + board[x, y, 1] + board[x, y, 2] == 3:  # Player wins
                return 1
            if board[x, y, 0] + board[x, y, 1] + board[x, y, 2] == -3:  # Opponent wins
                return -1

    # Check 3D diagonals
    if (board[0, 0, 0] + board[1, 1, 1] + board[2, 2, 2] == 3 or
        board[0, 2, 2] + board[1, 1, 1] + board[2, 0, 0] == 3):
        return 1
    if (board[0, 0, 0] + board[1, 1, 1] + board[2, 2, 2] == -3 or
        board[0, 2, 2] + board[1, 1, 1] + board[2, 0, 0] == -3):
        return -1

    # Check all possible 2D planes (12 planes in total)
    # Front, middle, back layers
    for layer in range(3):
        # Check rows
        for row in range(3):
            if board[:, row, layer].sum() == 3:  # Player wins
                return 1
            if board[:, row, layer].sum() == -3:  # Opponent wins
                return -1
        # Check columns
        for col in range(3):
            if board[:, col, layer].sum() == 3:  # Player wins
                return 1
            if board[:, col, layer].sum() == -3:  # Opponent wins
                return -1
        # Check diagonals
        if board[0, 0, layer] + board[1, 1, layer] + board[2, 2, layer] == 3:  # Player wins
            return 1
        if board[0, 0, layer] + board[1, 1, layer] + board[2, 2, layer] == -3:  # Opponent wins
            return -1
        if board[0, 2, layer] + board[1, 1, layer] + board[2, 0, layer] == 3:  # Player wins
            return 1
        if board[0, 2, layer] + board[1, 1, layer] + board[2, 0, layer] == -3:  # Opponent wins
            return -1

    return 0

This implementation:

1. Uses minimax with alpha-beta pruning to evaluate possible moves
2. Checks all possible winning conditions in 3D Tic Tac Toe (rows, columns, diagonals in all planes)
3. Prioritizes winning moves, then blocking opponent's winning moves
4. Has a special case for the first move (center position)
5. Uses numpy for efficient board manipulation

The algorithm will:
- First look for immediate winning moves
- Then block opponent's immediate winning moves
- Then look for moves that create the most favorable position
- Finally fall back to a reasonable heuristic if no clear winning/blocking moves exist

The depth of the search is limited by the game's complexity (27 cells), but the alpha-beta pruning helps make it more efficient than pure minimax.
