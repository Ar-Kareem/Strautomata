def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    # Precompute all winning lines in 3x3x3 Tic Tac Toe
    all_lines = []
    
    # Layer lines (z, x, y)
    for z in range(3):
        for x in range(3):
            for y in range(3):
                all_lines.append([(z, x, y)])
    
    # Rows (z, x, y)
    for z in range(3):
        for x in range(3):
            for y in range(3):
                all_lines.append([(z, x, y)])
    
    # Columns (z, x, y)
    for z in range(3):
        for x in range(3):
            for y in range(3):
                all_lines.append([(z, x, y)])
    
    # Diagonals in each layer (z, x, y)
    for z in range(3):
        all_lines.append([(z, 0, 0), (z, 1, 1), (z, 2, 2)])
        all_lines.append([(z, 0, 2), (z, 1, 1), (z, 2, 0)])
    
    # Pillars (x, y, z)
    for x in range(3):
        for y in range(3):
            all_lines.append([(x, y, 0), (x, y, 1), (x, y, 2)])
    
    # Space diagonals
    all_lines.append([(0, 0, 0), (1, 1, 1), (2, 2, 2)])
    all_lines.append([(0, 0, 2), (1, 1, 1), (2, 2, 0)])
    all_lines.append([(0, 2, 0), (1, 1, 1), (2, 0, 2)])
    all_lines.append([(0, 2, 2), (1, 1, 1), (2, 0, 0)])
    
    # Map each cell to the lines that contain it
    cell_to_lines = {}
    for line in all_lines:
        for cell in line:
            if cell not in cell_to_lines:
                cell_to_lines[cell] = []
            cell_to_lines[cell].append(line)
    
    # Check each empty cell in order
    for z in range(3):
        for x in range(3):
            for y in range(3):
                if board[z][x][y] == 0:
                    cell = (z, x, y)
                    
                    # Check for winning move (2 opponent's marks + 1 empty)
                    for line in cell_to_lines[cell]:
                        count_opponent = 0
                        for c in line:
                            if c == cell:
                                continue
                            if board[c[0]][c[1]][c[2]] == -1:
                                count_opponent += 1
                        if count_opponent == 2:
                            return (z, x, y)
                    
                    # Check for blocking move (2 opponent's marks + 1 empty)
                    for line in cell_to_lines[cell]:
                        count_opponent = 0
                        for c in line:
                            if c == cell:
                                continue
                            if board[c[0]][c[1]][c[2]] == -1:
                                count_opponent += 1
                        if count_opponent == 2:
                            return (z, x, y)
                    
                    # Check for creating a winning opportunity (1 opponent's mark + 1 empty)
                    for line in cell_to_lines[cell]:
                        count_opponent = 0
                        for c in line:
                            if c == cell:
                                continue
                            if board[c[0]][c[1]][c[2]] == -1:
                                count_opponent += 1
                        if count_opponent == 1:
                            return (z, x, y)
    
    # If no immediate threats, return first empty cell
    for z in range(3):
        for x in range(3):
            for y in range(3):
                if board[z][x][y] == 0:
                    return (z, x, y)
    
    # Fallback: return None if no empty cells (shouldn't happen in valid game)
    return (0, 0, 0)
