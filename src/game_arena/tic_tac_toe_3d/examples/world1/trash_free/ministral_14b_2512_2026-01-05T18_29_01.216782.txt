def minimax(board: np.ndarray, alpha: float, beta: float, is_maximizing: bool) -> int:
    """
    Minimax algorithm with alpha-beta pruning for 3D Tic Tac Toe.

    Args:
        board: Current game board.
        alpha: Alpha value for pruning.
        beta: Beta value for pruning.
        is_maximizing: Whether the current player is maximizing (True) or minimizing (False).

    Returns:
        The evaluated score of the current board position.
    """
    # Check if the game is over
    if is_game_over(board):
        if is_winner(board, 1):
            return 10  # Win score
        elif is_winner(board, -1):
            return -10  # Loss score
        else:
            return 0  # Draw score

    if is_maximizing:
        best_score = -float('inf')
        for move in np.argwhere(board == 0):
            new_board = board.copy()
            new_board[move[0], move[1], move[2]] = 1
            score = minimax(new_board, alpha, beta, False)
            best_score = max(score, best_score)
            alpha = max(alpha, best_score)
            if beta <= alpha:
                break  # Alpha-beta pruning
        return best_score
    else:
        best_score = float('inf')
        for move in np.argwhere(board == 0):
            new_board = board.copy()
            new_board[move[0], move[1], move[2]] = -1
            score = minimax(new_board, alpha, beta, True)
            best_score = min(score, best_score)
            beta = min(beta, best_score)
            if beta <= alpha:
                break  # Alpha-beta pruning
        return best_score

def is_game_over(board: np.ndarray) -> bool:
    """
    Checks if the game is over (either someone won or it's a draw).
    """
    return is_winner(board, 1) or is_winner(board, -1) or np.all(board != 0)

def is_winner(board: np.ndarray, player: int) -> bool:
    """
    Checks if the specified player has won the game.
    """

    for layer in range(3):
        for row in range(3):
            if all(board[layer, row, :] == player):
                return True
            if all(board[layer, :, row] == player):
                return True

    for layer in range(3):
        for col in range(3):
            if all(board[layer, :, col] == player):
                return True

        if all(board[layer, i, i] == player for i in range(3)):
            return True
        if all(board[layer, i, 2 - i] == player for i in range(3)):
            return True

    for x in range(3):
        for y in range(3):
            if all(board[z, x, y] == player for z in range(3)):
                return True

    if all(board[i, i, i] == player for i in range(3)):
        return True
    if all(board[i, 2 - i, 2 - i] == player for i in range(3)):
        return True
    if all(board[i, i, 2 - i] == player for i in range(3)):
        return True
    if all(board[i, 2 - i, i] == player for i in range(3)):

    # Check other 3D diagonals
    if all(board[i, i, 2 - i] == player for i in range(3)):
        return True
    if all(board[i, 2 - i, i] == player for i in range(3)):
        return True

    return False

### Explanation:

1. **Policy Function**:
   - The `policy` function evaluates all possible moves using the minimax algorithm with alpha-beta pruning.
   - It returns the best move (coordinates) based on the highest evaluation score.

2. **Minimax with Alpha-Beta Pruning**:
   - The `minimax` function recursively evaluates all possible game states, alternating between maximizing (current player) and minimizing (opponent) players.
   - Alpha-beta pruning is used to eliminate branches that cannot influence the final decision, significantly improving efficiency.

3. **Game State Checks**:
   - `is_game_over` checks if the game is won or drawn.
   - `is_winner` checks all possible winning conditions in 3D Tic Tac Toe, including:
     - Rows, columns, and diagonals in each 2D layer.
     - Vertical lines through layers.
     - Space diagonals and other 3D diagonals.

This implementation ensures that the AI makes optimal moves by considering all possible future game states and choosing the one with the highest probability of winning. The use of alpha-beta pruning makes it efficient enough to handle the game's complexity.
