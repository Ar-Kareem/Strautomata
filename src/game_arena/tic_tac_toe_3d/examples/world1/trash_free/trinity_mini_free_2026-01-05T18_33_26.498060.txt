import random

def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    # Generate all possible winning lines in 3D Tic Tac Toe
    lines = []
    # Along x-axis (fixed y, z)
    for y in range(3):
        for z in range(3):
            lines.append([(0, y, z), (1, y, z), (2, y, z)])
    # Along y-axis (fixed x, z)
    for x in range(3):
        for z in range(3):
            lines.append([(x, 0, z), (x, 1, z), (x, 2, z)])
    # Along z-axis (fixed x, y)
    for x in range(3):
        for y in range(3):
            lines.append([(x, y, 0), (x, y, 1), (x, y, 2)])
    # Diagonals in xy-plane (fixed z)
    for z in range(3):
        lines.append([(0, 0, z), (1, 1, z), (2, 2, z)])
        lines.append([(0, 2, z), (1, 1, z), (2, 0, z)])
    # Diagonals in xz-plane (fixed y)
    for y in range(3):
        lines.append([(0, y, 0), (1, y, 1), (2, y, 2)])
        lines.append([(0, y, 2), (1, y, 1), (2, y, 0)])
    # Diagonals in yz-plane (fixed x)
    for x in range(3):
        lines.append([(x, 0, 0), (x, 1, 1), (x, 2, 2)])
        lines.append([(x, 0, 2), (x, 1, 1), (x, 2, 0)])
    # Space diagonals
    lines.append([(0, 0, 0), (1, 1, 1), (2, 2, 2)])
    lines.append([(0, 0, 2), (1, 1, 1), (2, 2, 0)])
    lines.append([(0, 2, 0), (1, 1, 1), (2, 0, 2)])
    lines.append([(0, 2, 2), (1, 1, 1), (2, 0, 0)])

    # Helper function to check if a move wins
    def is_win(move):
        x, y, z = move
        for line in lines:
            if move in line:
                count = 0
                for cell in line:
                    if board[cell[0]][cell[1]][cell[2]] == 1:
                        count += 1
                if count == 3:
                    return True
        return False

    # Helper function to check if a move creates a fork
    def is_fork(move):
        x, y, z = move
        count = 0
        for line in lines:
            if move in line:
                all_ones = True
                for cell in line:
                    if board[cell[0]][cell[1]][cell[2]] != 1:
                        all_ones = False
                        break
                if all_ones:
                    count += 1
        return count >= 2

    # Find all empty cells
    empty_cells = []
    for x in range(3):
        for y in range(3):
            for z in range(3):
                if board[x][y][z] == 0:
                    empty_cells.append((x, y, z))

    # Check for winning moves
    for move in empty_cells:
        board[move[0]][move[1]][move[2]] = 1
        if is_win(move):
            board[move[0]][move[1]][move[2]] = 0
            return move
        board[move[0]][move[1]][move[2]] = 0

    # Check for opponent's winning moves (blocking)
    for move in empty_cells:
        board[move[0]][move[1]][move[2]] = -1
        if is_win(move):
            board[move[0]][move[1]][move[2]] = 0
            return move
        board[move[0]][move[1]][move[2]] = 0

    # Check for forks
    for move in empty_cells:
        board[move[0]][move[1]][move[2]] = 1
        if is_fork(move):
            board[move[0]][move[1]][move[2]] = 0
            return move
        board[move[0]][move[1]][move[2]] = 0

    # Choose a random move if no threats exist
    return random.choice(empty_cells)
