Here's a Python implementation for a basic 3D Tic Tac Toe policy function that uses a simple heuristic to choose the next move. This version prioritizes winning moves, then blocks opponent winning moves, then chooses the center of the board, and finally picks an empty cell randomly if no strategic move is available.

```python
import random
from typing import List, Tuple

def policy(board: List[List[List[int]]]) -> Tuple[int, int, int]:
    """
    Determines the next move for a 3D Tic Tac Toe game (3x3x3) using a simple heuristic.

    The heuristic prioritizes:
    1. Winning moves (if available)
    2. Blocking opponent winning moves (if available)
    3. Center of the board
    4. Random empty cell

    Args:
        board: A 3x3x3 list of lists containing 0 (empty), 1 (player), and -1 (opponent).

    Returns:
        A tuple (x, y, z) representing the chosen empty cell coordinates.
    """

    def check_line(line: List[int]) -> Tuple[int, int, int]:
        """Checks if a line has a winning move or is about to be won by the opponent."""
        if line[0] == line[1] == line[2] == 0:
            return (0, 0, 0)  # All empty - no move yet
        if line[0] == line[1] == line[2] == 1:
            return None  # Player already won this line
        if line[0] == line[1] == line[2] == -1:
            return None  # Opponent already won this line

        # Check for winning move (two player marks and one empty)
        if line[0] == line[1] == 1 and line[2] == 0:
            return (0, 0, 2)
        if line[0] == line[2] == 1 and line[1] == 0:
            return (0, 1, 1)
        if line[1] == line[2] == 1 and line[0] == 0:
            return (0, 2, 0)

        # Check for blocking move (two opponent marks and one empty)
        if line[0] == line[1] == -1 and line[2] == 0:
            return (0, 0, 2)
        if line[0] == line[2] == -1 and line[1] == 0:
            return (0, 1, 1)
        if line[1] == line[2] == -1 and line[0] == 0:
            return (0, 2, 0)

        return None

    def check_all_lines(board: List[List[List[int]]]) -> Tuple[int, int, int]:
        """Checks all possible lines (rows, columns, depth, diagonals) for winning/blocking moves."""
        # Check rows in each layer
        for z in range(3):
            for y in range(3):
                for x in range(3):
                    line = [board[x][y][z]]
                    if line[0] == 0:
                        continue
                    # Check row (fixed y, z, varying x)
                    row = [board[x][y][z] for x in range(3)]
                    move = check_line(row)
                    if move is not None:
                        return move
                    # Check column (fixed x, z, varying y)
                    col = [board[x][y][z] for y in range(3)]
                    move = check_line(col)
                    if move is not None:
                        return move
                    # Check depth (fixed x, y, varying z)
                    depth = [board[x][y][z] for z in range(3)]
                    move = check_line(depth)
                    if move is not None:
                        return move

        # Check space diagonals (all layers)
        for z in range(3):
            # Diagonal 1: (0,0,z) to (2,2,z)
            diag = [board[i][i][z] for i in range(3)]
            move = check_line(diag)
            if move is not None:
                return move
            # Diagonal 2: (0,2,z) to (2,0,z)
            diag = [board[i][2-i][z] for i in range(3)]
            move = check_line(diag)
            if move is not None:
                return move

        # Check layer diagonals (fixed z)
        for z in range(3):
            # Diagonal 1: (0,0,0) to (2,2,2)
            if z == 0:
                diag = [board[i][i][z] for i in range(3)]
                move = check_line(diag)
                if move is not None:
                    return move
            # Diagonal 2: (0,2,0) to (2,0,2)
            if z == 0:
                diag = [board[i][2-i][z] for i in range(3)]
                move = check_line(diag)
                if move is not None:
                    return move
            # Diagonal 3: (0,0,2) to (2,2,0)
            if z == 2:
                diag = [board[i][i][2-i] for i in range(3)]
                move = check_line(diag)
                if move is not None:
                    return move
            # Diagonal 4: (0,2,2) to (2,0,0)
            if z == 2:
                diag = [board[i][2-i][2-i] for i in range(3)]
                move = check_line(diag)
                if move is not None:
                    return move

        # Check all 3D space diagonals
        # Diagonal 1: (0,0,0) to (2,2,2)
        diag = [board[i][i][i] for i in range(3)]
        move = check_line(diag)
        if move is not None:
            return move
        # Diagonal 2: (0,2,2) to (2,0,0)
        diag = [board[i][2-i][2-i] for i in range(3)]
        move = check_line(diag)
        if move is not None:
            return move

        return None

    # Check for winning move first
    winning_move = check_all_lines(board)
    if winning_move is not None:
        return winning_move

    # Check for blocking move next
    # Create a copy of the board to simulate opponent move
    opponent_board = [row[:] for row in board]
    for x in range(3):
        for y in range(3):
            for z in range(3):
                if opponent_board[x][y][z] == 0:
                    opponent_board[x][y][z] = -1
                    blocking_move = check_all_lines(opponent_board)
                    if blocking_move is not None:
                        return blocking_move
                    opponent_board[x][y][z] = 0  # Revert

    # If no winning or blocking move, choose the center (1,1,1)
    if board[1][1][1] == 0:
        return (1, 1, 1)

    # Otherwise, pick a random empty cell
    empty_cells = [(x, y, z) for x in range(3) for y in range(3) for z in range(3) if board[x][y][z] == 0]
    return random.choice(empty_cells)
```

### Explanation:
1. **`check_line`**: Helper function to check if a 1D line (row, column, depth, or diagonal) has a winning move or can be blocked by the opponent.
2. **`check_all_lines`**: Checks all possible lines in the 3D board for winning/blocking moves. This includes:
   - Rows, columns, and depths in each layer.
   - Space diagonals (e.g., `(0,0,0)` to `(2,2,2)`).
   - Layer diagonals (e.g., `(0,0,0)` to `(2,2,0)`).
3. **Winning/Blocking Logic**:
   - First, the function checks if the current player (`1`) can win in the next move by occupying any line.
   - If no winning move, it checks if the opponent (`-1`) can win in their next move and blocks it.
4. **Center Preference**: If no strategic move is found, the center cell `(1,1,1)` is chosen as it's the most symmetric.
5. **Random Fallback**: If the center is taken, a random empty cell is selected.

### Note:
This is a **basic heuristic** and not optimal for all cases. For a more advanced AI, you could implement:
- Minimax with alpha-beta pruning.
- Evaluation functions based on board symmetry or potential future moves.
- Monte Carlo Tree Search for probabilistic move selection.
