def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    # Generate all possible winning lines
    lines = []
    
    # Rows (fixed x, y; vary z)
    for x in range(3):
        for y in range(3):
            lines.append([(x, y, z) for z in range(3)])
    
    # Columns (fixed x, z; vary y)
    for x in range(3):
        for z in range(3):
            lines.append([(x, y, z) for y in range(3)])
    
    # Pillars (fixed y, z; vary x)
    for y in range(3):
        for z in range(3):
            lines.append([(x, y, z) for x in range(3)])
    
    # Face diagonals in x-y planes (fixed z)
    for z in range(3):
        lines.append([(x, x, z) for x in range(3)])
        lines.append([(x, 2 - x, z) for x in range(3)])
    
    # Face diagonals in x-z planes (fixed y)
    for y in range(3):
        lines.append([(x, y, x) for x in range(3)])
        lines.append([(x, y, 2 - x) for x in range(3)])
    
    # Face diagonals in y-z planes (fixed x)
    for x in range(3):
        lines.append([(x, y, y) for y in range(3)])
        lines.append([(x, y, 2 - y) for y in range(3)])
    
    # Space diagonals
    lines.append([(0, 0, 0), (1, 1, 1), (2, 2, 2)])
    lines.append([(0, 0, 2), (1, 1, 1), (2, 2, 0)])
    lines.append([(0, 2, 0), (1, 1, 1), (2, 0, 2)])
    lines.append([(0, 2, 2), (1, 1, 1), (2, 0, 0)])
    
    # Check for immediate winning moves
    for line in lines:
        values = [board[x][y][z] for (x, y, z) in line]
        if sum(values) == 2 and 0 in values:
            for x, y, z in line:
                if board[x][y][z] == 0:
                    return (x, y, z)
    
    # Check for opponent's immediate threats
    blocking_moves = []
    for line in lines:
        values = [board[x][y][z] for (x, y, z) in line]
        if sum(values) == -2 and 0 in values:
            for x, y, z in line:
                if board[x][y][z] == 0:
                    blocking_moves.append((x, y, z))
                    break  # Only one empty in the line
    
    if blocking_moves:
        return blocking_moves[0]
    
    # Prioritize center cell
    if board[1][1][1] == 0:
        return (1, 1, 1)
    
    # Try corners next
    corners = [
        (0, 0, 0), (0, 0, 2), (0, 2, 0), (0, 2, 2),
        (2, 0, 0), (2, 0, 2), (2, 2, 0), (2, 2, 2)
    ]
    for x, y, z in corners:
        if board[x][y][z] == 0:
            return (x, y, z)
    
    # Choose any remaining empty cell
    for x in range(3):
        for y in range(3):
            for z in range(3):
                if board[x][y][z] == 0:
                    return (x, y, z)
    
    # No moves left (unreachable in valid games)
    raise ValueError("No empty cell available")
