def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    # Generate all possible winning lines
    lines = []
    
    # Rows in each layer
    for depth in range(3):
        for row in range(3):
            lines.append([(depth, row, col) for col in range(3)])
    
    # Columns in each layer
    for depth in range(3):
        for col in range(3):
            lines.append([(depth, row, col) for row in range(3)])
    
    # Depth pillars
    for row in range(3):
        for col in range(3):
            lines.append([(depth, row, col) for depth in range(3)])
    
    # 2D diagonals in each layer
    for depth in range(3):
        lines.append([(depth, 0, 0), (depth, 1, 1), (depth, 2, 2)])
        lines.append([(depth, 0, 2), (depth, 1, 1), (depth, 2, 0)])
    
    # 3D space diagonals
    lines.append([(0, 0, 0), (1, 1, 1), (2, 2, 2)])
    lines.append([(0, 0, 2), (1, 1, 1), (2, 2, 0)])
    lines.append([(0, 2, 0), (1, 1, 1), (2, 0, 2)])
    lines.append([(0, 2, 2), (1, 1, 1), (2, 0, 0)])
    
    # Check for winning move
    for line in lines:
        values = [board[d][r][c] for (d, r, c) in line]
        if sum(values) == 2 and 0 in values:  # Two 1s and one 0
            for (d, r, c) in line:
                if board[d][r][c] == 0:
                    return (d, r, c)
    
    # Check for blocking move
    for line in lines:
        values = [board[d][r][c] for (d, r, c) in line]
        if sum(values) == -2 and 0 in values:  # Two -1s and one 0
            for (d, r, c) in line:
                if board[d][r][c] == 0:
                    return (d, r, c)
    
    # Check center
    if board[1][1][1] == 0:
        return (1, 1, 1)
    
    # Check corners
    corners = [
        (0, 0, 0), (0, 0, 2), (0, 2, 0), (0, 2, 2),
        (2, 0, 0), (2, 0, 2), (2, 2, 0), (2, 2, 2)
    ]
    for d, r, c in corners:
        if board[d][r][c] == 0:
            return (d, r, c)
    
    # Finally, choose the first available cell
    for d in range(3):
        for r in range(3):
            for c in range(3):
                if board[d][r][c] == 0:
                    return (d, r, c)
    
    # Fallback (shouldn't be reached in a valid game state)
    return (0, 0, 0)
