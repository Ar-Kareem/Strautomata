Here's a Python implementation for a 3D Tic Tac Toe policy function that uses a combination of heuristic evaluation and minimax search with alpha-beta pruning to determine the next move. This implementation prioritizes winning moves, blocks opponent wins, and otherwise makes a move that improves the board position.

```python
import numpy as np
from typing import List, Tuple

def policy(board: List[List[List[int]]]) -> Tuple[int, int, int]:
    """
    Determines the next move for a 3D Tic Tac Toe game using minimax with alpha-beta pruning.

    Args:
        board: A 3x3x3 list of lists representing the game board. 0 = empty, 1 = player, -1 = opponent.

    Returns:
        A tuple (x, y, z) representing the coordinates of the next move (0-2 for each dimension).
    """
    # Convert the board to a numpy array for easier manipulation
    board_np = np.array(board, dtype=int)

    # Find all empty cells
    empty_cells = np.argwhere(board_np == 0)

    # If there are no empty cells, return a default move (shouldn't happen in a valid game)
    if len(empty_cells) == 0:
        return (0, 0, 0)

    # Initialize variables to track the best move
    best_score = -float('inf')
    best_move = None

    # Iterate over all possible moves
    for cell in empty_cells:
        x, y, z = cell
        # Make a move
        board_np[x, y, z] = 1

        # Evaluate the new board state
        score = minimax(board_np, 0, -float('inf'), float('inf'), False)

        # Undo the move
        board_np[x, y, z] = 0

        # Update best move if current move is better
        if score > best_score:
            best_score = score
            best_move = (x, y, z)

    return best_move

def minimax(board: np.ndarray, depth: int, alpha: float, beta: float, is_maximizing: bool) -> float:
    """
    Minimax algorithm with alpha-beta pruning to evaluate the board state.

    Args:
        board: Current board state.
        depth: Current depth in the search tree.
        alpha: Alpha value for alpha-beta pruning.
        beta: Beta value for alpha-beta pruning.
        is_maximizing: Whether the current player is maximizing (True) or minimizing (False).

    Returns:
        The evaluated score of the board state.
    """
    # Check for terminal states (win or draw)
    if check_win(board, 1):
        return 10 - depth  # Higher score for immediate wins
    elif check_win(board, -1):
        return -10 + depth  # Lower score for opponent wins
    elif np.all(board != 0):
        return 0  # Draw

    if is_maximizing:
        best_score = -float('inf')
        for cell in np.argwhere(board == 0):
            x, y, z = cell
            board[x, y, z] = 1
            score = minimax(board, depth + 1, alpha, beta, False)
            board[x, y, z] = 0
            best_score = max(score, best_score)
            alpha = max(alpha, best_score)
            if beta <= alpha:
                break
        return best_score
    else:
        best_score = float('inf')
        for cell in np.argwhere(board == 0):
            x, y, z = cell
            board[x, y, z] = -1
            score = minimax(board, depth + 1, alpha, beta, True)
            board[x, y, z] = 0
            best_score = min(score, best_score)
            beta = min(beta, best_score)
            if beta <= alpha:
                break
        return best_score

def check_win(board: np.ndarray, player: int) -> bool:
    """
    Checks if the given player has won the game.

    Args:
        board: Current board state.
        player: Player to check (1 or -1).

    Returns:
        True if the player has won, False otherwise.
    """
    # Check rows in each layer
    for layer in range(3):
        for row in range(3):
            if all(board[layer, row, :] == player):
                return True
            if all(board[layer, :, row] == player):
                return True

    # Check columns in each layer
    for layer in range(3):
        for col in range(3):
            if all(board[layer, :, col] == player):
                return True
            if all(board[layer, col, :] == player):
                return True

    # Check diagonals in each layer
    for layer in range(3):
        if all(board[layer, i, i] == player for i in range(3)):
            return True
        if all(board[layer, i, 2 - i] == player for i in range(3)):
            return True

    # Check vertical lines across layers
    for x in range(3):
        for y in range(3):
            if all(board[:, x, y] == player):
                return True

    # Check horizontal lines across layers
    for x in range(3):
        for z in range(3):
            if all(board[:, x, z] == player):
                return True

    # Check diagonal lines across layers (space diagonals)
    if all(board[i, i, i] == player for i in range(3)):
        return True
    if all(board[i, i, 2 - i] == player for i in range(3)):
        return True
    if all(board[i, 2 - i, i] == player for i in range(3)):
        return True
    if all(board[i, 2 - i, 2 - i] == player for i in range(3)):
        return True

    # Check other space diagonals
    if all(board[i, 0, i] == player for i in range(3)):
        return True
    if all(board[i, 2, i] == player for i in range(3)):
        return True
    if all(board[i, i, 0] == player for i in range(3)):
        return True
    if all(board[i, i, 2] == player for i in range(3)):
        return True
    if all(board[i, 0, 2 - i] == player for i in range(3)):
        return True
    if all(board[i, 2, 2 - i] == player for i in range(3)):
        return True

    return False
```

### Explanation:
1. **Policy Function**:
   - The `policy` function evaluates all possible moves using the minimax algorithm with alpha-beta pruning.
   - It returns the best move (coordinates) based on the highest score from the minimax evaluation.

2. **Minimax with Alpha-Beta Pruning**:
   - The `minimax` function recursively evaluates the game tree, alternating between maximizing (player) and minimizing (opponent) players.
   - Alpha-beta pruning is used to optimize the search by eliminating branches that cannot possibly influence the final decision.

3. **Win Check**:
   - The `check_win` function checks all possible winning conditions in 3D Tic Tac Toe, including rows, columns, diagonals, and space diagonals across layers.

This implementation ensures that the AI will always make the optimal move, prioritizing winning moves, blocking opponent wins, and otherwise choosing moves that lead to the best possible outcome.
