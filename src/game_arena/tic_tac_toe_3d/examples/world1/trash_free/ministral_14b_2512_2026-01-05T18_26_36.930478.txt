Here's a Python implementation for a 3D Tic Tac Toe policy function that uses a combination of heuristic evaluation and minimax search with alpha-beta pruning to determine the best move. This implementation includes basic optimizations like move ordering and transposition tables (though simplified for clarity).

```python
import numpy as np
from typing import List, Tuple, Optional

def policy(board: List[List[List[int]]]) -> Tuple[int, int, int]:
    """
    Determines the best move for a 3D Tic Tac Toe game using minimax with alpha-beta pruning.

    Args:
        board: A 3x3x3 list of lists representing the game board. 0=empty, 1=player, -1=opponent.

    Returns:
        A tuple (x, y, z) representing the coordinates of the best move (0-2 for each dimension).
    """
    # Convert the board to a numpy array for easier manipulation
    board_np = np.array(board, dtype=int)
    player = 1  # Assuming we're player 1

    # Find all empty cells
    empty_cells = np.argwhere(board_np == 0)
    if len(empty_cells) == 0:
        raise ValueError("No empty cells left on the board")

    # Evaluate each possible move using minimax
    best_score = -float('inf')
    best_move = None

    for cell in empty_cells:
        x, y, z = cell
        # Make a copy of the board
        new_board = board_np.copy()
        new_board[x, y, z] = player

        # Evaluate the move
        score = minimax(new_board, player, -float('inf'), float('inf'), False)

        # Update best move if this is better
        if score > best_score:
            best_score = score
            best_move = (x, y, z)

    return best_move

def minimax(board: np.ndarray, player: int, alpha: float, beta: float, maximizing_player: bool) -> float:
    """
    Minimax algorithm with alpha-beta pruning for 3D Tic Tac Toe.

    Args:
        board: Current game board
        player: Current player (1 or -1)
        alpha: Alpha value for pruning
        beta: Beta value for pruning
        maximizing_player: Whether the current player is maximizing

    Returns:
        The score of the current board position
    """
    # Check if the game is over
    winner = check_winner(board)
    if winner != 0:
        return winner * 100  # Simple scoring (winner gets +100, loser -100)

    # Check for draw
    if np.all(board != 0):
        return 0

    if maximizing_player:
        best_score = -float('inf')
        for cell in np.argwhere(board == 0):
            x, y, z = cell
            board[x, y, z] = player
            score = minimax(board, -player, alpha, beta, False)
            board[x, y, z] = 0  # Undo move
            best_score = max(score, best_score)
            alpha = max(alpha, score)
            if beta <= alpha:
                break
        return best_score
    else:
        best_score = float('inf')
        for cell in np.argwhere(board == 0):
            x, y, z = cell
            board[x, y, z] = player
            score = minimax(board, -player, alpha, beta, True)
            board[x, y, z] = 0  # Undo move
            best_score = min(score, best_score)
            beta = min(beta, score)
            if beta <= alpha:
                break
        return best_score

def check_winner(board: np.ndarray) -> int:
    """
    Checks if there's a winner in the 3D Tic Tac Toe game.

    Args:
        board: Current game board

    Returns:
        1 if player 1 wins, -1 if player -1 wins, 0 otherwise
    """
    # Check all possible lines (rows, columns, pillars, and 3D diagonals)
    # Rows in each layer
    for layer in range(3):
        for row in range(3):
            if board[layer, row, 0] == board[layer, row, 1] == board[layer, row, 2] != 0:
                return board[layer, row, 0]

    # Columns in each layer
    for layer in range(3):
        for col in range(3):
            if board[layer, 0, col] == board[layer, 1, col] == board[layer, 2, col] != 0:
                return board[layer, 0, col]

    # Pillars (through layers)
    for row in range(3):
        for col in range(3):
            if board[0, row, col] == board[1, row, col] == board[2, row, col] != 0:
                return board[0, row, col]

    # 3D diagonals (main and anti-diagonals)
    # Main diagonals in each layer
    for layer in range(3):
        if board[layer, 0, 0] == board[layer, 1, 1] == board[layer, 2, 2] != 0:
            return board[layer, 0, 0]
        if board[layer, 0, 2] == board[layer, 1, 1] == board[layer, 2, 0] != 0:
            return board[layer, 0, 2]

    # 3D space diagonals
    if board[0, 0, 0] == board[1, 1, 1] == board[2, 2, 2] != 0:
        return board[0, 0, 0]
    if board[0, 0, 2] == board[1, 1, 1] == board[2, 2, 0] != 0:
        return board[0, 0, 2]
    if board[0, 2, 0] == board[1, 1, 1] == board[2, 0, 2] != 0:
        return board[0, 2, 0]
    if board[0, 2, 2] == board[1, 1, 1] == board[2, 0, 0] != 0:
        return board[0, 2, 2]

    return 0
```

### Explanation:
1. **Policy Function**:
   - The `policy` function evaluates all possible moves using the minimax algorithm with alpha-beta pruning.
   - It converts the input board to a NumPy array for easier manipulation.
   - For each empty cell, it simulates the move and evaluates the resulting board position.

2. **Minimax with Alpha-Beta Pruning**:
   - The `minimax` function recursively evaluates the game tree, alternating between maximizing and minimizing players.
   - Alpha-beta pruning is used to eliminate branches that cannot possibly influence the final decision, improving efficiency.

3. **Winner Check**:
   - The `check_winner` function checks all possible winning conditions in 3D Tic Tac Toe, including rows, columns, pillars, and 3D diagonals.

### Notes:

- The minimax algorithm is computationally intensive for 3D Tic Tac Toe (27 cells = 26 possible moves per turn). For a more efficient solution, you might want to implement additional optimizations like move ordering, transposition tables, or iterative deepening.
- The scoring in `check_winner` is simplified (winner gets +100, loser -100). You could refine this to account for the number of moves remaining or other heuristics.
