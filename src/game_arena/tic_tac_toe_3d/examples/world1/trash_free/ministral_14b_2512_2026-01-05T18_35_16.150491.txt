import numpy as np
from typing import List, Tuple, Optional

def policy(board: List[List[List[int]]]) -> Tuple[int, int, int]:
    """
    Determines the next move for a 3x3x3 3D Tic Tac Toe game using minimax with alpha-beta pruning.

    Args:
        board: A 3x3x3 list of lists where 0 is empty, 1 is player, -1 is opponent.

    Returns:
        A tuple (x, y, z) representing the coordinates of the chosen empty cell.
    """
    # Convert the board to a numpy array for easier manipulation
    board_np = np.array(board, dtype=int)

    # Find all empty cells
    empty_cells = np.argwhere(board_np == 0)
    if len(empty_cells) == 0:
        raise ValueError("No empty cells left on the board")

    # If it's the first move, choose the center (1,1,1)
    if len(empty_cells) == 26:  # 27 cells total, so 26 empty means first move
        return (1, 1, 1)

    # Use minimax with alpha-beta pruning to find the best move
    best_score = -float('inf')
    best_move = None

    # Sort empty cells by heuristic (e.g., center first, then corners, then edges)
    def heuristic_sort(cell):
        x, y, z = cell
        # Prioritize center (1,1,1), then corners, then edges
        if (x, y, z) == (1, 1, 1):
            return 0
        if (x + y + z) % 2 == 0:  # corners
            return 1
        return 2

    # Sort empty cells by heuristic
    empty_cells = sorted(empty_cells, key=heuristic_sort)

    for cell in empty_cells:
        x, y, z = cell
        # Make a copy of the board
        new_board = board_np.copy()
        new_board[x, y, z] = 1  # Player's move

        # Evaluate the move using minimax
        score = minimax(new_board, 0, -float('inf'), float('inf'), False)

        # Update best move if this move is better
        if score > best_score:
            best_score = score
            best_move = (x, y, z)

    return best_move

def minimax(board: np.ndarray, depth: int, alpha: float, beta: float, is_maximizing: bool) -> float:
    """
    Minimax algorithm with alpha-beta pruning for 3D Tic Tac Toe.

    Args:
        board: Current game board.
        depth: Current depth in the search tree.
        alpha: Alpha value for alpha-beta pruning.
        beta: Beta value for alpha-beta pruning.
        is_maximizing: Whether the current player is maximizing (player) or minimizing (opponent).

    Returns:
        The evaluated score of the current board state.
    """
    # Check if the game is over
    result = check_winner(board)
    if result != 0:
        return evaluate_result(result, depth)

    # If no moves left, it's a draw
    if np.all(board != 0):
        return 0

    if is_maximizing:
        best_score = -float('inf')
        for x in range(3):
            for y in range(3):
                for z in range(3):
                    if board[x, y, z] == 0:
                        new_board = board.copy()
                        new_board[x, y, z] = 1
                        score = minimax(new_board, depth + 1, alpha, beta, False)
                        best_score = max(score, best_score)
                        alpha = max(alpha, best_score)
                        if beta <= alpha:
                            break
                if beta <= alpha:
                    break
        return best_score
    else:
        best_score = float('inf')
        for x in range(3):
            for y in range(3):
                for z in range(3):
                    if board[x, y, z] == 0:
                        new_board = board.copy()
                        new_board[x, y, z] = -1
                        score = minimax(new_board, depth + 1, alpha, beta, True)
                        best_score = min(score, best_score)
                        beta = min(beta, best_score)
                        if beta <= alpha:
                            break
                if beta <= alpha:
                    break
        return best_score

def check_winner(board: np.ndarray) -> int:
    """
    Checks if there is a winner on the 3D Tic Tac Toe board.

    Args:
        board: Current game board.

    Returns:
        1 if player wins, -1 if opponent wins, 0 otherwise.
    """
    # Check rows in each layer
    for layer in range(3):
        for row in range(3):
            if board[layer, row, :].sum() == 3:  # Player wins
                return 1
            if board[layer, row, :].sum() == -3:  # Opponent wins
                return -1

    # Check columns in each layer
    for layer in range(3):
        for col in range(3):
            if board[layer, :, col].sum() == 3:  # Player wins
                return 1
            if board[layer, :, col].sum() == -3:  # Opponent wins
                return -1

    # Check diagonals in each layer
    for layer in range(3):
        if board[layer, 0, 0] + board[layer, 1, 1] + board[layer, 2, 2] == 3:  # Player wins
            return 1
        if board[layer, 0, 0] + board[layer, 1, 1] + board[layer, 2, 2] == -3:  # Opponent wins
            return -1
        if board[layer, 0, 2] + board[layer, 1, 1] + board[layer, 2, 0] == 3:  # Player wins
            return 1
        if board[layer, 0, 2] + board[layer, 1, 1] + board[layer, 2, 0] == -3:  # Opponent wins
            return -1

    # Check vertical lines (through layers)
    for x in range(3):
        for y in range(3):
            if board[x, y, 0] + board[x, y, 1] + board[x, y, 2] == 3:  # Player wins
                return 1
            if board[x, y, 0] + board[x, y, 1] + board[x, y, 2] == -3:  # Opponent wins
                return -1

    # Check 3D diagonals
    if (board[0, 0, 0] + board[1, 1, 1] + board[2, 2, 2] == 3 or
        board[0, 0, 2] + board[1, 1, 1] + board[2, 2, 0] == 3 or
        board[0, 2, 0] + board[1, 1, 1] + board[2, 0, 2] == 3 or
        board[0, 2, 2] + board[1, 1, 1] + board[2, 0, 0] == 3):
        return 1
    if (board[0, 0, 0] + board[1, 1, 1] + board[2, 2, 2] == -3 or
        board[0, 0, 2] + board[1, 1, 1] + board[2, 2, 0] == -3 or
        board[0, 2, 0] + board[1, 1, 1] + board[2, 0, 2] == -3 or
        board[0, 2, 2] + board[1, 1, 1] + board[2, 0, 0] == -3):
        return -1

    return 0

def evaluate_result(result: int, depth: int) -> float:
    """
    Evaluates the result of a game state.

    Args:
        result: 1 for player win, -1 for opponent win, 0 for draw.
        depth: Depth in the search tree (used for depth-based evaluation).

    Returns:
        Evaluation score.
    """
    if result == 1:
        return 10 - depth  # Player wins, higher depth means less certain
    elif result == -1:
        return -10 + depth  # Opponent wins, higher depth means less certain
    else:
        return 0  # Draw

### Explanation:

   - Handles the first move by choosing the center (1,1,1).
   - Uses minimax with alpha-beta pruning to evaluate possible moves.
   - Sorts empty cells by heuristic (center first, then corners, then edges) to improve move ordering.

2. **minimax function**:
   - Recursively evaluates game states by simulating moves.
   - Uses alpha-beta pruning to eliminate branches that cannot influence the final decision.
   - Alternates between maximizing (player) and minimizing (opponent) players.

3. **check_winner function**:
   - Checks all possible winning conditions in 3D Tic Tac Toe (rows, columns, diagonals in each layer, vertical lines, and 3D diagonals).

4. **evaluate_result function**:
   - Assigns a score based on the game result (win, loss, or draw) and the depth in the search tree.
