def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    # Generate all possible winning lines
    lines = []
    
    # x-axis lines (fixed y and z)
    for z in range(3):
        for y in range(3):
            line = [(x, y, z) for x in range(3)]
            lines.append(line)
    
    # y-axis lines (fixed x and z)
    for z in range(3):
        for x in range(3):
            line = [(x, y, z) for y in range(3)]
            lines.append(line)
    
    # z-axis lines (fixed x and y)
    for x in range(3):
        for y in range(3):
            line = [(x, y, z) for z in range(3)]
            lines.append(line)
    
    # xy-diagonals (fixed z)
    for z in range(3):
        lines.append([(d, d, z) for d in range(3)])
        lines.append([(d, 2 - d, z) for d in range(3)])
    
    # xz-diagonals (fixed y)
    for y in range(3):
        lines.append([(d, y, d) for d in range(3)])
        lines.append([(d, y, 2 - d) for d in range(3)])
    
    # yz-diagonals (fixed x)
    for x in range(3):
        lines.append([(x, d, d) for d in range(3)])
        lines.append([(x, d, 2 - d) for d in range(3)])
    
    # Space diagonals
    lines.append([(0, 0, 0), (1, 1, 1), (2, 2, 2)])
    lines.append([(0, 0, 2), (1, 1, 1), (2, 2, 0)])
    lines.append([(0, 2, 0), (1, 1, 1), (2, 0, 2)])
    lines.append([(0, 2, 2), (1, 1, 1), (2, 0, 0)])
    
    # Check for winning move (two 1's and a 0)
    for line in lines:
        total = 0
        empty = None
        for (x, y, z) in line:
            val = board[z][y][x]
            total += val
            if val == 0:
                empty = (x, y, z)
        if total == 2 and empty is not None:
            return empty
    
    # Check for blocking move (two -1's and a 0)
    blocking_moves = []
    for line in lines:
        total = 0
        empty = None
        for (x, y, z) in line:
            val = board[z][y][x]
            total += val
            if val == 0:
                empty = (x, y, z)
        if total == -2 and empty is not None:
            blocking_moves.append(empty)
    if blocking_moves:
        return blocking_moves[0]
    
    # Choose center if available
    if board[1][1][1] == 0:
        return (1, 1, 1)
    
    # Check corners
    corners = [
        (0, 0, 0), (0, 0, 2),
        (0, 2, 0), (0, 2, 2),
        (2, 0, 0), (2, 0, 2),
        (2, 2, 0), (2, 2, 2)
    ]
    for corner in corners:
        x, y, z = corner
        if board[z][y][x] == 0:
            return corner
    
    # Choose any empty cell
    for z in range(3):
        for y in range(3):
            for x in range(3):
                if board[z][y][x] == 0:
                    return (x, y, z)
    
    # If no empty cell (shouldn't happen as per game rules)
    raise ValueError("No empty cell found on the board")
