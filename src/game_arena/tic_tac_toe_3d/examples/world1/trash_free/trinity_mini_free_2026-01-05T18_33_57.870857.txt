def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    # Precompute all possible lines in the 3x3x3 board
    all_lines = []
    # X-axis lines (fixed y, z; vary x)
    for y in range(3):
        for z in range(3):
            line = [(x, y, z) for x in range(3)]
            all_lines.append(line)
    # Y-axis lines (fixed x, z; vary y)
    for x in range(3):
        for z in range(3):
            line = [(x, y, z) for y in range(3)]
            all_lines.append(line)
    # Z-axis lines (fixed x, y; vary z)
    for x in range(3):
        for y in range(3):
            line = [(x, y, z) for z in range(3)]
            all_lines.append(line)
    # Diagonals in each layer (fixed x)
    for x in range(3):
        # Diagonal 1: y = z
        line1 = [(x, y, y) for y in range(3)]
        all_lines.append(line1)
        # Diagonal 2: y = 2 - z
        line2 = [(x, y, 2 - y) for y in range(3)]
        all_lines.append(line2)
    # Diagonals in each layer (fixed y)
    for y in range(3):
        # Diagonal 1: x = z
        line1 = [(x, y, x) for x in range(3)]
        all_lines.append(line1)
        # Diagonal 2: x = 2 - z
        line2 = [(x, y, 2 - x) for x in range(3)]
        all_lines.append(line2)
    # Diagonals in each layer (fixed z)
    for z in range(3):
        # Diagonal 1: x = y
        line1 = [(x, x, z) for x in range(3)]
        all_lines.append(line1)
        # Diagonal 2: x = 2 - y
        line2 = [(x, 2 - x, z) for x in range(3)]
        all_lines.append(line2)
    # Space diagonals
    all_lines.append([(0, 0, 0), (1, 1, 1), (2, 2, 2)])
    all_lines.append([(0, 0, 2), (1, 1, 1), (2, 2, 0)])
    all_lines.append([(0, 2, 0), (1, 1, 1), (2, 0, 2)])
    all_lines.append([(0, 2, 2), (1, 1, 1), (2, 0, 0)])
    
    # Check each empty cell in row-major order
    for i in range(3):
        for j in range(3):
            for k in range(3):
                if board[i][j][k] == 0:
                    # Check for winning move
                    for line in all_lines:
                        if (i, j, k) in line:
                            other_cells = [cell for cell in line if cell != (i, j, k)]
                            if board[other_cells[0][0]][other_cells[0][1]][other_cells[0][2]] == 1 and \
                               board[other_cells[1][0]][other_cells[1][1]][other_cells[1][2]] == 1:
                                return (i, j, k)
                    # Check for blocking opponent's win
                    for line in all_lines:
                        if (i, j, k) in line:
                            other_cells = [cell for cell in line if cell != (i, j, k)]
                            if board[other_cells[0][0]][other_cells[0][1]][other_cells[0][2]] == -1 and \
                               board[other_cells[1][0]][other_cells[1][1]][other_cells[1][2]] == -1:
                                return (i, j, k)
                    # Check for fork (creates two winning threats)
                    fork_count = 0
                    for line in all_lines:
                        if (i, j, k) in line:
                            other_cells = [cell for cell in line if cell != (i, j, k)]
                            if board[other_cells[0][0]][other_cells[0][1]][other_cells[0][2]] == 1 and \
                               board[other_cells[1][0]][other_cells[1][1]][other_cells[1][2]] == 1:
                                fork_count += 1
                                if fork_count >= 2:
                                    return (i, j, k)
    # Return first empty cell if no winning/blocking/fork moves found
    for i in range(3):
        for j in range(3):
            for k in range(3):
                if board[i][j][k] == 0:
                    return (i, j, k)
    # Return None if board is full (though game should end before this)
    return None
