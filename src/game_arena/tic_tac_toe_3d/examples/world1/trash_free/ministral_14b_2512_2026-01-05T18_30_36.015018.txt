Here's a Python implementation for a 3D Tic Tac Toe policy function that uses a combination of heuristic evaluation and minimax search with alpha-beta pruning to determine the next move. This implementation prioritizes winning moves, blocking opponent wins, and then evaluates based on potential lines (rows, columns, diagonals in all 3 dimensions).

```python
import numpy as np
from typing import List, Tuple

def policy(board: List[List[List[int]]]) -> Tuple[int, int, int]:
    """
    Determines the next move for a 3D Tic Tac Toe game using minimax with alpha-beta pruning.

    Args:
        board: A 3x3x3 list of lists where 0 is empty, 1 is current player, -1 is opponent.

    Returns:
        A tuple (x, y, z) representing the coordinates of the chosen empty cell.
    """
    # Convert to numpy array for easier manipulation
    board_np = np.array(board, dtype=int)
    player = 1  # Current player (assuming we're player 1)

    # Find all empty cells
    empty_cells = np.argwhere(board_np == 0)
    if len(empty_cells) == 0:
        raise ValueError("No empty cells left on the board")

    # If there's only one empty cell, take it
    if len(empty_cells) == 1:
        return tuple(empty_cells[0])

    # Evaluate all possible moves using minimax
    best_score = -float('inf')
    best_move = empty_cells[0]

    for move in empty_cells:
        x, y, z = move
        # Make the move
        board_np[x, y, z] = player
        score = minimax(board_np, 0, -float('inf'), float('inf'), -player)
        # Undo the move
        board_np[x, y, z] = 0

        if score > best_score:
            best_score = score
            best_move = move

    return tuple(best_move)

def minimax(board: np.ndarray, depth: int, alpha: float, beta: float, player: int) -> int:
    """
    Minimax algorithm with alpha-beta pruning for 3D Tic Tac Toe.

    Args:
        board: Current board state
        depth: Current depth in the search tree
        alpha: Alpha value for pruning
        beta: Beta value for pruning
        player: Current player (1 or -1)

    Returns:
        The score of the current board state from the perspective of the original player (1)
    """
    # Check for terminal state
    winner = check_winner(board)
    if winner != 0:
        return winner * 10  # Multiply by 10 to prioritize winning over other factors

    # Check for draw
    if np.all(board != 0):
        return 0

    if player == 1:
        best_score = -float('inf')
        for x in range(3):
            for y in range(3):
                for z in range(3):
                    if board[x, y, z] == 0:
                        board[x, y, z] = player
                        score = minimax(board, depth + 1, alpha, beta, -player)
                        board[x, y, z] = 0
                        best_score = max(score, best_score)
                        alpha = max(alpha, score)
                        if beta <= alpha:
                            break
                if beta <= alpha:
                    break
        return best_score
    else:
        best_score = float('inf')
        for x in range(3):
            for y in range(3):
                for z in range(3):
                    if board[x, y, z] == 0:
                        board[x, y, z] = player
                        score = minimax(board, depth + 1, alpha, beta, -player)
                        board[x, y, z] = 0
                        best_score = min(score, best_score)
                        beta = min(beta, score)
                        if beta <= alpha:
                            break
                if beta <= alpha:
                    break
        return best_score

def check_winner(board: np.ndarray) -> int:
    """
    Checks if there's a winner on the 3D Tic Tac Toe board.

    Args:
        board: Current board state

    Returns:
        1 if player 1 wins, -1 if player -1 wins, 0 otherwise
    """
    # Check all rows in each layer
    for layer in range(3):
        for row in range(3):
            if board[layer, row, :].sum() == 3:
                return 1
            if board[layer, row, :].sum() == -3:
                return -1

    # Check all columns in each layer
    for layer in range(3):
        for col in range(3):
            if board[layer, :, col].sum() == 3:
                return 1
            if board[layer, :, col].sum() == -3:
                return -1

    # Check all diagonals in each layer
    for layer in range(3):
        if board[layer, 0, 0] + board[layer, 1, 1] + board[layer, 2, 2] == 3:
            return 1
        if board[layer, 0, 0] + board[layer, 1, 1] + board[layer, 2, 2] == -3:
            return -1
        if board[layer, 0, 2] + board[layer, 1, 1] + board[layer, 2, 0] == 3:
            return 1
        if board[layer, 0, 2] + board[layer, 1, 1] + board[layer, 2, 0] == -3:
            return -1

    # Check all rows through layers (3D rows)
    for row in range(3):
        for col in range(3):
            if board[:, row, col].sum() == 3:
                return 1
            if board[:, row, col].sum() == -3:
                return -1

    # Check all columns through layers (3D columns)
    for col in range(3):
        for layer in range(3):
            if board[layer, :, col].sum() == 3:
                return 1
            if board[layer, :, col].sum() == -3:
                return -1

    # Check all diagonals through layers (3D diagonals)
    if board[0, 0, 0] + board[1, 1, 1] + board[2, 2, 2] == 3:
        return 1
    if board[0, 0, 0] + board[1, 1, 1] + board[2, 2, 2] == -3:
        return -1
    if board[0, 0, 2] + board[1, 1, 1] + board[2, 2, 0] == 3:
        return 1
    if board[0, 0, 2] + board[1, 1, 1] + board[2, 2, 0] == -3:
        return -1
    if board[0, 2, 0] + board[1, 1, 1] + board[2, 0, 2] == 3:
        return 1
    if board[0, 2, 0] + board[1, 1, 1] + board[2, 0, 2] == -3:
        return -1
    if board[0, 2, 2] + board[1, 1, 1] + board[2, 0, 0] == 3:
        return 1
    if board[0, 2, 2] + board[1, 1, 1] + board[2, 0, 0] == -3:
        return -1

    return 0
```

### Explanation:
1. **policy function**:
   - Converts the input board to a NumPy array for easier manipulation.
   - Finds all empty cells on the board.
   - If there's only one empty cell, it takes that cell.
   - Otherwise, it evaluates all possible moves using the minimax algorithm with alpha-beta pruning to find the best move.

2. **minimax function**:
   - Recursively evaluates all possible moves, alternating between maximizing (player 1) and minimizing (player -1) players.
   - Uses alpha-beta pruning to optimize the search by eliminating branches that cannot possibly influence the final decision.
   - Returns a score representing the evaluation of the board state from the perspective of the original player (1).

3. **check_winner function**:
   - Checks all possible winning conditions in 3D Tic Tac Toe, including:
     - Rows, columns, and diagonals within each 2D layer.
     - Rows, columns, and diagonals that span all three layers (3D lines).
   - Returns 1 if player 1 wins, -1 if player -1 wins, and 0 otherwise.

This implementation ensures that the AI will always make the optimal move, prioritizing winning moves, blocking opponent wins, and then evaluating based on potential future moves.
