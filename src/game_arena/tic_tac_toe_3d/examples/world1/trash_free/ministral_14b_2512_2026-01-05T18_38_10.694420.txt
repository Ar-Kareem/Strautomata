import numpy as np
from typing import Tuple, Optional

def policy(board: list[list[list[int]]]) -> Tuple[int, int, int]:
    """
    Determines the best move for a 3D Tic Tac Toe game using minimax with alpha-beta pruning.

    Args:
        board: A 3x3x3 list of lists where 0 is empty, 1 is current player, -1 is opponent.

    Returns:
        A tuple (x, y, z) representing the coordinates of the best move (0-2 for each dimension).
    """
    # Convert the board to a numpy array for easier manipulation
    board_np = np.array(board, dtype=int)
    current_player = 1  # Assuming the function is called for player 1

    # Find all empty cells
    empty_cells = np.argwhere(board_np == 0)

    if len(empty_cells) == 0:
        raise ValueError("No empty cells left on the board")

    # If it's the first move, choose the center
    if len(empty_cells) == 27:
        return (1, 1, 1)

    # Evaluate all possible moves using minimax with alpha-beta pruning
    best_score = -float('inf')
    best_move = None

    for cell in empty_cells:
        x, y, z = cell
        # Make a copy of the board
        new_board = board_np.copy()
        new_board[x, y, z] = current_player

        # Evaluate the move
        score = minimax(new_board, current_player, -float('inf'), float('inf'), False)

        # Update best move if this move is better
        if score > best_score:
            best_score = score
            best_move = (x, y, z)

    return best_move

def minimax(board: np.ndarray, player: int, alpha: float, beta: float, maximizing_player: bool) -> float:
    """
    Minimax algorithm with alpha-beta pruning for 3D Tic Tac Toe.

    Args:
        board: Current game board
        player: Current player (1 or -1)
        alpha: Alpha value for pruning
        beta: Beta value for pruning
        maximizing_player: Whether the current player is maximizing

    Returns:
        The score of the current board position
    """
    # Check if the game is over
    result = check_winner(board)
    if result != 0:
        return evaluate_result(result, player)

    if maximizing_player:
        value = -float('inf')
        for cell in np.argwhere(board == 0):
            x, y, z = cell
            board[x, y, z] = player
            value = max(value, minimax(board, -player, alpha, beta, False))
            board[x, y, z] = 0  # Undo move
            alpha = max(alpha, value)
            if beta <= alpha:
                break
        return value
    else:
        value = float('inf')
        for cell in np.argwhere(board == 0):
            x, y, z = cell
            board[x, y, z] = player
            value = min(value, minimax(board, -player, alpha, beta, True))
            board[x, y, z] = 0  # Undo move
            beta = min(beta, value)
            if beta <= alpha:
                break
        return value

def check_winner(board: np.ndarray) -> int:
    """
    Checks if there's a winner on the 3D Tic Tac Toe board.

    Args:
        board: Current game board

    Returns:
        1 if player 1 wins, -1 if player -1 wins, 0 if no winner
    """
    # Check rows in each layer
    for layer in range(3):
        for row in range(3):
            if board[layer, row, :].sum() == 3:
                return 1
            if board[layer, row, :].sum() == -3:
                return -1
            if board[layer, :, row].sum() == 3:
                return 1
            if board[layer, :, row].sum() == -3:
                return -1
            if board[layer, :, :].diagonal().sum() == 3:
                return 1
            if board[layer, :, :].diagonal().sum() == -3:
                return -1
            if np.fliplr(board[layer, :, :]).diagonal().sum() == 3:
                return 1
            if np.fliplr(board[layer, :, :]).diagonal().sum() == -3:
                return -1

    # Check columns across layers
    for col in range(3):
        for row in range(3):
            if board[:, row, col].sum() == 3:
                return 1
            if board[:, row, col].sum() == -3:
                return -1
            if board[:, col, :].sum() == 3:
                return 1
            if board[:, col, :].sum() == -3:
                return -1

    # Check 3D diagonals
    if board.diagonal().sum() == 3:
        return 1
    if board.diagonal().sum() == -3:
        return -1
    if np.fliplr(board).diagonal().sum() == 3:
        return 1
    if np.fliplr(board).diagonal().sum() == -3:
        return -1

    # Check space diagonals
    if board.trace() == 3:
        return 1
    if board.trace() == -3:
        return -1

    # Check if the board is full
    if np.all(board != 0):
        return 0  # Draw

    return 0  # No winner

def evaluate_result(result: int, player: int) -> float:
    """
    Evaluates the result of the game.

    Args:
        result: 1 for player 1 win, -1 for player -1 win, 0 for draw
        player: Current player (1 or -1)

    Returns:
        Evaluation score
    """
    if result == player:
        return 1000  # Win
    elif result == -player:
        return -1000  # Loss
    else:
        return 0  # Draw

   - Finds all empty cells on the board.

   - Uses minimax with alpha-beta pruning to evaluate all possible moves and selects the best one.

2. **minimax function**:
   - Recursively evaluates all possible moves using the minimax algorithm with alpha-beta pruning to optimize performance.
   - Alternates between maximizing and minimizing players to simulate both players' optimal play.

3. **check_winner function**:
   - Checks all possible winning conditions in 3D Tic Tac Toe, including:
     - Rows, columns, and diagonals in each 2D layer.
     - Columns and diagonals across layers.
     - 3D space diagonals.
   - Returns 1 for player 1 win, -1 for player -1 win, and 0 otherwise.

4. **evaluate_result function**:
   - Assigns a high score for a win, low score for a loss, and 0 for a draw.
