The strategy for this policy is to use the minimax algorithm with alpha-beta pruning to evaluate all possible future game states in the 3x3x3 3D Tic Tac Toe board, considering up to 9 moves ahead (since there are 27 cells but the game typically ends before filling all). This ensures the AI selects the optimal move that maximizes its chances of winning (returning a utility of 1 for a win for itself, -1 for a loss to the opponent, or 0 for a draw/tie). The policy first checks for any immediate threat (opponent about to win) or opportunity (itself about to win), then uses recursive minimax with pruning to explore the game tree, evaluating utility based on game outcomes. All 53 winning lines (straight lines in all dimensions, planes, and space) are precomputed and checked for termination. This approach makes the AI unbeatable in practice against simple or random opponents while fitting within the 1-second execution limit due to the small state space (27! / symmetries), and it always returns a legal move.

def get_winner(board):
    for win in wins:
        vals = set(board[i][j][k] for i, j, k in win)
        if len(vals) == 1 and 1 in vals:
            return 1
        elif len(vals) == 1 and -1 in vals:
            return -1
    return 0  # draw or ongoing

def minimax(curr_board, maximizer_player, alpha, beta):
    winner = get_winner(curr_board)
    if winner == 1 if maximizer_player else -1:
        # Win for the current maximizer player
        return 1 if winner == (1 if maximizer_player else -1) else -1
    if winner != 0:
        return winner  # opponent win: -1 for maximizer

    moves = [(i, j, k) for i in range(3) for j in range(3) for k in range(3) if curr_board[i][j][k] == 0]
    if not moves:  # unlikely tie before full, but safe
        return 0

    if maximizer_player:  # maximizing player (current turn player's move)
        max_eval = -math.inf
        for i, j, k in moves:
            curr_board[i][j][k] = 1 if maximizer_player else -1
            eval_score = minimax(curr_board, not maximizer_player, alpha, beta)
            curr_board[i][j][k] = 0  # undo
            max_eval = max(max_eval, eval_score)
            alpha = max(alpha, eval_score)
            if beta <= alpha:
                break
        return max_eval
    else:  # minimizing player
        min_eval = math.inf
        for i, j, k in moves:
            curr_board[i][j][k] = 1 if maximizer_player else -1
            eval_score = minimax(curr_board, not maximizer_player, alpha, beta)
            curr_board[i][j][k] = 0  # undo
            min_eval = min(min_eval, eval_score)
            beta = min(beta, eval_score)
            if beta <= alpha:
                break
        return min_eval

def policy(board):
    moves = [(i, j, k) for i in range(3) for j in range(3) for k in range(3) if board[i][j][k] == 0]
    if not moves:
        return (0, 0, 0)  # shouldn't happen

    # Check immediate win for me
    for i, j, k in moves:
        board[i][j][k] = 1
        if get_winner(board) == 1:
            board[i][j][k] = 0
            return (i, j, k)
        board[i][j][k] = 0

    # Check immediate win for opponent (block)
    for i, j, k in moves:
        board[i][j][k] = -1
        if get_winner(board) == -1:
            board[i][j][k] = 0
            return (i, j, k)  # block
        board[i][j][k] = 0

    # Use minimax for best move
    best_move = None
    best_value = -math.inf
    for i, j, k in moves:
        board[i][j][k] = 1  # my move
        move_value = minimax(board, False, -math.inf, math.inf)  # opponent to move next
        board[i][j][k] = 0  # undo
        if move_value > best_value:
            best_value = move_value
            best_move = (i, j, k)

    return best_move
