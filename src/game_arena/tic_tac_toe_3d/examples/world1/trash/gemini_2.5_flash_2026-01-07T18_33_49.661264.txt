A few considerations for the `policy` function:
1.  **Initial move**: Taking the center (1,1,1) is often a strong opening. I've added a check for this.
2.  **Shuffling**: Shuffling `empty_cells` before iterating ensures that if multiple moves yield the same best score, one is chosen randomly, making the AI less predictable over multiple games.
3.  **MAX_DEPTH**: I've set it to 6. This is a compromise. A full game search `depth=27` (or `len(empty_cells)`) would be ideal but probably too slow. `MAX_DEPTH=6` means it will look 6 moves deep (3 for me, 3 for opponent). This should be enough to see 2-move threats and some simpler forks. If a game can be won/lost in exactly 6 moves or less it will detect it perfectly. For longer games, it relies on the heuristic.
4.  **WIN_SCORE/LOSE_SCORE adjustment**: `- depth` and `+ depth` in the terminal state return values are to prefer faster wins and slower losses. This makes the AI "eager" to win quickly and "stubborn" to avoid losing.
5.  **Heuristic `_evaluate_board`**: This needs to be robust for states that are not terminal and are at the `MAX_DEPTH` limit. The current heuristic primarily focuses on immediate threats (2-in-a-row) and setting up lines (1-in-a-row). This is a decent starting point.

    # If no move was found (shouldn't happen with valid empty_cells), pick a random one
    if best_move is None:

import math
import random

# Pre-compute all 49 winning lines for a 3x3x3 board
WINNING_LINES = []

# --- 27 basic lines (rows, columns, stacks) ---
# Rows (fixed y, z; varying x)
for y in range(3):
    for z in range(3):
        WINNING_LINES.append([(x, y, z) for x in range(3)])

# Columns (fixed x, z; varying y)
for x in range(3):
    for z in range(3):
        WINNING_LINES.append([(x, y, z) for y in range(3)])

# Stacks (fixed x, y; varying z)
for x in range(3):
    for y in range(3):
        WINNING_LINES.append([(x, y, z) for z in range(3)])

# --- 18 two-dimensional diagonal lines (on each of the 3 planes) ---
# XY-planes (fixed z)
for z in range(3):
    WINNING_LINES.append([(i, i, z) for i in range(3)])      # Main diagonal in XY plane
    WINNING_LINES.append([(i, 2 - i, z) for i in range(3)])  # Anti-diagonal in XY plane

# XZ-planes (fixed y)
for y in range(3):
    WINNING_LINES.append([(i, y, i) for i in range(3)])      # Main diagonal in XZ plane
    WINNING_LINES.append([(i, y, 2 - i) for i in range(3)])  # Anti-diagonal in XZ plane

# YZ-planes (fixed x)
for x in range(3):
    WINNING_LINES.append([(x, i, i) for i in range(3)])      # Main diagonal in YZ plane
    WINNING_LINES.append([(x, i, 2 - i) for i in range(3)])  # Anti-diagonal in YZ plane

# --- 4 three-dimensional diagonal lines ---
WINNING_LINES.append([(i, i, i) for i in range(3)])          # Space diagonal (0,0,0) to (2,2,2)
WINNING_LINES.append([(i, i, 2 - i) for i in range(3)])      # Space diagonal (0,0,2) to (2,2,0)
WINNING_LINES.append([(i, 2 - i, i) for i in range(3)])      # Space diagonal (0,2,0) to (2,0,2)
WINNING_LINES.append([(2 - i, i, i) for i in range(3)])      # Space diagonal (2,0,0) to (0,2,2)


# Player constants
PLAYER = 1
OPPONENT = -1
EMPTY = 0

# Score constants for minimax
WIN_SCORE = 1000000  # A very large score for immediate wins
LOSE_SCORE = -1000000 # A very large negative score for immediate losses
DRAW_SCORE = 0

# Max depth for minimax search.
# A higher depth leads to better play but takes more time.
# For 3x3x3, a full depth search is computationally expensive (up to 27 moves).
# A value around 6-8 usually provides a good balance between performance and strength.
MAX_DEPTH = 6

def _is_winner(board, player: int) -> bool:
    """Checks if the given player has won the game."""
    for line in WINNING_LINES:
        if all(board[x][y][z] == player for x, y, z in line):
            return True
    return False

def _get_empty_cells(board: list[list[list[int]]]) -> list[tuple[int, int, int]]:
    """Returns a list of all empty (x, y, z) coordinates on the board."""
    empty_cells = []
    for x in range(3):
        for y in range(3):
            for z in range(3):
                if board[x][y][z] == EMPTY:
                    empty_cells.append((x, y, z))
    return empty_cells

def _evaluate_line(board: list[list[list[int]]], line: list[tuple[int, int, int]], player: int) -> int:
    """Evaluates a single line for the given player."""
    my_pieces = 0
    opponent_pieces = 0
    empty_cells = 0

    for x, y, z in line:
        if board[x][y][z] == player:
            my_pieces += 1
        elif board[x][y][z] == -player: # Opponent is -player based on game rules
            opponent_pieces += 1
        else: # EMPTY
            empty_cells += 1

    score = 0
    # Prioritize immediate wins/blocks at the highest level
    if my_pieces == 3:
        score += WIN_SCORE
    elif opponent_pieces == 3:
        score -= WIN_SCORE
    # Evaluate potential threats/opportunities without immediate win/loss
    elif my_pieces == 2 and empty_cells == 1:
        score += 100 # Two in a row for player, one spot open (strong threat)
    elif opponent_pieces == 2 and empty_cells == 1:
        score -= 100 # Opponent has two in a row, one spot open (must block)
    elif my_pieces == 1 and empty_cells == 2:
        score += 10  # One in a row for player, two spots open (potential strategic line)
    elif opponent_pieces == 1 and empty_cells == 2:
        score -= 10  # Opponent has one in a row, two spots open

    return score

def _evaluate_board(board: list[list[list[int]]], player: int) -> int:
    """
    Evaluates the entire board from the perspective of the given player.
    A positive total_score indicates a favorable board for 'player'.
    """
    total_score = 0
    for line in WINNING_LINES:
        total_score += _evaluate_line(board, line, player)
    return total_score

def _minimax(board: list[list[list[int]]], depth: int, is_maximizing_player: bool, alpha: float, beta: float) -> int:
    """
    Minimax algorithm with Alpha-Beta Pruning.
    `is_maximizing_player` is True for 'PLAYER' (our AI), False for 'OPPONENT'.
    """
    # Check for terminal states (win/loss/draw)
    if _is_winner(board, PLAYER):
        return WIN_SCORE - depth # Prefer faster wins by reducing score for deeper wins
    if _is_winner(board, OPPONENT):
        return LOSE_SCORE + depth # Prefer slower losses/quicker blocks by increasing score for deeper losses

    empty_cells = _get_empty_cells(board)
    if not empty_cells: # Draw
        return DRAW_SCORE

    # If depth limit is reached, use the heuristic evaluation
    if depth >= MAX_DEPTH:
        # Evaluate for the current player whose turn it is.
        # If it's maximizing player's turn, evaluate for PLAYER.
        # If it's minimizing player's turn, evaluate for OPPONENT and negate it.
        return _evaluate_board(board, PLAYER if is_maximizing_player else OPPONENT)

    if is_maximizing_player:
        max_eval = -math.inf
        for x, y, z in empty_cells:
            board[x][y][z] = PLAYER
            eval = _minimax(board, depth + 1, False, alpha, beta)
            board[x][y][z] = EMPTY # Undo move
            max_eval = max(max_eval, eval)
            alpha = max(alpha, eval)
            if beta <= alpha: # Beta cutoff
                break
        return max_eval
    else: # Minimizing player
        min_eval = math.inf
        for x, y, z in empty_cells:
            board[x][y][z] = OPPONENT
            eval = _minimax(board, depth + 1, True, alpha, beta)
            board[x][y][z] = EMPTY # Undo move
            min_eval = min(min_eval, eval)
            beta = min(beta, eval)
            if beta <= alpha: # Alpha cutoff
                break
        return min_eval

def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    """
    Determines the next optimal move for a 3x3x3 3D Tic Tac Toe game using Minimax
    with Alpha-Beta Pruning and a heuristic evaluation.
    """
    # Defensive check: Take the center cell (1,1,1) if available, as it is often optimal
    if board[1][1][1] == EMPTY:
        return (1, 1, 1)

    empty_cells = _get_empty_cells(board)
    if not empty_cells:
        # This case should ideally not be reached if a move is expected,
        # but provides a safe return if no moves are possible.
        return (-1, -1, -1) # Indicate no valid move available

    best_move = None
    best_score = -math.inf

    # Shuffle empty cells to introduce randomness when multiple moves have the same score.
    # This prevents the AI from being entirely predictable if equally good moves exist.
    random.shuffle(empty_cells)

    for x, y, z in empty_cells:
        board[x][y][z] = PLAYER # Make the hypothetical move on the board
        # Call minimax for the next turn, which is the opponent's.
        # Our minimax starts with depth 0 for the current "root" search level.
        score = _minimax(board, 0, False, -math.inf, math.inf)
        board[x][y][z] = EMPTY # Undo the move to restore the board state for the next iteration

        if score > best_score:
            best_score = score
            best_move = (x, y, z)
        # If scores are equal, the first encountered (due to shuffling) best move is kept.

    # Fallback: if no best_move was found (should not happen if empty_cells is not empty),
    # pick a random one.
    if best_move is None:
        return random.choice(empty_cells)

    return best_move
