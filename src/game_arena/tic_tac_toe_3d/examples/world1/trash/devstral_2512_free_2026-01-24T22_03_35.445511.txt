# Strategy for 3D Tic Tac Toe Policy

This policy implements a smart 3D Tic Tac Toe strategy with the following components:

1. **Winning Move Check**: First checks if there's an immediate winning move available
2. **Blocking Opponent**: If no winning move, checks if the opponent has a winning move that needs to be blocked
3. **Center Control**: Prioritizes controlling the center of the 3D space (1,1,1)
4. **Corner Control**: Prioritizes corners over edges
5. **Fork Creation**: Looks for opportunities to create multiple winning threats
6. **Random Fallback**: If no strategic move is found, selects a random empty cell

The policy uses a combination of immediate tactical responses and positional strategy to maximize winning chances while minimizing opponent opportunities.

import random
from typing import List, Tuple

def policy(board: List[List[List[int]]]) -> Tuple[int, int, int]:
    # Check for immediate winning move
    for i in range(3):
        for j in range(3):
            for k in range(3):
                if board[i][j][k] == 0:
                    # Simulate placing our mark
                    board[i][j][k] = 1
                    if check_win(board, 1):
                        board[i][j][k] = 0  # Undo simulation
                        return (i, j, k)
                    board[i][j][k] = 0  # Undo simulation

    # Check for opponent's winning move to block
    for i in range(3):
        for j in range(3):
            for k in range(3):
                if board[i][j][k] == 0:
                    # Simulate placing opponent's mark
                    board[i][j][k] = -1
                    if check_win(board, -1):
                        board[i][j][k] = 0  # Undo simulation
                        return (i, j, k)
                    board[i][j][k] = 0  # Undo simulation

    # Try to take the center if available
    if board[1][1][1] == 0:
        return (1, 1, 1)

    # Try to take corners (all combinations where coordinates are 0 or 2)
    corners = [(0,0,0), (0,0,2), (0,2,0), (0,2,2),
               (2,0,0), (2,0,2), (2,2,0), (2,2,2)]
    random.shuffle(corners)
    for i, j, k in corners:
        if board[i][j][k] == 0:
            return (i, j, k)

    # Try to create a fork (two potential winning lines)
    for i in range(3):
        for j in range(3):
            for k in range(3):
                if board[i][j][k] == 0:
                    # Simulate placing our mark
                    board[i][j][k] = 1
                    fork_count = 0
                    # Check all possible winning lines through this position
                    for line in get_all_lines():
                        if (i,j,k) in line:
                            # Check if this line would be a winning line
                            line_values = [board[x][y][z] for x,y,z in line]
                            if line_values.count(1) == 2 and line_values.count(0) == 1:
                                fork_count += 1
                    if fork_count >= 2:
                        board[i][j][k] = 0  # Undo simulation
                        return (i, j, k)
                    board[i][j][k] = 0  # Undo simulation

    # If no strategic move found, pick a random empty cell
    empty_cells = []
    for i in range(3):
        for j in range(3):
            for k in range(3):
                if board[i][j][k] == 0:
                    empty_cells.append((i, j, k))
    return random.choice(empty_cells)

def check_win(board: List[List[List[int]], player: int) -> bool:
    # Check all possible winning lines
    for line in get_all_lines():
        values = [board[i][j][k] for i,j,k in line]
        if all(v == player for v in values):
            return True
    return False

def get_all_lines() -> List[List[Tuple[int, int, int]]]:
    lines = []

    # Rows in each layer
    for i in range(3):
        for j in range(3):
            lines.append([(i, j, 0), (i, j, 1), (i, j, 2)])

    # Columns in each layer
    for i in range(3):
        for k in range(3):
            lines.append([(i, 0, k), (i, 1, k), (i, 2, k)])

    # Depth in each layer
    for j in range(3):
        for k in range(3):
            lines.append([(0, j, k), (1, j, k), (2, j, k)])

    # Diagonals in each layer
    for i in range(3):
        lines.append([(i, 0, 0), (i, 1, 1), (i, 2, 2)])
        lines.append([(i, 0, 2), (i, 1, 1), (i, 2, 0)])

    # Space diagonals
    lines.append([(0, 0, 0), (1, 1, 1), (2, 2, 2)])
    lines.append([(0, 0, 2), (1, 1, 1), (2, 2, 0)])
    lines.append([(0, 2, 0), (1, 1, 1), (2, 0, 2)])
    lines.append([(0, 2, 2), (1, 1, 1), (2, 0, 0)])

    return lines
