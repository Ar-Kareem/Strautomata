   - Center (1,1,1) if available.
   - Corners (cells where all coordinates are 0 or 2) if available.
   - Any available empty cell as a fallback.

def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    # Generate all winning lines for 3x3x3 Tic Tac Toe
    lines = []
    
    # Lines along the z-axis
    for x in range(3):
        for y in range(3):
            lines.append([(x, y, z) for z in range(3)])
    
    # Lines along the y-axis
    for x in range(3):
        for z in range(3):
            lines.append([(x, y, z) for y in range(3)])
    
    # Lines along the x-axis
    for y in range(3):
        for z in range(3):
            lines.append([(x, y, z) for x in range(3)])
    
    # Face diagonals (for x fixed)
    for x in [0, 2]:
        lines.append([(x, i, i) for i in range(3)])
        lines.append([(x, i, 2 - i) for i in range(3)])
    
    # Face diagonals (for y fixed)
    for y in [0, 2]:
        lines.append([(i, y, i) for i in range(3)])
        lines.append([(i, y, 2 - i) for i in range(3)])
    
    # Face diagonals (for z fixed)
    for z in [0, 2]:
        lines.append([(i, i, z) for i in range(3)])
        lines.append([(i, 2 - i, z) for i in range(3)])
    
    # Space diagonals
    lines.append([(0, 0, 0), (1, 1, 1), (2, 2, 2)])
    lines.append([(0, 0, 2), (1, 1, 1), (2, 2, 0)])
    lines.append([(0, 2, 0), (1, 1, 1), (2, 0, 2)])
    lines.append([(0, 2, 2), (1, 1, 1), (2, 0, 0)])
    
    # Find all empty cells
    empty_cells = [(x, y, z) for x in range(3) for y in range(3) for z in range(3) if board[x][y][z] == 0]
    
    # Check for a winning move
    for line in lines:
        values = [board[x][y][z] for (x, y, z) in line]
        if values.count(1) == 2 and values.count(0) == 1:
            for cell in line:
                if board[cell[0]][cell[1]][cell[2]] == 0:
                    return cell
    
    # Check for a blocking move
    for line in lines:
        values = [board[x][y][z] for (x, y, z) in line]
        if values.count(-1) == 2 and values.count(0) == 1:
            for cell in line:
                if board[cell[0]][cell[1]][cell[2]] == 0:
                    return cell
    
    # Prefer the center if available
    if board[1][1][1] == 0:
        return (1, 1, 1)
    
    # Prefer corners
    corners = [(0, 0, 0), (0, 0, 2), (0, 2, 0), (0, 2, 2),
               (2, 0, 0), (2, 0, 2), (2, 2, 0), (2, 2, 2)]
    for corner in corners:
        if board[corner[0]][corner[1]][corner[2]] == 0:
            return corner
    
    # Default to the first available cell
    return empty_cells[0]
