# Global variable to store winning lines once to avoid recomputing on every call

def _get_all_winning_lines() -> List[List[Tuple[int, int, int]]]:
    """

    Each line is a list of 3 (x, y, z) coordinate tuples.
    """
    lines = []

    # 1. Lines parallel to axes (27 lines)
    for i in range(3):  # Fixed coordinate for a plane or axis segment
        for j in range(3):  # Another fixed coordinate for a specific line
            # Lines along Z-axis (fixed x, y)
            lines.append([(i, j, k) for k in range(3)])
            # Lines along Y-axis (fixed x, z)
            lines.append([(i, k, j) for k in range(3)])
            # Lines along X-axis (fixed y, z)
            lines.append([(k, i, j) for k in range(3)])

    # 2. Planar diagonals (18 lines)
    for i in range(3): # The fixed coordinate for the plane
        # Diagonals on planes parallel to YZ (fixed x)
        lines.append([(i, k, k) for k in range(3)])       # (x,0,0), (x,1,1), (x,2,2)
        lines.append([(i, k, 2 - k) for k in range(3)])   # (x,0,2), (x,1,1), (x,2,0)

        # Diagonals on planes parallel to XZ (fixed y)
        lines.append([(k, i, k) for k in range(3)])       # (0,y,0), (1,y,1), (2,y,2)
        lines.append([(k, i, 2 - k) for k in range(3)])   # (0,y,2), (1,y,1), (2,y,0)

        # Diagonals on planes parallel to XY (fixed z)
        lines.append([(k, k, i) for k in range(3)])       # (0,0,z), (1,1,z), (2,2,z)
        lines.append([(k, 2 - k, i) for k in range(3)])   # (0,2,z), (1,1,z), (2,0,z)

    # 3. Main space diagonals (4 lines)
    lines.append([(k, k, k) for k in range(3)])       # (0,0,0) -> (2,2,2)
    lines.append([(k, k, 2 - k) for k in range(3)])   # (0,0,2) -> (2,2,0)
    lines.append([(k, 2 - k, k) for k in range(3)])   # (0,2,0) -> (2,0,2)
    lines.append([(2 - k, k, k) for k in range(3)])   # (2,0,0) -> (0,2,2)
    
    return lines

def _check_win(board_np: np.ndarray, player: int) -> bool:
    """
    Checks if the given player (1 for AI, -1 for opponent) has won on the board.
    """
    global _ALL_WINNING_LINES
    if _ALL_WINNING_LINES is None:
        _ALL_WINNING_LINES = _get_all_winning_lines()

    for line in _ALL_WINNING_LINES:
        # Get values for all cells in the current line
        line_values = [board_np[coord] for coord in line]
        # If all values in the line are equal to the player's value, that player wins
        if all(val == player for val in line_values):
            return True
    return False

def policy(board: List[List[List[int]]]) -> Tuple[int, int, int]:
    """
    Determines the next optimal move for a 3x3x3 3D Tic-Tac-Toe game.
    
    Args:
        board: A 3x3x3 list of lists representing the game board.
               0: empty, 1: AI's piece, -1: opponent's piece.
               
    Returns:
        A tuple of three integers (x, y, z) representing the chosen empty cell.
    """
    board_np = np.array(board, dtype=int)
    
    empty_cells: List[Tuple[int, int, int]] = []
    for x in range(3):
        for y in range(3):
            for z in range(3):
                if board_np[x, y, z] == 0:
                    empty_cells.append((x, y, z))

    # Phase 1: Check for an immediate winning move
    # If we can win on this turn, take the winning move.
    for (ex, ey, ez) in empty_cells:
        temp_board = np.copy(board_np)
        temp_board[ex, ey, ez] = 1 # Temporarily place our piece
        if _check_win(temp_board, 1):
            return (ex, ey, ez)

    # Phase 2: Check for an immediate blocking move
    # If the opponent can win on their next turn, block their winning move.
    for (ex, ey, ez) in empty_cells:
        temp_board = np.copy(board_np)
        temp_board[ex, ey, ez] = -1 # Temporarily place opponent's piece
        if _check_win(temp_board, -1):
            return (ex, ey, ez)

    # Phase 3: Prioritize the center cell if available
    # The center (1,1,1) is strategically very important in 3D Tic-Tac-Toe.
    if (1, 1, 1) in empty_cells:
        return (1, 1, 1)
        
    # Phase 4: Prioritize corner cells
    # Corners are generally strong positions for control and setting up future wins.
    corners = [
        (0,0,0), (0,0,2), (0,2,0), (0,2,2),
        (2,0,0), (2,0,2), (2,2,0), (2,2,2)
    ]
    for (cx, cy, cz) in corners:
        if board_np[cx, cy, cz] == 0:
            return (cx, cy, cz)

    # Phase 5: Fallback - choose the first available empty cell
    # If no strategic moves are found, pick the first empty cell.
    if empty_cells:
        return empty_cells[0]
    
    # This point should theoretically not be reached in a valid game unless the board is full
    # and no winner before this turn, but an empty_cells list ensures a move if accessible.
    raise Exception("No empty cells left, and no winning move identified for either player.")
