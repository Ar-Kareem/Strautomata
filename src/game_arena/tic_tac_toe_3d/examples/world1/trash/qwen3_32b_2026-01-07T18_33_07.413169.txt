If neither a winning nor blocking move is present, the strategy moves to heuristics:
1. **Center cell (1,1,1)** is prioritized.
2. **Corners** are next since they have the most lines.
3. **Edges** are selected as a fallback.

This approach ensures optimal play in critical situations and provides a balanced fallback when the board is not yet in a critical state.

The final policy function is written using these rules, leveraging the 3D cube's small size to precompute all important lines and quickly evaluate each candidate move.

def policy(board: list[list[list[int]]]) -> tuple[int, int, int]:
    lines = []
    # Generate lines along the x-axis (varying x)
    for y in range(3):
        for z in range(3):
            lines.append([(x, y, z) for x in range(3)])
    # Generate lines along the y-axis (varying y)
    for x in range(3):
        for z in range(3):
            lines.append([(x, y, z) for y in range(3)])
    # Generate lines along the z-axis (varying z)
    for x in range(3):
        for y in range(3):
            lines.append([(x, y, z) for z in range(3)])
    # Face diagonals in x-y planes (fixed z)
    for z in range(3):
        lines.append([(x, x, z) for x in range(3)])     # diagonal 1
        lines.append([(x, 2 - x, z) for x in range(3)])  # diagonal 2
    # Face diagonals in x-z planes (fixed y)
    for y in range(3):
        lines.append([(x, y, x) for x in range(3)])      # diagonal 1
        lines.append([(x, y, 2 - x) for x in range(3)])  # diagonal 2
    # Face diagonals in y-z planes (fixed x)
    for x in range(3):
        lines.append([(x, y, y) for y in range(3)])      # diagonal 1
        lines.append([(x, y, 2 - y) for y in range(3)])   # diagonal 2
    # Space diagonals
    lines.append([(i, i, i) for i in range(3)])
    lines.append([(i, i, 2 - i) for i in range(3)])
    lines.append([(i, 2 - i, i) for i in range(3)])
    lines.append([(i, 2 - i, 2 - i) for i in range(3)])

    empty_cells = [(x, y, z) for x in range(3) for y in range(3) for z in range(3) if board[x][y][z] == 0]

    # Check for winning move
    for (x, y, z) in empty_cells:
        board[x][y][z] = 1
        win = any(all(board[i][j][k] == 1 for (i, j, k) in line) for line in lines)
        if win:
            board[x][y][z] = 0
            return (x, y, z)
        board[x][y][z] = 0

    # Check for opponent's winning move
    for (x, y, z) in empty_cells:
        board[x][y][z] = -1
        win = any(all(board[i][j][k] == -1 for (i, j, k) in line) for line in lines)
        if win:
            board[x][y][z] = 0
            return (x, y, z)
        board[x][y][z] = 0

    # Prioritize center
    if board[1][1][1] == 0:
        return (1, 1, 1)

    # Prioritize corners
    corners = [(0,0,0), (0,0,2), (0,2,0), (0,2,2), (2,0,0), (2,0,2), (2,2,0), (2,2,2)]
    for corner in corners:
        x, y, z = corner
        if board[x][y][z] == 0:
            return (x, y, z)

    # Prioritize edges
    for (x, y, z) in empty_cells:
        if (x, y, z) != (1, 1, 1) and (x, y, z) not in corners:
            return (x, y, z)

    # Fallback to first empty cell
    return empty_cells[0]
