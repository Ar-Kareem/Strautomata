The strategy for this policy is to first check for an immediate winning move for the AI (player 1). If none exists, check for an immediate blocking move to prevent the opponent (player -1) from winning. If neither exists, use the Minimax algorithm with alpha-beta pruning to evaluate all possible game states and select the best move based on a scoring system where AI wins get +10, opponent wins get -10, and draws or non-terminal states get 0. This ensures optimal play while respecting the 1-second time limit on a modern 3x3x3 board.

import copy  # For deep copying the board

# All possible winning lines in 3D Tic Tac Toe (3x3x3 grid)
lines = [
    # Straight lines (varying each dimension)
    # Fixed j, k, vary i
    [ (0, j, k), (1, j, k), (2, j, k) ] for j in range(3) for k in range(3)
    # Fixed i, k, vary j
    [ (i, 0, k), (i, 1, k), (i, 2, k) ] for i in range(3) for k in range(3)
    # Fixed i, j, vary k
    [ (i, j, 0), (i, j, 1), (i, j, 2) ] for i in range(3) for j in range(3)
    # Diagonals
    # In depth planes (fixed i)
    [(i, 0, 0), (i, 1, 1), (i, 2, 2)] for i in range(3),
    [(i, 0, 2), (i, 1, 1), (i, 2, 0)] for i in range(3),
    # In height planes (fixed j)
    [(0, j, 0), (1, j, 1), (2, j, 2)] for j in range(3),
    [(0, j, 2), (1, j, 1), (2, j, 0)] for j in range(3),
    # In width planes (fixed k)
    [(0, 0, k), (1, 1, k), (2, 2, k)] for k in range(3),
    [(0, 2, k), (1, 1, k), (2, 0, k)] for k in range(3),
    # 3D diagonals
    [(0, 0, 0), (1, 1, 1), (2, 2, 2)],
    [(0, 0, 2), (1, 1, 1), (2, 2, 0)],
    [(0, 2, 0), (1, 1, 1), (2, 0, 2)],
    [(0, 2, 2), (1, 1, 1), (2, 0, 0)]
]

def flat_lines(lines):
    # Flatten the nested list of lines
    flattened = []
    for sublist in lines:
        flattened.extend(sublist)
    return flattened

lines = flat_lines(lines)  # Now lines is a list of lists of tuples

def has_won(board, player):
    for line in lines:
        if all(board[i][j][k] == player for i, j, k in line):
            return True
    return False

def is_full(board):
    return all(board[i][j][k] != 0 for i in range(3) for j in range(3) for k in range(3))

def get_legal_moves(board):
    return [(i, j, k) for i in range(3) for j in range(3) for k in range(3) if board[i][j][k] == 0]

def make_move(board, move, player):
    i, j, k = move
    new_board = copy.deepcopy(board)
    new_board[i][j][k] = player
    return new_board

def evaluate(board):
    if has_won(board, 1):
        return 10
    if has_won(board, -1):
        return -10
    if is_full(board):
        return 0
    return 0  # Neutral for non-terminal states

def minimax(board, player, alpha, beta):
    if has_won(board, 1) or has_won(board, -1) or is_full(board):
        return evaluate(board)
    if player == 1:
        max_eval = -float('inf')
        for move in get_legal_moves(board):
            new_board = make_move(board, move, player)
            eval_val = minimax(new_board, -player, alpha, beta)
            max_eval = max(max_eval, eval_val)
            alpha = max(alpha, eval_val)
            if beta <= alpha:
                break
        return max_eval
    else:
        min_eval = float('inf')
        for move in get_legal_moves(board):
            new_board = make_move(board, move, -player)
            eval_val = minimax(new_board, player, alpha, beta)  # Note: -player would make it 1 in recursion, but here player is the original min player
            min_eval = min(min_eval, eval_val)
            beta = min(beta, eval_val)
            if beta <= alpha:
                break
        return min_eval

def policy(board):
    legal_moves = get_legal_moves(board)
    
    # Check for immediate win
    for move in legal_moves:
        temp_board = make_move(board, move, 1)  # Simulate move for player 1
        if has_won(temp_board, 1):
            return move
    
    # Check for immediate block
    for move in legal_moves:
        temp_board = make_move(board, move, -1)  # Simulate move for opponent
        if has_won(temp_board, -1):
            return move
    
    # Use Minimax to choose the best move
    best_move = None
    best_score = -float('inf')
    for move in legal_moves:
        temp_board = make_move(board, move, 1)
        score = minimax(temp_board, -1, -float('inf'), float('inf'))  # Opponent's turn to minimize
        if score > best_score:
            best_score = score
            best_move = move
    
    return best_move
