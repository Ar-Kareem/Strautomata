def emergency(board):
    for r in range(8):
        for c in range(8):
            if board[r][c] == 1:
                return f"{r},{c}:{r},{c}"
    return "0,0:0,1"

def fmove(m):
    return f"{m[0]},{m[1]}:{m[2]},{m[3]}"

def domove(board, m):
    b = [row[:] for row in board]
    b[m[2]][m[3]] = b[m[0]][m[1]]
    b[m[0]][m[1]] = 0
