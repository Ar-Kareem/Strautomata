To solve this problem, we need to create a targeting policy for a Connect 4 game using a Minimax algorithm with Alpha-Beta pruning. The goal is to determine the optimal column to drop a disc such that the player can win the game or block the opponent from winning.

### Approach
1. **Immediate Win Check**: The first step is to check if dropping a disc in any available column results in an immediate win.If such a move exists, it is chosen immediately.
2. **Alpha-Beta Pruning**: If no immediate win is found, we use the Minimax algorithm with Alpha-Beta pruning to evaluate potential moves up to a certain depth. This helps in efficiently exploring possible moves and pruning less promising branches in the game tree.

### Solution Code

def check_win(board, row, col, disc):
    # Check horizontal
    count = 1
    c = col - 1
    while c >= 0 and board[row][c] == disc:
        count += 1
        c -= 1
    c = col + 1
    while c < 7 and board[row][c] == disc:
        count += 1
        c += 1
    if count >= 4:
        return True

    # Check vertical
    count = 1
    r = row - 1
    while r >= 0 and board[r][col] == disc:
        count += 1
        r -= 1
    r = row + 1
    while r < 6 and board[r][col] == disc:
        count += 1
        r += 1
    if count >= 4:
        return True

    # Check diagonal (top-left to bottom-right)
    count = 1
    r, c = row - 1, col - 1
    while r >= 0 and c >= 0 and board[r][c] == disc:
        count += 1
        r -= 1
        c -= 1
    r, c = row + 1, col + 1
    while r < 6 and c < 7 and board[r][c] == disc:
        count += 1
        r += 1
        c += 1
    if count >= 4:
        return True

    # Check diagonal (top-right to bottom-left)
    count = 1
    r, c = row - 1, col + 1
    while r >= 0 and c < 7 and board[r][c] == disc:
        count += 1
        r -= 1
        c += 1
    r, c = row + 1, col - 1
    while r < 6 and c >= 0 and board[r][c] == disc:
        count += 1
        r += 1
        c -= 1
    if count >= 4:
        return True

    return False

def get_available_row(board, col):
    for r in range(5, -1, -1):
        if board[r][col] == 0:
            return r
    return None

def evaluate(board):
    # Heuristic evaluation function
    # Calculate the number of potential lines for the player (1) and opponent (-1)
    player_lines = 0
    opponent_lines = 0

    # Check horizontal
    for row in range(6):
        for col in range(4):
            window = board[row][col:col+4]
            if window.count(1) == 3 and window.count(0) == 1:
                player_lines += 1
            if window.count(-1) == 3 and window.count(0) == 1:
                opponent_lines += 1

    # Check vertical
    for col in range(7):
        for row in range(3):
            window = [board[row][col], board[row+1][col], board[row+2][col], board[row+3][col]]
            if window.count(1) == 3 and window.count(0) == 1:
                player_lines += 1
            if window.count(-1) == 3 and window.count(0) == 1:
                opponent_lines += 1

    # Check diagonal 1 (top-left to bottom-right)
    for row in range(3):
        for col in range(4):
            if board[row][col] == 1 and board[row+1][col+1] == 1 and board[row+2][col+2] == 1 and board[row+3][col+3] == 0:
                player_lines += 1
            if board[row][col] == -1 and board[row+1][col+1] == -1 and board[row+2][col+2] == -1 and board[row+3][col+3] == 0:
                opponent_lines += 1

    # Check diagonal 2 (top-right to bottom-left)
    for row in range(3):
        for col in range(3, 7):
            if board[row][col] == 1 and board[row+1][col-1] == 1 and board[row+2][col-2] == 1 and board[row+3][col-3] == 0:
                player_lines += 1
            if board[row][col] == -1 and board[row+1][col-1] == -1 and board[row+2][col-2] == -1 and board[row+3][col-3] == 0:
                opponent_lines += 1

    return player_lines * 3 - opponent_lines * 3

def alpha_beta(board, alpha, beta, maximizing, depth):
    available_cols = [col for col in range(7) if board[0][col] == 0]
    if not available_cols:
        return 0  # Draw

    if depth == 0:
        return evaluate(board)

    for col in [3, 2, 4, 1, 5, 0, 6]:
        if board[0][col] != 0:
            continue
        row = get_available_row(board, col)
        new_board = [list(row) for row in board]
        new_board[row][col] = 1 if maximizing else -1

        if check_win(new_board, row, col, new_board[row][col]):
            if new_board[row][col] == 1:
                return 4
            else:
                return -4

        value = alpha_beta(new_board, alpha, beta, not maximizing, depth - 1)
        if maximizing:
            if value > best_value:
                best_value = value
            alpha = max(alpha, best_value)
        else:
            if value < best_value:
                best_value = value
            beta = min(beta, best_value)

        if beta <= alpha:
            break

    return best_value if maximizing else -best_value

def policy(board: list[list[int]]) -> int:
    # Check for immediate win
    for col in range(7):
        if board[0][col] != 0:
            continue
        row = get_available_row(board, col)
        new_board = [row.copy() for row in board]
        new_board[row][col] = 1
        if check_win(new_board, row, col, 1):
            return col

    best_col = -1
    best_val = float('-inf')
    alpha = float('-inf')
    beta = float('inf')

    for col in [3, 2, 4, 1, 5, 0, 6]:
        if board[0][col] != 0:
            continue
        row = get_available_row(board, col)
        new_board = [x.copy() for x in board]
        new_board[row][col] = 1

        if check_win(new_board, row, col, 1):
            return col

        value = alpha_beta(new_board, alpha, beta, False, 4)
        if value > best_val:
            best_val = value
            best_col = col
        alpha = max(alpha, best_val)
        if beta <= alpha:
            break

    return best_col

### Explanation
1. **check_win Function**: This function checks if placing a disc at a given position results in a win by checking horizontal, vertical, and both diagonal directions for four consecutive discs of the same value.
2. **get_available_row Function**: This helper function finds the first available row in a column from bottom to top where a disc can be placed.
3. **evaluate Function**: This function provides a heuristic score based on the number of potential three-in-a-row lines for the player minus the opponent's potential three-in-a-row lines. This heuristic helps in evaluating the board state when a terminal state is reached.
4. **alpha_beta Function**: This function implements the Minimax algorithm with Alpha-Beta pruning. It recursively evaluates possible moves, alternating between maximizing (player's move) and minimizing (opponent's move) phases, while pruning less promising branches to save computation.
5. **policy Function**: This is the main function that first checks for immediate winning moves. If none are found, it uses the Alpha-Beta pruning function to determine the best move based on the heuristic evaluation and recursive search.

This approach ensures that the AI efficiently finds the optimal move by combining immediate win checks with a depth-limited Alpha-Beta search, balancing between computational efficiency and strategic depth.
