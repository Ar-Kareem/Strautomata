```python
import random

def policy(board: list[list[int]]) -> int:
    """
    Chooses a column index to drop the disc based on the Connect 4 board.

    Args:
        board: A 6x7 grid of ints representing the Connect 4 board.
               0  -> empty cell.
               1  -> your disc (current player).
               -1 -> opponent's disc.

    Returns:
        An integer `col` in the range `0`-`6` representing the chosen column.
    """

    def is_valid_move(col: int) -> bool:
        """Checks if a move is valid."""
        if not (0 <= col <= 6):
            return False
        return board[0][col] == 0

    def make_move(board: list[list[int]], col: int, player: int) -> list[list[int]]:
        """Makes a move on a copy of the board."""
        new_board = [row[:] for row in board]
        for row in range(5, -1, -1):
            if new_board[row][col] == 0:
                new_board[row][col] = player
                return new_board
        return None  # Column is full

    def check_win(board: list[list[int]], player: int) -> bool:
        """Checks if the given player has won."""
        # Check horizontal
        for row in range(6):
            for col in range(4):
                if (board[row][col] == player and
                        board[row][col + 1] == player and
                        board[row][col + 2] == player and
                        board[row][col + 3] == player):
                    return True

        # Check vertical
        for row in range(3):
            for col in range(7):
                if (board[row][col] == player and
                        board[row + 1][col] == player and
                        board[row + 2][col] == player and
                        board[row + 3][col] == player):
                    return True

        # Check positive diagonal
        for row in range(3):
            for col in range(4):
                if (board[row][col] == player and
                        board[row + 1][col + 1] == player and
                        board[row + 2][col + 2] == player and
                        board[row + 3][col + 3] == player):
                    return True

        # Check negative diagonal
        for row in range(3):
            for col in range(3, 7):
                if (board[row][col] == player and
                        board[row + 1][col - 1] == player and
                        board[row + 2][col - 2] == player and
                        board[row + 3][col - 3] == player):
                    return True

        return False

    # 1. Check for winning move
    for col in range(7):
        if is_valid_move(col):
            temp_board = make_move(board, col, 1)
            if temp_board and check_win(temp_board, 1):
                return col

    # 2. Block opponent's winning move
    for col in range(7):
        if is_valid_move(col):
            temp_board = make_move(board, col, -1)
            if temp_board and check_win(temp_board, -1):
                return col

    # 3. Play center column
    if is_valid_move(3):
        return 3

    # 4. Play a random valid move
    valid_moves = [col for col in range(7) if is_valid_move(col)]
    if valid_moves:
        return random.choice(valid_moves)

    # Should never happen, but handle the case where no move is possible
    return -1
```
