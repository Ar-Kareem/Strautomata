import random

def policy(board: list[list[int]]) -> int:
    """
    Chooses a column to drop a disc in Connect 4.
    """
    def is_valid_move(board, col):
        """Checks if a move is valid."""
        return board[0][col] == 0

    def get_next_open_row(board, col):
        """Gets the next open row in a column."""
        for r in range(5, -1, -1):
            if board[r][col] == 0:
                return r
        return -1  # Column is full

    def check_win(board, player):
        """Checks for a win for the given player."""
        # Horizontal
        for c in range(4):
            for r in range(6):
                if board[r][c] == player and board[r][c + 1] == player and board[r][c + 2] == player and board[r][c + 3] == player:
                    return True

        # Vertical
        for c in range(7):
            for r in range(3):
                if board[r][c] == player and board[r + 1][c] == player and board[r + 2][c] == player and board[r + 3][c] == player:
                    return True

        # Positive Diagonal
        for c in range(4):
            for r in range(3):
                if board[r][c] == player and board[r + 1][c + 1] == player and board[r + 2][c + 2] == player and board[r + 3][c + 3] == player:
                    return True

        # Negative Diagonal
        for c in range(4):
            for r in range(3, 6):
                if board[r][c] == player and board[r - 1][c + 1] == player and board[r - 2][c + 2] == player and board[r - 3][c + 3] == player:
                    return True

        return False

    # Check for winning move
    for col in range(7):
        if is_valid_move(board, col):
            row = get_next_open_row(board, col)
            # Simulate the move
            temp_board = [row[:] for row in board]  # Create a copy
            temp_board[row][col] = 1
            if check_win(temp_board, 1):
                return col

    # Block opponent's winning move
    for col in range(7):
        if is_valid_move(board, col):
            row = get_next_open_row(board, col)
            # Simulate the move for opponent
            temp_board = [row[:] for row in board]
            temp_board[row][col] = -1
            if check_win(temp_board, -1):
                return col
            
    # Play in the center column with a small probability
    if is_valid_move(board, 3) and random.random() < 0.3:
           return 3

    # Choose a random valid column
    valid_cols = [col for col in range(7) if is_valid_move(board, col)]
    if valid_cols:
        return random.choice(valid_cols)
    else:
        # Should not happen in a standard game but needed for robustness
        return -1
