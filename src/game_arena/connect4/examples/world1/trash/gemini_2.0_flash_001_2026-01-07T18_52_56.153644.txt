import random

def policy(board: list[list[int]]) -> int:
    """
    A targeting policy for a Connect 4 game on a 6x7 grid.

    Args:
        board: A 6x7 grid of ints with the following encoding:
            0  → empty cell.
            1  → your disc (current player).
            -1 → opponent's disc.

    Returns:
        An integer `col` in the range `0`-`6` such that the column is not full.
    """

    def is_valid_move(board, col):
        return board[0][col] == 0

    def get_next_available_row(board, col):
        for row in range(5, -1, -1):
            if board[row][col] == 0:
                return row
        return -1  # Column is full

    def check_for_win(board, player):
        # Check horizontal
        for row in range(6):
            for col in range(4):
                if (board[row][col] == player and
                        board[row][col+1] == player and
                        board[row][col+2] == player and
                        board[row][col+3] == player):
                    return True

        # Check vertical
        for row in range(3):
            for col in range(7):
                if (board[row][col] == player and
                        board[row+1][col] == player and
                        board[row+2][col] == player and
                        board[row+3][col] == player):
                    return True

        # Check positive diagonal
        for row in range(3):
            for col in range(4):
                if (board[row][col] == player and
                        board[row+1][col+1] == player and
                        board[row+2][col+2] == player and
                        board[row+3][col+3] == player):
                    return True

        # Check negative diagonal
        for row in range(3):
            for col in range(3, 7):
                if (board[row][col] == player and
                        board[row+1][col-1] == player and
                        board[row+2][col-2] == player and
                        board[row+3][col-3] == player):
                    return True

        return False

    # Check for winning move
    for col in range(7):
        if is_valid_move(board, col):
            row = get_next_available_row(board, col)
            board[row][col] = 1
            if check_for_win(board, 1):
                board[row][col] = 0
                return col
            board[row][col] = 0

    # Check for opponent's winning move and block it
    for col in range(7):
        if is_valid_move(board, col):
            row = get_next_available_row(board, col)
            board[row][col] = -1
            if check_for_win(board, -1):
                board[row][col] = 0
                return col
            board[row][col] = 0

    # Play in the center column if possible
    if is_valid_move(board, 3):
        return 3

    # Play a random valid move
    valid_moves = [col for col in range(7) if is_valid_move(board, col)]
    if valid_moves:
        return random.choice(valid_moves)

    # If no valid moves, return -1 (should not happen in a valid Connect 4 game)
    return -1
