import random

def policy(board: list[list[int]]) -> int:
    """
    A Connect 4 AI policy that prioritizes winning, blocking the opponent,
    and then placing discs in center columns.

    Args:
        board: A 6x7 grid representing the Connect 4 board.
               0: empty cell
               1: your disc (current player)
               -1: opponent's disc

    Returns:
        An integer (0-6) representing the chosen column to drop a disc.
    """
    ROWS = 6
    COLS = 7
    MY_PIECE = 1
    OPPONENT_PIECE = -1

    def is_valid_location(board, col):
        """Checks if a column is not full (i.e., the top row is empty)."""
        return board[0][col] == 0

    def get_next_open_row(board, col):
        """Finds the lowest empty row for a given column."""
        for r in range(ROWS - 1, -1, -1):
            if board[r][col] == 0:
                return r
        return -1 # Should not be reached if is_valid_location is checked

    def check_win(board, player_piece):
        """Checks if the given player has won on the current board."""
        # Check horizontal locations for win
        for c in range(COLS - 3):
            for r in range(ROWS):
                if board[r][c] == player_piece and \
                   board[r][c+1] == player_piece and \
                   board[r][c+2] == player_piece and \
                   board[r][c+3] == player_piece:
                    return True

        # Check vertical locations for win
        for c in range(COLS):
            for r in range(ROWS - 3):
                if board[r][c] == player_piece and \
                   board[r+1][c] == player_piece and \
                   board[r+2][c] == player_piece and \
                   board[r+3][c] == player_piece:
                    return True

        # Check positively sloped diagonals
        for c in range(COLS - 3):
            for r in range(ROWS - 3):
                if board[r][c] == player_piece and \
                   board[r+1][c+1] == player_piece and \
                   board[r+2][c+2] == player_piece and \
                   board[r+3][c+3] == player_piece:
                    return True

        # Check negatively sloped diagonals
        for c in range(COLS - 3):
            for r in range(3, ROWS): # Start at row 3 and go up to ROWS-1
                if board[r][c] == player_piece and \
                   board[r-1][c+1] == player_piece and \
                   board[r-2][c+2] == player_piece and \
                   board[r-3][c+3] == player_piece:
                    return True
        return False

    # Get all valid columns where a disc can be dropped
    valid_cols = [c for c in range(COLS) if is_valid_location(board, c)]

    # --- Strategy Steps ---

    # 1. Check for immediate win: If I can win on this turn, make that move.
    for col in valid_cols:
        row = get_next_open_row(board, col)
        # Create a deep copy of the board to simulate the move
        temp_board = [r[:] for r in board]
        temp_board[row][col] = MY_PIECE
        if check_win(temp_board, MY_PIECE):
            return col

    # 2. Block opponent's immediate win: If the opponent can win on their next turn, block them.
    for col in valid_cols:
        row = get_next_open_row(board, col)
        # Create a deep copy of the board to simulate the opponent's move
        temp_board = [r[:] for r in board]
        temp_board[row][col] = OPPONENT_PIECE
        if check_win(temp_board, OPPONENT_PIECE):
            return col

    # 3. Prioritize center columns: If no immediate wins or blocks, choose a column
    #    from the center outwards, as these offer more strategic opportunities.
    #    Order: 3 (center), 2, 4, 1, 5, 0, 6
    preferred_cols = [3, 2, 4, 1, 5, 0, 6]
    for col in preferred_cols:
        if col in valid_cols:
            return col

    # must always be an available move if policy is called.
    # However, as a fallback, if for some reason valid_cols is empty (e.g., board is full),
    # it would raise an error.

    # If the board is full and no one has won, it's a draw, and `policy` shouldn't be called.
