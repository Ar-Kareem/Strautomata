import random

def policy(board: list[list[int]]) -> int:
    """
    Chooses a column to drop a disc into based on the Connect 4 board.

    Args:
        board: A 6x7 grid of ints representing the Connect 4 board.
               0: empty, 1: player's disc, -1: opponent's disc.

    Returns:
        The column index (0-6) to drop the disc into.
    """

    # 1. Check for winning moves for myself
    for col in range(7):
        # Check if column is full
        if board[0][col] != 0:
            continue
        
        # Simulate placing a disc in the column
        row = -1
        for r in range(5, -1, -1):
            if board[r][col] == 0:
                row = r
                break
                
        temp_board = [row[:] for row in board]
        temp_board[row][col] = 1
        
        # Check for win
        if check_win(temp_board, 1):
            return col

    # 2. Check for blocking moves to prevent opponent from winning
    for col in range(7):
        # Check if column is full
        if board[0][col] != 0:
            continue
            
        # Simulate placing a disc in the column
        row = -1
        for r in range(5, -1, -1):
            if board[r][col] == 0:
                row = r
                break

        temp_board = [row[:] for row in board]
        temp_board[row][col] = -1

        # Check for opponent winning
        if check_win(temp_board, -1):
            return col

    # 3. If no winning or blocking moves, choose a random valid column
    valid_cols = [col for col in range(7) if board[0][col] == 0]
    if valid_cols:
        return random.choice(valid_cols)
    else:
        # Should not happen in a normal game, but return a value if the board is full
        return -1


def check_win(board: list[list[int]], player: int) -> bool:
    """Checks if the given player has won the game on the board."""

    # Check horizontal
    for row in range(6):
        for col in range(4):
            if (board[row][col] == player and
                board[row][col+1] == player and
                board[row][col+2] == player and
                board[row][col+3] == player):
                return True

    # Check vertical
    for row in range(3):
        for col in range(7):
            if (board[row][col] == player and
                board[row+1][col] == player and
                board[row+2][col] == player and
                board[row+3][col] == player):
                return True

    # Check positive diagonal
    for row in range(3):
        for col in range(4):
            if (board[row][col] == player and
                board[row+1][col+1] == player and
                board[row+2][col+2] == player and
                board[row+3][col+3] == player):
                return True

    # Check negative diagonal
    for row in range(3):
        for col in range(3, 7):
            if (board[row][col] == player and
                board[row+1][col-1] == player and
                board[row+2][col-2] == player and
                board[row+3][col-3] == player):
                return True

    return False
