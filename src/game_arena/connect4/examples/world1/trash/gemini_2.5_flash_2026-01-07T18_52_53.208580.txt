        score += 100     # An open 3-in-a-row threat
    elif window.count(player) == 2 and window.count(0) == 2:
        score += 10      # An open 2-in-a-row

    # Penalize opponent's potential lines. These are less severe than our own
    # because immediate opponent wins are handled separately.
    if window.count(opponent) == 3 and window.count(0) == 1:
        score -= 90 # Opponent has an open 3-in-a-row threat if not blocked
    elif window.count(opponent) == 2 and window.count(0) == 2:
        score -= 5 # Opponent has an open 2-in-a-row

    return score

def score_position_for_player(board: list[list[int]], player: int) -> int:
    """
    Calculates a heuristic score for the entire board for a given player.
    Higher score indicates a better position for the player.
    """
    score = 0

    # Prioritize discs in the center column as they offer more connection opportunities
    center_column = [board[r][COLS // 2] for r in range(ROWS)]
    center_count = center_column.count(player)
    score += center_count * 3

    # Score Horizontal lines
    for r in range(ROWS):
        row_array = board[r]
        for c in range(COLS - 3):
            window = row_array[c:c+4]
            score += evaluate_window(window, player)

    # Score Vertical lines
    for c in range(COLS):
        col_array = [board[r][c] for r in range(ROWS)]
        for r in range(ROWS - 3):
            window = col_array[r:r+4]
            score += evaluate_window(window, player)

    # Score Positive Diagonals
    for r in range(ROWS - 3):
        for c in range(COLS - 3):
            window = [board[r+i][c+i] for i in range(4)]
            score += evaluate_window(window, player)

    # Score Negative Diagonals
    for r in range(3, ROWS):
        for c in range(COLS - 3):
            window = [board[r-i][c+i] for i in range(4)]
            score += evaluate_window(window, player)

    return score


def policy(board: list[list[int]]) -> int:
    """
    Defines the AI's targeting policy for Connect 4.
    The AI seeks to win, block opponent wins, avoid setting up opponents,
    and then optimize its position based on a heuristic score.
    """
    valid_moves = [c for c in range(COLS) if is_valid_location(board, c)]

    # If no valid moves, which should not happen in a typical Connect 4 game
    # (caller handles full board), return a random valid column if any, else 3.
    if not valid_moves:
        return 3 if is_valid_location(board, 3) else random.randint(0, COLS - 1)


    # 1. Check for immediate win (my turn)
    for col in valid_moves:
        temp_board = copy.deepcopy(board)
        r = get_next_open_row(temp_board, col)
        drop_piece(temp_board, r, col, PLAYER_ME)
        if check_winning_move(temp_board, PLAYER_ME):
            return col

    # 2. Check for immediate opponent win (block opponent)
    for col in valid_moves:
        temp_board = copy.deepcopy(board)
        r = get_next_open_row(temp_board, col)
        drop_piece(temp_board, r, col, PLAYER_OPPONENT) # Simulate opponent's potential move
        if check_winning_move(temp_board, PLAYER_OPPONENT):
            return col

    # 3. Evaluate other potential moves using a 1-ply lookahead with a 2-ply opponent trap check
    possible_moves_scores = {}

    for col in valid_moves:
        temp_board_mine_moved = copy.deepcopy(board)
        r_mine = get_next_open_row(temp_board_mine_moved, col)
        drop_piece(temp_board_mine_moved, r_mine, col, PLAYER_ME)

        # Check if this move allows the opponent to win on their *next* turn
        opponent_can_win_next_turn = False
        valid_moves_after_mine = [c_opp for c_opp in range(COLS) if is_valid_location(temp_board_mine_moved, c_opp)]
        
        # If the board becomes full (or nearly full) after my move and I didn't win,
        # still check for opponent's win on their hypothetical move.
        if valid_moves_after_mine:
            for opp_col in valid_moves_after_mine:
                temp_board_after_opp_move = copy.deepcopy(temp_board_mine_moved)
                r_opp = get_next_open_row(temp_board_after_opp_move, opp_col)
                drop_piece(temp_board_after_opp_move, r_opp, opp_col, PLAYER_OPPONENT)
                if check_winning_move(temp_board_after_opp_move, PLAYER_OPPONENT):
                    opponent_can_win_next_turn = True
                    break # Opponent found a winning move, so this 'col' is bad for me

        if opponent_can_win_next_turn:
            # Heavily penalize moves that lead to an immediate opponent win
            possible_moves_scores[col] = -1_000_000_000
        else:
            # If not a losing move, evaluate the resulting board state for PLAYER_ME
            score = score_position_for_player(temp_board_mine_moved, PLAYER_ME)
            possible_moves_scores[col] = score

    # Find the best move(s) based on computed scores
    max_score = -float('inf')
    best_cols = []

    # If all valid moves lead to a losing state (score of -1_000_000_000),
    # then max_score might remain -inf or be a very low number.
    # In such a scenario, we'll just pick *any* valid move.
    if not possible_moves_scores:
        # This case should ideally be caught by previous valid_moves check,
        # but as a safeguard, pick a center column or random.
        return 3 if is_valid_location(board, 3) else random.choice(valid_moves)

    for col, score in possible_moves_scores.items():
        if score > max_score:
            max_score = score
            best_cols = [col]
        elif score == max_score:
            best_cols.append(col)

    # Prioritize center columns among equally good moves to maximize flexibility
    # Sort by absolute distance from the center column (column 3)
    best_cols.sort(key=lambda c: abs(c - COLS // 2))

    # Return a random choice from the best-scoring, center-biased moves
    return random.choice(best_cols)
