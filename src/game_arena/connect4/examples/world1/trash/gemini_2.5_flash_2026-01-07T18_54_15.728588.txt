        start_r, start_c = r + offset, c - offset
        # Ensure the potential 4-disc segment is entirely within the board.
        # For this diagonal, start_c must be at least 3 to have 3 columns to its left.
        if 0 <= start_r <= ROWS - 4 and 3 <= start_c <= COLS - 1:
            if all(board[start_r + i][start_c - i] == player for i in range(4)):
                return True

    return False


def policy(board: list[list[int]]) -> int:
    """
    Implements a targeting policy for a Connect 4 AI.
    The AI prioritizes:
    1. Winning in the current turn.
    2. Blocking the opponent from winning in their next turn.
    3. Making a strategic move, favoring center columns.
    """
    my_player = 1
    opp_player = -1

    # Get a list of all columns that are currently valid moves
    valid_cols = [col for col in range(COLS) if is_valid_move(board, col)]

    # If, hypothetically, there are no valid moves left (board is full)
    # this condition should ideally not be met in a normal game flow.
    if not valid_cols:
        # As a fallback, if no columns are available, return the first one (which implies an error)
        # or raise an exception. For competitive arena, an int is expected.
        # This implies a game over state, which the arena should handle.
        return 3 # Return a central column, though it's invalid.

    # --- Strategy Step 1: Check for immediate win opportunities ---
    # Iterate through all possible moves (valid columns)
    for col in valid_cols:
        r = get_next_open_row(board, col)
        if r != -1: # Ensure a valid row was found (should always be true for valid_cols)
            # Simulate placing my disc in this column
            temp_board = copy.deepcopy(board) # Create a deep copy to avoid modifying the actual board
            temp_board[r][col] = my_player
            # Check if this move results in a win for me
            if check_potential_win(temp_board, r, col, my_player):
                return col  # If I can win, take the win immediately!

    # --- Strategy Step 2: Block opponent's immediate win opportunities ---
    # Iterate through all possible moves again
    for col in valid_cols:
        r = get_next_open_row(board, col)
        if r != -1:
            # Simulate opponent placing their disc in this column
            temp_board = copy.deepcopy(board)
            temp_board[r][col] = opp_player # Simulate opponent's move
            # Check if this move would result in a win for the opponent
            if check_potential_win(temp_board, r, col, opp_player):
                return col  # If opponent can win, block them!

    # --- Strategy Step 3: Make a strategic move (heuristic-based) ---
    # If no immediate wins or blocks are needed, choose a column based on a heuristic.
    # Prioritizing center columns is generally a good strategy in Connect 4,
    # as they offer more potential lines (horizontal, vertical, both diagonals).
    strategic_cols = [3, 2, 4, 1, 5, 0, 6] # Order of preference for columns

    # Filter the strategic columns to only include those that are currently valid moves.
    prioritized_valid_cols = [col for col in strategic_cols if col in valid_cols]

    # Return the first available column from the prioritized list.
    # This acts as a good default move if no immediate threats or wins are present.
    if prioritized_valid_cols:
        return prioritized_valid_cols[0]
    
    # Fallback: This case should only be reached if `valid_cols` is empty,
    # which is already handled above. It's a safety net.
    return valid_cols[0] # Return the first valid column if prioritized_valid_cols is empty for some reason.
