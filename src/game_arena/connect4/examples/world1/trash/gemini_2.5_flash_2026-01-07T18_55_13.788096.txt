    1. Iterate `COLS` times, calling `make_move` and `check_win`. -> `COLS * (ROWS*COLS)` approx.
    2. Iterate `COLS` times, calling `make_move` and `check_win`. -> `COLS * (ROWS*COLS)` approx.
    3. Iterate `COLS` times (my potential moves):
        `make_move`. `O(1)` basically.
        `evaluate_board`. `O(ROWS*COLS)`.
        Inner loop for opponent moves `COLS` times:
            `make_move`. `O(1)` basically.
            `check_win`. `O(ROWS*COLS)`.
            `evaluate_board`. `O(ROWS*COLS)`.
        Total for step 3: `COLS * (ROWS*COLS + COLS * (ROWS*COLS)) = COLS^2 * ROWS * COLS = COLS^3 * ROWS`.
    `7^3 * 6 = 343 * 6 = 2058`. Operations involving `copy` and `list slicing` might add constants.

```python
import copy

# Board dimensions
ROWS = 6
COLS = 7

# Scores for evaluation
# These values are chosen to prioritize actions:
# Winning immediately > Blocking opponent's win > Creating 3-in-a-row threat > Creating 2-in-a-row threat
# Opponent threats are penalized more than my opportunities are rewarded to encourage defensive play
IN_A_ROW_4_SCORE = 1000000  # Score for winning
IN_A_ROW_3_SCORE = 100       # Score for my open 3-in-a-row
IN_A_ROW_2_SCORE = 10        # Score for my open 2-in-a-row
OPPONENT_3_THREAT_PENALTY = 500 # Penalty for opponent's open 3-in-a-row (high priority to block)
OPPONENT_2_THREAT_PENALTY = 50 # Penalty for opponent's open 2-in-a-row
CENTER_PREFERENCE = 5        # Slight bonus for discs in the center column

def policy(board: list[list[int]]) -> int:

    def is_valid_move(board_state, col):
        """Checks if a column is a valid move (not full)."""
        return 0 <= col < COLS and board_state[0][col] == 0

    def get_next_open_row(board_state, col):
        """Returns the lowest available row in a given column."""
        for r in range(ROWS - 1, -1, -1):
            if board_state[r][col] == 0:
                return r
        return -1 # Should not be reached if is_valid_move returns True

    def make_move(board_state, col, player):
        """Returns a new board state after placing a disc."""
        new_board = [row[:] for row in board_state] # Create a deep copy of the board
        row = get_next_open_row(new_board, col)
        if row != -1: # Ensure move is valid before placing
            new_board[row][col] = player
        return new_board

    def check_win(board_state, player):
        """Checks if the given player has won on the current board."""
        # Check horizontal
        for r in range(ROWS):
            for c in range(COLS - 3):
                if all(board_state[r][c + i] == player for i in range(4)):
                    return True
        # Check vertical
        for r in range(ROWS - 3):
            for c in range(COLS):
                if all(board_state[r + i][c] == player for i in range(4)):
                    return True
        # Check positive diagonal
        for r in range(ROWS - 3):
            for c in range(COLS - 3):
                if all(board_state[r + i][c + i] == player for i in range(4)):
                    return True
        # Check negative diagonal
        for r in range(3, ROWS): # Start from r=3 up to ROWS-1
            for c in range(COLS - 3):
                 if all(board_state[r - i][c + i] == player for i in range(4)):
                    return True
        return False

    def evaluate_window(window, player_id):
        """Evaluates a 4-cell window for potential scores and threats."""
        score = 0
        my_count = window.count(player_id)
        opponent_count = window.count(-player_id)
        empty_count = window.count(0)

        # My opportunities
        if my_count == 4:
            score += IN_A_ROW_4_SCORE
        elif my_count == 3 and empty_count == 1:
            score += IN_A_ROW_3_SCORE
        elif my_count == 2 and empty_count == 2:
            score += IN_A_ROW_2_SCORE

        # Opponent threats (penalized more severely)
        if opponent_count == 4:
            score -= IN_A_ROW_4_SCORE * 10 # Opponent wins, this is devastating
        elif opponent_count == 3 and empty_count == 1:
            score -= OPPONENT_3_THREAT_PENALTY
        elif opponent_count == 2 and empty_count == 2:
            score -= OPPONENT_2_THREAT_PENALTY

        return score

    def evaluate_board(board_state, player_id):
        """Evaluates the entire board for a given player_id."""
        score = 0

        # Prioritize center column (more connecting opportunities)
        center_column_idx = COLS // 2
        for r in range(ROWS):
            if board_state[r][center_column_idx] == player_id:
                score += CENTER_PREFERENCE

        # Evaluate horizontal windows
        for r in range(ROWS):
            for c in range(COLS - 3):
                window = board_state[r][c:c+4]
                score += evaluate_window(window, player_id)

        # Evaluate vertical windows
        for c in range(COLS):
            for r in range(ROWS - 3):
                window = [board_state[r + i][c] for i in range(4)]
                score += evaluate_window(window, player_id)

        # Evaluate positive diagonal windows
        for r in range(ROWS - 3):
            for c in range(COLS - 3):
                window = [board_state[r + i][c + i] for i in range(4)]
                score += evaluate_window(window, player_id)

        # Evaluate negative diagonal windows
        for r in range(3, ROWS): # Start from r=3 up to ROWS-1
            for c in range(COLS - 3):
                window = [board_state[r - i][c + i] for i in range(4)]
                score += evaluate_window(window, player_id)

        return score

    # --- Policy logic ---

    valid_moves = [col for col in range(COLS) if is_valid_move(board, col)]

    # If no valid moves, this function should technically not be called (game already ended)
    # But as a failsafe, return a default if somehow no moves are left.
    if not valid_moves:
        return COLS // 2 

    # 1. Check for immediate win for self
    for col in valid_moves:
        temp_board = make_move(board, col, 1)
        if check_win(temp_board, 1):
            return col

    # 2. Check for opponent's immediate win and block it
    for col in valid_moves:
        temp_board = make_move(board, col, -1) # Simulate opponent's move
        if check_win(temp_board, -1):
            return col # Block the opponent's winning move

    # 3. Use 1-ply minimax-like evaluation (consider my move, then opponent's best response)
    best_score = -float('inf')
    # Default to center column if all else is equal, or if no better move is found.
    best_col = COLS // 2 if COLS // 2 in valid_moves else valid_moves[0] 

    for col in valid_moves:
        # Simulate my move
        my_temp_board = make_move(board, col, 1)
        
        # Calculate my immediate board score after my hypothetical move
        current_move_evaluation = evaluate_board(my_temp_board, 1)

        # Assume opponent plays optimally to minimize my score (maximize their score)
        min_score_for_me_after_opponent_turn = float('inf') # Initialize with a very high value
        opponent_valid_moves = [c for c in range(COLS) if is_valid_move(my_temp_board, c)]
        
        # If there are no opponent moves, then a draw or I won (already checked above), so no penalty from opponent.
        if opponent_valid_moves:
            for opp_col in opponent_valid_moves:
                opp_response_board = make_move(my_temp_board, opp_col, -1)
                
                if check_win(opp_response_board, -1):
                    # Opponent wins if I make this move and they respond with opp_col.
                    # This path is extremely bad for me.
                    min_score_for_me_after_opponent_turn = -IN_A_ROW_4_SCORE * 10 
                    break # Opponent found a winning move, no need to check other opponent columns.
                
                # Evaluate the board from my perspective after opponent's potential best move
                score_if_this_opp_move = evaluate_board(opp_response_board, 1)
                min_score_for_me_after_opponent_turn = min(min_score_for_me_after_opponent_turn, score_if_this_opp_move)
            
            # The score for my current move is the score after my move, considering the worst opponent response.
            # If opponent can win, my `current_move_evaluation` is superseded by the `min_score_for_me_after_opponent_turn`.
            current_move_evaluation = min_score_for_me_after_opponent_turn

        if current_move_evaluation > best_score:
            best_score = current_move_evaluation
            best_col = col
        # Tie-breaking: if scores are equal, prefer the center or smaller index columns.
        # This implicitly gives priority to smaller indices because `valid_moves` goes from left to right.
        # Unless `best_col` was initially set to a center, which is good.
        elif current_move_evaluation == best_score:
            if abs(col - COLS // 2) < abs(best_col - COLS // 2):
                best_col = col

    return best_col
```
