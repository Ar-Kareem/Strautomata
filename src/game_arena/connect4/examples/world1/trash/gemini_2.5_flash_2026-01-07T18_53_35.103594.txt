def get_next_empty_row(board: list[list[int]], col: int) -> int | None:
    """
    Finds the lowest empty row in a given column.
    Returns the row index or None if the column is full.
    """
    for r in range(ROW_COUNT - 1, -1, -1):  # Iterate from bottom to top
        if board[r][col] == 0:
            return r
    return None  # Column is full

    """
    Checks if a column is a valid move (i.e., not full).

    return board[0][col] == 0

def drop_disc(board: list[list[int]], col: int, player: int) -> list[list[int]]:
    """
    Simulates dropping a disc into a column and returns the new board state.
    Assumes the move is valid.
    """
    new_board = [row[:] for row in board]  # Create a deep copy of the board
    row = get_next_empty_row(new_board, col)
    if row is not None:

    """
    Evaluates the board state using a heuristic approach.
    Positive scores favor the current player, negative favor the opponent.
    """
    score = 0
    opponent = -player

    # Score for current player's potential lines
    score += count_potential_lines(board, player, 3) * 100  # High value for three-in-a-row

    # Subtract score for opponent's potential lines (defensive play)

    score -= count_potential_lines(board, opponent, 2) * 5  # Blocking opponent's two-in-a-row

def policy(board: list[list[int]]) -> int:
    """
    Determines the best column to drop a disc for the current player.
    """
    # 1. Check for immediate win (my turn)
    for col in range(COLUMN_COUNT):
        if is_valid_move(board, col):
            next_board = drop_disc(board, col, PLAYER_ME)
            if check_win(next_board, PLAYER_ME):
                return col

    # 2. Check for immediate opponent win and block it
    for col in range(COLUMN_COUNT):
        if is_valid_move(board, col):
            # Simulate opponent's move in this column
            next_board_opponent = drop_disc(board, col, PLAYER_OPPONENT)
            if check_win(next_board_opponent, PLAYER_OPPONENT):
                return col

    # 3. Evaluate other moves using a heuristic
    best_score = -float('inf')
    best_col = -1
    
    # Order columns to prioritize center (e.g., 3, 2, 4, 1, 5, 0, 6)
    # This helps with tie-breaking and generally leads to stronger play
    column_order = [3, 2, 4, 1, 5, 0, 6] 

    for col in column_order:
        if is_valid_move(board, col):
            current_move_score = 0
            
            # Simulate my move in the current column
            board_after_my_move = drop_disc(board, col, PLAYER_ME)
            
            # Critical defensive check: Does this move directly allow the opponent to win on their next turn?
            opponent_can_win_after_my_move = False
            for opp_col_option in range(COLUMN_COUNT):
                if is_valid_move(board_after_my_move, opp_col_option):
                    temp_board_after_opp = drop_disc(board_after_my_move, opp_col_option, PLAYER_OPPONENT)
                    if check_win(temp_board_after_opp, PLAYER_OPPONENT):
                        opponent_can_win_after_my_move = True
                        break # Found a way for opponent to win, so this 'col' is bad for me

            if opponent_can_win_after_my_move:
                current_move_score = -1_000_000 # Severely penalize moves that directly lead to opponent's win
            else:
                # If not a losing move, evaluate its heuristic potential
                current_move_score = evaluate_board_heuristic(board_after_my_move, PLAYER_ME)
                
                # Add bonuses for central column play, as it generally offers more opportunities
                if col == 3: current_move_score += 5
                elif col == 2 or col == 4: current_move_score += 3
                elif col == 1 or col == 5: current_move_score += 1
            
            if current_move_score > best_score:
                best_score = current_move_score
                best_col = col

    # or if all moves are equally bad (e.g., all lead to opponent winning next turn, which is handled

    # For robustnes, if `best_col` is still -1, fall back to the first available column.
