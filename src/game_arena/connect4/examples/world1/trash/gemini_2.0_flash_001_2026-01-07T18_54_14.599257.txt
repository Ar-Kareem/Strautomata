```python
import random

def policy(board: list[list[int]]) -> int:
    """
    Chooses a column to play in Connect 4.

    Args:
        board: A 6x7 grid representing the Connect 4 board.

    Returns:
        The column index (0-6) to play in.
    """

    def is_valid_move(col: int) -> bool:
        """Checks if a column is a valid move."""
        return board[0][col] == 0

    def make_move(board: list[list[int]], col: int, player: int) -> list[list[int]]:
        """Makes a move on the board and returns the new board."""
        new_board = [row[:] for row in board]
        for row in range(5, -1, -1):
            if new_board[row][col] == 0:
                new_board[row][col] = player
                break
        return new_board

    def check_win(board: list[list[int]], player: int) -> bool:
        """Checks if the given player has won the game."""
        # Check horizontal
        for row in range(6):
            for col in range(4):
                if (board[row][col] == player and
                        board[row][col + 1] == player and
                        board[row][col + 2] == player and
                        board[row][col + 3] == player):
                    return True

        # Check vertical
        for row in range(3):
            for col in range(7):
                if (board[row][col] == player and
                        board[row + 1][col] == player and
                        board[row + 2][col] == player and
                        board[row + 3][col] == player):
                    return True

        # Check positive diagonal
        for row in range(3):
            for col in range(4):
                if (board[row][col] == player and
                        board[row + 1][col + 1] == player and
                        board[row + 2][col + 2] == player and
                        board[row + 3][col + 3] == player):
                    return True

        # Check negative diagonal
        for row in range(3):
            for col in range(3, 7):
                if (board[row][col] == player and
                        board[row + 1][col - 1] == player and
                        board[row + 2][col - 2] == player and
                        board[row + 3][col - 3] == player):
                    return True

        return False
    

    # 1. Check for winning moves
    for col in range(7):
        if is_valid_move(col):
            temp_board = make_move(board, col, 1)
            if check_win(temp_board, 1):
                return col

    # 2. Check for opponent's winning moves and block them
    for col in range(7):
        if is_valid_move(col):
            temp_board = make_move(board, col, -1)
            if check_win(temp_board, -1):
                return col

    # 3. Play center column if possible
    if is_valid_move(3):
        return 3

    # 4. Play a random valid move
    valid_moves = [col for col in range(7) if is_valid_move(col)]
    if valid_moves:
        return random.choice(valid_moves)

    # If no valid moves, return -1 (should not happen in a normal game)
    return -1
```
