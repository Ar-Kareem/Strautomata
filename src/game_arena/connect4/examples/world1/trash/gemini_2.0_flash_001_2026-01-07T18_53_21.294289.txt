import random

def policy(board: list[list[int]]) -> int:
    """
    A targeting policy for a Connect 4 game on a 6x7 grid.

    Args:
        board: A 6x7 grid of ints representing the Connect 4 board.
            0  → empty cell.
            1  → your disc (current player).
            -1 → opponent's disc.

    Returns:
        An integer col in the range 0-6 such that the column is not full.
    """

    def is_valid_move(board, col):
        """Checks if a move is valid."""
        if not (0 <= col <= 6):
            return False
        if board[0][col] != 0:
            return False
        return True

    def get_next_available_row(board, col):
        """Gets the next available row in a column."""
        for row in range(5, -1, -1):
            if board[row][col] == 0:
                return row
        return -1  # Column is full

    def check_winning_move(board, col, player):
        """Checks if a move wins the game."""
        row = get_next_available_row(board, col)
        if row == -1:
            return False  # Column is full, move is invalid

        # Check horizontal
        count = 0
        for c in range(7):
            if board[row][c] == player:
                count += 1
                if count >= 4:
                    return True
            else:
                count = 0

        # Check vertical
        count = 0
        for r in range(6):
            if board[r][col] == player:
                count += 1
                if count >= 4:
                    return True
            else:
                count = 0

        # Check diagonal (positive slope)
        count = 0
        for i in range(-3, 4):
            r = row + i
            c = col + i
            if 0 <= r < 6 and 0 <= c < 7:
                if board[r][c] == player:
                    count += 1
                    if count >= 4:
                        return True
                else:
                    count = 0

        # Check diagonal (negative slope)
        count = 0
        for i in range(-3, 4):
            r = row - i
            c = col + i
            if 0 <= r < 6 and 0 <= c < 7:
                if board[r][c] == player:
                    count += 1
                    if count >= 4:
                        return True
                else:
                    count = 0

        return False

    # 1. Check for winning move
    for col in range(7):
        if is_valid_move(board, col):
            if check_winning_move(board, col, 1):
                return col

    # 2. Block opponent's winning move
    for col in range(7):
        if is_valid_move(board, col):
            # Temporarily place opponent's move to check if it's a winning move.

            row = get_next_available_row(board, col)  #row to check.

            temp_board = [row[:] for row in board]  #deepcopy.

            temp_board[row][col] = -1 #temporarily assign to opponent's move.

            if check_winning_move(temp_board, col, -1): #If opponent placing at col would win.
                return col

    # 3. Play center column
    if is_valid_move(board, 3):
        return 3

    # 4. Play a random valid move
    valid_moves = [col for col in range(7) if is_valid_move(board, col)]
    if valid_moves:
        return random.choice(valid_moves)

    return -1 # should not reach here
