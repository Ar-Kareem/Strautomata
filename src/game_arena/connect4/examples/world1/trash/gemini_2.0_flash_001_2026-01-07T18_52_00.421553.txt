import random

def policy(board: list[list[int]]) -> int:
    """
    Selects a column to drop a disc in Connect 4 using a simple policy.

    The policy prioritizes winning moves, then blocking opponent's winning moves,
    and finally chooses a random valid move.

    Args:
        board: A 6x7 grid representing theConnect 4 board. 0=empty, 1=me, -1=opponent.

    Returns:
        An integer representing the column to drop the disc (0-6).
    """

    def is_valid_move(board, col):
        """Checks if a move (column) is valid on the given board."""
        return board[0][col] == 0

    def get_next_available_row(board, col):
        """Returns the next available row in a column, or -1 if the column is full."""
        for row in range(5, -1, -1):
            if board[row][col] == 0:
                return row
        return -1

    def check_win(board, player):
        """Checks if the given player has won the game."""
        # Check horizontal
        for row in range(6):
            for col in range(4):
                if (board[row][col] == player and
                        board[row][col + 1] == player and
                        board[row][col + 2] == player and
                        board[row][col + 3] == player):
                    return True

        # Check vertical
        for row in range(3):
            for col in range(7):
                if (board[row][col] == player and
                        board[row + 1][col] == player and
                        board[row + 2][col] == player and
                        board[row + 3][col] == player):
                    return True

        # Check positive diagonal
        for row in range(3):
            for col in range(4):
                if (board[row][col] == player and
                        board[row + 1][col + 1] == player and
                        board[row + 2][col + 2] == player and
                        board[row + 3][col + 3] == player):
                    return True

        # Check negative diagonal
        for row in range(3):
            for col in range(3, 7):
                if (board[row][col] == player and
                        board[row + 1][col - 1] == player and
                        board[row + 2][col - 2] == player and
                        board[row + 3][col - 3] == player):
                    return True

        return False

    # Check for winning moves
    for col in range(7):
        if is_valid_move(board, col):
            row = get_next_available_row(board, col)
            temp_board = [row[:] for row in board]  # Create a copy
            temp_board[row][col] = 1
            if check_win(temp_board, 1):  # I am player 1
                return col

    # Check for blocking moves
    for col in range(7):
        if is_valid_move(board, col):
            row = get_next_available_row(board, col)
            temp_board = [row[:] for row in board]  # Create a copy
            temp_board[row][col] = -1
            if check_win(temp_board, -1):  # Opponent is player -1
                return col

    # Choose a random valid move
    valid_moves = [col for col in range(7) if is_valid_move(board, col)]
    if valid_moves:
        return random.choice(valid_moves)
    else:
        # Should never happen in a valid game state.  Return arbitrarily.
        return 0
