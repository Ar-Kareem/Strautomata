ROWS = 6
COLS = 7

OPPONENT = -1 # Opponent's disc

def get_next_open_row(board, col):
    """
    Returns the lowest available row in a given column, or -1 if the column is full.
    """
    for r in range(ROWS - 1, -1, -1):
        if board[r][col] == EMPTY:
            return r
    return -1

def is_valid_move(board, col):
    """
    Checks if a column is a valid move (not full).

    return 0 <= col < COLS and board[0][col] == EMPTY # Check if top row is empty

def drop_disc(board, col, player):
    """
    Returns a new board state after dropping a disc in the specified column.
    Does not modify the original board.
    """
    new_board = [row[:] for row in board] # Deep copy of the board
    row = get_next_open_row(new_board, col)
    if row != -1:
        new_board[row][col] = player
    return new_board

def check_win(board, player):
    """
    Checks if the given player has won on the current board.
    """
    # Check horizontal
    for r in range(ROWS):
        for c in range(COLS - 3):
            if all(board[r][c+i] == player for i in range(4)):
                return True

    # Check vertical
    for c in range(COLS):
        for r in range(ROWS - 3):
            if all(board[r+i][c] == player for i in range(4)):
                return True

            if all(board[r-i][c+i] == player for i in range(4)):
