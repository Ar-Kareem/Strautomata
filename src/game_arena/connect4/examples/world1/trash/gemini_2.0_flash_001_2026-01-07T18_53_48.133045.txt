```python
import random

def policy(board: list[list[int]]) -> int:
    """
    Chooses a valid column index for a Connect 4 move.

    Args:
        board: A 6x7 grid representing the Connect 4 board.

    Returns:
        An integer representing the column index (0-6) for the move.
    """

    def is_valid_move(board, col):
        """Checks if a move in the given column is valid."""
        return board[0][col] == 0

    def get_possible_moves(board):
        """Returns a list of valid column indices."""
        possible_moves = []
        for col in range(7):
            if is_valid_move(board, col):
                possible_moves.append(col)
        return possible_moves

    possible_moves = get_possible_moves(board)

    if not possible_moves:
        return -1  # Should not happen in a standard game, but handle it anyway

    # Simple strategy: Choose a random valid move
    return random.choice(possible_moves)
```
