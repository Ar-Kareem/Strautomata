To solve this problem, we need to implement a move-selection policy for the game of Pentago. The goal is to determine the best possible move for the player to either win immediately or set up a strong position for future moves. The solution involves checking all possible moves for an immediate win and, if none are found, selecting a move based on a heuristic that evaluates potential future winning opportunities.

### Approach
1. **Check for Immediate Winning Moves**: Iterate through all possible moves (placing a marble in an empty cell and rotating any quadrant in either direction) and check if any move results in an immediate win (5 consecutive marbles in a row, column, or diagonal).
2. **Heuristic Evaluation for Non-Winning Moves**: If no immediate win is found, evaluate each move based on a heuristic score that considers the potential for creating 4 consecutive marbles, which can lead to a win in the next turn.

### Solution Code
```python
import numpy as np
from numpy.lib.stride_tricks import sliding_window_view

def check_win(board):
    # Check rows using sliding windows
    rows_windows = sliding_window_view(board, (1, 5))
    if np.any(np.all(rows_windows == 1, axis=1)):
        return True

    # Check columns
    cols_windows = sliding_window_view(board, (5, 1))
    if np.any(np.all(cols_windows == 1, axis=0)):
        return True

    # Check main diagonals (top-left to bottom-right)
    for i in range(0, 2):
        for j in range(0, 2):
            window = board[i:i+5, j:j+5]
            if np.all(window.diagonal() == 1):
                return True

    # Check anti-diagonals (top-right to bottom-left)
    for i in range(0, 2):
        for j in range(2, 5):
            if j >= 4:
                window = board[i:i+5, j-4:j+1]
                diag = np.fliplr(window).diagonal()
                if np.all(diag == 1):
                    return True

    return False

def policy(you, opponent):
    you_arr = np.array(you, dtype=np.int8)
    opp_arr = np.array(opponent, dtype=np.int8)

    # Get all empty cells
    empty_cells = []
    for r in range(6):
        for c in range(6):
            if you_arr[r, c] == 0 and opp_arr[r, c] == 0:
                empty_cells.append((r, c))

    # Define quadrant starting coordinates
    quad_starts = [(0, 0), (0, 3), (3, 0), (3, 3)]
    all_moves = []
    for r, c in empty_cells:
        for quad in range(4):
            for direction in ['L', 'R']:
                all_moves.append((r, c, quad, direction))

    # Check all moves for a winning condition
    for move in all_moves:
        r, c, quad, direction = move
        sim_you = you_arr.copy()
        sim_opp = opp_arr.copy()
        sim_you[r, c] = 1  # Place marble

        # Rotate the specified quadrant
        qr, qc = quad_starts[quad]
        quad_you = sim_you[qr:qr+3, qc:qc+3].copy()
        quad_opp = sim_opp[qr:qr+3, qc:qc+3].copy()

        # Apply rotation
        if direction == 'L':
            rotated_you = np.transpose(quad_you, (1, 0))[::-1, :]
            rotated_opp = np.transpose(quad_opp, (1, 0))[::-1, :]
        else:
            rotated_you = np.transpose(quad_you, (1, 0))[:, ::-1]
            rotated_opp = np.transpose(quad_opp, (1, 0))[:, ::-1]

        sim_you[qr:qr+3, qc:qc+3] = rotated_you

        if check_win(sim_you):
            return f"{r+1},{c+1},{quad},{direction}"

    # If no winning move found, select based on heuristic
    best_score = -1
    best_move = None
    for move in all_moves:
        r, c, quad, direction = move
        sim_you = you_arr.copy()
        sim_opp = opp_arr.copy()
        sim_you[r, c] = 1

        qr, qc = quad_starts[quad]
        quad_you = sim_you[qr:qr+3, qc:qc+3].copy()
        quad_opp = sim_opp[qr:qr+3, qc:qc+3].copy()

        if direction == 'L':
            rotated_you = np.transpose(quad_you, (1, 0))[::-1, :]
            rotated_opp = np.transpose(quad_opp, (1, 0))[::-1, :]
        else:
            rotated_you = np.transpose(quad_you, (1, 0))[:, ::-1]
            rotated_opp = np.transpose(quad_opp, (1, 0))[:, ::-1]

        sim_you[qr:qr+3, qc:qc+3] = rotated_you

        # Calculate score based on number of 4-marble lines
        score = 0
        directions = [(0, 1), (1, 0), (1, 1), (1, -1)]
        for dx, dy in directions:
            for i in range(6):
                for j in range(6):
                    valid = True
                    for k in range(4):
                        nr = i + dx * k
                        nc = j + dy * k
                        if nr < 0 or nr >= 6 or nc < 0 or nc >= 6:
                            valid = False
                            break
                    if not valid:
                        continue
                    if np.all(sim_you[i:i+4, j:j+4] == 1) if dx == 0 else True:
                        # Check the line in this direction
                        all_ones = True
                        for step in range(4):
                            nr_step = i + dx * step
                            nc_step = j + dy * step
                            if sim_you[nr_step, nc_step] != 1:
                                all_ones = False
                                break
                        if all_ones:
                            score += 100  # High score for 4-marble align

        # Simple tiebreaker for same score: prefer lower row, column, then quad, then 'L'
        if score > best_score or (score == best_score and best_move is None):
            best_score = score
            best_move = move

    if best_move:
        r, c, quad, dir = best_move
        return f"{r+1},{c+1},{quad},{dir}"
    else:
        # Fallback to the first possible move
        r, c = empty_cells[0]
        quad = 0
        direction = 'L'
        return f"{r+1},{c+1},{quad},{direction}"
```

### Explanation
1. **Immediate Win Check**: The code first checks all possible moves to see if any of them result in an immediate win. This is done using the `check_win` function, which checks for 5 consecutive marbles in any direction.
2. **Heuristic Evaluation**: If no winning moves are found, the code evaluates each move based on the number of 4-marble lines created, which could lead to a win in the next turn. The move with the highest score is selected.
3. **Move Generation**: The code generates all possible moves by iterating over all empty cells and all possible quadrants and directions for rotation.

This approach ensures that the code efficiently checks for immediate wins and selects the best possible move based on potential future wins, adhering to the game's rules and constraints.
