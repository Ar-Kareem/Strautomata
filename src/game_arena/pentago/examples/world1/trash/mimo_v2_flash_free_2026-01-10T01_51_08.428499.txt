To implement a strong agent for Pentago, I will use a Minimax algorithm with Alpha-Beta pruning and Depth-Limited Search (DLS). Pentago is a game with a relatively small state space (6x6 board), allowing us to search several plies ahead within the time limit.

**Strategy Overview:**
1.  **Representation**: The board state is encoded using a single 36-bit integer for each player (2x `int`). This allows for extremely fast state transitions and hashing.
2.  **Move Generation**: I will implement a generator that yields valid moves (placement + rotation). The move order is optimized by prioritizing moves that capture the center or allow immediate wins, which helps the Alpha-Beta pruning cut off branches faster.
3.  **Evaluation Function**: A heuristic function assesses the board state. It looks for:
    *   Immediate wins (5-in-a-row).
    *   Potential threats (4-in-a-row with open ends).
    *   Center control (squares that can form multiple lines).
    *   Mobility (number of available moves).
4.  **Search Algorithm**: I will use Iterative Deepening Search. The depth of the search increases until the time limit (computed dynamically based on the number of empty cells) is approached. At each turn, the algorithm evaluates all possible moves to find the one that maximizes the player's advantage while minimizing the opponent's.

Here is the complete Python code:

```python
import time
import sys

# Increase recursion depth just in case, though we use iterative deepening
sys.setrecursionlimit(2000)

def policy(you, opponent):
    """
    Pentago move selection policy using Iterative Deepening Alpha-Beta Search.
    """
    start_time = time.time()
    
    # 1. Convert board state to integer bitmasks for efficient processing
    me = 0
    opp = 0
    for r in range(6):
        for c in range(6):
            bit = 1 << (r * 6 + c)
            if you[r][c]:
                me |= bit
            if opponent[r][c]:
                opp |= bit
    
    # 2. Precompute global adjacency and line masks
    if not hasattr(policy, 'all_lines'):
        policy.all_lines = []
        # Horizontal
        for r in range(6):
            for c in range(2):
                policy.all_lines.append(((r * 6 + c) << 0) | ((r * 6 + c + 1) << 5) | ((r * 6 + c + 2) << 10) | ((r * 6 + c + 3) << 15) | ((r * 6 + c + 4) << 20))
        # Vertical
        for r in range(2):
            for c in range(6):
                policy.all_lines.append(((r * 6 + c) << 0) | (((r + 1) * 6 + c) << 5) | (((r + 2) * 6 + c) << 10) | (((r + 3) * 6 + c) << 15) | (((r + 4) * 6 + c) << 20))
        # Diagonals
        for r in range(2):
            for c in range(2):
                # TL to BR
                policy.all_lines.append(((r * 6 + c) << 0) | (((r + 1) * 6 + c + 1) << 5) | (((r + 2) * 6 + c + 2) << 10) | (((r + 3) * 6 + c + 3) << 15) | (((r + 4) * 6 + c + 4) << 20))
                # TR to BL
                policy.all_lines.append(((r * 6 + (5 - c)) << 0) | (((r + 1) * 6 + (4 - c)) << 5) | (((r + 2) * 6 + (3 - c)) << 10) | (((r + 3) * 6 + (2 - c)) << 15) | (((r + 4) * 6 + (1 - c)) << 20))

        # Bitmask for 5 consecutive bits (for checking 5-in-a-row)
        policy.mask_5 = (1 << 5) - 1

        # Quadrant definitions (1-indexed coordinates -> 0-indexed bits)
        # Quad 0: TL (0-2, 0-2), Quad 1: TR (0-2, 3-5), Quad 2: BL (3-5, 0-2), Quad 3: BR (3-5, 3-5)
        policy.quad_masks = []
        policy.quad_coords = []
        for q in range(4):
            q_rows = [0, 1, 2] if q < 2 else [3, 4, 5]
            q_cols = [0, 1, 2] if q % 2 == 0 else [3, 4, 5]
            mask = 0
            coords = []
            for r in q_rows:
                for c in q_cols:
                    bit_pos = r * 6 + c
                    mask |= (1 << bit_pos)
                    coords.append((r, c))
            policy.quad_masks.append(mask)
            policy.quad_coords.append(coords)

        # Rotation mappings: src bit -> dst bit for Left (CCW) and Right (CW)
        policy.rot_L = {}
        policy.rot_R = {}
        for q in range(4):
            coords = policy.quad_coords[q]
            # Extract 3x3 grid coordinates relative to quad
            grid = {(r%3, c%3): (r, c) for r, c in coords} # (local_r, local_c) -> (global_r, global_c)
            for lr in range(3):
                for lc in range(3):
                    gr, gc = grid[(lr, lc)]
                    src_bit = gr * 6 + gc
                    # Right (CW): (r, c) -> (c, 2-r)
                    tr, tc = lc, 2 - lr
                    gr_dst, gc_dst = grid[(tr, tc)]
                    dst_bit_R = gr_dst * 6 + gc_dst
                    policy.rot_R[src_bit] = dst_bit_R
                    # Left (CCW): (r, c) -> (2-c, r)
                    tr, tc = 2 - lc, lr
                    gr_dst, gc_dst = grid[(tr, tc)]
                    dst_bit_L = gr_dst * 6 + gc_dst
                    policy.rot_L[src_bit] = dst_bit_L

    # 3. Heuristic Evaluation
    def evaluate(board_me, board_opp):
        # Check immediate win
        if check_win(board_me): return 100000
        if check_win(board_opp): return -100000
        
        score = 0
        
        # Analyze lines
        for l in policy.all_lines:
            me_bits = bin(board_me & l).count('1')
            opp_bits = bin(board_opp & l).count('1')
            if me_bits > 0 and opp_bits == 0:
                if me_bits >= 4: score += 10000 + (me_bits * 1000)
                elif me_bits == 3: score += 500
                elif me_bits == 2: score += 50
            elif opp_bits > 0 and me_bits == 0:
                if opp_bits >= 4: score -= 10000 + (opp_bits * 1000)
                elif opp_bits == 3: score -= 500
                elif opp_bits == 2: score -= 50
        
        # Center control bonus (squares involved in max lines)
        center_bonus = 0
        full = board_me | board_opp
        if not (full & 0x0000333300000000): center_bonus += 20 # Very center 4 empty
        score += bin(board_me & 0x00003C3C00000000).count('1') * 5
        score -= bin(board_opp & 0x00003C3C00000000).count('1') * 5
        
        return score

    def check_win(board):
        for l in policy.all_lines:
            if (board & l) == l: return True
        return False

    # 4. Alpha-Beta Search
    def alpha_beta(board_me, board_opp, depth, alpha, beta, maximizing, quad_rot_cache=None, turn_start=0):
        # Time check
        if time.time() - turn_start > 0.45: 
            raise TimeoutError
            
        # Check full board draw
        if (board_me | board_opp) == 0xFFFFFFFFFFFFFF:
            return 0

        # Leaf node or depth limit
        if depth == 0:
            return evaluate(board_me, board_opp)

        # Determine move order to improve pruning
        moves = []
        empty = ~(board_me | board_opp) & 0xFFFFFFFFFFFFFF
        
        # Heuristic: Center placements first
        # Priority mask: bits 15, 16, 20, 21 (center 2x2)
        center_bits = empty & 0x0000000000181800
        other_bits = empty & ~center_bits
        
        # Generate moves
        def gen(bits):
            temp = bits
            while temp:
                b = temp & -temp
                idx = (b.bit_length() - 1)
                r, c = divmod(idx, 6)
                # Quad index
                q = 2 * (r // 3) + (c // 3)
                # Rotations
                # Prefer non-rotations (q, 'R' to not waste rotation) or specific rotations?
                # Standard simple heuristic: just generate all
                # Order: R then L (arbitrary preference)
                moves.append((r+1, c+1, q, 'R'))
                moves.append((r+1, c+1, q, 'L'))
                temp ^= b

        if center_bits: gen(center_bits)
        if other_bits: gen(other_bits)
        
        # If depth is shallow or we have specific rotation caches, we can filter, 
        # but here we iterate.

        if maximizing:
            max_eval = -float('inf')
            for (r, c, q, d) in moves:
                # Apply move
                bit_idx = (r-1)*6 + (c-1)
                # Check if cell empty (double check)
                if (board_me >> bit_idx) & 1 or (board_opp >> bit_idx) & 1: continue
                
                n_me = board_me | (1 << bit_idx)
                n_opp = board_opp
                
                # Rotation
                if d == 'R':
                    mapping = policy.rot_R
                else:
                    mapping = policy.rot_L
                
                # Apply rotation to quadrant q only
                qmask = policy.quad_masks[q]
                
                bits_me = n_me & qmask
                bits_opp = n_opp & qmask
                n_me = (n_me & ~qmask)
                n_opp = (n_opp & ~qmask)
                
                # Rotate bits
                # Since we have mapping src->dst, we iterate src
                # But building dst directly is faster
                # Let's use a precomputed array of (src_bit, dst_bit) for the quadrant
                # To do this efficiently in loop without static data structure change:
                # We iterate bits of 'bits_me' and set in n_me
                temp = bits_me
                while temp:
                    b = temp & -temp
                    src = b.bit_length() - 1
                    dst = mapping[src]
                    n_me |= (1 << dst)
                    temp ^= b
                
                temp = bits_opp
                while temp:
                    b = temp & -temp
                    src = b.bit_length() - 1
                    dst = mapping[src]
                    n_opp |= (1 << dst)
                    temp ^= b
                
                # Check Win immediately after move (Placement then Rotation)
                if check_win(n_me):
                    return 100000 # Winning move
                
                # Check if opponent wins after our move (should be avoided)
                # This is handled by minimizing player response, but checking here speeds up pruning
                if check_win(n_opp):
                    # This move is bad, but maybe we can't avoid it.
                    # We continue searching but score very low? 
                    # No, let minimizer find it.
                    pass 

                val = alpha_beta(n_me, n_opp, depth - 1, alpha, beta, False, None, turn_start)
                if val > max_eval: max_eval = val
                if max_eval > alpha: alpha = max_eval
                if beta <= alpha: break
            return max_eval
        else:
            min_eval = float('inf')
            for (r, c, q, d) in moves:
                bit_idx = (r-1)*6 + (c-1)
                if (board_me >> bit_idx) & 1 or (board_opp >> bit_idx) & 1: continue
                
                n_opp = board_opp | (1 << bit_idx)
                n_me = board_me
                
                if d == 'R': mapping = policy.rot_R
                else: mapping = policy.rot_L
                
                qmask = policy.quad_masks[q]
                bits_me = n_me & qmask
                bits_opp = n_opp & qmask
                n_me = (n_me & ~qmask)
                n_opp = (n_opp & ~qmask)
                
                temp = bits_me
                while temp:
                    b = temp & -temp
                    src = b.bit_length() - 1
                    n_me |= (1 << mapping[src])
                    temp ^= b
                temp = bits_opp
                while temp:
                    b = temp & -temp
                    src = b.bit_length() - 1
                    n_opp |= (1 << mapping[src])
                    temp ^= b

                if check_win(n_opp):
                    return -100000
                
                val = alpha_beta(n_me, n_opp, depth - 1, alpha, beta, True, None, turn_start)
                if val < min_eval: min_eval = val
                if min_eval < beta: beta = min_eval
                if beta <= alpha: break
            return min_eval

    # 5. Iterative Deepening
    best_move = None
    empty_count = bin(me | opp).count('1')
    empty_count = 36 - empty_count
    
    # Time management: Depth depends on empty squares
    # If early game, fewer moves to consider, we can go deeper.
    # If late game, fewer empty spots, also deeper.
    # We use a scaled depth.
    # 0-12 moves: Depth 4/5, 12-24: 3, 24-36: 2-3
    if empty_count > 28: max_depth = 2
    elif empty_count > 18: max_depth = 3
    elif empty_count > 10: max_depth = 4
    else: max_depth = 5
    
    # Try to find winning move at depth 1 first (fast)
    try:
        alpha = -float('inf')
        best_move_str = None
        
        # Generator for root moves (optimized order)
        empty = ~(me | opp) & 0xFFFFFFFFFFFFFF
        root_moves = []
        center_bits = empty & 0x0000000000181800
        other_bits = empty & ~center_bits
        
        def gen_root(bits):
            temp = bits
            while temp:
                b = temp & -temp
                idx = b.bit_length() - 1
                r, c = divmod(idx, 6)
                q = 2 * (r // 3) + (c // 3)
                # Priority: R then L, or just both
                root_moves.append((r+1, c+1, q, 'R'))
                root_moves.append((r+1, c+1, q, 'L'))
                temp ^= b
        
        if center_bits: gen_root(center_bits)
        if other_bits: gen_root(other_bits)

        for depth in range(1, max_depth + 1):
            current_best = -float('inf')
            local_best_move = None
            
            # Sort root moves based on previous iteration? (Not implemented for simplicity, center first is good)
            # We use simple iterative loop
            for (r, c, q, d) in root_moves:
                # Apply move to check win immediately
                bit_idx = (r-1)*6 + (c-1)
                n_me = me | (1 << bit_idx)
                n_opp = opp
                if d == 'R': mapping = policy.rot_R
                else: mapping = policy.rot_L
                qmask = policy.quad_masks[q]
                bits_me = n_me & qmask
                bits_opp = n_opp & qmask
                n_me = (n_me & ~qmask)
                n_opp = (n_opp & ~qmask)
                temp = bits_me
                while temp:
                    b = temp & -temp
                    n_me |= (1 << mapping[b.bit_length() - 1])
                    temp ^= b
                temp = bits_opp
                while temp:
                    b = temp & -temp
                    n_opp |= (1 << mapping[b.bit_length() - 1])
                    temp ^= b
                
                # Quick check
                if check_win(n_me):
                    return f"{r},{c},{q},{d}"
                
                # Search
                val = alpha_beta(n_me, n_opp, depth - 1, -float('inf'), float('inf'), False, None, start_time)
                
                if val > current_best:
                    current_best = val
                    local_best_move = (r, c, q, d)
                
                # Pruning check at root? Hard to prune effectively without sorting, 
                # but we check time.
                if time.time() - start_time > 0.4:
                    raise TimeoutError
            
            if local_best_move:
                best_move = local_best_move
                if current_best > 90000: # Found a forced win
                    return f"{best_move[0]},{best_move[1]},{best_move[2]},{best_move[3]}"

    except TimeoutError:
        pass
    
    if best_move is None:
        # Fallback: just take center if available, else first empty
        # This should rarely happen due to the "assume legal move exists" 
        # and the basic generation, but for safety.
        empty = ~(me | opp) & 0xFFFFFFFFFFFFFF
        if empty:
            idx = (empty & -empty).bit_length() - 1
            r, c = divmod(idx, 6)
            q = 2 * (r // 3) + (c // 3)
            return f"{r+1},{c+1},{q},R"
        # Should not happen
        return "1,1,0,R"
        
    return f"{best_move[0]},{best_move[1]},{best_move[2]},{best_move[3]}"
```
