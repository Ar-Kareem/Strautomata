To solve this problem, we need to develop an AI that can play the game Pentago by selecting optimal moves. The AI should evaluate the board state to find the best possible move, prioritizing moves that can lead to an immediate win or maximize the potential for future lines of 5 marbles.

### Approach
1. **Check for Immediate Winning Moves**: The AI first checks if placing a marble in any empty cell followed by rotating any quadrant can result in an immediate win. This involves simulating each possible move and checking if it creates a line of 5 marbles for the current player.
2. **Heuristic Evaluation**: If no immediate winning move is found, the AI evaluates each possible move using a heuristic. The heuristic prioritizes moves that create new lines of 3 marbles, as these are more likely to be extended into a winning line in future moves.
3. **Default Move**: If no promising moves are found, the AI defaults to placing the marble in the safest available spot, typically the bottom-left cell of the bottom-left quadrant, and rotating the same quadrant to the right.

```python
import numpy as np

def has_win(board):
    # Check for any 5-in-a-row for either player
    lines = []
    
    # Horizontal lines
    for r in range(6):
        for c in range(2):
            line = [[board[r][c + i] for i in range(5)]]
            if len(line[0]) == 5 and (all(cell == 1 for cell in line[0]) or all(cell == -1 for cell in line[0])):
                return True
    
    # Vertical lines
    for c in range(6):
        for r in range(2):
            line = [board[r + i][c] for i in range(5)]
            if len(line) == 5 and (all(cell == 1 for cell in line) or all(cell == -1 for cell in line)):
                return True
    
    # Diagonals (main)
    for r in range(2):
        for c in range(2):
            line = [board[r + i][c + i] for i in range(5)]
            if len(line) == 5 and (all(cell == 1 for cell in line) or all(cell == -1 for cell in line)):
                return True
    
    # Diagonals (anti)
    for r in range(2):
        for c in range(4, 6):
            line = [board[r + i][c - i] for i in range(5)]
            if len(line) == 5 and (all(cell == 1 for cell in line) or all(cell == -1 for cell in line)):
                return True
    
    return False

def evaluate_line(board, r, c, dr, dc):
    count = 0
    first = r
    last = c
    # Check in direction (dr, dc)
    while 0 <= r < 6 and 0 <= c < 6 and (board[r][c] == 1 or (board[r][c] == 0 and (r, c) == (first, last))):
        count += 1
        r += dr
        c += dc
    # Check in opposite direction
    r, c = first, last
    while 0 <= r < 6 and 0 <= c < 6 and (board[r][c] == 1 or (board[r][c] == 0 and (r, c) == (first, last))):
        r -= dr
        c -= dc
        count += 1
    return count - 1  # subtract duplicate count at starting cell

def evaluate_line_potential(you_arr, r, c):
    count = 0
    directions = [(0, 1), (1, 0), (1, 1), (1, -1)]
    for dr, dc in directions:
        line_length = evaluate_line(you_arr, r, c, dr, dc)
        count += max(0, line_length - 1)  # -1 to exclude self-single count
    return count

def policy(you, opponent):
    you_arr = np.array(you, dtype=int)
    opponent_arr = np.array(opponent, dtype=int)
    empty_cells = np.argwhere((not you_arr) & (not opponent_arr))
    
    # Check for any winning moves
    for (r, c) in empty_cells:
        new_you = you_arr.copy()
        new_you[r, c] = 1
        
        for quad in range(4):
            qr, qc = [(0, 0), (0, 3), (3, 0), (3, 3)][quad]
            for direction in ['L', 'R']:
                temp_you = new_you.copy()
                temp_opponent = opponent_arr.copy()
                
                # Rotate the selected quadrant
                quad_area = temp_you[qr:qr+3, qc:qc+3]
                if direction == 'L':
                    quad_rot = np.rot90(quad_area, 1)
                else:
                    quad_rot = np.rot90(quad_area, -1)
                
                temp_you[qr:qr+3, qc:qc+3] = quad_rot
                
                if has_win(temp_you):
                    return f"{r+1},{c+1},{quad},{direction}"
    
    # No winning move found, evaluate best heuristic move
    best_score = -1
    best_move = None
    quadrant_coords = {
        0: (0, 0),
        1: (0, 3),
        2: (3, 0),
        3: (3, 3)
    }
    
    quad_line_count = [sum(has_win((you_arr == 1).astype(int)) for _ in range(4))]
    
    for (r, c) in empty_cells:
        current_quad = np.searchsorted(np.array([0, 3, 3, 3]), r)
        current_quad += np.searchsorted(np.array([0, 3, 3, 3]), c) > 2
        current_quad = min(current_quad, 3)
        
        qr, qc = quadrant_coords[current_quad]
        adjacent_you = sum(you_arr[qr:qr+3, qc:qc+3])
        
        for quad in range(4):
            qr, qc = quadrant_coords[quad]
            line_count_you = 0
            for i in range(3):
                for j in range(3):
                    line_count_you += evaluate_line_potential(you_arr, qr + i, qc + j)
            
            for direction in ['L', 'R']:
                temp_you = you_arr.copy()
                temp_opponent = opponent_arr.copy()
                temp_you[r, c] = 1
                
                # Apply rotation to quadrant
                q_area = temp_you[qr:qr+3, qc:qc+3]
                if direction == 'L':
                    rotated_quad = np.rot90(q_area, 1)
                else:
                    rotated_quad = np.rot90(q_area, -1)
                temp_you[qr:qr+3, qc:qc+3] = rotated_quad
                
                score = sum(evaluate_line_potential(temp_you, i, j) for i in range(6) for j in range(6))
                if score > best_score:
                    best_score = score
                    best_move = (r, c, quad, direction)
    
    r, c, quad, direction = best_move
    return f"{r+1},{c+1},{quad},{direction}"
```

### Explanation
1. **Immediate Win Check**: The AI first checks each possible move to see if placing a marble and rotating any quadrant results in an immediate win. This is done by simulating each move and checking if it creates a line of 5 marbles.
2. **Heuristic Evaluation**: If no immediate win is found, the AI evaluates each possible move by placing the marble in an empty cell and rotating a quadrant. The evaluation function checks potential lines of 3 or more marbles in all directions, giving higher scores to moves that create more such lines.
3. **Default Move**: If no promising moves are found, the AI selects the first available empty cell and rotates the bottom-left quadrant to the left, ensuring a valid move while trying to strategically place the marble.

This approach balances between checking for immediate wins and using heuristics to maximize the potential for future wins, ensuring efficient and effective move selection within the given constraints.
