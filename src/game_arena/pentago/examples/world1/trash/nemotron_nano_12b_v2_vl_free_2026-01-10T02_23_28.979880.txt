The strategy involves checking for immediate winning moves by simulating each possible placement and rotation. If a move is found that results in a 5-in-a-row for the current player, it is returned. If no such move exists, the first available empty cell is chosen and the quadrant is rotated left.

import numpy as np

def has_5_in_a_row(board):
    for i in range(6):
        for j in range(2):
            if all(board[i][j + k] for k in range(5)):
                return True
    for j in range(6):
        for i in range(2):
            if all(board[i + d][j] for d in range(5)):
                return True
    for i in range(2):
        for j in range(2):
            if all(board[i + k][j + k] for k in range(5)):
                return True
    for i in range(2):
        for j in range(4, 6):
            if all(board[i + k][j - k] for k in range(5)):
                return True
    return False

def policy(you, opponent):
    import numpy as np

    you = np.array(you)
    opponent = np.array(opponent)

    for r in range(6):
        for c in range(6):
            if not you[r, c] and not opponent[r, c]:
                q = ((r // 3) * 2) + (c // 3)
                start_row = (r // 3) * 3
                start_col = (c // 3) * 3

                # Check L rotation
                quad_you = you[start_row:start_row+3, start_col:start_col+3]
                quad_opponent = opponent[start_row:start_row+3, start_col:start_col+3]
                rotated_you_l = np.rot90(quad_you, 3)
                rotated_opponent_l = np.rot90(quad_opponent, 3)
                new_you_l = you.copy()
                new_opponent_l = opponent.copy()
                new_you_l[start_row:start_row+3, start_col:start_col+3] = rotated_you_l
                new_opponent_l[start_row:start_row+3, start_col:start_col+3] = rotated_opponent_l
                if has_5_in_a_row(new_you_l):
                    return f"{r+1},{c+1},{q},'L'}"

                # Check R rotation
                rotated_you_r = np.rot90(quad_you, 1)
                rotated_opponent_r = np.rot90(quad_opponent, 1)
                new_you_r = you.copy()
                new_opponent_r = opponent.copy()
                new_you_r[start_row:start_row+3, start_col:start_col+3] = rotated_you_r
                new_opponent_r[start_row:start_row+3, start_col:start_col+3] = rotated_opponent_r
                if has_5_in_a_row(new_you_r):
                    return f"{r+1},{c+1},{q},'R'}"

    for r in range(6):
        for c in range(6):
            if not you[r, c] and not opponent[r, c]:
                q = ((r // 3) * 2) + (c // 3)
                return f"{r+1},{c+1},{q},'L'"
