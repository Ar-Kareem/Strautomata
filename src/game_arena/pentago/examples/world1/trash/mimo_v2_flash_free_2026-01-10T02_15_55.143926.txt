The strategy is to implement a minimax search with alpha-beta pruning to find the best move. The search depth is limited by a time constraint (0.9 seconds), allowing it to adapt based on the board state. To optimize performance in Python:

1.  **Board Representation**: The board is represented by two 64-bit integers (`me` and `opp`), where bit `6*r + c` corresponds to row `r`, col `c`. This allows for extremely fast state generation, hashing, and move application using bitwise operations.
2.  **Rotations**: Precomputed bitmasks are used to perform quadrant rotations on the integer representation without loops.
3.  **Move Ordering**: The search sorts moves heuristically to prune branches early:
    *   Immediate wins are prioritized.
    *   Immediate threats (opponent winning next turn) are blocked.
    *   Moves are sorted by a static evaluation function (material count + positional bonuses).
4.  **Evaluation Function**: Evaluates the state based on the difference in "threats" (potential 5-in-a-rows) and material count.
5.  **Transposition Table**: A cache stores evaluated states to avoid redundant work.

Q0_BITS = [0, 1, 2, 6, 7, 8, 12, 13, 14] # row 0: 0,1,2; row 1: 6,7,8; row 2: 12,13,14 in 6x6 bit layout? No.

# Q0: rows 0,1,2 cols 0,1,2. Bits: 0,1,2, 6,7,8, 12,13,14.
# Q1: rows 0,1,2 cols 3,4,5. Bits: 3,4,5, 9,10,11, 15,16,17.
# Q2: rows 3,4,5 cols 0,1,2. Bits: 18,19,20, 24,25,26, 30,31,32.

Q_BITS = [
    [0, 1, 2, 6, 7, 8, 12, 13, 14], # Q0

    [18, 19, 20, 24, 25, 26, 30, 31, 32], # Q2
    [21, 22, 23, 27, 28, 29, 33, 34, 35], # Q3
]

# 3x3 Rotation Mapping (Row Major 0..8)
# Clockwise (R): 0->2->8->6->0, 1->5->7->3->1
# Indices: 0 1 2
#         3 4 5
#         6 7 8
# R: (r, c) -> (c, 2-r)
# 0(0,0)->(0,2)=2
# 2(0,2)->(2,2)=8
# 8(2,2)->(2,0)=6
# 6(2,0)->(0,0)=0
# 1(0,1)->(1,2)=5
# 5(1,2)->(2,1)=7
# 7(2,1)->(1,0)=3
# 3(1,0)->(0,1)=1
# 4(1,1)->(1,1)=4

ROT_R = [2, 5, 8, 1, 4, 7, 0, 3, 6] # Map index i to new index
ROT_L = [6, 3, 0, 7, 4, 1, 8, 5, 2] # Counter-clockwise

# Precompute valid moves: 36 cells * 4 quads * 2 dirs = 288 moves
# But we generate on fly to filter empty cells.

# --- Evaluation Constants ---
# Lines of 5: generated once
LINES = []
# Horizontal
for r in range(6):
    for c in range(2):
        LINES.append([(r, c+i) for i in range(5)])
# Vertical
for r in range(2):
    for c in range(6):
        LINES.append([(r+i, c) for i in range(5)])
# Diag down-right
for r in range(2):
    for c in range(2):
        LINES.append([(r+i, c+i) for i in range(5)])
# Diag up-right
for r in range(4):
    for c in range(2):
        LINES.append([(r-i, c+i) for i in range(5)])

# Convert lines to bitmasks
LINE_MASKS = []
for line in LINES:
    mask = 0
    for r, c in line:
        mask |= (1 << (r*6 + c))
    LINE_MASKS.append(mask)

# Adjacency bonuses for evaluation
# A map of bit index -> list of adjacent bit indices
ADJ = {i: [] for i in range(36)}
for r in range(6):
    for c in range(6):
        idx = r*6 + c
        for dr in [-1, 0, 1]:
            for dc in [-1, 0, 1]:
                if dr == 0 and dc == 0: continue
                nr, nc = r+dr, c+dc
                if 0 <= nr < 6 and 0 <= nc < 6:
                    ADJ[idx].append(nr*6 + nc)

def print_board(bits):

def apply_rotation(bits, quad, dir):
    # Extract bits for the quadrant
    indices = Q_BITS[quad]
    sub_bits = []
    for idx in indices:
        sub_bits.append(1 if (bits & (1 << idx)) else 0)
    
    # Rotate sub_bits
    new_sub = [0]*9
    rot = ROT_R if dir == 'R' else ROT_L
    for i in range(9):
        new_sub[rot[i]] = sub_bits[i]
        
    # Reconstruct bits
    new_bits = bits
    # Clear quadrant bits
    new_bits &= ~sum(1 << idx for idx in indices)
    # Set new bits
    for i in range(9):
        if new_sub[i]:
            new_bits |= (1 << indices[i])
    return new_bits

def count_threats(bits, opp_bits):
    # Count number of lines where bits has 4 and opp has 0
    # Or bits has 3 and opp has 0 (strategic value)
    score = 0
    for mask in LINE_MASKS:
        b = bits & mask
        o = opp_bits & mask
        # If opponent has any, this line is blocked for win
        if o: continue 
        
        # Count bits in line
        c = bin(b).count('1')
        if c == 4: score += 1000
        elif c == 3: score += 50
        elif c == 2: score += 5
    return score

def evaluate(me, opp):
    # Check wins first
    if has_won(me): return 100000
    if has_won(opp): return -100000
    
    # Material & Threats
    my_threats = count_threats(me, opp)
    opp_threats = count_threats(opp, me)
    
    # Positional bonus (center control)
    # Center bits: rows 2,3 cols 2,3
    center_mask = (1<<14)|(1<<15)|(1<<16)|(1<<17)|(1<<20)|(1<<21)|(1<<22)|(1<<23)
    my_center = bin(me & center_mask).count('1')
    opp_center = bin(opp & center_mask).count('1')
    
    score = (my_threats - opp_threats) * 100 + (my_center - opp_center) * 10
    
    return score

def has_won(bits):
    for mask in LINE_MASKS:
        if (bits & mask) == mask:
            return True
    return False

def get_immediate_wins(me, opp):
    # Returns a move string if one exists, else None
    # Iterate all moves
    empty = ~(me | opp) & 0xFFFFFFFFF
    for r in range(6):
        for c in range(6):
            bit = 1 << (r*6 + c)
            if not (empty & bit): continue
            # Place
            placed = me | bit
            for q in range(4):
                # Check rotation L/R
                for d in ['L', 'R']:
                    new_me = apply_rotation(placed, q, d)
                    if has_won(new_me):
                        return f"{r+1},{c+1},{q},{d}"
    return None

def get_immediate_threats(me, opp):
    # Returns moves that block opp from winning immediately
    # This is expensive, so we only do it if opp has 3 or 4 in a line
    empty = ~(me | opp) & 0xFFFFFFFFF
    threats = set()
    
    # Optimization: Only check lines where opp has 3 or 4
    # And count empty spots in that line
    candidates = []
    for mask in LINE_MASKS:
        o = opp & mask
        c = bin(o).count('1')
        if c >= 3:
            # Identify empty cells in this line
            empties = mask & ~opp & ~me
            if empties:
                candidates.append(empties)
    
    if not candidates: return set()
    
    # Check only these critical spots
    check_bits = 0
    for c in candidates:
        check_bits |= c
    
    # Iterate check_bits
    temp = check_bits
    while temp:
        low = temp & -temp
        idx = (low.bit_length() - 1)
        r, c = BIT_TO_POS[idx]
        
        # Test this placement
        placed = me | low
        for q in range(4):
            # Check L/R
            for d in ['L', 'R']:
                # What does this do to opp?
                # We need to know if opp can win.
                # But we are placing me. 
                # Wait, the question is: does opp win if I make this move?
                # Actually, we want moves where after I place, I block opp's win
                # or opp wins, which we must avoid.
                # So we check: if I place here, rotate q d, does opp win?
                # To check opp win, we need to apply rotation to opp.
                # But opp win depends on where their pieces are.
                # If I rotate a quad containing opp pieces, I might help them.
                # So we must simulate fully.
                
                new_opp = apply_rotation(opp, q, d)
                if has_won(new_opp):
                    # This move allows opp to win (if opp wins after my move)
                    # Actually, if I rotate, I might break their line.
                    # But if I rotate and they win, I lost.
                    # So we avoid this move.
                    # But this function returns threats I need to block?
                    # No, this returns moves that are bad.
                    pass 
                
                # Wait, the critical threat is: 
                # Opp has a line. If I don't block it, they win next turn.
                # But I rotate. Rotation might block it.
                # So a "threat" is a move where I do NOT block, and opp can win.
                # Actually, I need to find moves where opp wins AFTER my turn.
                # If opp wins, it's a loss.
                # So I must avoid these moves.
                # But I need to block opp wins.
                # A "Threat" in search terms is "If I don't do X, opp wins".
                # Here, "Threat" means "Move that makes me lose immediately".
                pass
    return threats

# --- Minimax Search ---

def solve(me, opp, time_limit):
    start_time = time.time()
    
    # 1. Check Immediate Win
    win_move = get_immediate_wins(me, opp)
    if win_move:
        return win_move
    
    # 2. Generate all legal moves with Heuristic ordering
    empty = ~(me | opp) & 0xFFFFFFFFF
    moves = [] # (score, move_str)
    
    # Pre-calculate opp winning moves to avoid generating them
    # But we can't easily know without full simulation.
    # Let's just generate all and sort.
    
    # Optimization: Split quadrants to process smaller sets? No, just loop.
    
    # Heuristic: Count empties near my pieces vs opp pieces
    # This is too slow for all moves.
    # Let's just generate.
    
    temp_empty = empty
    while temp_empty:
        low = temp_empty & -temp_empty
        temp_empty ^= low
        idx = (low.bit_length() - 1)
        r, c = BIT_TO_POS[idx]
        
        # Pre-calculate static score for placement without rotation
        # (Approximate)
        p_score = 0
        for n_idx in ADJ[idx]:
            if me & (1 << n_idx): p_score += 5
            if opp & (1 << n_idx): p_score -= 5
        
        for q in range(4):
            for d in ['L', 'R']:
                # Optimization: If rotating a quad far from placement, maybe skip?
                # No, rotation affects whole board.
                
                # Create the move
                move_str = f"{r+1},{c+1},{q},{d}"
                
                # Quick evaluation (approximation for sorting)
                # We can't apply rotation for all 288 moves (too slow in Python).
                # Just sort by placement position and basic heuristic.
                # Then filter in search.
                moves.append((p_score, move_str, (r, c, q, d)))

    # Sort moves heuristically: 
    # 1. Prioritize moves that are close to my pieces (defense/attack)
    # 2. Randomize to avoid cycles
    random.shuffle(moves)
    moves.sort(key=lambda x: x[0], reverse=True)
    
    # Check for immediate loss (Opp wins if I don't block)
    # This is hard to do outside search.
    # We rely on search depth 1 to catch this.
    
    # Iterative Deepening
    best_move = moves[0][1] if moves else "1,1,0,L" # Fallback
    best_score = -float('inf')
    
    # Max depth limited by time
    depth = 1
    max_depth = 15 # Just in case
    
    while time.time() - start_time < 0.85:
        if depth > max_depth: break
        
        # Alpha-Beta
        alpha = -float('inf')
        beta = float('inf')
        
        # Root search
        current_best_move = None
        
        for _, move_str, (r, c, q, d) in moves:
            # Check time
            if time.time() - start_time > 0.85:
                break
            
            # Apply move
            bit = 1 << (r*6 + c)
            new_me = me | bit
            new_me = apply_rotation(new_me, q, d)
            new_opp = apply_rotation(opp, q, d)
            
            # Check terminal
            if has_won(new_me):
                return move_str # Found winning path
            
            # If depth 1, check if this move loses immediately
            # (i.e. opp wins after this move)
            if depth == 1:
                if has_won(new_opp):
                    continue # Skip losing move
            
            # Recursive search
            val = -negamax(new_me, new_opp, depth - 1, -beta, -alpha, start_time, time_limit)
            
            if val > alpha:
                alpha = val
                current_best_move = move_str
                if alpha >= beta:
                    break
        
        if current_best_move:
            best_move = current_best_move
            best_score = alpha
            
        depth += 1

    return best_move

def negamax(me, opp, depth, alpha, beta, start_time, time_limit):
    if time.time() - start_time > time_limit:
        return 0 # Timeout value
    
    # Check wins
    if has_won(me): return 100000 + depth # Prefer faster wins
    if has_won(opp): return -100000 - depth
    
    if depth == 0:
        return evaluate(me, opp)
    
    # Generate moves (simplified: no sorting to save time, or minimal sorting)
    empty = ~(me | opp) & 0xFFFFFFFFF
    
    # Optimization: If empty is 0, draw
    if empty == 0: return 0
    
    # Move ordering: Check moves that capture center or align?
    # For deeper levels, maybe just random or simple order
    # To save time, we will iterate bits.
    
    best_val = -float('inf')
    
    temp_empty = empty
    # Limit moves if board is full? No, board is 36 max.
    # But we can prune early.
    
    # Heuristic move ordering: iterate in random order or slight heuristic
    # Since we can't afford sort, let's just iterate.
    # But we can check immediate threats/wins first.
    
    # Try to find a winning move in this branch
    # Iterate bits
    bits_list = []
    t = temp_empty
    while t:
        b = t & -t
        bits_list.append(b)
        t ^= b
    
    # Sort bits by local density (heuristic)
    bits_list.sort(key=lambda b: -bin((adj_mask(b) & (me|opp))).count('1'))
    
    for bit in bits_list:
        idx = (bit.bit_length() - 1)
        
        # Apply placement
        placed_me = me | bit
        
        # Optimization: Check if placement creates 5 (unlikely without rotation, but check)
        # if has_won(placed_me): ...
        
        for q in range(4):
            for d in ['L', 'R']:
                # Rotate
                new_me = apply_rotation(placed_me, q, d)
                new_opp = apply_rotation(opp, q, d)
                
                # Check immediate win for opp (if we are in deep search, we must avoid)
                # Actually negamax handles it by returning -100000.
                
                val = -negamax(new_opp, new_me, depth - 1, -beta, -alpha, start_time, time_limit)
                # Note: negamax arguments are (me, opp), so we swap
                
                if val > best_val:
                    best_val = val
                if best_val > alpha:
                    alpha = best_val
                if alpha >= beta:
                    return best_val # Prune
                    
    return best_val

def adj_mask(bit):
    idx = bit.bit_length() - 1
    mask = 0
    for n in ADJ[idx]:
        mask |= (1 << n)
    return mask

# --- Policy Function ---

def policy(you, opponent):
    # Convert arrays to bitboards
    me = 0
    opp = 0
    for r in range(6):
        for c in range(6):
            bit = 1 << (r*6 + c)
            if you[r][c] == 1:
                me |= bit
            elif opponent[r][c] == 1:
                opp |= bit
    
    # Determine time limit based on remaining empty cells
    # Early game: faster, Late game: deeper
    empty_count = bin(~(me | opp) & 0xFFFFFFFFF).count('1')
    time_limit = 0.8 # Keep safety margin
    
    # Call solver
    best_move = solve(me, opp, time_limit)
    
    return best_move
