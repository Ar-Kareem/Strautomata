

The strategyinvolves prioritizing immediate wins by checking all possible moves that result in a 5-in-a-row for the current player. If no such move exists, the policy selects a random legal move from available empty cells. This approach ensures compliance with game rules while providing a basic yet functional policy.

```python
def rotate_3x3(matrix, direction):
    if direction == 'L':
        transposed = list(zip(*matrix))
        rotated = [list(row)[::-1] for row in transposed]
        return rotated
    elif direction == 'R':
        reversed_rows = [row[::-1] for row in matrix]
        transposed = list(zip(*reversed_rows))
        return transposed

def check_win(board, player):
    for r in range(6):
        for c in range(2):
            if all(board[r][c+i] == player for i in range(5)):
                return True
    for c in range(6):
        for r in range(2):
            if all(board[r+i][c] == player for i in range(5)):
                return True
    for r in range(2):
        for c in range(2):
            if all(board[r+i][c+i] == player for i in range(5)):
                return True
    for r in range(2):
        for c in range(4, 2, -1):
            if all(board[r+i][c-i] == player for i in range(5)):
                return True
    return False

def simulate_move(you, opponent, move):
    r, c, quad, dir = move
    r0 = r - 1
    c0 = c - 1
    new_you = [row[:] for row in you]
    new_opponent = [row[:] for row in opponent]
    new_you[r0][c0] = 1
    if quad == 0:
        rows, cols = [0,1,2], [0,1,2]
    elif quad == 1:
        rows, cols = [0,1,2], [3,4,5]
    elif quad == 2:
        rows, cols = [3,4,5], [0,1,2]
    else:
        rows, cols = [3,4,5], [3,4,5]
    quadrant = [new_you[r][cols] for r in rows]
    rotated_quadrant = rotate_3x3(quadrant, dir)
    for i, r_idx in enumerate(rows):
        for j, c_idx in enumerate(cols):
            new_you[r_idx][c_idx] = rotated_quadrant[i][j]
    return new_you, new_opponent

def policy(you, opponent):
    for r in range(6):
        for c in range(6):
            if you[r][c] == 0 and opponent[r][c] == 0:
                for quad in range(4):
                    for dir in ['L', 'R']:
                        move = (r+1, c+1, quad, dir)
                        new_you, _ = simulate_move(you, opponent, move)
                        if check_win(new_you, 1):
                            return move
    import random
    empty_cells = [(r, c) for r in range(6) for c in range(6) if you[r][c] == 0 and opponent[r][c] == 0]
    r, c = random.choice(empty_cells)
    quad = random.randint(0, 3)
    dir = random.choice(['L', 'R'])
    return f"{r+1},{c+1},{quad},{dir}"
```
