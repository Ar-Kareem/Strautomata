def rotate_clockwise(board, r0, c0):
    new_quad = [[0]*3 for _ in range(3)]
    for i in range(3):
        for j in range(3):
            original_r = r0 + i
            original_c = c0 + j
            new_r = r0 + (2 - j)
            new_c = c0 + i
            new_quad[new_r - r0][new_c - c0] = board[original_r][original_c]
    for i in range(3):
        for j in range(3):
            board[r0 + i][c0 + j] = new_quad[i][j]

def rotate_anticlockwise(board, r0, c0):

def simulate_move(you, opponent, move):
    r, c, quad, direction = move
    you_copy = [row[:] for row in you]
    opp_copy = [row[:] for row in opponent]
    you_copy[r][c] = 1
    quad_start = {0: (0,0), 1: (0,3), 2: (3,0), 3: (3,3)}
    r0, c0 = quad_start[quad]
    if direction == 'L':
        rotate_anticlockwise(you_copy, r0, c0)
    else:
        rotate_clockwise(you_copy, r0, c0)
    win_you = check_win(you_copy)
    win_opp = check_win(opp_copy)
    if win_you and win_opp:
        return 'draw'
    elif win_you:
        return 'win'
    elif win_opp:
        return 'loss'
    else:
        return 'continue'

def policy(you, opponent):
    empty_cells = []
    for r in range(6):
        for c in range(6):
            if you[r][c] == 0 and opponent[r][c] == 0:
                empty_cells.append((r, c))
    moves = []
    for (r, c) in empty_cells:
        for quad in range(4):
            for direction in ['L', 'R']:
                moves.append((r, c, quad, direction))
    winning_moves = []
    for move in moves:
        result = simulate_move(you, opponent, move)
        if result == 'win':
            winning_moves.append(move)
        elif result == 'draw':
            pass
        else:
            continue
    if winning_moves:
        return f"{winning_moves[0][0]+1},{winning_moves[0][1]+1},{winning_moves[0][2]},{winning_moves[0][3]}"
    else:
        best_move = None
        best_score = -float('inf')
        for move in moves:
            you_copy = [row[:] for row in you]
            opp_copy = [row[:] for row in opponent]
            r, c, quad, direction = move
            you_copy[r][c] = 1
            quad_start = {0: (0,0), 1: (0,3), 2: (3,0), 3: (3,3)}
            r0, c0 = quad_start[quad]

                rotate_clockwise(you_copy, r0, c0)
