3. Strategy should look for winning moves, blocking opponent wins, and general strategic placement

Here's my approach:
1. Check if there's an immediate winning move (5-in-a-row) for me
2. If not, check if opponent has a winning move and block it
3. Otherwise, use a heuristic to choose a good move (central positions, potential lines)

def policy(you, opponent) -> str:
    import numpy as np
    
    # Convert inputs to numpy arrays for easier manipulation
    you_board = np.array(you)
    opponent_board = np.array(opponent)
    
    # Helper function to check if a player has won
    def check_winner(board):
        # Check all possible 5-in-a-row lines
        # Horizontal
        for r in range(6):
            for c in range(2):
                if all(board[r][c+i] == 1 for i in range(5)):
                    return True
        
        # Vertical
        for c in range(6):
            for r in range(2):
                if all(board[r+i][c] == 1 for i in range(5)):
                    return True
        
        # Diagonal (top-left to bottom-right)
        for r in range(2):
            for c in range(2):
                if all(board[r+i][c+i] == 1 for i in range(5)):
                    return True
        
        # Diagonal (top-right to bottom-left)
        for r in range(2):
            for c in range(4, 6):
                if all(board[r+i][c-i] == 1 for i in range(5)):
                    return True
        
        return False

        elif quad == 1:  # Top-right
            r1, r2, r3 = 0, 1, 2

            r1, r2, r3 = 3, 4, 5

        else:  # Bottom-right
            r1, r2, r3 = 3, 4, 5

        if dir == 'R':

            new_board[r3][c1] = temp
            
            temp = new_board[r1][c2]
            new_board[r1][c2] = new_board[r2][c3]
            new_board[r2][c3] = new_board[r3][c2]
            new_board[r3][c2] = new_board[r2][c1]
            new_board[r2][c1] = temp
            
        return new_board
    
    # Check if a cell is empty

        return you_board[row-1][col-1] == 0 and opponent_board[row-1][col-1] == 0
    
    # Generate all legal moves
    legal_moves = []
    
    # Try all empty positions for placement
    for row in range(1, 7):
        for col in range(1, 7):
            if is_empty(row, col):
                # Try all quadrants
                for quad in range(4):

                        legal_moves.append((row, col, quad, dir))
    
    # Evaluate each move and choose the best
    # Strategy: Prefer moves that result in winning, then blocking opponent,
    # then corner positions, then center positions

    best_score = -1
    
    for row, col, quad, dir in legal_moves:
        # Apply move to temp board
        temp_you = you_board.copy()
        temp_opponent = opponent_board.copy()
        
        # Place marble on temporary board
        temp_you[row-1][col-1] = 1
        
        # Rotate quadrant on temporary board
        # We'll need to handle rotation properly
        new_board = apply_move(temp_you, row, col, quad, dir)
        
        # Check if this move wins for us
        if check_winner(new_board):
            return f"{row},{col},{quad},{dir}"

        temp_opponent[row-1][col-1] = 1  # Place opponent marble for validation

        temp_board = apply_move(temp_opponent, row, col, quad, dir)

                score += 10

    first_move = legal_moves[0]
    return f"{first_move[0]},{first_move[1]},{first_move[2]},{first_move[3]}"
