The strategy is to implement a greedy policy that selects a legal move for an Amazon and shoots an arrow. The policy will focus on:
1. Selecting my Amazon (value 1) to move
2. Finding a valid queen-like movement (orthogonal or diagonal in any direction) to an empty square (value 0)
3. From the new position, finding a valid arrow shot (also queen-like movement) to an empty square
4. Prioritizing moves that block the opponent's amazons and open more paths for my own amazons

Here's the Python source code for our policy:

import numpy as np
from typing import Tuple

def policy(board: np.ndarray) -> str:
    # Search for a valid move
    for from_row, from_col in get_my_amazons(board):
        # Get all possible positions to move to
        for to_row, to_col in get_valid_moves(board, from_row, from_col):
            # Assume we've moved the amazon
            temp_board = board.copy()
            temp_board[from_row, from_col] = 0
            temp_board[to_row, to_col] = 1

            # Get valid arrow shots from the new position
            for arrow_row, arrow_col in get_valid_moves(temp_board, to_row, to_col):
                # Return the first legal move we find in a deterministic pattern
                return f"{from_row},{from_col}:{to_row},{to_col}:{arrow_row},{arrow_col}"
    
    # This should never be reached as the environment promises a legal move exists
    raise Exception("No legal move found")

def get_my_amazons(board: np.ndarray) -> list[Tuple[int, int]]:
    # Find coordinates of all my amazons
    return list(zip(*np.where(board == 1)))

def get_valid_moves(board: np.ndarray, r: int, c: int) -> list[Tuple[int, int]]:
    # Get all positions reachable by queen-like movement (orthogonal and diagonal)
    valid_moves = []
    
    directions = [(-1, 0), (1, 0), (0, -1), (0, 1),  # orthogonal
                  (-1, -1), (-1, 1), (1, -1), (1, 1)]  # diagonal

    for dr, dc in directions:
        for step in range(1, 6):
            nr, nc = r + dr * step, c + dc * step
            
            # Check if the new position is within the board
            if not (0 <= nr < 6 and 0 <= nc < 6):
                break
                
            # Check if the new position is empty
            if board[nr, nc] == 0:
                valid_moves.append((nr, nc))
            # Stop if we encounter a blocker
            elif board[nr, nc] != 0:
                break

    return valid_moves
