def policy(pieces: dict[str, str], to_play: str, memory: dict) -> tuple[str, dict]:
    # Helper function to check if a square is under attack
    def is_attacked(square: str, attacker_color: str) -> bool:
        # Iterate over all enemy pieces
        for sq, piece in pieces.items():
            if piece[0] == attacker_color:
                piece_type = piece[1]
                # Check if enemy piece can attack the given square
                if piece_type == 'N':  # Knight
                    knight_moves = {'b3', 'g3', 'b4', 'g4', 'c2', 'f2', 'c6', 'f6'}
                    if (ord(sq[0]) - ord('a') + 1, int(sq[1])) in [(ord(k[0]) - ord('a') + 1, int(k[1])) for k in knight_moves]:
                        return True
                elif piece_type == 'B':  # Bishop
                    if (abs(ord(sq[0]) - ord('a') - ord(piece[0]) - ord('a')) == abs(int(sq[1]) - int(piece[1]))):
                        return True
                elif piece_type == 'R':  # Rook
                    if sq[0] == piece[0] or sq[1] == piece[1]:
                        return True
                elif piece_type == 'Q':  # Queen
                    if (sq[0] == piece[0] or sq[1] == piece[1] or 
                        abs(ord(sq[0]) - ord('a') - ord(piece[0]) - ord('a')) == abs(int(sq[1]) - int(piece[1]))):
                        return True
                elif piece_type == 'K':  # King
                    if abs(ord(sq[0]) - ord('a') - ord(piece[0]) - ord('a')) <= 1 and abs(int(sq[1]) - int(piece[1])) <= 1:
                        return True
        return False

    # Evaluate material balance
    def material_score() -> int:
        score = 0
        for sq, piece in pieces.items():
            if piece[0] == 'w':
                if piece[1] == 'P': score += 1
                elif piece[1] == 'N': score += 3
                elif piece[1] == 'B': score += 3
                elif piece[1] == 'R': score += 5
                elif piece[1] == 'Q': score += 9
            else:
                if piece[1] == 'P': score -= 1
                elif piece[1] == 'N': score -= 3
                elif piece[1] == 'B': score -= 3
                elif piece[1] == 'R': score -= 5
                elif piece[1] == 'Q': score -= 9
        return score

    # Evaluate king safety
    def king_safety() -> int:
        king_sq = next(sq for sq, piece in pieces.items() if piece == 'wK' if to_play == 'white')

            opponent = 'b'
        else:
            opponent = 'w'

            return -10
        return 0

                    if sq[1] == '2' or sq[1] == '7': score += 1
