# Strategy

This policy implements a multi-layered chess strategy that prioritizes immediate threats and tactical opportunities while maintaining a long-term positional advantage. The key components are:

1. **Checkmate Priority**: Always play a checkmate move if available.
2. **Material Gain**: Prioritize captures that win material (using piece values: Q=9, R=5, B=3, N=3, P=1).
3. **Piece Development**: Favor moves that develop pieces toward the center or active squares.
4. **King Safety**: Prefer moves that improve king safety (castling, moving away from threats).
5. **Pawn Structure**: Avoid moves that weaken pawn structure or create isolated pawns.
6. **Tempo**: Prefer moves that maintain or gain tempo (like checks or threats).
7. **Randomness**: When moves are equally good, add slight randomness to avoid predictability.

The policy uses the memory dictionary to track:
- Previous board states to detect repetitions
- Material balance
- Development scores for both sides
- King safety metrics

import random
import chess
from collections import defaultdict

# Piece values for material evaluation
PIECE_VALUES = {
    'Q': 9, 'R': 5, 'B': 3, 'N': 3, 'P': 1, 'K': 0
}

# Center squares (for development evaluation)
CENTER_SQUARES = {'d4', 'e4', 'd5', 'e5'}

def evaluate_move(move_str, pieces, to_play, legal_moves, memory):
    """Evaluate a move based on multiple chess heuristics"""
    board = chess.Board()
    # Reconstruct board state from pieces dictionary
    for square, piece in pieces.items():
        if piece[0] == 'w':
            board.set_piece_at(chess.SQUARE_NAMES.index(square), chess.Piece.from_symbol(piece[1].upper()))
        else:
            board.set_piece_at(chess.SQUARE_NAMES.index(square), chess.Piece.from_symbol(piece[1].lower()))

    move = chess.Move.from_uci(move_str)
    board.push(move)

    score = 0

    # 1. Checkmate is best
    if board.is_checkmate():
        return float('inf')

    # 2. Material gain
    if board.is_capture(move):
        captured_piece = board.piece_at(move.to_square)
        if captured_piece:
            score += PIECE_VALUES[captured_piece.symbol().upper()]

    # 3. Development (moving toward center)
    from_square = chess.SQUARE_NAMES[move.from_square]
    to_square = chess.SQUARE_NAMES[move.to_square]
    piece_type = pieces[from_square][1]

    if piece_type in ['N', 'B']:
        if to_square in CENTER_SQUARES:
            score += 0.5

    # 4. King safety
    if piece_type == 'K':
        # Castling is good
        if move_str in ['O-O', 'O-O-O']:
            score += 1.0
        # Moving king away from center is good
        elif to_square in ['a1', 'a8', 'h1', 'h8']:
            score += 0.3

    # 5. Checks
    if board.is_check():
        score += 0.5

    # 6. Pawn structure (avoid moving pawns that create weaknesses)
    if piece_type == 'P':
        # Avoid moving pawns that create isolated pawns
        file = to_square[0]
        adjacent_files = [chr(ord(file)-1), chr(ord(file)+1)] if file in 'bcdefg' else []
        for adj_file in adjacent_files:
            if f"{adj_file}{to_square[1]}" in pieces and pieces[f"{adj_file}{to_square[1]}"][1] == 'P':
                score -= 0.2

    return score

def policy(pieces: dict[str, str], to_play: str, legal_moves: list[str], memory: dict) -> tuple[str, dict]:
    # Initialize memory if empty
    if not memory:
        memory = {
            'previous_states': [],
            'material_balance': 0,
            'development_score': {'white': 0, 'black': 0},
            'king_safety': {'white': 0, 'black': 0}
        }

    # Evaluate all legal moves
    move_scores = []
    for move in legal_moves:
        score = evaluate_move(move, pieces, to_play, legal_moves, memory)
        move_scores.append((score, move))

    # Sort moves by score (descending)
    move_scores.sort(reverse=True, key=lambda x: x[0])

    # If we have a clear best move, play it
    if len(move_scores) > 0 and move_scores[0][0] > move_scores[1][0] if len(move_scores) > 1 else True:
        best_move = move_scores[0][1]
    else:
        # If multiple moves have same score, add slight randomness
        top_moves = [m for s, m in move_scores if s == move_scores[0][0]]
        best_move = random.choice(top_moves)

    # Update memory with current state
    memory['previous_states'].append(pieces.copy())
    if len(memory['previous_states']) > 10:  # Keep only last 10 states
        memory['previous_states'].pop(0)

    return best_move, memory
