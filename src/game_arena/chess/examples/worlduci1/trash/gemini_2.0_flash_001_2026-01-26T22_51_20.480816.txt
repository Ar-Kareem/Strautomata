The strategy for this policy is to implement a basic chess engine with the following features:

1.  **Legal Move Generation:** Use the python-chess library for legal move generation.
2.  **Evaluation Function:** A simple evaluation function that considers piece values and a few other factors.
3.  **Minimax Search:** A minimax search algorithm with alpha-beta pruning to a limited depth (e.g., depth 2).
4.  **Move Ordering:** Order legal moves to try captures and checks first, hoping to improve alpha-beta pruning efficiency.
5.  **Checkmate detection:** Prioritize checkmating moves above all else.
6.  **Material advantage:** Favor moves that capture opponent's pieces and gain material advantage.
7.  **King safety:** Penalize moves that expose the king to checks.
8.  **Simple Threat Detection:** Identify immediate threats to important pieces and attempt to counter them.

```python
import chess
import random

piece_values = {
    chess.PAWN: 100,
    chess.KNIGHT: 320,
    chess.BISHOP: 330,
    chess.ROOK: 500,
    chess.QUEEN: 900,
    chess.KING: 20000,
}

def evaluate_board(board, to_play):
    """Simple evaluation function based on piece values."""
    score = 0
    for square in chess.SQUARES:
        piece = board.piece_at(square)
        if piece:
            if piece.color == board.turn:  # to_play:
                score += piece_values[piece.piece_type]
            else:
                score -= piece_values[piece.piece_type]
    if not board.is_game_over():
        return score if board.turn == (to_play == 'white') else -score
    elif board.is_checkmate():
        return float('inf') if board.turn != (to_play == 'white') else float('-inf')
    else:  # Stalemate or draw
        return 0

def minimax(board, depth, alpha, beta, maximizing_player, to_play):
    """Minimax search with alpha-beta pruning."""
    if depth == 0 or board.is_game_over():
        return evaluate_board(board, to_play), None

    if maximizing_player:
        max_eval = float('-inf')
        best_move = None
        for move in board.legal_moves:
            board.push(move)
            eval, _ = minimax(board, depth - 1, alpha, beta, False, to_play)
            board.pop()
            if eval > max_eval:
                max_eval = eval
                best_move = move
            alpha = max(alpha, eval)
            if beta <= alpha:
                break
        return max_eval, best_move
    else:
        min_eval = float('inf')
        best_move = None
        for move in board.legal_moves:
            board.push(move)
            eval, _ = minimax(board, depth - 1, alpha, beta, True, to_play)
            board.pop()
            if eval < min_eval:
                min_eval = eval
                best_move = move
            beta = min(beta, eval)
            if beta <= alpha:
                break
        return min_eval, best_move

def order_moves(board):
    """Order moves to prioritize captures and checks."""
    def move_value(move):
        if board.is_capture(move):
            return 2
        elif board.gives_check(move):
            return 1
        else:
            return 0
    return sorted(board.legal_moves, key=move_value, reverse=True)

def policy(pieces: dict[str, str], to_play: str) -> str:
    """
    Selects the next move based on the given board state and player turn.
    """
    board = chess.Board()
    board.clear()
    for square, piece in pieces.items():
        square_index = chess.SQUARE_NAMES.index(square)
        piece_type = piece[1]
        color = piece[0]
        piece_symbol = piece[1]
        if color == 'w':
            piece_color = chess.WHITE
        else:
            piece_color = chess.BLACK

        if piece_symbol == 'P':
            piece_val = chess.PAWN
        elif piece_symbol == 'N':
            piece_val = chess.KNIGHT
        elif piece_symbol == 'B':
            piece_val = chess.BISHOP
        elif piece_symbol == 'R':
            piece_val = chess.ROOK
        elif piece_symbol == 'Q':
            piece_val = chess.QUEEN
        elif piece_symbol == 'K':
            piece_val = chess.KING
        board.set_piece_at(square_index, chess.Piece(piece_val, piece_color))

    board.turn = to_play == 'white'
    board.castling_rights = chess.BB_A1 | chess.BB_H1 | chess.BB_A8 | chess.BB_H8  # Assume all castling rights initially

    # Detect checkmates and return immediately, if found.
    for move in board.legal_moves:
        board.push(move)
        if board.is_checkmate():
            board.pop()
            return move.uci()
        board.pop()

    # Use minimax to find the best move
    _, best_move = minimax(board, depth=2, alpha=float('-inf'), beta=float('inf'), maximizing_player=(to_play == 'white'), to_play=to_play)

    if best_move:
        return best_move.uci()
    else:
        # If minimax fails, return a random legal move
        legal_moves = list(board.legal_moves)
        if legal_moves:
           return random.choice(legal_moves).uci()
        else:
            return "a7a8q" # no legal move, game over, will be invalid, trigger loss, but guaranteed to return a string.
```
