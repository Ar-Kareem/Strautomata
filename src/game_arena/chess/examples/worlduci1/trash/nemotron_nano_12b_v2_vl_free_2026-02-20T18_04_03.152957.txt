### Approach
1. **Move Generation**: The AI generates all possible legal moves for each of its pieces. This includes pawns, rooks, knights, bishops, queens, and kings, including castling for the king.
2. **Legal Move Validation**: After generating potential moves, each move is checked to ensure it does not leave the king in check.
3. **Evaluation**: Legal moves are evaluated based on a heuristic that prioritizes material gains, promotes pawns, and checks for king safety.

def generate_pawn_moves(board, start, color, legal_moves):
    direction = 1 if color == 'w' else -1
    rank = int(start[1])
    # Forward non-capturing move
    new_rank = rank + direction
    new_sq = start[0] + str(new_rank)
    if not board.get(new_sq):
        legal_moves.append(new_sq)
        # Two-square move if possible
        if (rank == 2 and direction == 1) or (rank == 7 and direction == -1):
            new_square_2 = start[0] + str(rank + 2 * direction)
            if not board.get(new_square_2):
                legal_moves.append(new_square_2)
    # Capturing moves
    for dx in [-1, 1]:
        new_col = chr(ord(start[0]) + dx)
        if new_col < 'a' or new_col > 'h':
            continue
        new_sq = new_col + str(rank + direction)
        target = board.get(new_sq)
        if target and target[0] != color:
            cap_promote = f"{color.upper()}{'q'}{target[1]:upper()}"
            legal_moves.append((new_sq, cap_promote))
            for promote in 'qrnb':
                legal_moves.append(f"{new_sq}{promote}")
        # Handle promotion
        if (color == 'w' and new_rank == 8) or (color == 'b' and new_rank == 1):
            for promote in 'qrnb':
                legal_moves.append((new_sq, promote))

def generate_rook_moves(board, square, color, legal_moves):
    directions = [(0, 1), (0, -1), (1, 0), (-1, 0)]
    for dx, dy in directions:
        file, rank = ord(square[0]), int(square[1])
        step_x, step_y = (dx, 0) if dx else (0, dy)
        x, y = file, rank
        while True:
            x += step_x
            y += step_y
            if not ('a' <= chr(x) <= 'h' and 1 <= y <= 8):
                break
            new_sq = chr(x) + str(y)
            if new_sq in board:
                if board[new_sq][0] != color:
                    # Capture
                    legal_moves.append(new_sq)
                break
            legal_moves.append(new_sq)

def generate_bishop_moves(board, square, color, legal_moves):
    directions = [ (1,1), (1,-1), (-1,1), (-1,-1) ]
    for dx, dy in directions:
        x, y = ord(square[0]) - ord('a'), int(square[1]) - 1
        while True:
            x += dx
            y += dy
            if not (0 <= x < 8 and 0 <= y < 8):
                break
            new_sq = chr(x + ord('a')) + str(y + 1)
            if new_sq in board:
                if board[new_sq][0] != color:
                    legal_moves.append(new_sq)
                break
            legal_moves.append(new_sq)
