The strategy for this policy is to use the python-chess library to generate all legal moves from the current board state. Then, it evaluates each move based on a heuristic that prioritizes checkmate if available, followed by checking moves, captures that win material, promotions, and avoiding king moves. This approach ensures that the selected move is legal and follows standard chess objectives.

The policy converts the pieces dictionary into a FEN string, creates a board object, and then applies the move selection heuristic to choose the best move.

import chess
import random

def policy(pieces, to_play):
    def square_to_coords(square):
        file = square[0]
        rank = square[1]
        col = ord(file) - ord('a')
        row = int(rank) - 1
        return row, col

    def pieces_to_fen(pieces):
        board_grid = [[None for _ in range(8)] for _ in range(8)]
        for square, piece_code in pieces.items():
            row, col = square_to_coords(square)
            board_grid[row][col] = piece_code

        fen_rows = []
        for r in range(7, -1, -1):
            fen_row = ""
            empty_count = 0
            for c in range(8):
                piece = board_grid[r][c]
                if piece is None:
                    empty_count += 1
                else:
                    if empty_count > 0:
                        fen_row += str(empty_count)
                        empty_count = 0
                    color = piece[0]
                    piece_type = piece[1]
                    fen_symbol = piece_type
                    if color == 'b':
                        fen_symbol = fen_symbol.lower()
                    fen_row += fen_symbol
            if empty_count > 0:
                fen_row += str(empty_count)
            fen_rows.append(fen_row)
        return '/'.join(fen_rows)

    def compute_castling_rights(pieces):
        rights = ""
        # White castling
        if pieces.get('e1') == 'wK':
            if pieces.get('h1') == 'wR' and pieces.get('f1') is None and pieces.get('g1') is None:
                rights += 'K'
            if pieces.get('a1') == 'wR' and pieces.get('b1') is None and pieces.get('c1') is None and pieces.get('d1') is None:
                rights += 'Q'
        # Black castling
        if pieces.get('e8') == 'bK':
            if pieces.get('h8') == 'bR' and pieces.get('f8') is None and pieces.get('g8') is None:
                rights += 'k'
            if pieces.get('a8') == 'bR' and pieces.get('b8') is None and pieces.get('c8') is None and pieces.get('d8') is None:
                rights += 'q'
        return rights if rights != "" else "-"

    def select_move(legal_moves, board):
        if not legal_moves:
            return random.choice(legal_moves) if legal_moves else "e2e4"  # Fallback, though should not happen

        piece_values = {
            'p': 1, 'n': 3, 'b': 3, 'r': 5, 'q': 9, 'k': 0,
            'P': 1, 'N': 3, 'B': 3, 'R': 5, 'Q': 9, 'K': 0
        }

        # Check for checkmate moves
        for move in legal_moves:
            temp_board = board.copy()
            temp_board.push_uci(move)
            if temp_board.is_checkmate():
                return move

        # Check for checking moves
        checking_moves = []
        for move in legal_moves:
            temp_board = board.copy()
            temp_board.push_uci(move)
            if temp_board.is_check():
                checking_moves.append(move)
        if checking_moves:
            return random.choice(checking_moves)

        # Check for captures that win material
        capture_moves = []
        for move in legal_moves:
            if board.is_capture(move):
                to_square = move[2:4]
                square = chess.parse_square(to_square)
                piece = board.piece_at(square)
                if piece is not None:
                    value = piece_values.get(piece.symbol(), 0)
                    capture_moves.append((move, value))
        if capture_moves:
            capture_moves.sort(key=lambda x: x[1], reverse=True)
            return capture_moves[0][0]

        # Check for promotions
        promotion_moves = [move for move in legal_moves if len(move) == 5]
        if promotion_moves:
            return random.choice(promotion_moves)

        # Avoid moving the king if possible
        non_king_moves = []
        for move in legal_moves:
            from_square = move[0:2]
            from_piece = board.piece_at(chess.parse_square(from_square))
            if from_piece is not None and from_piece.piece_type != chess.KING:
                non_king_moves.append(move)
        if non_king_moves:
            return random.choice(non_king_moves)
        else:
            return random.choice(legal_moves)

    piece_placement = pieces_to_fen(pieces)
    active_color = 'w' if to_play == 'white' else 'b'
    castling_rights = compute_castling_rights(pieces)
    en_passant = '-'
    halfmove = '0'
    fullmove = '1'
    fen = f"{piece_placement} {active_color} {castling_rights} {en_passant} {halfmove} {fullmove}"

    board = chess.Board(fen)
    legal_moves = [move.uci() for move in board.legal_moves]

    return select_move(legal_moves, board)
