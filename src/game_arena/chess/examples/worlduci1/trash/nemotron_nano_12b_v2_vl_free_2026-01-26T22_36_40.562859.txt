### 🧠 **Strategic Heuristic Overview**

This AI prioritizes moves based on the following **game theory principles**:

1. **Check or Checkmate**: If the player's king is in check, the AI tries to:
   - Move the king to a safe square.
   - Capture the attacking piece.
   - Block the attack (if possible).

2. **Material Gain**: If no check exists, the AI looks for moves that:
   - Capture the most valuable enemy piece.
   - Prioritize by value: `Queen > Rook > Bishop/Knight > Pawn`.

3. **Development and Control**: If no captures are available, the AI tries to:
   - Move knights and bishops toward the center.
   - Develop the queen if safe.
   - Avoid exposing the king.

4. **Stalemate Avoidance**: Avoid moves that lead to a stalemate (e.g., unnecessary king moves).

This heuristic-based approach is not perfect but is fast, reliable, and effective for early to mid-game positions.

from copy import deepcopy

# Helper function to parse a square into (file, rank) coordinates
def parse_square(square):
    return square[0], int(square[1:])

# Helper function to get the king's square for a given color
def find_king(pieces, color):
    for square, piece in pieces.items():
        if piece == f'{color}K':
            return square
    return None

# Helper function to check if a square is occupied by a friend/opponent
def is_occupied(board, square):
    return square in board

# Helper function to evaluate a piece for capturing
def get_piece_value(piece_type):
    values = {'P': 1, 'N': 3, 'B': 3, 'R': 5, 'Q': 9, 'K': 0}
    return values.get(piece_type, 0)

# Helper function to generate all possible moves for a piece
def generate_piece_moves(piece_type, from_sq, pieces, color):
    moves = []
    file, rank = parse_square(from_sq)
    rank -= 1  # convert to 0-based index

    if piece_type == 'P':
        # Pawn movement (white pawn)
        if color == 'w':
            # Move forward
            next_rank = rank + 1
            if next_rank < 8:
                to_sq = f'{file}{next_rank + 1}'
                if not is_occupied(pieces, to_sq):
                    moves.append(to_sq)
            # Initial two squares
            if rank == 1:
                to_sq = f'{file}{next_rank + 1}'
                if not is_occupied(pieces, to_sq):
                    moves.append(to_sq)
            # Capture diagonally
            for df in (-1, 1):
                to_file = chr(ord(file) + df)
                to_sq = f'{to_file}{next_rank + 1}'
                if is_occupied(pieces, to_sq) and pieces[to_sq][0] != color:
                    moves.append(to_sq)
        else:
            # Black pawn movement (similar to white but moving down)
            next_rank = rank - 1
            if next_rank >= 0:
                to_sq = f'{file}{next_rank + 1}'
                if not is_occupied(pieces, to_sq):
                    moves.append(to_sq)
            # Initial two squares
            if rank == 6:
                to_sq = f'{file}{next_rank - 1}'
                if not is_occupied(pieces, to_sq):
                    moves.append(to_sq)
            # Capture diagonally
            for df in (-1, 1):
                to_file = chr(ord(file) + df)
                to_sq = f'{to_file}{next_rank - 1}'
                if is_occupied(pieces, to_sq) and pieces[to_sq][0] != color:

    elif piece_type == 'N':
        # Knight moves
        knight_moves = [
            (file, rank - 2), (file, rank - 1),
            (chr(ord(file) + 1), rank - 2), (chr(ord(file) + 2), rank - 1),
            (chr(ord(file) + 2), rank + 1), (chr(ord(file) + 1), rank + 2),
            (file, rank + 2), (chr(ord(file) - 1), rank + 2)
        ]
        for to_sq in knight_moves:
            if to_sq[0] in 'abcdefgh' and to_sq[1] in '12345678':
                moves.append(to_sq)
    elif piece_type == 'B':
        # Bishop moves
        for dx, dy in [(1, 1), (1, -1), (-1, 1), (-1, -1)]:
            current_file, current_rank = file, rank
            while True:
                current_rank += dy
                current_file = chr(ord(current_file) + dx)
                if current_file < 'a' or current_file > 'h' or current_rank < 0 or current_rank > 7:
                    break
                to_sq = f'{current_file}{current_rank + 1}'
                if is_occupied(pieces, to_sq):
                    if pieces[to_sq][0] != color:
                        moves.append(to_sq)
                    break
                else:
                    moves.append(to_sq)
    elif piece_type == 'R':
        # Rook moves
        moves = []
        for dx, dy in [(0, 1), (0, -1), (1, 0), (-1, 0)]:
            current_file, current_rank = file, rank
            while True:
                current_rank += dy
                if current_file[0] == 'a' and dx == -1:
                    break
                if current_rank < 0 or current_rank >= 8:
                    break
                to_sq = f'{current_file}{current_rank + 1}'
                if is_occupied(pieces, to_sq):
                    if pieces[to_sq][0] != color:
                        moves.append(to_sq)
                    break
                else:
                    moves.append(to_sq)
    elif piece_type == 'Q':
        moves = generate_piece_moves('B', from_sq, pieces, color)
        moves += generate_piece_moves('R', from_sq, pieces, color)
    elif piece_type == 'K':
        # King moves
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                new_file = chr(ord(file) + dx)
                new_rank = rank + dy
                if 'a' <= new_file <= 'h' and 0 <= new_rank <= 7:
                    to_sq = f'{new_file}{new_rank + 1}'
                    if not is_occupied(pieces, to_sq) or pieces.get(to_sq, '')[0] != color:
                        moves.append(to_sq)
    return moves

def evaluate_move(move, pieces, to_play):

    captured = pieces.get(to_sq, '')

# Main policy function
def policy(pieces: dict[str, str], to_play: str) -> str:
    current_color = 'w' if to_play == 'white' else 'b'
    opponent_color = 'b' if current_color == 'w' else 'w'
    legal_moves = []
    king_sq = find_king(pieces, current_color)

    # Step 1: If the king is in check, generate escape moves
    if is_in_check(pieces, to_play):
        for square in pieces:
            if pieces[square][0] == current_color:
                for move in generate_piece_moves(pieces[square][1], square, pieces, current_color):
                    new_pieces = deepcopy(pieces)
                    new_pieces[move] = pieces[square]
                    del new_pieces[square]
                    if not is_in_check(new_pieces, to_play):
                        legal_moves.append(move)
        if legal_moves:
            return max(legal_moves, key=lambda m: evaluate_move(m, pieces, to_play))
        else:
            return 'e1g1'  # Default move if no valid escape

    # Step 2: Otherwise, find the best move based on value
    best_move = ''
    best_score = -float('inf')
    for square in pieces:
        if pieces[square][0] != current_color:
            continue
        for move in generate_piece_moves(pieces[square][1], square, pieces, current_color):
            new_pieces = deepcopy(pieces)
            new_pieces[move] = pieces[square]
            del new_pieces[square]
            # Simulate move and check if it's legal
            if not is_in_check(new_pieces, to_play):
                score = evaluate_move(move, pieces, to_play)
                if score > best_score:
                    best_move = move
                    best_score = score
    return best_move if best_move else 'e1c1'  # default move
