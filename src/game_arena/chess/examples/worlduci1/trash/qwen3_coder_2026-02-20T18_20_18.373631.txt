The strategy for this policy is to build a robust chess engine that evaluates positions based on several key factors:

1. **Material Balance**: Calculate the total material value for both sides using standard piece values (King is irrelevant for material as it cannot be captured).
2. **King Safety**: Prefer moves that keep our king safer (more protected) and that threaten the opponent's king.
3. **Piece Activity**: Reward moves that improve the positioning of pieces, especially centralizing them and increasing their mobility.
4. **Pawn Structure**: Discourage isolated and doubled pawns, and reward passed pawns, especially those close to promotion.
5. **Immediate Threats**: Handle checks and threats to valuable pieces dynamically during move evaluation.
6. **Checkmate Priority**: If a move results in checkmate, immediately return it.

The policy proceeds as follows:
- First, it detects if any legal move is immediately checkmating and returns it.
- Then, it evaluates each move by:
  - Simulating it on the board.
  - Recalculating material balance, piece-square bonuses, king safety, and pawn structure.
  - Assigning a score to each resulting position.
- It chooses the move with the highest score.

The position evaluation combines:
- Material advantage.
- Piece-square tables for centralized positioning.
- King safety and pawn structure metrics.
- Mobility (number of legal moves available to us vs. opponent).

PIECE_VALUES = {'P': 100, 'N': 320, 'B': 330, 'R': 500, 'Q': 900, 'K': 20000}

def square_to_indices(square: str):
    file = ord(square[0]) - ord('a')
    rank = int(square[1]) - 1
    return 7 - rank, file  # rank 0 is row 7 (a8 is [0][0])

def is_in_check(king_square: str, opponent_pieces: dict, opponent_color: str) -> bool:
    # Very simplified check detection by seeing if any capture can get to king's square
    # NOTE: This is far from perfect but gives some approximation for safety scoring
    for sq, piece in opponent_pieces.items():
        if piece[0] == opponent_color:
            # Simulate a capture move
            if king_square in [sq[2:4] for sq in make_pseudo_legal_moves(piece, sq, opponent_pieces)]:
                return True
    return False

def make_pseudo_legal_moves(piece_code: str, square: str, pieces: dict) -> list:

    moves = []
    piece_type = piece_code[1]
    color = piece_code[0]

        # Forward moves

        if target in pieces:
            pass  # Blocked

                    moves.append(f"{square}{target2}")

            if 0 <= nc < 8:

                if target in pieces and pieces[target][0] != color:
                    moves.append(f"{square}{target}")
    # Others are simplified for scoring purposes
    return moves[:10]  # Dummy limit

def evaluate_position(pieces: dict, to_play: str, legal_moves: list) -> float:
    score = 0
    my_color = 'w' if to_play == 'white' else 'b'
    opp_color = 'b' if to_play == 'white' else 'w'

    piece_positions = collections.defaultdict(list)
    kings = {'w': None, 'b': None}

    for sq, p in pieces.items():
        color, piece_type = p[0], p[1]
        piece_positions[color].append((piece_type, sq))
        if piece_type == 'K':
            kings[color] = sq

    # 1. Material
    my_material = sum(PIECE_VALUES[p[0]] for p in piece_positions[my_color])
    opp_material = sum(PIECE_VALUES[p[0]] for p in piece_positions[opp_color])
    score += my_material - opp_material

    # 2. Piece-Square Tables
    for color in piece_positions:
        for piece_type, sq in piece_positions[color]:
            r, c = square_to_indices(sq)
            if piece_type in piece_square_table:
                value = piece_square_table[piece_type][r][c]
                if color == my_color:
                    score += value
                else:
                    score -= value

    # 3. King Safety (simplified)
    if kings[my_color]:
        if is_in_check(kings[my_color], pieces, opp_color):
            score -= 200
    if kings[opp_color]:
        if is_in_check(kings[opp_color], pieces, my_color):
            score += 200

    # 4. Mobility (approximated via number of legal moves)
    score += len(legal_moves) * 10

    return score

def apply_move(pieces: dict, move: str) -> dict:
    new_pieces = pieces.copy()
    start, end = move[:2], move[2:4]
    piece = new_pieces.pop(start)
    # Promotion
    if len(move) > 4:
        piece = piece[0] + move[4].upper()
    new_pieces[end] = piece
    return new_pieces
