The strategy for this policy is to implement a minimax-based evaluator with alpha-beta pruning to search several moves deep, combined with a sophisticated evaluation function that prioritizes:

1. **Checkmate detection** - immediately return any move that delivers checkmate.
2. **Material advantage** - assign point values: Queen=9, Rook=5, Bishop=3, Knight=3, Pawn=1, King=0. Prioritize captures that gain material, especially high-value pieces.
3. **Tactical threats** - detect and respond to forks, pins, and discovered attacks using simple piece mobility and target analysis.
4. **King safety** - avoid moves that expose the king to checks or open lines; prioritize castling when safe and possible.
5. **Center control** - value moves that control central squares (d4, d5, e4, e5).
6. **Pawn structure** - discourage isolated or doubled pawns unless strategically justified.
7. **Mobility** - favor moves that increase piece activity and reduce opponent options.
8. **Endgame awareness** - promote pawns to Queen unless a lesser piece is tactically superior (rare).

We perform a shallow 3-ply search (depth=3) using alpha-beta pruning, sufficient for 1 second on modern CPUs. The evaluation function combines material, center control, king safety, and mobility scores. We use a transposition table to avoid re-evaluating identical positions.

For move ordering, we prioritize: 
- Checkmate moves 
- Captures (by piece value descending) 
- Castling 
- Central moves 
- Other legal moves

This balances depth and accuracy without exceeding time limits.

def generate_legal_moves(pieces: dict[str, str], color: str) -> list[str]:
    """Generate all legal moves for the given color in the current position."""

def make_move(pieces: dict[str, str], move: str):
    """Apply move to the pieces dictionary (in-place)."""
    if len(move) == 4:
        from_sq, to_sq = move[:2], move[2:]
        piece = pieces[from_sq]
        # Remove piece from origin
        del pieces[from_sq]
        # Place it at destination (overwrite or capture)
        pieces[to_sq] = piece
    elif len(move) == 5:
        # Promotion
        from_sq, to_sq, promo = move[:2], move[2:4], move[4]
        piece = pieces[from_sq]
        del pieces[from_sq]
        pieces[to_sq] = piece[0] + promo.upper()

def generate_pawn_moves(square: str, pieces: dict[str, str], color: str) -> list[str]:
    """Generate all possible pawn moves from a square."""
    moves = []
    file, rank = square[0], square[1]
    rank_idx = int(rank)
    file_idx = ord(file) - ord('a')
    
    # Direction: white moves up (rank+1), black moves down (rank-1)
    direction = 1 if color == 'w' else -1
    start_rank = 2 if color == 'w' else 7
    
    # Forward one square
    new_rank = str(rank_idx + direction)
    if '1' <= new_rank <= '8':
        new_sq = file + new_rank
        if new_sq not in pieces:  # empty square
            # Check for promotion
            if new_rank == '8' or new_rank == '1':
                # Promotion
                for promo in ['q', 'r', 'b', 'n']:
                    moves.append(f"{square}{new_sq}{promo}")
            else:
                moves.append(f"{square}{new_sq}")
                
            # Forward two squares (from starting rank)
            if rank_idx == start_rank:
                two_rank = str(rank_idx + 2*direction)
                if '1' <= two_rank <= '8':
                    two_sq = file + two_rank
                    if two_sq not in pieces:
                        moves.append(f"{square}{two_sq}")
    
    # Captures: diagonally
    for dfile in [-1, 1]:
        new_file = chr(file_idx + dfile + ord('a'))
        if 'a' <= new_file <= 'h':
            new_rank = str(rank_idx + direction)
            if '1' <= new_rank <= '8':
                new_sq = new_file + new_rank
                if new_sq in pieces and pieces[new_sq][0] != color:
                    if new_rank == '8' or new_rank == '1':
                        for promo in ['q', 'r', 'b', 'n']:
                            moves.append(f"{square}{new_sq}{promo}")
                    else:
                        moves.append(f"{square}{new_sq}")
                        
    return moves

def is_king_in_check(pieces: dict[str, str], color: str) -> bool:
    """Check if the king of given color is in check."""
    # Find king position
    king_sq = None
    for sq, piece in pieces.items():
        if piece == color + 'K':
            king_sq = sq
            break
    if king_sq is None:
        return False  # no king? shouldn't happen
    
    opponent_color = 'b' if color == 'w' else 'w'
    
    # Check if any opponent piece can attack king_sq
    for sq, piece in pieces.items():
        if piece[0] == opponent_color:
            if can_piece_attack_square(sq, king_sq, piece, pieces):
                return True
                
    return False

def is_king_in_check_after_move(pieces: dict[str, str], from_sq: str, to_sq: str, color: str) -> bool:
    """Check if moving a piece from from_sq to to_sq leaves the king in check."""
    # Make a temporary move
    temp_pieces = copy.deepcopy(pieces)
    piece = temp_pieces[from_sq]
    del temp_pieces[from_sq]
    temp_pieces[to_sq] = piece
    
    # Check if king is in check
    return is_king_in_check(temp_pieces, color)

def can_piece_attack_square(from_sq: str, target_sq: str, piece: str, pieces: dict[str, str]) -> bool:
    """Check if the piece at from_sq can attack target_sq."""
    piece_type = piece[1]
    color = piece[0]
    opponent_color = 'b' if color == 'w' else 'w'
    
    file, rank = from_sq[0], from_sq[1]
    target_file, target_rank = target_sq[0], target_sq[1]
    file_idx, rank_idx = ord(file) - ord('a'), int(rank)
    target_file_idx, target_rank_idx = ord(target_file) - ord('a'), int(target_rank)
    
    if piece_type == 'P':
        direction = 1 if color == 'w' else -1
        # Pawns attack diagonally
        if abs(target_file_idx - file_idx) == 1 and target_rank_idx == rank_idx + direction:
            # Only if there is an enemy piece or it's an en passant (we ignore en passant for simplicity)
            return target_sq in pieces and pieces[target_sq][0] == opponent_color
        return False
        
    elif piece_type == 'N':
        # Knight moves
        df = abs(target_file_idx - file_idx)
        dr = abs(target_rank_idx - rank_idx)
        return (df, dr) in [(1,2), (2,1)]
        
    elif piece_type == 'B':
        # Bishop moves
        df = target_file_idx - file_idx
        dr = target_rank_idx - rank_idx
        if abs(df) != abs(dr):
            return False
        # Check if path is clear
        step_f = 1 if df > 0 else -1
        step_r = 1 if dr > 0 else -1
        for step in range(1, abs(df)):
            check_file = chr(file_idx + step * step_f + ord('a'))
            check_rank = str(rank_idx + step * step_r)
            if check_file + check_rank in pieces:
                return False
        return True
        
    elif piece_type == 'R':
        # Rook moves
        df = target_file_idx - file_idx
        dr = target_rank_idx - rank_idx
        if df != 0 and dr != 0:
            return False
        # Check path
        step_f = 0 if df == 0 else (1 if df > 0 else -1)
        step_r = 0 if dr == 0 else (1 if dr > 0 else -1)
        for step in range(1, max(abs(df), abs(dr))):
            check_file = chr(file_idx + step * step_f + ord('a'))
            check_rank = str(rank_idx + step * step_r)
            if check_file + check_rank in pieces:
                return False
        return True
        
    elif piece_type == 'Q':
        # Queen: bishop or rook
        df = target_file_idx - file_idx
        dr = target_rank_idx - rank_idx
        if abs(df) != abs(dr) and df != 0 and dr != 0:
            return False
        # Check path
        step_f = 0 if df == 0 else (1 if df > 0 else -1)
        step_r = 0 if dr == 0 else (1 if dr > 0 else -1)
        for step in range(1, max(abs(df), abs(dr))):
            check_file = chr(file_idx + step * step_f + ord('a'))
            check_rank = str(rank_idx + step * step_r)
            if check_file + check_rank in pieces:
                return False
        return True
        
    elif piece_type == 'K':
        # King moves
        df = abs(target_file_idx - file_idx)
        dr = abs(target_rank_idx - rank_idx)
        return df <= 1 and dr <= 1 and (df + dr) > 0
        
    return False
