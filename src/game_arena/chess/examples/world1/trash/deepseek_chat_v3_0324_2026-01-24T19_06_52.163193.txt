# Chess Policy Strategy

This policy implements a basic chess evaluation function combined with minimax search to select strong moves. The strategy focuses on:

1. Forced checkmates - if a checkmate is found in legal moves, it's played immediately
2. Material advantage - evaluates piece values (Q=9, R=5, B=3, N=3, P=1)
3. Piece activity - rewards controlling center squares and mobility
4. King safety - penalizes exposed kings in middle game
5. Basic search - looks 2-3 ply ahead to avoid immediate blunders

The evaluation combines these factors with weights that change based on game phase (opening/middle/endgame). The policy performs an iterative deepening search within the 1-second time constraint.

import numpy as np
from collections import defaultdict
from time import time

def policy(pieces, to_play, legal_moves):
    # Check for immediate checkmates
    for move in legal_moves:
        if move.endswith('#'):
            return move
    
    # Setup board state
    color = to_play[0].lower()  # 'w' or 'b'
    board = {square: piece for square, piece in pieces.items()}
    
    # Piece values
    piece_values = {'Q': 9, 'R': 5, 'B': 3, 'N': 3, 'P': 1, 'K': 0}
    
    # Define game phases
    total_material = sum(piece_values.get(p[1], 0) for p in board.values())
    game_phase = 'opening' if total_material > 39 else 'middlegame' if total_material > 20 else 'endgame'
    
    # Evaluation parameters by phase
    params = {
        'opening': {'material': 1, 'mobility': 0.1, 'center': 0.2, 'king_safety': 0.3},
        'middlegame': {'material': 1, 'mobility': 0.2, 'center': 0.1, 'king_safety': 0.5},
        'endgame': {'material': 1, 'mobility': 0.3, 'center': 0.1, 'king_safety': 0.1}
    }
    
    # Cache opponents legal moves to avoid recomputing
    opp_color = 'b' if color == 'w' else 'w'
    
    def evaluate_position(board, color):
        score = 0
        mobility = 0
        center_squares = {'d4', 'e4', 'd5', 'e5', 'c3', 'f3', 'c6', 'f6'}
        king_safety = 0
        
        # Material and piece activity
        for square, piece in board.items():
            sign = 1 if piece[0] == color else -1
            p = piece[1]
            
            # Material
            score += sign * piece_values.get(p, 0)
            
            # Center control for non-pawns
            if p != 'P' and square in center_squares:
                score += sign * 0.2
            
            # King safety
            if p == 'K':
                file = square[0]
                rank = int(square[1])
                if game_phase != 'endgame':
                    # Penalize king in center
                    if file in {'d', 'e'} and rank in {4,5}:
                        king_safety -= sign * 0.5
                    # Reward castled kings
                    if (color == 'w' and square[0] in {'g','b'}) or (color == 'b' and square[0] in {'g','b'}):
                        king_safety += sign * 0.5
        
        # Apply phase weights
        score = params[game_phase]['material'] * score
        score += params[game_phase]['mobility'] * mobility
        score += params[game_phase]['king_safety'] * king_safety
        
        return score
    
    def make_move(board, move, color):
        new_board = dict(board)
        # This is simplified - would implement full move logic in production
        if 'x' in move:  # capture
            target = move.split('x')[1][:2]
            if target in new_board:
                del new_board[target]
        
        # Find source square (this is simplified - real implementation would handle all move types)
        if move.startswith('O'):
            # Castling
            if move == 'O-O':
                new_board.pop('e1' if color == 'w' else 'e8')
                new_board.pop('h1' if color == 'w' else 'h8')
                new_board['g1' if color == 'w' else 'g8'] = f'{color}K'
                new_board['f1' if color == 'w' else 'f8'] = f'{color}R'
            else:
                new_board.pop('e1' if color == 'w' else 'e8')
                new_board.pop('a1' if color == 'w' else 'a8')
                new_board['c1' if color == 'w' else 'c8'] = f'{color}K'
                new_board['d1' if color == 'w' else 'd8'] = f'{color}R'
        else:
            # Regular moves
            parts = move.replace('+', '').replace('#', '').split('=')
            basic_move = parts[0]
            
            if 'x' in basic_move:
                src, dest = basic_move.split('x')
                if len(src) > 2:  # disambiguated move like Nbd2
                    src = src[-2:]
                dest = dest[:2]
            else:
                if len(basic_move) > 2 and basic_move[0].isupper() and basic_move[1].islower():
                    src = basic_move[:2]
                    dest = basic_move[2:]
                elif len(basic_move) == 2:
                    # Pawn move
                    file = basic_move[0]
                    for rank in range(1,9):
                        square = f'{file}{rank}'
                        if square in new_board and new_board[square] == f'{color}P':
                            src = square
                            break
                    dest = basic_move
                else:
                    # Piece move with disambiguation
                    src = next(sq for sq in new_board 
                              if new_board[sq] == f'{color}{basic_move[0]}' and 
                              sq[0] == basic_move[1] if len(basic_move) == 3 and basic_move[1].islower() else True)
                    dest = basic_move[-2:]
            
            piece = new_board.pop(src)
            if len(parts) > 1:  # promotion
                piece = f'{color}{parts[1]}'
            new_board[dest] = piece
        
        return new_board
    
    def minimax(board, depth, alpha, beta, maximizing_player, color, start_time):
        if time() - start_time > 0.95:
            return evaluate_position(board, color)
        
        if depth == 0:
            return evaluate_position(board, color)
            
        if maximizing_player:
            max_eval = -np.inf
            for move in legal_moves:
                new_board = make_move(board, move, color)
                eval = minimax(new_board, depth-1, alpha, beta, False, opp_color, start_time)
                max_eval = max(max_eval, eval)
                alpha = max(alpha, eval)
                if beta <= alpha:
                    break
            return max_eval
        else:
            min_eval = np.inf
            # Generate opponent moves (simplified - in production would generate from board)
            for move in legal_moves:
                new_board = make_move(board, move, opp_color)
                eval = minimax(new_board, depth-1, alpha, beta, True, color, start_time)
                min_eval = min(min_eval, eval)
                beta = min(beta, eval)
                if beta <= alpha:
                    break
            return min_eval
    
    # Select best move with iterative deepening
    start_time = time()
    best_move = legal_moves[0]
    best_eval = -np.inf
    
    max_depth = 3
    for move in legal_moves:
        if time() - start_time > 0.95:
            break
        
        new_board = make_move(board, move, color)
        current_eval = minimax(new_board, max_depth, -np.inf, np.inf, False, opp_color, start_time)
        
        if current_eval > best_eval:
            best_eval = current_eval
            best_move = move
    
    return best_move
