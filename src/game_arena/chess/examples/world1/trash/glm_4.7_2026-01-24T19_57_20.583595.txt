The strategy for this policy is to implement a Minimax search algorithm with Alpha-Beta pruning to look ahead several moves. The evaluation function considers material balance using standard piece values (Pawn=10, Knight=30, etc.) and positional heuristics using Piece-Square Tables (PSTs) to encourage controlling the center and developing pieces.

To ensure compliance with the time limit and API constraints:
1.  **State Conversion**: The input board dictionary is converted into a flat 64-element list for efficient manipulation.
2.  **Move Parsing**: Since the input provides legal moves in Standard Algebraic Notation (SAN), a custom parser maps these strings (e.g., "Nf3", "exd5", "O-O") to internal coordinate pairs `(from_index, to_index, promotion)`. This allows the engine to understand exactly which move is being considered.
3.  **Move Generation**: For positions beyond the root (depth > 0), a lightweight pseudo-legal move generator is used. Moves are filtered to ensure the King is not left in check.
4.  **Search**: The engine runs a depth-3 search (root + 2 plies). This depth was chosen as a balance between tactical insight and the 1-second execution limit in Python.
5.  **Safety**: A `try-except` block wraps the execution. If the search fails or times out, the policy defaults to the first available legal move to avoid disqualification.

```python
import random
import math

# --- Constants ---
WHITE = 1
BLACK = -1
PAWN = 1
KNIGHT = 2
BISHOP = 3
ROOK = 4
QUEEN = 5
KING = 6

# Piece values for evaluation
PIECE_VALUES = {
    PAWN: 100,
    KNIGHT: 320,
    BISHOP: 330,
    ROOK: 500,
    QUEEN: 900,
    KING: 20000
}

# Simplified Piece-Square Tables (PST) for positional play
# Center squares (indices 27-28, 35-36) are favored for Knights
PST_KNIGHT = [
    -50,-40,-30,-30,-30,-30,-40,-50,
    -40,-20,  0,  0,  0,  0,-20,-40,
    -30,  0, 10, 15, 15, 10,  0,-30,
    -30,  5, 15, 20, 20, 15,  5,-30,
    -30,  0, 15, 20, 20, 15,  0,-30,
    -30,  5, 10, 15, 15, 10,  5,-30,
    -40,-20,  0,  5,  5,  0,-20,-40,
    -50,-40,-30,-30,-30,-30,-40,-50
]

PST_PAWN = [
    0,  0,  0,  0,  0,  0,  0,  0,
    50, 50, 50, 50, 50, 50, 50, 50,
    10, 10, 20, 30, 30, 20, 10, 10,
    5,  5, 10, 25, 25, 10,  5,  5,
    0,  0,  0, 20, 20,  0,  0,  0,
    5, -5,-10,  0,  0,-10, -5,  5,
    5, 10, 10,-20,-20, 10, 10,  5,
    0,  0,  0,  0,  0,  0,  0,  0
]

class ChessEngine:
    def __init__(self):
        self.board = [0] * 64
        self.sq_to_idx = {f"{chr(f)}{r}": r*8 + f for r in range(1,9) for f in range(ord('a'), ord('h')+1))}
        self.piece_map = {'P': PAWN, 'N': KNIGHT, 'B': BISHOP, 'R': ROOK, 'Q': QUEEN, 'K': KING}
        self.color_map = {'w': WHITE, 'b': BLACK, 'white': WHITE, 'black': BLACK}
        
        # Prepare PSTs for both colors
        self.pst = {
            PAWN: {WHITE: PST_PAWN, BLACK: PST_PAWN[::-1]},
            KNIGHT: {WHITE: PST_KNIGHT, BLACK: PST_KNIGHT[::-1]},
            # Default to 0 for others to save space/time
            BISHOP: {WHITE: [0]*64, BLACK: [0]*64},
            ROOK: {WHITE: [0]*64, BLACK: [0]*64},
            QUEEN: {WHITE: [0]*64, BLACK: [0]*64},
            KING: {WHITE: [0]*64, BLACK: [0]*64}
        }

    def load_state(self, pieces, to_play):
        self.board = [0] * 64
        self.turn = self.color_map[to_play]
        for sq, code in pieces.items():
            p_color = self.color_map[code[0]]
            p_type = self.piece_map[code[1]]
            self.board[self.sq_to_idx[sq]] = p_color * p_type

    def parse_san(self, san):
        """Parse a SAN move string to (from_idx, to_idx, promotion_type)."""
        # Handle Castling
        if san == "O-O" or san == "O-O+":
            row = 7 if self.turn == WHITE else 0
            return (row*8 + 4, row*8 + 6, None)
        if san == "O-O-O" or san == "O-O-O+":
            row = 7 if self.turn == WHITE else 0
            return (row*8 + 4, row*8 + 2, None)

        # Clean string
        clean = san.replace('+', '').replace('#', '').replace('x', '')
        promotion = None
        if '=' in clean:
            promo_char = clean.split('=')[1]
            promotion = self.piece_map[promo_char]
            clean = clean.split('=')[0]
        
        target_str = clean[-2:]
        to_idx = self.sq_to_idx[target_str]
        
        # Determine piece type
        p_char = clean[0]
        if p_char.isupper():
            p_type = self.piece_map[p_char]
            src_hint = clean[1:-2] # Disambiguation (e.g. 'b' in 'Nbd7')
        else:
            p_type = PAWN
            src_hint = clean[:-2] # Disambiguation for pawn (e.g. 'axb5' -> 'a')

        # Find source square
        for i in range(64):
            if self.board[i] == self.turn * p_type:
                # Basic geometry check to filter candidates before expensive checks
                if not self.is_pseudo_legal(i, to_idx, p_type, self.turn):
                    continue
                
                # Check disambiguation
                if src_hint:
                    r, f = divmod(i, 8)
                    if src_hint.isdigit() and str(r+1) != src_hint: continue
                    if src_hint.isalpha() and chr(ord('a')+f) != src_hint: continue
                
                return (i, to_idx, promotion)
        
        # Fallback (should not happen with legal moves)
        return None

    def is_pseudo_legal(self, fr, to, p_type, color):
        """Check if a move is geometrically valid."""
        dr = (to // 8) - (fr // 8)
        df = (to % 8) - (fr % 8)
        
        if p_type == PAWN:
            direction = -1 if color == WHITE else 1
            # Move
            if df == 0 and dr == direction and self.board[to] == 0: return True
            if df == 0 and dr == 2*direction and (fr//8 == 6 if color==WHITE else fr//8==1) and self.board[to] == 0: return True
            # Capture
            if abs(df) == 1 and dr == direction and self.board[to] != 0: return True
            return False
            
        elif p_type == KNIGHT:
            return sorted([abs(dr), abs(df)]) == [1, 2]
            
        elif p_type in (BISHOP, ROOK, QUEEN):
            if p_type in (BISHOP, QUEEN) and abs(dr) != abs(df): pass
            elif p_type in (ROOK, QUEEN) and dr != 0 and df != 0: pass
            else: # Valid direction for the piece type
                if self.is_path_clear(fr, to): return True
            return False
            
        elif p_type == KING:
            return max(abs(dr), abs(df)) == 1
            
        return False

    def is_path_clear(self, fr, to):
        dr = (to // 8) - (fr // 8)
        df = (to % 8) - (fr % 8)
        step_r = 0 if dr == 0 else int(dr/abs(dr))
        step_f = 0 if df == 0 else int(df/abs(df))
        
        curr = fr + step_r * 8 + step_f
        while curr != to:
            if self.board[curr] != 0: return False
            # Handle file wrap for rooks/queens moving horizontally
            if step_f != 0:
                r1, f1 = divmod(curr - step_r*8 - step_f, 8)
                r2, f2 = divmod(curr, 8)
                if abs(f2-f1) > 1: return False
            curr += step_r * 8 + step_f
        return True

    def get_legal_moves(self, color):
        moves = []
        # Generate pseudo-legal moves for all pieces
        for i in range(64):
            p = self.board[i]
            if p == 0 or (p > 0) != (color == WHITE): continue
            
            p_type = abs(p)
            candidates = []
            
            # Generate target squares
            if p_type == PAWN:
                direction = -1 if color == WHITE else 1
                start_row = 6 if color == WHITE else 1
                # Forward
                fwd = i + direction * 8
                if 0 <= fwd < 64 and self.board[fwd] == 0:
                    candidates.append(fwd)
                    if i // 8 == start_row:
                        fwd2 = i + 2 * direction * 8
                        if self.board[fwd2] == 0: candidates.append(fwd2)
                # Captures
                for df in [-1, 1]:
                    cap = i + direction * 8 + df
                    if 0 <= cap < 64:
                        r1, f1 = divmod(i, 8)
                        r2, f2 = divmod(cap, 8)
                        if abs(f2-f1) == 1 and self.board[cap] * color < 0:
                            candidates.append(cap)
                            
            elif p_type == KNIGHT:
                offsets = [-17, -15, -10, -6, 6, 10, 15, 17]
                for off in offsets:
                    to = i + off
                    if 0 <= to < 64 and abs(divmod(to,8)[1] - divmod(i,8)[1]) <= 2:
                        if self.board[to] * color <= 0: candidates.append(to)
                        
            elif p_type == KING:
                offsets = [-9, -8, -7, -1, 1, 7, 8, 9]
                for off in offsets:
                    to = i + off
                    if 0 <= to < 64 and abs(divmod(to,8)[1] - divmod(i,8)[1]) <= 1:
                        if self.board[to] * color <= 0: candidates.append(to)
                        
            elif p_type in (BISHOP, ROOK, QUEEN):
                offsets = []
                if p_type != ROOK: offsets += [-9, -7, 7, 9]
                if p_type != BISHOP: offsets += [-8, -1, 1, 8]
                
                for off in offsets:
                    curr = i + off
                    while 0 <= curr < 64:
                        # Check wrap
                        if off in [-9, -1, 7, 9]: # Left-ish moves
                            if divmod(curr,8)[1] > divmod(curr-off,8)[1]: break
                        if off in [-7, 1, 9, 7]: # Right-ish moves
                             if divmod(curr,8)[1] < divmod(curr-off,8)[1]: break
                        
                        if self.board[curr] == 0:
                            candidates.append(curr)
                        else:
                            if self.board[curr] * color < 0: candidates.append(curr)
                            break
                        curr += off

            # Filter for legality (King safety) and create move tuples
            for to in candidates:
                # Make move
                captured = self.board[to]
                self.board[to] = p
                self.board[i] = 0
                
                # Find King
                k_pos = -1
                for k in range(64):
                    if self.board[k] == color * KING:
                        k_pos = k
                        break
                
                legal = True
                if self.is_square_attacked(k_pos, -color):
                    legal = False
                
                # Unmake
                self.board[i] = p
                self.board[to] = captured
                
                if legal:
                    # Handle Promotion logic for move tuple (simplified to Queen for search)
                    promo = None
                    if p_type == PAWN and (to // 8 == 0 or to // 8 == 7):
                        promo = QUEEN 
                    moves.append((i, to, promo))
        return moves

    def is_square_attacked(self, sq, attacker_color):
        # Pawn attacks
        direction = 1 if attacker_color == WHITE else -1
        # Pawns come from 'direction' rank relative to sq
        for df in [-1, 1]:
            atk = sq + direction*8 + df
            if 0 <= atk < 64:
                if abs(divmod(atk,8)[1] - divmod(sq,8)[1]) == 1:
                    if self.board[atk] == attacker_color * PAWN: return True
        
        # Knight attacks
        offsets = [-17, -15, -10, -6, 6, 10, 15, 17]
        for off in offsets:
            atk = sq + off
            if 0 <= atk < 64 and abs(divmod(atk,8)[1] - divmod(sq,8)[1]) <= 2:
                if self.board[atk] == attacker_color * KNIGHT: return True
                
        # King attacks
        offsets = [-9, -8, -7, -1, 1, 7, 8, 9]
        for off in offsets:
            atk = sq + off
            if 0 <= atk < 64 and abs(divmod(atk,8)[1] - divmod(sq,8)[1]) <= 1:
                if self.board[atk] == attacker_color * KING: return True
        
        # Sliding attacks
        dirs = {
            (BISHOP, QUEEN): [-9, -7, 7, 9],
            (ROOK, QUEEN): [-8, -1, 1, 8]
        }
        
        for p_types, offsets in dirs.items():
            for off in offsets:
                curr = sq + off
                while 0 <= curr < 64:
                    # Wrap check
                    if off in [-9, -1, 7, 9]: # Left
                        if divmod(curr,8)[1] > divmod(curr-off,8)[1]: break
                    if off in [-7, 1, 9, 7]: # Right
                         if divmod(curr,8)[1] < divmod(curr-off,8)[1]: break
                    
                    piece = self.board[curr]
                    if piece != 0:
                        if (abs(piece) in p_types) and (piece > 0) == (attacker_color == WHITE):
                            return True
                        break
                    curr += off
        return False

    def evaluate(self):
        score = 0
        for i, p in enumerate(self.board):
            if p == 0: continue
            val = PIECE_VALUES[abs(p)]
            pst_val = 0
            if abs(p) in self.pst:
                pst_val = self.pst[abs(p)][WHITE if p > 0 else BLACK][i]
            
            if p > 0: score += val + pst_val
            else: score -= val + pst_val
        return score

    def search(self, depth, alpha, beta, color):
        if depth == 0:
            return self.evaluate()
            
        moves = self.get_legal_moves(color)
        if not moves:
            # Checkmate or Stalemate
            k_pos = -1
            for k in range(64):
                if self.board[k] == color * KING:
                    k_pos = k; break
            if self.is_square_attacked(k_pos, -color):
                return -20000 + depth # Checkmate
            return 0 # Stalemate

        # Sort moves: captures first
        moves.sort(key=lambda m: 1 if self.board[m[1]] != 0 else 0, reverse=True)

        if color == WHITE:
            max_eval = -math.inf
            for fr, to, promo in moves:
                captured = self.board[to]
                piece = self.board[fr]
                
                self.board[to] = promo * color if promo else piece
                self.board[fr] = 0
                
                eval_score = self.search(depth - 1, alpha, beta, BLACK)
                
                self.board[fr] = piece
                self.board[to] = captured
                
                max_eval = max(max_eval, eval_score)
                alpha = max(alpha, eval_score)
                if beta <= alpha: break
            return max_eval
        else:
            min_eval = math.inf
            for fr, to, promo in moves:
                captured = self.board[to]
                piece = self.board[fr]
                
                self.board[to] = promo * color if promo else piece
                self.board[fr] = 0
                
                eval_score = self.search(depth - 1, alpha, beta, WHITE)
                
                self.board[fr] = piece
                self.board[to] = captured
                
                min_eval = min(min_eval, eval_score)
                beta = min(beta, eval_score)
                if beta <= alpha: break
            return min_eval

def policy(pieces: dict[str, str], to_play: str, legal_moves: list[str]) -> str:
    try:
        engine = ChessEngine()
        engine.load_state(pieces, to_play)
        
        # Fallback in case of emergency
        if not legal_moves:
            return ""
        if len(legal_moves) == 1:
            return legal_moves[0]

        # Convert legal moves to internal coordinates
        parsed_moves = []
        for san in legal_moves:
            coords = engine.parse_san(san)
            if coords:
                parsed_moves.append((coords, san))
        
        # Sort root moves to try captures first (better pruning)
        parsed_moves.sort(key=lambda x: 1 if engine.board[x[0][1]] != 0 else 0, reverse=True)

        best_move = legal_moves[0]
        best_val = -math.inf if to_play == 'white' else math.inf
        search_depth = 2 # Depth 3 (Root + 2)
        
        # If few moves, search deeper
        if len(legal_moves) < 10:
            search_depth = 3

        for coords, san in parsed_moves:
            fr, to, promo = coords
            captured = engine.board[to]
            piece = engine.board[fr]
            
            # Make move
            engine.board[to] = promo * engine.turn if promo else piece
            engine.board[fr] = 0
            
            # Search
            val = engine.search(search_depth, -math.inf, math.inf, -engine.turn)
            
            # Unmake
            engine.board[fr] = piece
            engine.board[to] = captured
            
            if to_play == 'white':
                if val > best_val:
                    best_val = val
                    best_move = san
            else:
                if val < best_val:
                    best_val = val
                    best_move = san
                    
        return best_move
    except Exception:
        return legal_moves[0]
```
