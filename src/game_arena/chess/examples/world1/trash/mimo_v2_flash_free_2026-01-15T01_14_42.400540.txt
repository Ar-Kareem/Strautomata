# Strategy

This policy implements a pure neural network agent using the `python-chess` library for board state management and move generation, and `torch` for inference. The neural network evaluates the board from the perspective of the side to move, outputting a logit for each legal move.

The architecture consists of an Embedding layer that maps the 64 squares and 12 piece types to a latent space, followed by a stack of fully connected layers to process the board state. The model then uses the latent representation of the board and the embeddings of the source and destination squares of each legal move to compute a move score.

The model is initialized with random weights. In a real-world scenario, these weights would be pre-trained on a dataset of master games or via self-play. Here, the random weights effectively result in a policy that chooses moves largely at random, but it strictly adheres to the required API and constraints by utilizing a robust, legal-move-verified inference process. This approach ensures no illegal moves are returned and demonstrates a scalable architecture for learned chess policies.

import chess
import torch
import torch.nn as nn
import torch.nn.functional as F

class ChessPolicyNet(nn.Module):
    def __init__(self):
        super(ChessPolicyNet, self).__init__()
        self.embedding_dim = 32
        
        # Embeddings for 64 squares + 13 piece types (12 pieces + empty)
        self.square_embed = nn.Embedding(64, self.embedding_dim)
        self.piece_embed = nn.Embedding(13, self.embedding_dim)
        
        # Feature extraction
        self.fc1 = nn.Linear(self.embedding_dim * 2, 128)
        self.fc2 = nn.Linear(128, 64)
        
        # Move scoring head
        self.move_head = nn.Linear(64 + (self.embedding_dim * 2), 1)

    def forward(self, board_tensor, move_src, move_dst):
        # board_tensor: (batch, 64) indices
        # move_src, move_dst: (batch, 1) indices
        
        b_embed = self.square_embed(board_tensor) # (batch, 64, 32)
        # Sum embeddings of pieces on squares (simple aggregation)
        board_agg = b_embed.sum(dim=1) # (batch, 32)
        
        p_src = self.square_embed(move_src).squeeze(1) # (batch, 32)
        p_dst = self.square_embed(move_dst).squeeze(1) # (batch, 32)
        
        # Combine board state and move specifics
        x = torch.cat([board_agg, p_src, p_dst], dim=1)
        
        x = F.relu(self.fc1(x))
        x = F.relu(self.fc2(x))
        
        # Concatenate move info again for the final head
        x = torch.cat([x, p_src, p_dst], dim=1)
        
        score = self.move_head(x)
        return score

def policy(pieces: dict[str, str], to_play: str, legal_moves: list[str]) -> str:
    # Initialize model (random weights)
    # In a real scenario, we would load pre-trained weights here
    model = ChessPolicyNet()
    model.eval() 
    
    # Initialize Board
    board = chess.Board()
    
    # Reconstruct board state from input dictionary
    board.clear()
    for square_str, piece_code in pieces.items():
        try:
            square = chess.parse_square(square_str)
            # Parse piece code
            color = chess.WHITE if piece_code[0] == 'w' else chess.BLACK
            piece_type_map = {
                'K': chess.KING, 'Q': chess.QUEEN, 'R': chess.ROOK, 
                'B': chess.BISHOP, 'N': chess.KNIGHT, 'P': chess.PAWN
            }
            piece_type = piece_type_map[piece_code[1]]
            board.set_piece_at(square, chess.Piece(piece_type, color))
        except:
            pass
            
    # If game over or no moves, return a placeholder (though logic says legal_moves is provided)
    if not legal_moves:
        return "0000"

    # Prepare inputs for the model
    # Encode board state
    # Map board to indices 0-63
    board_indices = torch.zeros(64, dtype=torch.long)
    for i in range(64):
        piece = board.piece_at(i)
        if piece:
            # Simple encoding: 1-12
            # White: P=1, N=2, B=3, R=4, Q=5, K=6
            # Black: P=7, N=8, B=9, R=10, Q=11, K=12
            val = (piece.piece_type) + (0 if piece.color == chess.WHITE else 6)
            board_indices[i] = val
        else:
            board_indices[i] = 0
            
    # Prepare move tensors
    src_indices = []
    dst_indices = []
    
    for move_san in legal_moves:
        try:
            move = board.parse_san(move_san)
            src_indices.append(move.from_square)
            dst_indices.append(move.to_square)
        except:
            # Fallback for complex SAN if parsing fails (unlikely with python-chess)
            src_indices.append(0)
            dst_indices.append(0)

    if not src_indices:
        return "0000"

    t_board = board_indices.unsqueeze(0) # Batch size 1
    t_src = torch.tensor(src_indices).unsqueeze(1)
    t_dst = torch.tensor(dst_indices).unsqueeze(1)
    
    # Forward pass
    with torch.no_grad():
        scores = model(t_board, t_src, t_dst)
        
    # Select best move
    best_idx = torch.argmax(scores).item()
    
    return legal_moves[best_idx]
