```python
import math

def calculate_hand_value(hand_cards: list[str], target: int) -> int:
    """
    Calculates the value of the hand according to the game rules.
    'A' is 11 when hand_value <= T, or 1 when hand_value > T.
    This effectively means calculating the highest hand value <= target.
    If all possible values are > target, it returns the lowest possible value > target (i.e., a bust value).
    """
    value_without_aces = 0
    num_aces = 0

    for card_str in hand_cards:
        if card_str.isdigit():
            value_without_aces += int(card_str)
        elif card_str in ("J", "Q", "K"):
            value_without_aces += 10
        elif card_str == "A":
            num_aces += 1

    # Start by assuming all aces are 1. This is the lowest possible score.
    current_value = value_without_aces + num_aces

    # Now, try to convert aces from 1 to 11 (by adding 10) greedily,
    # as long as the total value does not exceed the target.
    # This ensures we get the highest possible score without busting (if possible).
    for _ in range(num_aces):
        if current_value + 10 <= target: # If we can make an ace 11 without busting
            current_value += 10
        # Else, the ace must remain 1 (already counted above), or
        #
        # Note on rule interpretation: "A" -> 11 (when the hand value is <= T) or 1 (when the hand value is > T)
        # This implies we get the maximum value for the hand which is <= T.
        # If no such value exists (even with all aces as 1, we are > T), then the lowest possible value > T is the result.
        # The current logic correctly implements this:
        # 1. Start with all aces as 1.
        # 2. Add 10 for each ace if `current_value + 10 <= target`.
        # This ensures the maximum score that is <= target.
        # If `current_value` is already > target (meaning even with all aces as 1 we busted), then this loop won't change it,
        # which is the correct "lowest possible value > T".
        # E.g. hand = ["K", "Q", "A"], target = 15.
        # value_without_aces = 20, num_aces = 1.
        # current_value = 20 + 1 = 21.
        # Loop for ace: current_value + 10 <= 15 (21+10 <= 15) is False.
        # Returns 21. This correctly reflects the bust state.

    return current_value


def policy(hand: list[str], target: int) -> str:
    """
    Implements the policy for the Blackjack-like game.
    Decides whether to "HIT" or "STAY" based on the current hand and target.
    """
    my_score = calculate_hand_value(hand, target)

    # If already busted (my_score > target), then always STAY.
    # The game rules state: "If at any point the sum of your card values is greater than target, you bust and immediately lose."
    # So, if my_score is already > target, the game is effectively over for me. Hitting would be meaningless or worse.
    if my_score > target:
        return "STAY"

    # Define a threshold for staying.
    # This threshold represents how close to the target we want to be before stopping.
    # A value of 6 for 'X' in 'target - X' means:
    # If my score is T, T-1, T-2, T-3, T-4, T-5, T-6, then STAY.
    # Otherwise, HIT.
    # This balances the risk of busting with the need to get closer to the target.
    # Smallest card is '2'. So if target - my_score < 2, hitting guarantees a bust.
    # If target - my_score is 2, hitting '2' is perfect, but other cards bust you.
    # If target - my_score is 3, hitting '2' or '3' or 'A'(1) are okay, but '4' or higher busts you. This is a high bust risk.
    # A threshold of 6 aims for ~50% chance of not busting when you decide to hit.
    # Specifically, if my_score == target - 7, I HIT.
    # Cards 2,3,4,5,6,7,A(1) (7 cards) would not bust or bring me to T. Cards 8,9,10,J,Q,K (6 cards) would bust.
    # This is roughly (7/13) non-bust chance.
    STAY_THRESHOLD_DIFFERENCE = 6 # Stay if current score is within this difference to target (inclusive)

    if my_score >= target - STAY_THRESHOLD_DIFFERENCE:
        return "STAY"
    else:
        return "HIT"
```

```python
def policy(hand: list[str], target: int) -> str:
    """
    Implements a policy for a simplified Blackjack-like game.

    The policy decides whether to "HIT" (draw another card) or "STAY" (stop drawing cards)
    based on the current hand and the target score.

    Args:
        hand (list[str]): A list of strings representing the cards currently held.
                          Example: ["A", "5"], ["10", "K"]
        target (int): The target score to approach without exceeding.
                      This is a random integer between 10 and 30.

    Returns:
        str: "HIT" to draw another card, or "STAY" to stop drawing cards.
    """

    def calculate_hand_value(hand_cards: list[str], current_target: int) -> int:
        """
        Calculates the value of the hand according to the game rules,
        dynamically adjusting 'A' (Ace) values.

        Card values:
        - "2" to "10": face value
        - "J", "Q", "K": 10
        - "A": 11 if the hand value is <= current_target, or 1 if the hand value is > current_target.
               This means the Ace will be counted as 11 if it doesn't cause the total to exceed the target.
               If multiple Aces are present, they are adjusted optimally to yield the highest possible score
               that is less than or equal to the target. If even with all Aces counted as 1, the score
               still exceeds the target, then the lowest possible (busting) score is returned.

        Args:
            hand_cards (list[str]): The cards in the hand.
            current_target (int): The target score.

        Returns:
            int: The calculated value of the hand.
        """
        value_sum = 0
        num_aces = 0

        # First, sum all non-Ace cards and count Aces
        for card_str in hand_cards:
            if card_str.isdigit():
                value_sum += int(card_str)
            elif card_str in ("J", "Q", "K"):
                value_sum += 10
            elif card_str == "A":
                num_aces += 1

        # Calculate an initial total, treating all Aces as 11 (the "soft" total)
        soft_total = value_sum + num_aces * 11

        # Adjust Aces: If the soft_total exceeds the target, convert Aces from 11 to 1
        # one by one until the total is no longer greater than the target, or no more
        # Aces are available to convert. This ensures the highest possible score <= target.
        while soft_total > current_target and num_aces > 0:
            soft_total -= 10  # Convert an Ace from 11 to 1
            num_aces -= 1

        # The resulting soft_total is either:
        # 1. The highest possible hand value that is less than or equal to the target.
        # 2. If even with all Aces counted as 1 (i.e., num_aces becomes 0 and soft_total is still > current_target),
        #    then it represents the lowest possible busting score.
        return soft_total

    my_score = calculate_hand_value(hand, target)

    # Strategy:
    # 1. If the current score already exceeds the target, we have busted.
    #    In this scenario, playing "STAY" is the only reasonable action as "HIT" would
    #    not change the bust condition and the game implies an immediate loss upon busting.
    if my_score > target:
        return "STAY"

    # 2. Define a threshold for when to stop hitting.
    #    This threshold (target - X) determines how close to the target we want to get
    #    before we decide to "STAY".
    #    A `STAY_THRESHOLD_DIFFERENCE` of 6 means we will "STAY" if our score is
    #    `target`, `target-1`, `target-2`, `target-3`, `target-4`, `target-5`, or `target-6`.
    #    Otherwise, we will "HIT".
    #
    #    Rationale for 6:
    #    - If `target - my_score` is less than 2, any card ('2' is the minimum value) will cause a bust.
    #      So, if `my_score >= target - 1`, we must STAY.
    #    - If `target - my_score` is 2 or 3, there's a very high probability (over 70%) of busting
    #      if we "HIT" (e.g., for diff=3, cards '4' through 'K' and 'A'(11) would bust). Staying is safer.
    #    - If `target - my_score` is 4 or 5, the bust probability remains high (over 50-60%).
    #    - If `target - my_score` is 6 (i.e., `my_score = target - 6`), hitting implies:
    #      - Safe cards (2,3,4,5,6, and A if it evaluates to 1, leading to T-5): These are 6 + 1 = 7 cards. (~7/13 chance)
    #      - Busting cards (7,8,9,10, J,Q,K, and A if it evaluates to 11 and causes bust): These are 7 cards. (~6/13 chance)
    #      The exact probability varies based on Ace evaluation, but it becomes "more" favorable to hit.
    #    - Setting the threshold at 6 prioritizes not busting while still allowing hits for significantly lower scores,
    #      especially important for higher 'target' values where more cards are generally needed.
    #    - This conservative approach minimizes the primary losing condition (busting).
    
    STAY_THRESHOLD_DIFFERENCE = 6

    if my_score >= target - STAY_THRESHOLD_DIFFERENCE:
        return "STAY"
    else:
        return "HIT"
```
