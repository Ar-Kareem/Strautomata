import random

def policy(hand: list[str], target: int) -> str:
    """
    A policy for a simplified Blackjack-like game.

    Args:
        hand: The list of strings representing the cards you currently hold.
        target: The integer T you are trying to approach.

    Returns:
        "HIT" if you want to draw the next card, "STAY" otherwise.
    """

    def calculate_hand_value(hand: list[str], target: int) -> int:
        """Calculates the value of the hand."""
        hand_value = 0
        ace_count = 0
        for card in hand:
            if card.isdigit():
                hand_value += int(card)
            elif card in ("J", "Q", "K"):
                hand_value += 10
            elif card == "A":
                ace_count += 1
                hand_value += 11

        while hand_value > target and ace_count > 0:
            hand_value -= 10
            ace_count -= 1

        return hand_value

    hand_value = calculate_hand_value(hand, target)

    if hand_value >= target:
        return "STAY"

    # Define a dynamic hit threshold based on the target.
    # The higher the target, the more willing we are to hit.
    # The lower the target, the less willing we are to hit.
    hit_threshold = target - 7 + (target - 10) // 5 

    if hand_value < hit_threshold:
        return "HIT"
    else:
        return "STAY"
