The strategy for this policy is to use a combination of alpha-beta pruning with a sophisticated evaluation function optimized for Havannah's unique winning conditions (ring, bridge, fork). Since the game state space is large, we prioritize moves that:

1. **Immediately threaten a win** — check if placing a stone completes a ring, bridge, or fork.
2. **Block opponent threats** — if the opponent is one move away from completing any of the three win conditions, play the blocking move.
3. **Extend strong patterns** — prioritize moves that contribute to multiple potential paths (high connectivity), especially near edges and corners for bridge/fork chances.
4. **Use heuristic board evaluation** — score positions based on:
   - Number of stones connected to edges (for fork/bridge)
   - Distance to opponent’s key stones
   - Potential ring formations (circular patterns around empty cells)
   - Liberty count (adjacent empty spaces) — more liberties mean more flexibility
5. **Search depth**: Use iterative deepening with alpha-beta pruning to depth 3-4 (within 1s timeframe on 3GHz CPU), accelerated by move ordering (prioritizing captures, threats, and central plays).

- Use bitboards or sets for fast membership testing.
- Use a transposition table to avoid re-evaluating identical states.

def analyze_position(me_stones: Set[Tuple[int, int]], opp_stones: Set[Tuple[int, int]]) -> int:

    Heuristic evaluation function: higher score is better for current player.

    """

    if is_bridge_complete(me_stones, opp_stones) or is_fork_complete(me_stones) or is_ring_complete(me_stones):
        return 1000000
    if is_bridge_complete(opp_stones, me_stones) or is_fork_complete(opp_stones) or is_ring_complete(opp_stones):
        return -1000000

    score = 0
    # Add score for: number of stones
    score += len(me_stones) * 10
    score -= len(opp_stones) * 10
    
    # Add score for connection to edges (fork potential)
    edge_connections, components = get_connected_components(me_stones, set(range(6)), EDGES)
    score += len(edge_connections) * 50
    
    # Add score based on how many edge connections opponent has
    opp_edge_connections, _ = get_connected_components(opp_stones, set(range(6)), EDGES)
    score -= len(opp_edge_connections) * 40
    
    # Add score for liberty count (adjacent empty cells)
    me_liberties = set()
    for r, c in me_stones:
        for nr, nc in NEIGHBORS[(r, c)]:
            if (nr, nc) not in me_stones and (nr, nc) not in opp_stones:
                me_liberties.add((nr, nc))
    score += len(me_liberties) * 5

    # Penalty for opponent liberties
    opp_liberties = set()
    for r, c in opp_stones:
        for nr, nc in NEIGHBORS[(r, c)]:
            if (nr, nc) not in me_stones and (nr, nc) not in opp_stones:
                opp_liberties.add((nr, nc))
    score -= len(opp_liberties) * 4

    # Central control
    center = (7, 7)
    me_center_dist = 0
    for r, c in me_stones:
        d = abs(r - 7) + abs(c - 7)
        me_center_dist += max(0, 14 - d)  # bonus for being central
    score += me_center_dist
    
    opp_center_dist = 0
    for r, c in opp_stones:
        d = abs(r - 7) + abs(c - 7)
        opp_center_dist += max(0, 14 - d)
    score -= opp_center_dist

    return score

def get_potential_win_moves(me_stones: Set[Tuple[int, int]], opp_stones: Set[Tuple[int, int]], valid_mask) -> List[Tuple[int, int]]:
    """
    Returns a list of moves that would complete a win for me or block a win for opponent.
    """
    win_moves = []
    
    # Check for immediate win for me
    for r in range(15):
        for c in range(15):
            if not valid_mask[r][c]:
                continue
            if (r, c) in me_stones or (r, c) in opp_stones:
                continue
            
            # Simulate placing a stone here
            me_stones.add((r, c))
            if is_bridge_complete(me_stones, opp_stones) or is_fork_complete(me_stones) or is_ring_complete(me_stones):
                win_moves.append((r, c))
            me_stones.remove((r, c))
    
    # Check for blocking opponent win
    for r in range(15):
        for c in range(15):
            if not valid_mask[r][c]:
                continue
            if (r, c) in me_stones or (r, c) in opp_stones:
                continue
            
            # Simulate opponent placing a stone here
            opp_stones.add((r, c))
            if is_bridge_complete(opp_stones, me_stones) or is_fork_complete(opp_stones) or is_ring_complete(opp_stones):
                win_moves.append((r, c))
            opp_stones.remove((r, c))
    
    return win_moves

def alpha_beta(me_stones: Set[Tuple[int, int]], opp_stones: Set[Tuple[int, int]], depth: int, alpha: int, beta: int, maximizing: bool, valid_mask: np.ndarray, cache: Dict[str, int]) -> Tuple[int, Tuple[int, int]]:
    """
    Alpha-beta pruning with iterative deepening. Returns (score, move)
    """
    if depth == 0:
        return analyze_position(me_stones, opp_stones), None
    
    # Check for win/loss immediately
    if is_bridge_complete(me_stones, opp_stones) or is_fork_complete(me_stones) or is_ring_complete(me_stones):
        return 1000000, None
    if is_bridge_complete(opp_stones, me_stones) or is_fork_complete(opp_stones) or is_ring_complete(opp_stones):
        return -1000000, None
    
    # Generate moves
    moves = []
    for r in range(15):
        for c in range(15):
            if valid_mask[r][c] and (r, c) not in me_stones and (r, c) not in opp_stones:
                moves.append((r, c))
    
    if not moves:
        return analyze_position(me_stones, opp_stones), None

    # Sort moves: prioritize win/block and central moves
    win_moves = get_potential_win_moves(me_stones, opp_stones, valid_mask)
    # Reorder: put win/block moves first
    if win_moves:
        ordered_moves = win_moves
        # Then others: prioritize center
        others = [m for m in moves if m not in win_moves]
        others.sort(key=lambda m: (abs(m[0]-7)**2 + abs(m[1]-7)**2))  # closer to center first
        ordered_moves.extend(others)
    else:
        others = moves
        others.sort(key=lambda m: (abs(m[0]-7)**2 + abs(m[1]-7)**2))
        ordered_moves = others

    best_move = None
    if maximizing:
        max_eval = -float('inf')
        for move in ordered_moves:
            me_stones.add(move)
            eval_score, _ = alpha_beta(me_stones, opp_stones, depth - 1, alpha, beta, False, valid_mask, cache)
            me_stones.remove(move)
            
            if eval_score > max_eval:
                max_eval = eval_score
                best_move = move
            alpha = max(alpha, eval_score)
            if beta <= alpha:
                break  # beta cutoff
        return max_eval, best_move
    else:
        min_eval = float('inf')
        for move in ordered_moves:
            opp_stones.add(move)
            eval_score, _ = alpha_beta(me_stones, opp_stones, depth - 1, alpha, beta, True, valid_mask, cache)
            opp_stones.remove(move)
            
            if eval_score < min_eval:
                min_eval = eval_score
                best_move = move
            beta = min(beta, eval_score)
            if beta <= alpha:
                break  # alpha cutoff
        return min_eval, best_move

def policy(me: List[Tuple[int, int]], opp: List[Tuple[int, int]], valid_mask) -> Tuple[int, int]:
