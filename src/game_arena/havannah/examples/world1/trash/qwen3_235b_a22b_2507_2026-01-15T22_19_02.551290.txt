The strategy for this policy is to combine several heuristics inspired by strong human play and classical game AI techniques for Havannah. The policy prioritizes moves based on a scoring system that evaluates both offensive and defensive opportunities across the three winning conditions: **ring**, **bridge**, and **fork**.

The board is hexagonal in connectivity but laid out in a rhombus-shaped 15x15 grid. Each cell (r, c) has up to 6 neighbors, determined by axial coordinates converted to offset (row, col). We precompute neighbor mappings to efficiently navigate the hex grid.

Key components of the strategy:
1. **Threat Detection**: Evaluate if any move completes or blocks a win (ring, bridge, or fork) for either player.
2. **Pattern-Based Evaluation**: Score each valid move based on:
   - Proximity to edges (for bridge and fork potential)
   - Connectivity to own stones
   - Potential to form enclosures (ring candidates)
3. **Control of Corners and Edges**: Prioritize corners for bridge threats and mid-edge locations for fork threats.
4. **Defensive Safety**: Penalize moves that strongly help the opponent’s connectivity.
5. **Monte Carlo Light Rollout Simulation**: For top candidate moves, perform shallow simulations to estimate success likelihood.

However, due to the 1-second constraint and the complexity of full Havannah AI, we implement a **fast greedy policy with threat detection and prioritization**, focusing on:
- Immediate win
- Block opponent’s win
- Extend own groups
- Target strategic positions (edges/corners)

We use `numpy` for efficient mask and coordinate operations.

The policy proceeds as:
1. Find all legal moves via `valid_mask`.
2. Check if any move completes a ring/bridge/fork for `me` → play and win.
3. Check if any opponent move would complete their win in the next turn → block.
4. Score remaining moves based on:
   - Distance to forming bridges, forks, rings
   - Group cohesion and liberty
   - Board control (center vs edges)
5. Return the highest-scoring legal move.

We rely on efficient hexagonal neighbor logic and precomputed edge/corner definitions.

def get_neighbors(r: int, c: int) -> List[Tuple[int, int]]:
    """Get valid neighboring cells in hex grid using offset coordinates (odd-r)."""
    if r % 2 == 0:
        dirs = HEX_DIRS['even']
    else:
        dirs = HEX_DIRS['odd']
    neighbors = []
    for dr, dc in dirs:
        nr, nc = r + dr, c + dc
        if 0 <= nr < N and 0 <= nc < N:
            neighbors.append((nr, nc))
    return neighbors

def in_board(r: int, c: int) -> bool:
    return 0 <= r < N and 0 <= c < N

def flood_fill(start: Tuple[int, int], stones: Set[Tuple[int, int]], visited: Set[Tuple[int, int]]):
    """DFS to get connected group."""
    if start in visited or start not in stones:
        return
    visited.add(start)
    for nb in get_neighbors(start[0], start[1]):
        if nb in stones and nb not in visited:
            flood_fill(nb, stones, visited)

def get_groups(stones: List[Tuple[int, int]]) -> List[Set[Tuple[int, int]]]:
    """Get connected groups (sets of stones)."""
    stone_set = set(stones)
    visited = set()
    groups = []
    for stone in stones:
        if stone not in visited:
            group = set()
            flood_fill(stone, stone_set, group)
            visited |= group
            groups.append(group)
    return groups

def is_connected(group: Set[Tuple[int, int]], target_cells: List[Tuple[int, int]]) -> bool:
    """Check if any cell in group touches any cell in target_cells."""
    for r, c in group:
        for nr, nc in get_neighbors(r, c):
            if (nr, nc) in target_cells:
                return True
    return False

def completes_ring(me_stones: List[Tuple[int, int]], move: Tuple[int, int]) -> bool:
    """Heuristic: check if adding `move` completes a ring around any cell."""
    # A ring is a loop around at least one cell. We check if the move closes a loop.
    # We can try: after placing move, is there a group that encircles a cell?
    # But this is complex. Approximate: check if the move increases group's ability to surround.
    # Instead, we use a simple local check: is the move part of a ring of 6 stones around a central cell?
    # More accurately: for each empty cell enclosed by same color?
    # Due to time, we use a limited check: for each cell adjacent to the move, check if it is surrounded.
    me_set = set(me_stones) | {move}
    # Check cells that are fully enclosed by a connected group
    for r in range(N):
        for c in range(N):
            if (r, c) in me_set:
                continue
            # Check if (r,c) is surrounded by a single connected group of me
            surrounding = 0
            group_id = None
            valid_ring = True
            for nr, nc in get_neighbors(r, c):
                if (nr, nc) in me_set:
                    surrounding += 1
                    nb_group = None
                    for g in get_groups(me_stones):
                        if (nr, nc) in g:
                            nb_group = g
                            break
                    # All surrounding stones must be in one connected group
                    if group_id is None:
                        group_id = nb_group
                    elif group_id != nb_group:
                        valid_ring = False
                        break
                else:
                    valid_ring = False
            if valid_ring and surrounding == 6:
                return True
    return False

def completes_bridge(me_stones: List[Tuple[int, int]], move: Tuple[int, int]) -> bool:
    """Check if move completes a bridge between any two corners."""
    stones_with_move = me_stones + [move]
    groups = get_groups(stones_with_move)
    corner_set = set(CORNERS)
    corners_occupied = [c for c in CORNERS if c in set(stones_with_move)]
    if len(corners_occupied) < 2:
        return False
    # Check if any group connects at least two corners
    for group in groups:
        occupied_in_group = [c for c in corners_occupied if c in group]
        if len(occupied_in_group) >= 2:
            return True
    return False

def completes_fork(me_stones: List[Tuple[int, int]], move: Tuple[int, int]) -> bool:
    """Check if move completes a fork connecting three edges."""
    stones_with_move = me_stones + [move]
    groups = get_groups(stones_with_move)
    for group in groups:
        edges_touched = set()
        for edge_def in EDGES:
            if is_connected(group, edge_def['cells']):
                edges_touched.add(edge_def['name'])
        if len(edges_touched) >= 3:
            return True
    return False

def will_win(me_stones: List[Tuple[int, int]], move: Tuple[int, int]) -> bool:
    """Check if playing `move` completes any winning structure."""
    return (completes_ring(me_stones, move) or
            completes_bridge(me_stones, move) or
            completes_fork(me_stones, move))

def evaluate_move(move: Tuple[int, int], me_stones: List[Tuple[int, int]], opp_stones: List[Tuple[int, int]]) -> float:
    """Evaluate move based on heuristics."""
    score = 0.0
    me_set = set(me_stones)
    opp_set = set(opp_stones)

    r, c = move

    # Proximity to own stones
    near_own = 0
    near_opp = 0
    for nr, nc in get_neighbors(r, c):
        if (nr, nc) in me_set:
            near_own += 1
        elif (nr, nc) in opp_set:
            near_opp += 1
    score += near_own * 10
    score -= near_opp * 5  # pressure

    # Edge/corner bonus
    if (r, c) in ALL_CORNERS:
        score += 8
    elif (r, c) in ALL_EDGE_CELLS:
        score += 5

    # Center control
    dr, dc = r - 7, c - 7
    dist_to_center = abs(dr) + abs(dc)  # rough
    score += max(0, 7 - dist_to_center)  # bonus for central, up to 7

    # Potential for fork or bridge
    if (r, c) in ALL_CORNERS:
        score += 15  # corner is critical for bridge
    # For fork: being near multiple edges
    edge_count = 0
    for edge in EDGES:
        # distance to this edge
        min_dist = min([abs(r - er) + abs(c - ec) for (er, ec) in edge['cells']], default=100)
        if min_dist == 0:
            edge_count += 1
        elif min_dist == 1:
            score += 3
    if edge_count >= 2:
        score += 10  # near two edges, potential for fork

    return score

def policy(me: List[Tuple[int, int]], opp: List[Tuple[int, int]], valid_mask) -> Tuple[int, int]:
