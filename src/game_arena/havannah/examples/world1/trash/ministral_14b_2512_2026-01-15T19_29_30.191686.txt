### **Strategy for Havannah 15x15 Policy**

My policy is designed to **maximize control of the board while aggressively pursuing winning conditions** (ring, bridge, or fork). The approach combines:

1. **Immediate Win Detection**: If the current move can complete a ring, bridge, or fork, the policy prioritizes that move.
2. **Block Opponent’s Threats**: If the opponent has a near-complete winning structure, the policy blocks it with highest priority.
3. **Center Control**: Early moves focus on controlling the center (rows 4-10, columns 4-10) to limit opponent mobility.
4. **Edge & Corner Dominance**: Later moves prioritize edges and corners, especially if they help form bridges or forks.
5. **Hexagonal Connectivity**: The policy evaluates potential moves by considering **6-directional connectivity** (unlike standard Go, where only 4 directions matter).
6. **Fork & Bridge Opportunities**: If no immediate threats exist, the policy looks for moves that create **forks** (3+ edge connections) or **bridges** (2+ corner connections).
7. **Ring Formation**: If a move can start or extend a loop (ring), it is considered highly valuable.
8. **Defensive Play**: If the opponent has a strong potential fork/bridge, the policy blocks it.
9. **Randomness for Uncertainty**: If no clear winning/blocking move is found, the policy selects a move randomly from high-priority regions (center, edges, corners).

The policy uses **depth-limited search (DLS)** to evaluate future threats and opportunities, but due to the 1-second constraint, it restricts search depth to **3-4 plies** while focusing on **local board patterns** (e.g., potential loops, bridges, and forks).

    1. Check for immediate win (ring, bridge, or fork).
    2. Block opponent's immediate threats.
    3. Prioritize center control.
    4. Look for fork/bridge opportunities.
    5. Randomly select from high-priority regions if no threats.

    board_size = len(valid_mask)

        board[r][c] = True

        return tuple(edge_moves[np.random.choice(len(edge_moves))])

        return tuple(possible_moves[np.random.choice(len(possible_moves))])

def hex_neighbors(r: int, c: int, board_size: int) -> List[Tuple[int, int]]:
    """Returns 6 hexagonal neighbors of (r, c) in a 15x15 board."""
    neighbors = [
        (r + 1, c),       # Down
        (r - 1, c),       # Up
        (r, c + 1),       # Right
        (r, c - 1),       # Left
        (r + 1, c - 1),  # Down-left
        (r - 1, c + 1)   # Up-right
    ]
    valid_neighbors = []
    for nr, nc in neighbors:
        if 0 <= nr < board_size and 0 <= nc < board_size:
            valid_neighbors.append((nr, nc))
    return valid_neighbors

def is_connected(stones: List[Tuple[int, int]], board: np.ndarray) -> bool:
    """Checks if all stones in 'stones' are connected via hexagonal adjacency."""
    if not stones:
        return False
    visited = set()
    stack = [stones[0]]
    visited.add(stones[0])
    while stack:
        r, c = stack.pop()
        for nr, nc in hex_neighbors(r, c, len(board)):
            if board[nr][nc] and (nr, nc) in stones and (nr, nc) not in visited:
                visited.add((nr, nc))
                stack.append((nr, nc))
    return len(visited) == len(stones)

def find_immediate_win(board: np.ndarray, me: List[Tuple[int, int]], possible_moves: np.ndarray) -> Tuple[int, int]:

    board_size = len(board)

    return None

def check_bridge(stones: List[Tuple[int, int]], board: np.ndarray) -> bool:
    """Checks if 'stones' form a bridge between two corners."""
    corners = [(0, 0), (0, board_size - 1), (board_size - 1, 0), (board_size - 1, board_size - 1)]
    # Check if any two corners are connected via stones
    for i in range(len(corners)):
        for j in range(i + 1, len(corners)):
            corner1 = corners[i]
            corner2 = corners[j]
            if corner1 in stones and corner2 in stones:

                path_stones = [s for s in stones if s not in corners]

                    return True
    return False

def check_fork(stones: List[Tuple[int, int]], board: np.ndarray) -> bool:
    """Checks if 'stones' connect three edges (not corners)."""
    edges = []
    # Top edge (row 0, columns 1..board_size-2)
    edges.extend([(0, c) for c in range(1, board_size - 1)])
    # Bottom edge (row board_size-1, columns 1..board_size-2)
    edges.extend([(board_size - 1, c) for c in range(1, board_size - 1)])
    # Left edge (columns 0, rows 1..board_size-2)
    edges.extend([(r, 0) for r in range(1, board_size - 1)])
    # Right edge (columns board_size-1, rows 1..board_size-2)
    edges.extend([(r, board_size - 1) for r in range(1, board_size - 1)])
    # Check if any three edges are connected via stones
    for i in range(len(edges)):
        for j in range(i + 1, len(edges)):
            for k in range(j + 1, len(edges)):
                edge1, edge2, edge3 = edges[i], edges[j], edges[k]
                if edge1 in stones and edge2 in stones and edge3 in stones:
                    # Check if there's a path connecting all three edges
                    combined = stones.copy()
                    if is_connected(combined, board):
                        return True
    return False

def find_block_threat(board: np.ndarray, me: List[Tuple[int, int]], opp: List[Tuple[int, int]], possible_moves: np.ndarray) -> Tuple[int, int]:
    """Blocks opponent's near-complete ring, bridge, or fork."""

def find_fork_opportunity(board: np.ndarray, me: List[Tuple[int, int]], possible_moves: np.ndarray) -> Tuple[int, int]:
    """Finds moves that create a fork (connect 3 edges)."""
    edges = []
    board_size = len(board)
    # Top edge (row 0, columns 1..board_size-2)
    edges.extend([(0, c) for c in range(1, board_size - 1)])
    # Bottom edge (row board_size-1, columns 1..board_size-2)
    edges.extend([(board_size - 1, c) for c in range(1, board_size - 1)])
    # Left edge (columns 0, rows 1..board_size-2)
    edges.extend([(r, 0) for r in range(1, board_size - 1)])
    # Right edge (columns board_size-1, rows 1..board_size-2)
    edges.extend([(r, board_size - 1) for r in range(1, board_size - 1)])

    for move in possible_moves:
        r, c = move
        new_me = me.copy()
        new_me.append((r, c))
        # Check if this move connects at least 3 edges
        connected_edges = set()

            if (er, ec) in new_me:
                connected_edges.add((er, ec))
        # Check if the move is adjacent to at least 3 edges

        for nr, nc in neighbors:
            if (nr, nc) in edges:
                connected_edges.add((nr, nc))
        if len(connected_edges) >= 3:

    return None

def find_bridge_opportunity(board: np.ndarray, me: List[Tuple[int, int]], possible_moves: np.ndarray) -> Tuple[int, int]:
    """Finds moves that create a bridge between two corners."""
    corners = [(0, 0), (0, board_size - 1), (board_size - 1, 0), (board_size - 1, board_size - 1)]
    board_size = len(board)
    for move in possible_moves:
        r, c = move
        new_me = me.copy()
        new_me.append((r, c))
        # Check if this move connects two corners via a path
        for i in range(len(corners)):
            for j in range(i + 1, len(corners)):
                corner1 = corners[i]
                corner2 = corners[j]
                if corner1 in new_me or corner2 in new_me:
                    # Check if the move is adjacent to a corner

                    return tuple(move)
    return None
