The logic for this policy is to prioritize moving checkers into the home board, bearing off when possible, and handling checkers on the bar first. It uses a greedy heuristic to select the highest-indexed points for movement and assumes that higher dice values are used first (`H` order). The code also handles basic bearing off strategies and minimizes exposure during non-bearoff phases.

def policy(state: dict) -> str:
    my_pts = state['my_pts']
    opp_pts = state['opp_pts']
    my_bar = state['my_bar']
    dice = state['dice']

    if my_bar > 0:
        valid_dices = []
        for d in set(dice):
            dest = 24 - d
            if 0 <= dest < 24 and opp_pts[dest] < 2:
                valid_dices.append(d)
        valid_dices.sort(reverse=True)
        if not valid_dices:
            return "H:P,P"
        if len(valid_dices) >= 2:
            return "H:B,B"
        else:
            return "H:B,P"
    else:
        all_in_home = True
        for i in range(6, 24):
            if my_pts[i] > 0:
                all_in_home = False
                break
        if all_in_home:
            possible_bear_off = []
            for d in dice:
                for point in range(6):
                    if my_pts[point] > 0 and d >= (point + 1):
                        possible_bear_off.append(point)
            if possible_bear_off:
                possible_bear_off.sort(reverse=True)
                if len(dice) == 2:
                    from1 = possible_bear_off[0]
                    from2 = possible_bear_off[1] if len(possible_bear_off) > 1 else 'P'
                    return f"H:A{from1},A{from2}"
                else:
                    return f"H:A{possible_bear_off[0]},P"
            else:
                for point in range(5, -1, -1):
                    if my_pts[point] > 0:
                        for d in sorted(dice, reverse=True):
                            dest_point = point - d
                            if dest_point >= 0 and opp_pts[dest_point] < 2:
                                return f"H:A{point},P"
                return "H:P,P"
        else:
            for point in range(23, 5, -1):
                if my_pts[point] > 0:
                    for d in sorted(dice, reverse=True):
                        dest_point = point - d
                        if dest_point >= 0 and opp_pts[dest_point] < 2:
                            if len(dice) == 2:
                                return f"H:A{point},A{point}"
                            else:
                                return f"H:A{point},P"
            for point in range(5, -1, -1):
                if my_pts[point] > 0:
                    for d in sorted(dice, reverse=True):
                        dest_point = point - d
                        if dest_point >= 0 and opp_pts[dest_point] < 2:
                            return f"H:A{point},P"
            return "H:P,P"
