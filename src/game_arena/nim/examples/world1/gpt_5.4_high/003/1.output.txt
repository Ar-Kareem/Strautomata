The strategy for this policy is exact misère Nim play.

For misère Nim, the losing positions are:
- if every nonempty pile has size 1: positions with an odd number of nonempty piles;
- otherwise: positions whose pile xor is 0.

So the policy is:
1. If all nonempty piles are 1, remove one from any nonempty pile.
2. If exactly one pile is larger than 1, reduce that pile to:
   - 1 if the number of 1-piles is even,
   - 0 if the number of 1-piles is odd.
3. If at least two piles are larger than 1, play normal Nim: make the xor of all piles equal to 0.

That is perfect play. In positions that are already losing under perfect play, the code still guarantees a legal move and biases toward leaving a tricky misère endgame for imperfect opponents.

<code>
from typing import List


def _first_legal_move(piles: List[int]) -> str:
    for i, h in enumerate(piles):
        if h > 0:
            return f"{i},1"
    # Terminal state should not be queried, but return a well-formed fallback.
    return "0,1"


def _safe_move(piles: List[int], pile: int, take: int) -> str:
    if 0 <= pile < len(piles) and 1 <= take <= piles[pile]:
        return f"{pile},{take}"
    return _first_legal_move(piles)


def policy(piles: List[int]) -> str:
    xor_sum = 0
    big_indices = []
    ones = 0

    for i, h in enumerate(piles):
        xor_sum ^= h
        if h > 1:
            big_indices.append(i)
        elif h == 1:
            ones += 1

    big_count = len(big_indices)

    # Case 1: all nonempty piles are size 1
    if big_count == 0:
        return _first_legal_move(piles)

    # Case 2: exactly one pile > 1
    if big_count == 1:
        i = big_indices[0]
        target = 1 if (ones % 2 == 0) else 0
        return _safe_move(piles, i, piles[i] - target)

    # Case 3: at least two piles > 1
    # Play normal Nim: make xor 0.
    if xor_sum != 0:
        for i, h in enumerate(piles):
            target = h ^ xor_sum
            if target < h:
                return _safe_move(piles, i, h - target)

    # Already losing under perfect play.
    # Still return a legal move; reduce the smallest big heap to 1,
    # which often leaves a trickier misère endgame for weak opponents.
    i = min(big_indices, key=lambda idx: piles[idx])
    return _safe_move(piles, i, piles[i] - 1)
