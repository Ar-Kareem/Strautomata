To build a competitive agent for Dots and Boxes on a 4x4 grid, we implement a game-theoretic heuristic search based on the mathematical principles of Dots and Boxes (Elwyn Berlekamp's *The Winning Ways*):

### Strategy Overview
1. **Board Structure & Boundary Alignment**:
   - The grid has $4 \times 4$ boxes indexed $(r, c)$ for $r, c \in \{0, 1, 2, 3\}$.
   - Box $(r, c)$ is bounded by:
     - Top: `horizontal[r, c]`
     - Bottom: `horizontal[r+1, c]`
     - Left: `vertical[r, c]`
     - Right: `vertical[r, c+1]`
   - Any edge that bounds a box is a playable edge. Unused entries (like `horizontal[:, 4]` or `vertical[4, :]`) are avoided unless no standard moves exist.

2. **Move Categorization**:
   - **Captures (3-sided boxes)**: If a move completes a box, drawing it awards points and grants an immediate bonus turn.
   - **Safe Moves (0-sided or 1-sided boxes)**: Moves that do not create any 3-sided boxes (i.e., do not offer free captures to the opponent).
   - **Sacrifices / Double-Crosses**: When forced to give away boxes, give away the smallest possible chain or loop.

3. **Chain & Mini-Max / Greedy Evaluation**:
   - If one or more boxes have 3 edges filled, capture them immediately.
   - If capturing completes a chain, take all boxes in the chain. When finishing a long chain, prioritize leaving a double-cross if advantageous, or cleanly claiming all boxes.
   - When no captures are available, select among all safe moves (moves that do not result in any box having 3 drawn edges). Safe moves are prioritized by those that share edges with 0-sided boxes.
   - When no safe moves remain, compute the consequence of each sacrifice (the size of the component/chain gifted to the opponent) and choose the move that surrenders the minimum number of boxes.

import numpy as np

def _get_box_edges(r: int, c: int):
    """Returns the four edges bounding box (r, c)."""
    return [
        (r, c, 'H'),      # top
        (r + 1, c, 'H'),  # bottom
        (r, c, 'V'),      # left
        (r, c + 1, 'V')   # right
    ]

def _is_edge_drawn(horizontal: np.ndarray, vertical: np.ndarray, r: int, c: int, orientation: str) -> bool:
    if orientation == 'H':
        return horizontal[r, c] != 0
    else:
        return vertical[r, c] != 0

def _get_boxes_for_edge(r: int, c: int, orientation: str):
    """Returns the coordinates of boxes adjacent to a given edge in a 4x4 grid."""
    boxes = []
    if orientation == 'H':
        if r > 0:
            boxes.append((r - 1, c))
        if r < 4:
            boxes.append((r, c))
    else:  # 'V'
        if c > 0:
            boxes.append((r, c - 1))
        if c < 4:
            boxes.append((r, c))
    return boxes

def _count_box_edges(horizontal: np.ndarray, vertical: np.ndarray, r: int, c: int) -> int:
    """Counts how many of the 4 edges of box (r, c) have been drawn."""
    count = 0
    if horizontal[r, c] != 0:
        count += 1
    if horizontal[r + 1, c] != 0:
        count += 1
    if vertical[r, c] != 0:
        count += 1
    if vertical[r, c + 1] != 0:
        count += 1
    return count

def _simulate_chain_length(h: np.ndarray, v: np.ndarray, start_move):
    """
    Simulates greedy capture sequence starting from a sacrifice move
    to estimate the number of boxes conceded to the opponent.
    """
    h_copy = h.copy()
    v_copy = v.copy()
    
    r, c, orient = start_move
    if orient == 'H':
        h_copy[r, c] = 1
    else:
        v_copy[r, c] = 1

    total_conceded = 0
    while True:
        found_box = False
        for br in range(4):
            for bc in range(4):
                if _count_box_edges(h_copy, v_copy, br, bc) == 3:
                    # Find missing edge and fill it
                    for er, ec, eo in _get_box_edges(br, bc):
                        if not _is_edge_drawn(h_copy, v_copy, er, ec, eo):
                            if eo == 'H':
                                h_copy[er, ec] = 1
                            else:
                                v_copy[er, ec] = 1
                            total_conceded += 1
                            found_box = True
                            break
                    if found_box:
                        break
            if found_box:
                break
        if not found_box:
            break

    return total_conceded

def policy(horizontal: np.ndarray, vertical: np.ndarray, capture: np.ndarray) -> str:
    """
    Dots and Boxes policy for 4x4 box grid.
    Returns move formatted as 'row,col,dir'.
    """
    # 1. Identify all valid playable grid edges (5x4 horizontal, 4x5 vertical)
    legal_moves = []
    
    for r in range(5):
        for c in range(4):
            if horizontal[r, c] == 0:
                legal_moves.append((r, c, 'H'))

    for r in range(4):
        for c in range(5):
            if vertical[r, c] == 0:
                legal_moves.append((r, c, 'V'))

    # Fallback to any remaining 0 in arrays if standard set is somehow empty
    if not legal_moves:
        for r in range(horizontal.shape[0]):
            for c in range(horizontal.shape[1]):
                if horizontal[r, c] == 0:
                    return f"{r},{c},H"
        for r in range(vertical.shape[0]):
            for c in range(vertical.shape[1]):
                if vertical[r, c] == 0:
                    return f"{r},{c},V"

    # 2. Priority 1: Captures (edges completing a 3-sided box)
    captures = []
    for move in legal_moves:
        r, c, orient = move
        adj_boxes = _get_boxes_for_edge(r, c, orient)
        boxes_completed = sum(1 for br, bc in adj_boxes if _count_box_edges(horizontal, vertical, br, bc) == 3)
        if boxes_completed > 0:
            captures.append((boxes_completed, move))

    if captures:
        # Prefer moves capturing 2 boxes simultaneously
        captures.sort(key=lambda x: x[0], reverse=True)
        best_move = captures[0][1]
        return f"{best_move[0]},{best_move[1]},{best_move[2]}"

    # 3. Priority 2: Safe Moves (moves that do not create any 3-sided box)
    safe_moves = []
    for move in legal_moves:
        r, c, orient = move
        adj_boxes = _get_boxes_for_edge(r, c, orient)
        # Safe if none of the adjacent boxes currently have 2 edges filled
        is_safe = all(_count_box_edges(horizontal, vertical, br, bc) < 2 for br, bc in adj_boxes)
        if is_safe:
            # Score safe moves: prefer edges bounded by boxes with 0 sides filled
            edge_score = sum(1 for br, bc in adj_boxes if _count_box_edges(horizontal, vertical, br, bc) == 0)
            safe_moves.append((edge_score, move))

    if safe_moves:
        safe_moves.sort(key=lambda x: x[0], reverse=True)
        best_move = safe_moves[0][1]
        return f"{best_move[0]},{best_move[1]},{best_move[2]}"

    # 4. Priority 3: Sacrifices (minimize boxes surrendered)
    best_move = None
    min_conceded = 999

    for move in legal_moves:
        conceded = _simulate_chain_length(horizontal, vertical, move)
        if conceded < min_conceded:
            min_conceded = conceded
            best_move = move

    if best_move is None:
        best_move = legal_moves[0]

    return f"{best_move[0]},{best_move[1]},{best_move[2]}"
