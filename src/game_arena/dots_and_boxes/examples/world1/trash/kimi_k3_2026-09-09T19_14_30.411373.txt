The policy treats the current captured boxes as a constant and searches only over future box swings using a 40-bit edge mask蝎. It uses time-bounded iterative-deepening alpha-beta with a transposition table. Capturing moves are “free” extra turns, so they do not consume search depth, and the search is allowed to decline a capture when continuing would be worse. Move ordering prefers captures, then safe edges that create the fewest two-sided threats, then the smallest forced sacrifices. A capture-run heuristic resolves forced scoring sequences at leaf nodes, and a final fallback always returns an empty edge cell.

def _greedy_run(mask, mine):
    """Net future boxes if the side to move greedily takes all currently forced boxes."""
    m = mask
    score = 0
    for _ in range(16):
        avail = _FULL ^ m
        best_nm = m
        best_gain = 0
        best_cont = -1
        while avail:
            lsb = avail & -avail
            i = lsb.bit_length() - 1
            avail ^= lsb
            nm = m | lsb
            gain = 0
            cont = 0
            for b in _EDGE_BOXES[i]:
                bm = _BOX_MASKS[b]
                after = nm & bm
                if after == bm:
                    gain += 1
                elif after.bit_count() == 3:
                    cont += 1
            if gain and (gain > best_gain or (gain == best_gain and cont > best_cont)):
                best_gain = gain
                best_cont = cont
                best_nm = nm
        if best_gain == 0:
            break
        score += best_gain if mine else -best_gain
        m = best_nm
    return score, m

def _search(mask, to_move, depth, alpha, beta):
    """Future net boxes (for the root player) from mask, excluding already captured boxes."""
    global _NODES
    _NODES += 1
    if (_NODES & 1023) == 0 and time.perf_counter() > _DEADLINE:
        raise _SearchTimeout()

    if mask == _FULL:
        return 0

    key = (mask, to_move, depth)
    ent = _TT.get(key)
    if ent is not None:
        val, flag = ent
        if flag == 0:
            return val
        if flag == 1:
            if val > alpha:
                alpha = val
        else:
            if val < beta:
                beta = val
        if alpha >= beta:
            return val

    a0 = alpha
    b0 = beta
    moves = _gen_moves(mask)

    if to_move:
        best = -_INF
        for _, nm, i, gain in moves:
            if gain:
                # I captured, so it is still my turn; adjust the window by the free gain.
                val = gain + _search(nm, True, depth, alpha - gain, beta - gain)
            elif depth > 0:
                val = _search(nm, False, depth - 1, alpha, beta)
            else:
                val = _heuristic(nm, False)
            if val > best:
                best = val
            if best > alpha:
                alpha = best
            if alpha >= beta:
                break
    else:
        best = _INF刮刮卡
        for _, nm, i, gain in moves:
            if gain:
                # Opponent captured and moves again; my future net drops by gain.
                val = -gain + _search(nm, False, depth, alpha + gain, beta + gain)
            elif depth > 0:
                val = _search(nm, True, depth - 1, alpha, beta)
            else:
                val = _heuristic(nm, True)

def _root_search(mask, depth):
    moves = _gen_moves(mask)
    if not moves:
        return None
    best_i = moves[0][2]
    best = -_INF
    for _, nm, i, gain in moves:
        if (_NODES & 255) == 0 and time.perf_counter() > _DEADLINE:
            raise _SearchTimeout()
        if gain:
            val = gain + _search(nm, True, depth, best - gain, _INF)
        elif depth > 0:
            val = _search(nm, False, depth - 1, best, _INF)
        else:
            val = _heuristic(nm, False)
        if val > best:
            best = val
            best_i = i
    return best_i
