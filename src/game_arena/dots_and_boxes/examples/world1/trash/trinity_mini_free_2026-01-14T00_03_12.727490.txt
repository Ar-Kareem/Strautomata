

Thelogic for this policy is to prioritize moves that capture boxes without completing the opponent's boxes. If no such moves exist, it selects a move that blocks the opponent's potential captures. If all moves are blocked, it chooses any safe move that doesn't create a box with three sides.

def policy(horizontal: np.ndarray, vertical: np.ndarray, capture: np.ndarray) -> str:
    # Generate all possible moves (edges that are empty)
    moves = []
    for i in range(5):
        for j in range(5):
            if horizontal[i][j] == 0:
                moves.append((i, j, 'H'))
            if vertical[i][j] == 0:
                moves.append((i, j, 'V'))
    
    best_move = None
    best_boxes = -1
    
    for move in moves:
        i, j, d = move
        valid = True
        boxes_completed = 0
        if d == 'H':
            # Check box above (i-1, j)
            if i > 0:
                box_i = i - 1
                box_j = j
                # Check if the box is completed by this move
                top_ok = (box_i > 0 and horizontal[box_i][box_j] != 0)
                bottom_ok = True  # because we're drawing this edge
                left_ok = (box_j > 0 and vertical[box_i][box_j] != 0)
                right_ok = (box_j < 3 and vertical[box_i][box_j+1] != 0)
                if top_ok and left_ok and right_ok:
                    if capture[box_i][box_j] == -1:
                        valid = False
                    else:
                        boxes_completed += 1
            # Check box below (i, j)
            if i < 4:
                box_i = i
                box_j = j
                top_ok = (box_i > 0 and horizontal[box_i-1][box_j] != 0)
                bottom_ok = True
                left_ok = (box_j > 0 and vertical[box_i][box_j] != 0)
                right_ok = (box_j < 3 and vertical[box_i][box_j+1] != 0)
                if top_ok and left_ok and right_ok:
                    if capture[box_i][box_j] == -1:
                        valid = False
                    else:
                        boxes_completed += 1
        else:  # vertical edge
            # Check box to the left (i, j-1)
            if j > 0:
                box_i = i
                box_j = j - 1
                top_ok = (box_i > 0 and horizontal[box_i][box_j] != 0)
                bottom_ok = (box_i < 3 and horizontal[box_i+1][box_j] != 0)
                left_ok = True
                right_ok = (box_j < 3 and vertical[box_i][box_j+1] != 0)
                if top_ok and bottom_ok and right_ok:
                    if capture[box_i][box_j] == -1:
                        valid = False
                    else:
                        boxes_completed += 1
            # Check box to the right (i, j)
            if j < 4:
                box_i = i
                box_j = j
                top_ok = (box_i > 0 and horizontal[box_i][box_j] != 0)
                bottom_ok = (box_i < 3 and horizontal[box_i+1][box_j] != 0)
                left_ok = (box_j > 0 and vertical[box_i][box_j] != 0)
                right_ok = True
                if top_ok and bottom_ok and left_ok:
                    if capture[box_i][box_j] == -1:
                        valid = False
                    else:
                        boxes_completed += 1
        
        if valid:
            if boxes_completed > best_boxes:
                best_boxes = boxes_completed
                best_move = move
    
    if best_move is None:
        # No valid moves found, choose a move that doesn't create a box with three sides
        for move in moves:
            i, j, d = move
            valid = True
            if d == 'H':
                # Check box above (i-1, j)
                if i > 0:
                    box_i = i - 1
                    box_j = j
                    top_ok = (box_i > 0 and horizontal[box_i][box_j] != 0)
                    bottom_ok = True
                    left_ok = (box_j > 0 and vertical[box_i][box_j] != 0)
                    right_ok = (box_j < 3 and vertical[box_i][box_j+1] != 0)
                    if top_ok and left_ok and right_ok:
                        if capture[box_i][box_j] == 0:
                            valid = False
                # Check box below (i, j)
                if i < 4:
                    box_i = i
                    box_j = j
                    top_ok = (box_i > 0 and horizontal[box_i-1][box_j] != 0)
                    bottom_ok = True
                    left_ok = (box_j > 0 and vertical[box_i][box_j] != 0)
                    right_ok = (box_j < 3 and vertical[box_i][box_j+1] != 0)
                    if top_ok and left_ok and right_ok:
                        if capture[box_i][box_j] == 0:
                            valid = False
            else:  # vertical
                # Check box to the left (i, j-1)
                if j > 0:
                    box_i = i
                    box_j = j - 1
                    top_ok = (box_i > 0 and horizontal[box_i][box_j] != 0)
                    bottom_ok = (box_i < 3 and horizontal[box_i+1][box_j] != 0)
                    left_ok = True
                    right_ok = (box_j < 3 and vertical[box_i][box_j+1] != 0)
                    if top_ok and bottom_ok and right_ok:
                        if capture[box_i][box_j] == 0:
                            valid = False
                # Check box to the right (i, j)
                if j < 4:
                    box_i = i
                    box_j = j
                    top_ok = (box_i > 0 and horizontal[box_i][box_j] != 0)
                    bottom_ok = (box_i < 3 and horizontal[box_i+1][box_j] != 0)

                    if top_ok and bottom_ok and left_ok:
                        if capture[box_i][box_j] == 0:
                            valid = False
            if valid:
                best_move = move
                break
        if best_move is None:
            best_move = moves[0]
    
    return f"{best_move[0]},{best_move[1]},{best_move[2]}"
