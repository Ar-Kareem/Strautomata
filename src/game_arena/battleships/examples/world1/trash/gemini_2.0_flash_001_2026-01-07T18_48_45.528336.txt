```python
import random
import numpy as np

def policy(board: list[list[int]]) -> tuple[int, int]:
    """
    Chooses a cell to target in a Battleship-like game.

    Args:
        board: A 10x10 list of lists representing the current knowledge of the opponent's grid.
            0: unknown cell
            -1: miss
            1: hit

    Returns:
        A tuple (row, col) with integers in the range 0-9, representing the chosen cell to target.
    """

    board_np = np.array(board)
    hits = np.argwhere(board_np == 1)

    if hits.size > 0:
        # Target adjacent cells to existing hits
        row, col = hits[0]
        
        possible_targets = []
        
        for r, c in hits:
            adjacent_cells = [(r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)]

            for nr, nc in adjacent_cells:
                if 0 <= nr < 10 and 0 <= nc < 10 and board[nr][nc] == 0:
                    possible_targets.append((nr, nc))
        
        if possible_targets:
            return random.choice(possible_targets)

    # If no hits, target random cells
    unknown_cells = np.argwhere(board_np == 0)
    if unknown_cells.size > 0:
        row, col = random.choice(unknown_cells)
        return (row, col)

    # If no unknown cells, return a random cell (should not happen)
    return (random.randint(0, 9), random.randint(0, 9))
```
