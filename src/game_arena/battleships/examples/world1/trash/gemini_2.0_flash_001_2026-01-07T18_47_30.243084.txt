import random

def policy(board: list[list[int]]) -> tuple[int, int]:
    """
    A targeting policy for a Battleship-like game on a 10x10 grid.

    Args:
        board: A 10x10 list of lists representing the current knowledge of the opponent grid.
               0  → unknown cell (you have not fired here).
               -1 → miss (you fired here and hit water).
               1  → hit (you fired here and hit a ship cell).

    Returns:
        A tuple (row, col) with integers in the range 0-9, choosing a cell you have not previously fired at.
    """

    # Prioritize targeting cells around known hits
    for r in range(10):
        for c in range(10):
            if board[r][c] == 1:
                # Check adjacent cells
                adjacent_cells = [(r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)]
                valid_adjacent_cells = []
                for row, col in adjacent_cells:
                    if 0 <= row < 10 and 0 <= col < 10 and board[row][col] == 0:
                        valid_adjacent_cells.append((row, col))

                if valid_adjacent_cells:
                    return random.choice(valid_adjacent_cells)

    # If no adjacent cells available, fire randomly at unknown cells
    unknown_cells = []
    for r in range(10):
        for c in range(10):
            if board[r][c] == 0:
                unknown_cells.append((r, c))

    if unknown_cells:
        return random.choice(unknown_cells)

    # If no unknown cells left (which should be impossible), return a random cell that has been targetted already
    for r in range(10):
        for c in range(10):
          if board[r][c] == -1: # check for misses
            return (r,c)

    for r in range(10):
        for c in range(10):
          if board[r][c] == 1: # check for hits
            return (r,c)

    return (0,0) #Should never reach here.
