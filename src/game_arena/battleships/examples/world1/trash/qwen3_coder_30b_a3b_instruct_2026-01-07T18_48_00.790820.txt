import random

def policy(board: list[list[int]]) -> tuple[int, int]:
    # Find all un-fired positions
    un-fired = []
    hits = []
    
    for i in range(10):
        for j in range(10):
            if board[i][j] == 0:
                un-fired.append((i, j))
            elif board[i][j] == 1:
                hits.append((i, j))
    
    # If we have hits, try to expand around them
    if hits:
        # Look for adjacent cells to hits that haven't been fired
        adjacent_unfired = []
        # Check all 4 directions
        for i, j in hits:
            # Up
            if i > 0 and board[i-1][j] == 0:
                adjacent_unfired.append((i-1, j))
            # Down  
            if i < 9 and board[i+1][j] == 0:
                adjacent_unfired.append((i+1, j))
            # Left
            if j > 0 and board[i][j-1] == 0:
                adjacent_unfired.append((i, j-1))
            # Right
            if j < 9 and board[i][j+1] == 0:
                adjacent_unfired.append((i, j+1))
        
        # If there are adjacent unfired cells, fire at one of them
        if adjacent_unfired:
            return random.choice(adjacent_unfired)
    
    # If no hits or no adjacent unfired cells, fire randomly at any un-fired cell
    if un-fired:
        return random.choice(un-fired)
