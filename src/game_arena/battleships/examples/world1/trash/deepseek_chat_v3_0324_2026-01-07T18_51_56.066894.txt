def policy(board: list[list[int]]) -> tuple[int, int]:
    # Convert the board to a numpy array for easier handling
    board_np = np.array(board)
    
    # Get the dimensions of the board
    rows, cols = board_np.shape
    
    # Find all the hits (1s) on the board
    hits = np.argwhere(board_np == 1)
    
    # If there are hits, switch to targeting mode
    if len(hits) > 0:
        # For each hit, check adjacent cells
        for (r, c) in hits:
            # Check all four possible directions
            for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
                nr, nc = r + dr, c + dc
                # If the adjacent cell is within bounds and unknown, target it
                if 0 <= nr < rows and 0 <= nc < cols and board_np[nr, nc] == 0:
                    return (nr, nc)
    
    # If no adjacent cells to hits are found, or no hits, use a checkerboard pattern
    # Create a checkerboard pattern to maximize coverage
    for r in range(rows):
        for c in range(cols):
            if (r + c) % 2 == 0 and board_np[r, c] == 0:
                return (r, c)
    
    # If all checkerboard cells are exhausted, target any remaining unknown cell
    for r in range(rows):

            if board_np[r, c] == 0:
