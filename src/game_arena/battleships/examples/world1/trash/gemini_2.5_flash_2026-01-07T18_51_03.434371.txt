                    if 0 <= min_c - 1 < 10 and board[r_common][min_c - 1] == 0:
                        _global_target_candidates.append((r_common, min_c - 1))
                    # Try extending right
                    if 0 <= max_c + 1 < 10 and board[r_common][max_c + 1] == 0:
                        _global_target_candidates.append((r_common, max_c + 1))
                else: # Must be vertical
                    c_common = _global_hits_pending_sinking[0][1]
                    min_r = min(r for r, _ in _global_hits_pending_sinking)
                    max_r = max(r for r, _ in _global_hits_pending_sinking)

                    # Try extending up
                    if 0 <= min_r - 1 < 10 and board[min_r - 1][c_common] == 0:
                        _global_target_candidates.append((min_r - 1, c_common))
                    # Try extending down
                    if 0 <= max_r + 1 < 10 and board[max_r + 1][c_common] == 0:
                        _global_target_candidates.append((max_r + 1, c_common))

        # If we have candidates, pop one and return it
        if _global_target_candidates:
            return _global_target_candidates.popleft() # Use popleft for deque

        # If we are here, it means _global_hits_pending_sinking is not empty,
        # but no valid target candidates could be generated or found.
        # This implies the current segment is "closed" (sunk or surrounded).
        # We should clear _global_hits_pending_sinking for the next turn
        # to effectively "give up" on this segment and look for new ships.
        # This will move us to the hunting phase.
        _global_hits_pending_sinking = [] # This is an important step to prevent getting stuck

    # Hunting Phase: If no active hits or no target candidates, hunt for new ships
    # Use a checkerboard pattern.
    
    # Try cells where (r+c) is even first (pattern 1)
    for r in range(10):
        for c in range(10):

        for c in range(10):

                return (r, c)

    raise Exception("No unfired cells left on the board!")

The `_global_hits_pending_sinking = []` at the end of the targeting phase inside the `if _global_hits_pending_sinking:` block is crucial.
Let's trace it:
1. Hit (0,0). `_global_hits_pending_sinking = [(0,0)]`. Candidates `(0,1),(1,0),(-1,0), (0,-1)`. Returns `(0,1)`.
2. Shot `(0,1)`:
   *   If `HIT`, `_global_hits_pending_sinking = [(0,0),(0,1)]`. Candidates re-generated for extension: `(0,-1),(0,2)`. Returns `(0,-1)`.
   *   If `MISS`, `_global_hits_pending_sinking = [(0,0)]`. Candidates filtered: `(1,0),(-1,0),(0,-1)`. Returns `(1,0)`.
3. Shot `(0,-1)`:
   *   If `MISS`, `_global_hits_pending_sinking = [(0,0),(0,1)]`. Candidates filtered: `(0,2)`. Returns `(0,2)`.
4. Shot `(0,2)`:
   *   If `HIT`, `_global_hits_pending_sinking = [(0,0),(0,1),(0,2)]`. Candidates re-generated: `(0,3)`. Returns `(0,3)`.
5. Shot `(0,3)`:
   *   If `MISS`, `_global_hits_pending_sinking = [(0,0),(0,1),(0,2)]`. Candidates filtered -> `_global_target_candidates` becomes empty.
   *   Since `_global_target_candidates` is now empty, and `_global_hits_pending_sinking` is *not* empty, the code block `_global_hits_pending_sinking = []` is executed.
   *   On the *next* turn, `_global_hits_pending_sinking` will be recreated from `current_board_hits = [(0,0),(0,1),(0,2)]`. But since no candidates can be generated anymore, it will again hit `_global_hits_pending_sinking = []` and fall back to hunting.

This means if a ship is hit and fully surrounded by misses or edges, the AI will eventually give up trying to extend its current hits. This is the desired behavior when the game doesn't explicitly tell us a ship is sunk. It essentially "sinks" the ship from the AI's perspective and moves on. This is a common and effective heuristic.

import collections

# Global variables to maintain state between policy calls.
# These will persist throughout the game simulation.
_global_hits_pending_sinking: list[tuple[int, int]] = []
# A deque is used for _global_target_candidates for efficient popping from the left (FIFO).
_global_target_candidates: collections.deque[tuple[int, int]] = collections.deque()

    """
    Implements a targeting policy for a Battleship-like game.

    The policy prioritizes sinking active ships, then uses a checkerboard pattern
    to hunt for new ships.

    Args:
        board: A 10x10 list of lists representing the current knowledge of the opponent's grid.
               0: Unknown cell (not fired)
              -1: Miss (fired and hit water)
               1: Hit (fired and hit a ship cell)

        A tuple (row, col) representing the next target cell to fire at.
    """

    current_board_hits = []
    for r in range(10):
        for c in range(10):
            if board[r][c] == 1:
                current_board_hits.append((r, c))
    current_board_hits.sort() # Sort for consistent comparison and processing

    # 2. Check if the set of active hits has changed significantly.
    # This indicates either a ship was sunk, or a new ship segment was hit,
    # or the current segment grew/shrank.
    if _global_hits_pending_sinking != current_board_hits:
        # If the hits we were tracking are now different, clear previous target candidates.
        # This prevents trying to sink a ship that's already gone or changed.
        _global_target_candidates.clear()
    
    _global_hits_pending_sinking = current_board_hits # Update global state with current hits


    # 3. Targeting Phase: If we have active hits, prioritize sinking the ship(s).
    if _global_hits_pending_sinking:
        # Filter out _global_target_candidates that are no longer '0' (already fired).
        # This keeps the candidate list clean and prevents re-firing at known spots.
        _global_target_candidates = collections.deque([
            (r, c) for r, c in _global_target_candidates
            if board[r][c] == 0
        ])

        # If no valid candidates remain after filtering, generate new ones based on current hits.
        if not _global_target_candidates:
            if len(_global_hits_pending_sinking) == 1:
                # Case: Only one hit. Try all four cardinal neighbors to determine orientation.
                hr, hc = _global_hits_pending_sinking[0]

                    if 0 <= nr < 10 and 0 <= nc < 10 and board[nr][nc] == 0:
                        _global_target_candidates.append((nr, nc))
            else:

                # Assuming all hits in _global_hits_pending_sinking belong to one contiguous ship.
                
                # Check if it's horizontal (all row coordinates are the same)
                is_horizontal = all(h[0] == _global_hits_pending_sinking[0][0] for h in _global_hits_pending_sinking)
                
                if is_horizontal:
                    r_common = _global_hits_pending_sinking[0][0]
                    # Find min and max column indices to extend from ends
                    min_c = min(c for _, c in _global_hits_pending_sinking)
                    max_c = max(c for _, c in _global_hits_pending_sinking)
                    
                    # Try extending to the left
                    if 0 <= min_c - 1 < 10 and board[r_common][min_c - 1] == 0:
                        _global_target_candidates.append((r_common, min_c - 1))
                    # Try extending to the right
                    if 0 <= max_c + 1 < 10 and board[r_common][max_c + 1] == 0:
                        _global_target_candidates.append((r_common, max_c + 1))
                else: # Must be vertical (all column coordinates are the same)
                    c_common = _global_hits_pending_sinking[0][1]
                    # Find min and max row indices to extend from ends
                    min_r = min(r for r, _ in _global_hits_pending_sinking)
                    max_r = max(r for r, _ in _global_hits_pending_sinking)

                    # Try extending upwards
                    if 0 <= min_r - 1 < 10 and board[min_r - 1][c_common] == 0:
                        _global_target_candidates.append((min_r - 1, c_common))
                    # Try extending downwards
                    if 0 <= max_r + 1 < 10 and board[max_r + 1][c_common] == 0:
                        _global_target_candidates.append((max_r + 1, c_common))

        # If we have generated valid candidates, pop the first one and return it.
        if _global_target_candidates:
            return _global_target_candidates.popleft()

        # If we reach this point, it means _global_hits_pending_sinking is not empty,
        # but no valid next shot candidates could be generated (e.g., ship is surrounded by misses/edges).
        # In this scenario, we assume the current ship segment is fully explored/sunk from our perspective.
        # We clear _global_hits_pending_sinking to switch back to hunting mode for other ships.
        _global_hits_pending_sinking = [] 

    # 4. Hunting Phase: If no active ships are being targeted, hunt for new ones.
    # A checkerboard pattern is used to efficiently cover the board and increase hit probability.
    # We prioritize cells where (row + col) is even, as this covers 50% of the board with maximal
