                    potential_targets.append((r+1, c)) # Extend down
                if r > 1 and board[r-2][c] == 0:
                    potential_targets.append((r-2, c)) # Extend up

    1. Sinking Phase: If any ship cells have been hit (value 1),
       the policy attempts to sink that ship by prioritizing shots
       that extend known ship segments. If no segments can be extended,
       it fires at immediate neighbors of any hit cell.
    2. Hunting Phase: If no ships are currently hit, the policy
       searches for new ships using a checkerboard pattern for efficiency.

    Args:
        board: A 10x10 list of lists representing the current knowledge of the opponent's grid.
               0: Unknown cell (never fired)
               -1: Miss (fired here and hit water)
               1: Hit (fired here and hit a ship cell)

    Returns:
        A tuple (row, col) with integers in the range 0-9, representing
        the chosen cell to fire at.
    """
    BOARD_SIZE = 10

    # 1. Identify all hit cells (cells with value 1)
    hit_cells = []
    for r in range(BOARD_SIZE):
        for c in range(BOARD_SIZE):
            if board[r][c] == 1:
                hit_cells.append((r, c))

    # 2. Sinking Phase: If there are any active hits
    if hit_cells:
        potential_targets = []

        # Prioritize extending existing ship segments (two or more adjacent 1s)
        # We iterate through all hit cells and check their neighbors to identify segments.
        # This approach ensures we look for potential shots at both ends of a segment.
        for r, c in hit_cells:
            # Check for horizontal segments
            # Case 1: (r, c-1) and (r, c) are hits. Try extending to (r, c+1) or (r, c-2).
            if c > 0 and board[r][c-1] == 1: # Found a horizontal segment like '11' ending at (r,c)
                if c < BOARD_SIZE - 1 and board[r][c+1] == 0:
                    potential_targets.append((r, c+1)) # Try extending to the right
                if c > 1 and board[r][c-2] == 0:
                    potential_targets.append((r, c-2)) # Try extending to the left

            # Case 2: (r, c) and (r, c+1) are hits. Try extending to (r, c-1) or (r, c+2).
            # This check covers segments like '11' starting at (r,c)
            if c < BOARD_SIZE - 1 and board[r][c+1] == 1:
                if c > 0 and board[r][c-1] == 0:
                    potential_targets.append((r, c-1)) # Try extending to the left
                if c < BOARD_SIZE - 2 and board[r][c+2] == 0:
                    potential_targets.append((r, c+2)) # Try extending to the right

            # Check for vertical segments (symmetric logic to horizontal)
            # Case 1: (r-1, c) and (r, c) are hits. Try extending to (r+1, c) or (r-2, c).
            if r > 0 and board[r-1][c] == 1: # Found a vertical segment ending at (r,c)
                if r < BOARD_SIZE - 1 and board[r+1][c] == 0:
                    potential_targets.append((r+1, c)) # Try extending down
                if r > 1 and board[r-2][c] == 0:
                    potential_targets.append((r-2, c)) # Try extending up

            # Case 2: (r, c) and (r+1, c) are hits. Try extending to (r-1, c) or (r+2, c).
            # This covers segments starting at (r,c)
            if r < BOARD_SIZE - 1 and board[r+1][c] == 1:
                if r > 0 and board[r-1][c] == 0:
                    potential_targets.append((r-1, c)) # Try extending up
                if r < BOARD_SIZE - 2 and board[r+2][c] == 0:
                    potential_targets.append((r+2, c)) # Try extending down
        
        # Remove duplicate targets (if any) and shuffle to add some unpredictability
        potential_targets = list(set(potential_targets))
        
        # If we found valid targets from extending segments, pick one randomly
        if potential_targets:
            return random.choice(potential_targets)

        # If no segments could be extended (e.g., only single hits, or segments are
        # surrounded by misses/board edges), try firing at immediate neighbors of *any* hit cell.
        # This handles the initial phase of finding ship orientation after a first hit.
        single_hit_neighbors = []
        for r, c in hit_cells:
            # Check cardinal neighbors
            for dr, dc in [(0, 1), (0, -1), (1, 0), (-1, 0)]:
                nr, nc = r + dr, c + dc
                # Ensure neighbor is within bounds and has not been fired at yet (value 0)
                if 0 <= nr < BOARD_SIZE and 0 <= nc < BOARD_SIZE and board[nr][nc] == 0:
                    single_hit_neighbors.append((nr, nc))
        
        # If valid neighbors were found, pick one randomly
        if single_hit_neighbors:
            return random.choice(single_hit_neighbors)
            
        # Edge case: All known hit cells are completely surrounded by misses or board edges,
        # but the game state doesn't explicitly mark ships as 'sunk'. This implies that
        # effectively, no further moves can be made to sink these 'hit' ships.
        # In such a scenario, the policy should fall back to the hunting phase.

    # 3. Hunting Phase: No active hits, or no valid targets could be found around hits.
    # Use a checkerboard pattern for efficient searching. This strategy helps cover the board
    # effectively and increases the probability of hitting a ship of length >= 2 quickly.
    
    empty_cells_even_sum = [] # Cells where (row + col) is an even number
    empty_cells_odd_sum = []  # Cells where (row + col) is an odd number

    for r in range(BOARD_SIZE):
        for c in range(BOARD_SIZE):
            if board[r][c] == 0: # Only consider un-fired cells
                if (r + c) % 2 == 0:
                    empty_cells_even_sum.append((r, c))
                else:
                    empty_cells_odd_sum.append((r, c))

    # Prioritize firing at checkerboard cells with an even sum of coordinates
    # This ensures a systematic and spread-out search pattern.
    if empty_cells_even_sum:
        return random.choice(empty_cells_even_sum)
    # If all even-sum cells have been fired at, fall back to odd-sum cells.
    # This covers the remaining cells on the board.
    elif empty_cells_odd_sum:
        return random.choice(empty_cells_odd_sum)
    
    # Fallback: This point should theoretically not be reached in a valid game state,
    # as it implies no empty cells are left on the board, meaning the game should be over.
    return (0, 0)
