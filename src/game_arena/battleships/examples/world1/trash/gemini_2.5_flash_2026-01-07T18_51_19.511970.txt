    _hunting_idx_parity = 1 - _hunting_idx_parity # Flip 0 to 1, or 1 to 0
    _hunting_idx_row = 0
    _hunting_idx_col = 0
    
    # Search again with the new parity, starting from (0,0)
    for r in range(10):
        for c in range(10):
            if (r + c) % 2 == _hunting_idx_parity:
                if board[r][c] == 0:
                    # Update global state for the next call
                    _hunting_idx_row, _hunting_idx_col = r, c + 1
                    if _hunting_idx_col == 10:
                        _hunting_idx_col = 0
                        _hunting_idx_row += 1
                    return r, c

    # Fallback: If no ships are found in either checkerboard pattern (should only happen if game is over)
    # This also acts as a final sweep for any very small ships that might have been missed by strict checkerboard
    # (e.g., a single-cell ship, though not in this game variant) or just to wrap up.
    for r in range(10):
        for c in range(10):
            if board[r][c] == 0:
                return r, c
    
    # Should not be reached in a normal game. All cells must be known.
    # Return a dummy value in case of an unforeseen scenario (e.g., board full)
    return 0, 0
