```python
    Implements a targeting policy for a Battleship-like game.

    The policy prioritizes sinking damaged ships, then finding ship segments,
    and finally uses a checkerboard pattern for efficient hunting.

    Args:
        board: A 10x10 list of lists representing the current knowledge of the

               0  -> unknown cell (not yet fired)

               1  -> hit (fired here and hit a ship cell)

                    # Try to extend to the right of the segment.
                    if is_valid(r, c + 2) and board[r][c + 2] == EMPTY:
                        potential_line_extension_targets.append((r, c + 2))
                
                # Check for vertical segment (e.g., (r,c) and (r+1,c) are hits)
                # We only check (r+1,c) to avoid adding segments like (r,c) and (r-1,c)
                # that would already be considered when processing (r-1,c).
                if is_valid(r + 1, c) and board[r + 1][c] == HIT:
                    # Try to extend upwards from the segment.
                    if is_valid(r - 1, c) and board[r - 1][c] == EMPTY:
                        potential_line_extension_targets.append((r - 1, c))
                    # Try to extend downwards from the segment.
                    if is_valid(r + 2, c) and board[r + 2][c] == EMPTY:
                        potential_line_extension_targets.append((r + 2, c))

    if potential_line_extension_targets:
        # Remove duplicates (e.g., if multiple adjacent hits suggest the same target)
        unique_targets = list(set(potential_line_extension_targets))
        return random.choice(unique_targets)

    # --- Phase 2: Target isolated hits ---
    # If no clear ship lines can be extended (i.e., no two consecutive hits,
    # or no valid EMPTY cells to extend from existing lines), we look for
    # any 'HIT' cells and try to hit their immediate 'EMPTY' neighbors.
    # This helps in finding the orientation of a ship or sinking small ships
    # that have only been hit once.
    
    potential_adjacent_targets = []
    for r in range(GRID_SIZE):
        for c in range(GRID_SIZE):
            if board[r][c] == HIT:
                # Check all 4 orthogonal neighbors
                for dr, dc in [(0, 1), (0, -1), (1, 0), (-1, 0)]:
                    nr, nc = r + dr, c + dc
                    if is_valid(nr, nc) and board[nr][nc] == EMPTY:
                        potential_adjacent_targets.append((nr, nc))

    if potential_adjacent_targets:
        unique_targets = list(set(potential_adjacent_targets))
        return random.choice(unique_targets)

    # --- Phase 3: Hunt mode (checkerboard pattern) ---
    # If no ships are currently hit or being targeted, switch to hunting mode.
    # A checkerboard pattern (every other square) is a statistically efficient
    # way to find ships, especially given the ship lengths (smallest is 2).
    
    hunt_targets = []

    # First, try to hit cells where (row + col) is even.
    # This creates a primary checkerboard pattern.
    for r in range(GRID_SIZE):
        for c in range(GRID_SIZE):
            if board[r][c] == EMPTY and (r + c) % 2 == 0:
                hunt_targets.append((r, c))

    if hunt_targets:
        return random.choice(hunt_targets)

    # If all (row + col) % 2 == 0 cells have been shot at,
    # then try the remaining cells where (row + col) is odd.
    # This ensures that all cells are eventually covered.
    hunt_targets.clear() # Clear the list for new targets
    for r in range(GRID_SIZE):
        for c in range(GRID_SIZE):
            if board[r][c] == EMPTY and (r + c) % 2 == 1:
                hunt_targets.append((r, c))

    if hunt_targets:
        return random.choice(hunt_targets)

    # --- Fallback: If somehow nothing found ---
    # This section should theoretically not be reached in a typical game,
    # as the `policy` function expects to return an unshot cell.
    # It would only be hit if all cells on the board are already marked
    # as 'HIT' (1) or 'MISS' (-1), meaning there are no 'EMPTY' (0) cells left.
    # In such a scenario, the game should already be over.
    
    # As a safeguard, iterate and return the first EMPTY cell found.
    # If even this loop finds nothing, it indicates a highly unusual state,
    # possibly an error in game logic or the board state passed.
    for r in range(GRID_SIZE):
        for c in range(GRID_SIZE):
            if board[r][c] == EMPTY:
                return (r, c)
    
    # If no EMPTY cells are found whatsoever, return (0,0) as a last resort.
    # This implies all cells are already shot.
    return (0, 0)

```
