import random

def policy(board: list[list[int]]) -> tuple[int, int]:
    """
    Chooses a target on the Battleship grid based on hit/miss information.

    Args:
        board: A 10x10 list of lists representing the game board.
               0: unknown, -1: miss, 1: hit

    Returns:
        A tuple (row, col) representing the coordinates of the chosen target.
    """

    rows = len(board)
    cols = len(board[0])

    # Prioritize cells around hits
    for r in range(rows):
        for c in range(cols):
            if board[r][c] == 1:
                # Check adjacent cells
                neighbors = [(r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)]
                possible_targets = []
                for nr, nc in neighbors:
                    if 0 <= nr < rows and 0 <= nc < cols and board[nr][nc] == 0:
                        possible_targets.append((nr, nc))
                if possible_targets:
                    return random.choice(possible_targets)

    # If no hits, choose randomly from unknown cells
    unknown_cells = []
    for r in range(rows):
        for c in range(cols):
            if board[r][c] == 0:
                unknown_cells.append((r, c))

    if unknown_cells:
        return random.choice(unknown_cells)

    # If no unknown cells (shouldn't happen in a standard game), return a random cell
    return (random.randint(0, rows - 1), random.randint(0, cols - 1))
