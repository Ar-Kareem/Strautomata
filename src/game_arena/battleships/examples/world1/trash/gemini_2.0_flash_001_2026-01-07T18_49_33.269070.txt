import random

def policy(board: list[list[int]]) -> tuple[int, int]:
    """
    A targeting policy for a Battleship-like game on a 10x10 grid.

    Args:
        board: A 10x10 list of lists representing the current knowledge of the opponent grid.
            0: unknown cell
            -1: miss
            1: hit

    Returns:
        A tuple (row, col) with integers in the range 0-9, choosing a cell you have not previously fired at.
    """

    # Prioritize cells adjacent to hits to finish sinking ships
    for row in range(10):
        for col in range(10):
            if board[row][col] == 1:
                # Check adjacent cells
                adjacent_cells = []
                if row > 0 and board[row - 1][col] == 0:
                    adjacent_cells.append((row - 1, col))
                if row < 9 and board[row + 1][col] == 0:
                    adjacent_cells.append((row + 1, col))
                if col > 0 and board[row][col - 1] == 0:
                    adjacent_cells.append((row, col - 1))
                if col < 9 and board[row][col + 1] == 0:
                    adjacent_cells.append((row, col + 1))

                if adjacent_cells:
                    return random.choice(adjacent_cells)

    # If no hits, target random cells
    unknown_cells = []
    for row in range(10):
        for col in range(10):
            if board[row][col] == 0:
                unknown_cells.append((row, col))

    if unknown_cells:
        return random.choice(unknown_cells)
    else:
        # If no unknown cells, just return something valid (shouldn't happen)
        return (0, 0)
