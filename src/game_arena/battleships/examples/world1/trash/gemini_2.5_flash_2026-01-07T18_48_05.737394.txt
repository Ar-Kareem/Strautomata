    # We iterate through active_hits. If any hit is part of a line, we try to extend it.
    for r_hit, c_hit in active_hits:
        # Check for horizontal line existence (at least two '1's horizontally adjacent)
        is_horizontal_segment = \
            (is_valid(r_hit, c_hit - 1) and board[r_hit][c_hit - 1] == 1) or \
            (is_valid(r_hit, c_hit + 1) and board[r_hit][c_hit + 1] == 1)

        if is_horizontal_segment:
            # Try to extend along the line in both directions
            # Search to the left
            for c_scan in range(c_hit - 1, -1, -1):
                if board[r_hit][c_scan] == 0:
                    return (r_hit, c_scan)
                if board[r_hit][c_scan] == -1: # Hit water, this end of the ship is closed off
                    break
            # Search to the right
            for c_scan in range(c_hit + 1, GRID_SIZE):
                if board[r_hit][c_scan] == 0:
                    return (r_hit, c_scan)
                if board[r_hit][c_scan] == -1: # Hit water, this end of the ship is closed off
                    break

        # Check for vertical line existence (at least two '1's vertically adjacent)
        is_vertical_segment = \
            (is_valid(r_hit - 1, c_hit) and board[r_hit - 1][c_hit] == 1) or \
            (is_valid(r_hit + 1, c_hit) and board[r_hit + 1][c_hit] == 1)

        if is_vertical_segment:
            # Try to extend along the line in both directions
            # Search upwards
            for r_scan in range(r_hit - 1, -1, -1):
                if board[r_scan][c_hit] == 0:
                    return (r_scan, c_hit)
                if board[r_scan][c_hit] == -1: # Hit water, this end of the ship is closed off
                    break
            # Search downwards
            for r_scan in range(r_hit + 1, GRID_SIZE):
                if board[r_scan][c_hit] == 0:
                    return (r_scan, c_hit)
                if board[r_scan][c_hit] == -1: # Hit water, this end of the ship is closed off
                    break

    # If no clear line extension from *any* hit, it means all '1's are either isolated
    # or their direct extensions are already -1 (miss) or 1 (part of the same ship).
    # In this case, we look for any unknown cell adjacent to *any* hit.
    candidates_around_hits = []
    for r_hit, c_hit in active_hits:
        # Explore 4 orthogonal neighbors
        for dr, dc in [(0, 1), (0, -1), (1, 0), (-1, 0)]:
            nr, nc = r_hit + dr, c_hit + dc
            if is_valid(nr, nc) and board[nr][nc] == 0:
                candidates_around_hits.append((nr, nc))

    if candidates_around_hits:
        # Randomly choose one of the potential targets around hits
        return random.choice(candidates_around_hits)

    # If active_hits exist but no valid '0' target could be found (e.g., ship is effectively sunk,
    # or completely surrounded by misses), then return None, signaling to switch to hunt phase.
    return None

def hunt_for_ship(board: list[list[int]]) -> tuple[int, int]:
    """
    Hunts for ships using a checkerboard pattern to efficiently cover the board.
    Randomizes choice among valid checkerboard cells.
    """
    untried_checkerboard_cells = []
    for r in range(GRID_SIZE):
        for c in range(GRID_SIZE):
            # Use a checkerboard pattern (r + c) % 2 == 0
            # This ensures that any ship of length 2 or more will eventually be hit.
            if (r + c) % 2 == 0 and board[r][c] == 0:
                untried_checkerboard_cells.append((r, c))

    if untried_checkerboard_cells:
        # Choose a random cell from the available checkerboard cells
        return random.choice(untried_checkerboard_cells)

    # Fallback: If all checkerboard cells are tried (e.g., end of game, or unusually placed ships),
    # try any remaining unknown cell. This should rarely be needed.
    all_untried_cells = []
    for r in range(GRID_SIZE):
        for c in range(GRID_SIZE):
            if board[r][c] == 0:
                all_untried_cells.append((r, c))

    if all_untried_cells:
        return random.choice(all_untried_cells)

    # This case should ideally only be reached when the game is over
    # and all cells have been either hit or missed.
    # If it's reached during an active game, it indicates an issue or a very full board.
    # Return a default to prevent error, though it shouldn't be executed in practical play.
    return (0, 0) # Harmless default.


def policy(board: list[list[int]]) -> tuple[int, int]:
    """
    Determines the next shot (row, col) in a Battleship game.

    It implements a two-phase strategy:
    1. Target Phase: If there are any active hits ('1's) on the board, it tries to
       sink the ship by extending known segments or exploring adjacent cells around hits.
    2. Hunt Phase: If no active hits, or if the target phase cannot find a valid
       cell to shoot, it switches to hunting for new ships using a checkerboard
       pattern.
    """
    # Attempt to sink a ship if there are active hits
    target_from_hits = find_target_from_hits(board)
    if target_from_hits:
        return target_from_hits

    # If no active hits or no clear targets from hits, hunt for a new ship
    return hunt_for_ship(board)
