import random

def policy(board: list[list[int]]) -> tuple[int, int]:
    """
    A Battleship targeting policy.

    Args:
        board: A 10x10 grid representing the current knowledge of the opponent grid.
               0  → unknown cell (you have not fired here).
               -1 → miss (you fired here and hit water).
               1  → hit (you fired here and hit a ship cell).

    Returns:
        A tuple (row, col) with integers in the range 0-9, choosing a cell you have not previously fired at.
    """
    rows = len(board)
    cols = len(board[0])

    # Prioritize attacking adjacent cells to known hits
    for r in range(rows):
        for c in range(cols):
            if board[r][c] == 1:
                # Check adjacent cells
                adjacent_cells = [(r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)]
                valid_adjacent_cells = []
                for adj_r, adj_c in adjacent_cells:
                    if 0 <= adj_r < rows and 0 <= adj_c < cols and board[adj_r][adj_c] == 0:
                        valid_adjacent_cells.append((adj_r, adj_c))
                if valid_adjacent_cells:
                    return random.choice(valid_adjacent_cells)

    # If no adjacent cells to hit, choose a random unknown cell
    unknown_cells = []
    for r in range(rows):
        for c in range(cols):
            if board[r][c] == 0:
                unknown_cells.append((r, c))

    if unknown_cells:
        return random.choice(unknown_cells)
    else:
        # Should not occur, but return (0, 0) if the board is full.
        return (0, 0)
