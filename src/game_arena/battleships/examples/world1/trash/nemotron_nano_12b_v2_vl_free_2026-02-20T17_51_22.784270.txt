# Strategy

The policy combines systematic exploration with adaptive targeting around detected ships. During initial exploration (global phase), it uses a spiral pattern to scan the board methodically, ensuring no cell is missed. Once a ship is detected via a hit, the policy switches to local phase, prioritizing cells adjacent to known hits to sink the ship efficiently. It calculates a score for each unclicked cell based on the number of adjacent hits, favoring cells near existing hits to maximize the chance of sinking ships quickly. The spiral order ensures thorough global search, while the score-based selection in local phase optimizes ship destruction.

def policy(board):
    # Determine if any ship cells have been hit
    any_hit = any(cell == 1 for row in board for cell in row)

    if any_hit:
        # Local phase: attack cells adjacent to known hits
        candidates = []
        for x in range(10):
            for y in range(10):
                if board[x][y] == 0:
                    # Count adjacent hits to prioritize promising cells
                    cnt = 0
                    for dx, dy in [(-1,0), (1,0), (0,-1), (0,1)]:
                        nx, ny = x+dx, y+dy
                        if 0 <= nx < 10 and 0 <= ny < 10 and board[nx][ny] == 1:
                            cnt += 1
                    # Store with negative score for ascending sort
                    candidates.append( (-cnt, x, y) )
        
        # Return first cell in optimized attack order
        if candidates:
            candidates.sort()
            return (candidates[0][1], candidates[0][2])
        return None  # No attack possible (unlikely)
    
    else:
        # Global phase: attack in spiral search order
        spiral = []
        def recursive_spiral(x_min, x_max, y_min, y_max):
            if x_min > x_max or y_min > y_max:
                return
            # Top row left to right
            for y in range(y_min, y_max + 1):
                spiral.append((x_min, y))
            # Right column top to bottom
            if x_min < x_max:
                for x in range(x_min + 1, x_max + 1):
                    spiral.append((x, y_max))
            # Bottom row right to left
            if x_min != x_max:
                for y in range(y_max - 1, y_min - 1, -1):
                    spiral.append((x_max, y))
            # Left column bottom to top
            if y_min != y_max:
                for x in range(x_max - 1, x_min, -1):
                    spiral.append((x, y_min))
            # Recurse inward
            recursive_spiral(x_min + 1, x_max - 1, y_min + 1, y_max - 1)
        
        recursive_spiral(0, 9, 0, 9)
        # Find first unattacked cell in spiral order
        for pos in spiral:
            x, y = pos
            if board[x][y] == 0:
                return (x, y)
        return None  # All cells attacked
