import numpy as np
import random

def policy(board: list[list[int]]) -> tuple[int, int]:
    """
    A targeting policy for a Battleship-like game on a 10x10 grid.

    Args:
        board: A 10x10 list of lists representing the current knowledge of the opponent grid.
            0: unknown cell
            -1: miss
            1: hit

    Returns:
        A tuple (row, col) with integers in the range 0-9, choosing a cell to fire at.
    """

    board_np = np.array(board)
    hits = np.argwhere(board_np == 1)

    if hits.size > 0:
        # Prioritize cells around hits
        for row, col in hits:
            potential_targets = [(row - 1, col), (row + 1, col), (row, col - 1), (row, col + 1)]
            valid_targets = []
            for r, c in potential_targets:
                if 0 <= r < 10 and 0 <= c < 10 and board_np[r, c] == 0:
                    valid_targets.append((r, c))
            if valid_targets:
                return random.choice(valid_targets)

    # If no hits, choose randomly among unknown cells
    unknown_cells = np.argwhere(board_np == 0)
    if unknown_cells.size > 0:
        row, col = random.choice(unknown_cells)
        return (row, col)

    # If no unknown cells, choose randomly among missed cells (shouldn't happen)
    missed_cells = np.argwhere(board_np == -1)
    if missed_cells.size > 0:
        row, col = random.choice(missed_cells)
        return (row, col)

    # If everything has been tried (shouldn't happen), return (0, 0)
    return (0, 0)
