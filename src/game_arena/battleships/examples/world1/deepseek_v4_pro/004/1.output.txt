The strategy for this policy is based on a probabilistic heatmap approach (often called the "probability density function" method). It enumerates all valid placements for every remaining ship (lengths 5, 4, 3, 3, 2) that are consistent with the current board (no placements covering misses). For each unknown cell, the algorithm counts how many of these valid ship placements would cover that cell. The cell with the highest total count is chosen as the next target. This method efficiently balances exploration and exploitation: it naturally prioritizes finishing partially hit ships (because adjacent cells receive many counts from possible extensions) while also seeking new ships in high-probability regions. Ties are broken randomly to avoid being predictable.

import random

def policy(board: list[list[int]]) -> tuple[int, int]:
    ships = [5, 4, 3, 3, 2]
    scores = [[0 for _ in range(10)] for _ in range(10)]
    
    # Accumulate scores from all valid placements of all ships
    for L in ships:
        # Horizontal placements
        for r in range(10):
            for c in range(10 - L + 1):
                # Check if this placement is valid (contains no -1)
                valid = True
                for dc in range(L):
                    if board[r][c + dc] == -1:
                        valid = False
                        break
                if valid:
                    # Add to score for each unknown cell covered
                    for dc in range(L):
                        if board[r][c + dc] == 0:
                            scores[r][c + dc] += 1
                            
        # Vertical placements
        for c in range(10):
            for r in range(10 - L + 1):
                valid = True
                for dr in range(L):
                    if board[r + dr][c] == -1:
                        valid = False
                        break
                if valid:
                    for dr in range(L):
                        if board[r + dr][c] == 0:
                            scores[r + dr][c] += 1

    # Find the unknown cell(s) with the maximum score
    max_score = -1
    best_moves = []
    for r in range(10):
        for c in range(10):
            if board[r][c] == 0:
                if scores[r][c] > max_score:
                    max_score = scores[r][c]
                    best_moves = [(r, c)]
                elif scores[r][c] == max_score:
                    best_moves.append((r, c))

    # Tie-broken randomly to avoid predictable patterns
    return random.choice(best_moves)
