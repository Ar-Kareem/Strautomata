The strategy is to make a valid move, prioritizing moves that lead to immediate captures of opponent pieces in the next turn. If no such move exists, return the first valid move found. Let's implement this strategy.

```python
import numpy as np

def policy(you: list[int], opponent: list[int]) -> str:
    """
    Selects a move for the Clobber game.

    Args:
        you: A 5x6 array representing your pieces (1) and empty cells (0).
        opponent: A 5x6 array representing the opponent's pieces (1) and empty cells (0).

    Returns:
        A string representing the move in the format "row,col,dir".
    """
    you = np.array(you)
    opponent = np.array(opponent)

    def get_valid_moves(you, opponent):
        moves = []
        for row in range(5):
            for col in range(6):
                if you[row, col] == 1:
                    # Check Up
                    if row > 0 and opponent[row-1, col] == 1:
                        moves.append((row, col, 'U'))
                    # Check Down
                    if row < 4 and opponent[row+1, col] == 1:
                        moves.append((row, col, 'D'))
                    # Check Left
                    if col > 0 and opponent[row, col-1] == 1:
                        moves.append((row, col, 'L'))
                    # Check Right
                    if col < 5 and opponent[row, col+1] == 1:
                        moves.append((row, col, 'R'))
        return moves

    valid_moves = get_valid_moves(you, opponent)

    if not valid_moves:
        raise ValueError("No valid moves found, but this shouldnt be happening in the evaluation")

    return f"{valid_moves[0][0]},{valid_moves[0][1]},{valid_moves[0][2]}"
```
