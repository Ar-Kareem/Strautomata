def policy(you: list[int], opponent: list[int]) -> int:
    ROWS = 5
    COLS = 6
    
    # Direction vectors and characters
    dirs_vec = [(-1, 0), (1, 0), (0, -1), (0, 1)]  # U, D, L, R
    dirs_char = ['U', 'D', 'L', 'R']
    dir_map = dict(zip(dirs_vec, dirs_char))

    # Parse board state into sets of coordinates
    my_pieces = set()
    opp_pieces = set()

    # Detect if input is flattened (length 30) or 2D (length 5)
    # The prompt implies 5x6 arrays. 
    if len(you) == ROWS * COLS and not isinstance(you[0], list):
        # Flattened: 1D list of 30 ints
        for i in range(ROWS * COLS):
            r, c = divmod(i, COLS)
            if you[i] == 1:
                my_pieces.add((r, c))
            if opponent[i] == 1:
                opp_pieces.add((r, c))
    else:
        # 2D: List of lists (standard)
        for r in range(ROWS):
            for c in range(COLS):
                # Safety checks for varying list structures
                if r < len(you) and c < len(you[r]) and you[r][c] == 1:
                    my_pieces.add((r, c))
                if r < len(opponent) and c < len(opponent[r]) and opponent[r][c] == 1:
                    opp_pieces.add((r, c))

    def get_moves(pieces, opps):
        """Generate all valid capture moves for a set of pieces."""
        moves = []
        for r, c in pieces:
            for dr, dc in dirs_vec:
                nr, nc = r + dr, c + dc
                if 0 <= nr < ROWS and 0 <= nc < COLS:
                    if (nr, nc) in opps:
                        moves.append(((r, c), (nr, nc), (dr, dc)))
        return moves

    # Generate my legal moves
    legal_moves = get_moves(my_pieces, opp_pieces)
    
    best_move = None
    best_score = -1e9

    for start, end, dvec in legal_moves:
        # Simulate the move
        new_my_pieces = my_pieces.copy()
        new_my_pieces.remove(start)
        new_my_pieces.add(end)
        
        new_opp_pieces = opp_pieces.copy()
        new_opp_pieces.remove(end)

        # 1. Check for Immediate Win
        # If opponent has no moves after my move, I win.
        opp_moves_next = get_moves(new_opp_pieces, new_my_pieces)
        if not opp_moves_next:
            return f"{start[0]},{start[1]},{dir_map[dvec]}"

        # 2. Heuristic Scoring (if no immediate win)
        score = 0
        
        # Safety: Count enemies adjacent to my new position
        adj_opps = 0
        for dr, dc in dirs_vec:
            if (end[0] + dr, end[1] + dc) in new_opp_pieces:
                adj_opps += 1

        if adj_opps == 0:
            score += 5  # Safe square
        elif adj_opps == 1:
            score -= 15 # Dangerous (likely trade)
        else:
            score -= 30 # Very dangerous (likely loss of piece)

        score -= len(opp_moves_next) * 0.5

        if score > best_score:

            best_move = (start, dvec)

    # Fallback (should not be needed if legal moves exist, but required for safety)
    if best_move is None:
        start, end, dvec = legal_moves[0]
        return f"{start[0]},{start[1]},{dir_map[dvec]}"

    start, dvec = best_move
