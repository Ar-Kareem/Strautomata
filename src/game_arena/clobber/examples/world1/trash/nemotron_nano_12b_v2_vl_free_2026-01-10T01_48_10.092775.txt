The strategy is to iterate through all the AI's pieces in row-major order and check each possible orthogonal direction in a fixed priority (U, D, L, R). The first valid move found is returned, ensuring legality while favoring earlier check order. This approach balances simplicity and efficiency.

def policy(you, opponent):
    for r in range(5):
        for c in range(6):
            if you[r][c]:
                for dr, dc, dir in [(-1, 0, 'U'), (1, 0, 'D'), (0, -1, 'L'), (0, 1, 'R')]:
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < 5 and 0 <= nc < 6 and opponent[nr][nc]:
                        return f"{r},{c},{dir}"
    return "0,0,U"  # Fallback, though unreachable per problem constraints
