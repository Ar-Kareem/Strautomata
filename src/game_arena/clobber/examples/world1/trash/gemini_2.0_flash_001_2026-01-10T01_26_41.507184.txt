The strategy is to find a valid move and return it as quickly as possible. We will iterate through the board, and if we find a piece of ours, we will check the four possible directions and return the first valid move.
```python
import numpy as np

def policy(you: np.ndarray, opponent: np.ndarray) -> str:
    """
    Selects a legal move for the Clobber game.

    Args:
        you: A 5x6 numpy array representing the board state for the current player.
        opponent: A 5x6 numpy array representing the board state for the opponent.

    Returns:
        A string representing the chosen move in the format "row,col,dir".
    """
    rows, cols = you.shape

    for row in range(rows):
        for col in range(cols):
            if you[row, col] == 1:
                # Check up
                if row > 0 and opponent[row - 1, col] == 1:
                    return f"{row},{col},U"
                # Check down
                if row < rows - 1 and opponent[row + 1, col] == 1:
                    return f"{row},{col},D"
                # Check left
                if col > 0 and opponent[row, col - 1] == 1:
                    return f"{row},{col},L"
                # Check right
                if col < cols - 1 and opponent[row, col + 1] == 1:
                    return f"{row},{col},R"

    return "0,0,U"  # Should never happen, but return a default to avoid errors.
