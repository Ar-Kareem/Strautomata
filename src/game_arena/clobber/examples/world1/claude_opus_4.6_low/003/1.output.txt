Minimax with alpha-beta pruning, prioritizing moves that minimize opponent mobility.

import numpy as np

def policy(you, opponent):
    you = np.array(you)
    opponent = np.array(opponent)
    dirs = [('U', -1, 0), ('D', 1, 0), ('L', 0, -1), ('R', 0, 1)]
    
    def get_moves(mine, theirs):
        moves = []
        for r in range(5):
            for c in range(6):
                if mine[r][c]:
                    for name, dr, dc in dirs:
                        nr, nc = r+dr, c+dc
                        if 0 <= nr < 5 and 0 <= nc < 6 and theirs[nr][nc]:
                            moves.append((r, c, nr, nc, name))
        return moves
    
    def apply_move(mine, theirs, m):
        r, c, nr, nc, _ = m
        mine2 = mine.copy()
        theirs2 = theirs.copy()
        mine2[r][c] = 0
        mine2[nr][nc] = 1
        theirs2[nr][nc] = 0
        return mine2, theirs2
    
    def minimax(mine, theirs, depth, alpha, beta, maximizing):
        moves = get_moves(mine, theirs)
        if not moves:
            return -1000 if maximizing else 1000, None
        if depth == 0:
            opp_moves = len(get_moves(theirs, mine))
            return len(moves) - opp_moves, None
        if maximizing:
            best = -10000
            best_m = moves[0]
            for m in moves:
                new_mine, new_theirs = apply_move(mine, theirs, m)
                val, _ = minimax(new_theirs, new_mine, depth-1, alpha, beta, False)
                if val > best:
                    best = val
                    best_m = m
                alpha = max(alpha, val)
                if alpha >= beta:
                    break
            return best, best_m
        else:
            best = 10000
            best_m = moves[0]
            for m in moves:
                new_mine, new_theirs = apply_move(mine, theirs, m)
                val, _ = minimax(new_theirs, new_mine, depth-1, alpha, beta, True)
                if val < best:
                    best = val
                    best_m = m
                beta = min(beta, val)
                if alpha >= beta:
                    break
            return best, best_m
    
    moves = get_moves(you, opponent)
    if not moves:
        return "0,0,U"
    
    # Adaptive depth
    n_moves = len(moves)
    if n_moves <= 5:
        depth = 6
    elif n_moves <= 10:
        depth = 4
    else:
        depth = 3
    
    _, best_m = minimax(you, opponent, depth, -10000, 10000, True)
    if best_m is None:
        best_m = moves[0]
    return f"{best_m[0]},{best_m[1]},{best_m[4]}"
