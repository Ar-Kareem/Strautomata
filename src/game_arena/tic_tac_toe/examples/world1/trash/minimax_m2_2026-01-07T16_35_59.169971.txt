import sys
from functools import lru_cache

sys.setrecursionlimit(10000)

@lru_cache(maxsize=None)
def _minimax(state, current_player):
    lines = []
    for i in range(3):
        lines.append(state[i])
    for j in range(3):
        lines.append((state[0][j], state[1][j], state[2][j]))
    lines.append((state[0][0], state[1][1], state[2][2]))
    lines.append((state[0][2], state[1][1], state[2][0]))
    
    for line in lines:
        if line == (1, 1, 1):
            return 1
        if line == (-1, -1, -1):
            return -1
            
    if all(cell != 0 for row in state for cell in row):
        return 0
        
    empties = []
    for i in range(3):
        for j in range(3):
            if state[i][j] == 0:
                empties.append((i, j))
                
    if current_player == 1:
        best = -2
        for (i, j) in empties:
            new_row = list(state[i])
            new_row[j] = 1
            new_state = list(state)
            new_state[i] = tuple(new_row)
            new_state = tuple(new_state)
            value = _minimax(new_state, -1)
            if value > best:
                best = value
        return best
    else:
        best = 2
        for (i, j) in empties:
            new_row = list(state[i])
            new_row[j] = -1
            new_state = list(state)
            new_state[i] = tuple(new_row)
            new_state = tuple(new_state)
            value = _minimax(new_state, 1)
            if value < best:
                best = value
        return best

def policy(board: list[list[int]]) -> tuple[int, int]:
    board_state = tuple(tuple(row) for row in board)
    
    count1 = 0
    count_neg = 0
    for i in range(3):
        for j in range(3):
            if board[i][j] == 1:
                count1 += 1
            elif board[i][j] == -1:
                count_neg += 1
                
    if count1 != count_neg:
        for i in range(3):
            for j in range(3):
                if board[i][j] == 0:
                    return (i, j)
        return (0, 0)
        
    best_move = None
    best_value = -2
    
    empties = []
    for i in range(3):
        for j in range(3):
            if board_state[i][j] == 0:
                empties.append((i, j))
                
    for move in empties:
        i, j = move
        new_row = list(board_state[i])
        new_row[j] = 1
        new_state = list(board_state)
        new_state[i] = tuple(new_row)
        new_state = tuple(new_state)
        
        value = _minimax(new_state, -1)
        
        if value > best_value:
            best_value = value
            best_move = move
            
    if best_move is None:
        for i in range(3):
            for j in range(3):
                if board[i][j] == 0:
                    return (i, j)
        return (0, 0)
        
    return best_move
