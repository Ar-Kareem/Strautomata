def _check_win_or_block(board: list[list[int]], player_to_check: int) -> tuple[int, int] | None:
    """
    Helper function to find a winning or blocking move.

    It checks all rows, columns, and diagonals for a line where 'player_to_check'
    has two marks and one cell is empty.

    Args:
        board: The current 3x3 game board.
        player_to_check: The player (1 or -1) for whom to find a winning/blocking move.

    Returns:
        A tuple (row, col) of the empty cell if a win/block move is found,
        otherwise None.
    """

        # Rows
        [(0, 0), (0, 1), (0, 2)],
        [(1, 0), (1, 1), (1, 2)],
        [(2, 0), (2, 1), (2, 2)],
        # Columns
        [(0, 0), (1, 0), (2, 0)],
        [(0, 1), (1, 1), (2, 1)],
        [(0, 2), (1, 2), (2, 2)],
        # Diagonals
        [(0, 0), (1, 1), (2, 2)],
        [(0, 2), (1, 1), (2, 0)],
    ]

    for line in lines:
        player_count = 0
        empty_count = 0
        empty_cell_coords = None

        for r, c in line:
            if board[r][c] == player_to_check:
                player_count += 1
            elif board[r][c] == 0:
                empty_count += 1
                empty_cell_coords = (r, c)

        # this is the move that either wins or blocks.
        if player_count == 2 and empty_count == 1:
            return empty_cell_coords

    Determines the next move for a 3x3 Tic Tac Toe game based on a heuristic strategy.

    The strategy prioritizes moves in the following order:
    1.  **Win:** If there's a move that wins the game for the current player (1), take it.
    2.  **Block:** If the opponent (-1) can win on their next turn, block that move.
    3.  **Center:** If the center square (1,1) is empty, take it as it's strategically valuable.
    4.  **Opposite Corner:** If the opponent has occupied a corner and the opposite corner is free, take the opposite corner. This is a strong tactic.
    5.  **Empty Corner:** If no win, block, center, or opposite corner move is available, take any empty corner.
    6.  **Empty Side:** If none of the above applies, take any empty side (edge) square.

    Args:
        board: A 3x3 list of lists representing the game board.
               Each element can be:
               0: an empty cell
               1: the current player's mark
               -1: the opponent's mark

    Returns:
        A tuple of two integers (row, col) indicating the chosen move, where
        row and col are integers from 0 to 2 inclusive.
    """

    # 1. Check if the current player (1) can win in this turn
    win_move = _check_win_or_block(board, 1)
    if win_move:
        return win_move

    # 2. Check if the opponent (-1) can win in the next turn and block them
    block_move = _check_win_or_block(board, -1)
    if block_move:
        return block_move

    # 3. Take the center if it's empty (strategically advantageous)
    if board[1][1] == 0:
        return (1, 1)

    # 4. Strategic "opposite corner" play
    # This heuristic exploits opponent's corner placement. If opponent plays a corner,
    # taking the opposite corner (if free) sets up potential forks or wins.
    if board[0][0] == -1 and board[2][2] == 0: return (2, 2)
    if board[2][2] == -1 and board[0][0] == 0: return (0, 0)
    if board[0][2] == -1 and board[2][0] == 0: return (2, 0)
    if board[2][0] == -1 and board[0][2] == 0: return (0, 2)

    # 5. Take any empty corner
    # If specific opposite corner moves were not applicable, prioritize any available corner.
    corners = [(0, 0), (0, 2), (2, 0), (2, 2)]
    for r, c in corners:
        if board[r][c] == 0:

    # take the next available side square.

                return (r, c)

    # If no empty cell is found, it implies the board is full and no further move can be made.
    # This return value is a safeguard and indicates an unexpected state or a completed game.
    return (-1, -1)
