Here's the planned strategy:
1.  **Check for an immediate win:** If the AI can win in the next move, take it.
2.  **Block opponent's win:** If the opponent can win in the next move, block it.
3.  **Prioritize center:** If the center is open, take it.
4.  **Prioritize corners:** If corners are open, take one.
5.  **Take any open side:** Otherwise, take any available side.

Let's refine this with Minimax. Minimax will inherently handle winning and blocking.

**Minimax Algorithm:**
-   `minimax(board, depth, is_maximizing_player)`:
    -   Base cases:
        -   If game is won by AI (1), return a high score (e.g., 10 - depth).
        -   If game is won by Opponent (-1), return a low score (e.g., -10 + depth).
        -   If game is a draw, return 0.
    -   If `is_maximizing_player` (AI's turn):
        -   Initialize `max_eval = -infinity`.
        -   For each empty cell:
            -   Make the move.
            -   `eval = minimax(new_board, depth + 1, False)`
            -   `max_eval = max(max_eval, eval)`
            -   Undo the move.
        -   Return `max_eval`.
    -   If `is_maximizing_player` (Opponent's turn):
        -   Initialize `min_eval = +infinity`.
        -   For each empty cell:
            -   Make the move.
            -   `eval = minimax(new_board, depth + 1, True)`
            -   `min_eval = min(min_eval, eval)`
            -   Undo the move.
        -   Return `min_eval`.

The `policy` function will iterate through all possible moves, call `minimax` for each, and pick the move that leads to the best score.

import math

def check_win(board):
    # Check rows
    for row in board:
        if all(cell == 1 for cell in row):
            return 1
        if all(cell == -1 for cell in row):
            return -1

    # Check columns
    for col in range(3):
        if all(board[row][col] == 1 for row in range(3)):
            return 1
        if all(board[row][col] == -1 for row in range(3)):
            return -1

    # Check diagonals
    if (board[0][0] == 1 and board[1][1] == 1 and board[2][2] == 1) or \
       (board[0][2] == 1 and board[1][1] == 1 and board[2][0] == 1):
        return 1
    if (board[0][0] == -1 and board[1][1] == -1 and board[2][2] == -1) or \
       (board[0][2] == -1 and board[1][1] == -1 and board[2][0] == -1):
        return -1

    return 0 # No winner yet

def is_board_full(board):
    for r in range(3):
        for c in range(3):
            if board[r][c] == 0:
                return False
    return True

def minimax(current_board, depth, is_maximizing_player):
    winner = check_win(current_board)
    if winner == 1:
        return 1 # AI wins
    if winner == -1:
        return -1 # Opponent wins
    if is_board_full(current_board):
        return 0 # Draw

    if is_maximizing_player: # AI's turn
        best_score = -math.inf
        for r in range(3):
            for c in range(3):
                if current_board[r][c] == 0:
                    current_board[r][c] = 1 # Make move for AI
                    score = minimax(current_board, depth + 1, False)
                    current_board[r][c] = 0 # Undo move
                    best_score = max(best_score, score)
        return best_score
    else: # Opponent's turn
        best_score = math.inf
        for r in range(3):
            for c in range(3):
                if current_board[r][c] == 0:
                    current_board[r][c] = -1 # Make move for Opponent
                    score = minimax(current_board, depth + 1, True)
                    current_board[r][c] = 0 # Undo move
                    best_score = min(best_score, score)
        return best_score


def policy(board: list[list[int]]) -> tuple[int, int]:
    best_move = (-1, -1)
    best_score = -math.inf

    # Iterate through all possible moves
    for r in range(3):
        for c in range(3):
            if board[r][c] == 0: # If cell is empty
                # Try this move
                board[r][c] = 1
                # Calculate the score for this move using minimax (assuming opponent plays optimally after this)
                score = minimax(board, 0, False) # False because next turn is opponent's (minimizing player)
                # Undo the move to restore the board state for other evaluations
                board[r][c] = 0

                if score > best_score:
                    best_score = score
                    best_move = (r, c)
    
    # If no move found, it means the board is full or something went wrong.
    # This shouldn't happen in a valid game state where it's your turn.
    # But as a fallback, return the first empty cell found.
    if best_move == (-1, -1):
        for r in range(3):
            for c in range(3):
                if board[r][c] == 0:
                    return (r, c)

    return best_move

import math

# --- Helper Functions ---

def check_win(board: list[list[int]]) -> int:
    """
    Checks if a player has won the game.
    Returns 1 if player 1 wins, -1 if player -1 wins, 0 otherwise.
    """
    # Check rows
    for row in board:
        if all(cell == 1 for cell in row):
            return 1
        if all(cell == -1 for cell in row):
            return -1

    # Check columns
    for col in range(3):
        if all(board[row][col] == 1 for row in range(3)):
            return 1
        if all(board[row][col] == -1 for row in range(3)):
            return -1

    # Check diagonals
    if (board[0][0] == 1 and board[1][1] == 1 and board[2][2] == 1) or \
       (board[0][2] == 1 and board[1][1] == 1 and board[2][0] == 1):
        return 1
    if (board[0][0] == -1 and board[1][1] == -1 and board[2][2] == -1) or \
       (board[0][2] == -1 and board[1][1] == -1 and board[2][0] == -1):
        return -1

    return 0 # No winner yet

def is_board_full(board: list[list[int]]) -> bool:
    """
    Checks if the board is completely filled.
    """
    for r in range(3):
        for c in range(3):
            if board[r][c] == 0:
                return False
    return True

def minimax_alpha_beta(current_board: list[list[int]], depth: int, is_maximizing_player: bool, alpha: float, beta: float) -> int:
    """
    Implements the Minimax algorithm with Alpha-Beta Pruning.
    Returns the best score achievable from the current_board state.
    """
    winner = check_win(current_board)
    if winner != 0:
        return winner # Game won by 1 or -1
    if is_board_full(current_board):
        return 0 # Draw

    if is_maximizing_player: # AI's turn (player 1)
        best_score = -math.inf
        for r in range(3):
            for c in range(3):
                if current_board[r][c] == 0:
                    current_board[r][c] = 1 # Make the move
                    score = minimax_alpha_beta(current_board, depth + 1, False, alpha, beta)
                    current_board[r][c] = 0 # Undo the move
                    best_score = max(best_score, score)
                    alpha = max(alpha, best_score)
                    if beta <= alpha:
                        break # Beta cut-off
            if beta <= alpha:
                break
        return best_score
    else: # Opponent's turn (player -1)
        best_score = math.inf
        for r in range(3):
            for c in range(3):
                if current_board[r][c] == 0:
                    current_board[r][c] = -1 # Make the move
                    score = minimax_alpha_beta(current_board, depth + 1, True, alpha, beta)
                    current_board[r][c] = 0 # Undo the move
                    best_score = min(best_score, score)
                    beta = min(beta, best_score)
                    if beta <= alpha:
                        break # Alpha cut-off
            if beta <= alpha:
                break
        return best_score

# --- Policy Function ---

def policy(board: list[list[int]]) -> tuple[int, int]:
    """
    Determines the optimal next move for the AI (player 1) in a Tic Tac Toe game.
    Uses the Minimax algorithm with Alpha-Beta Pruning to find the best move.

    Args:
        board: A 3x3 list of lists representing the Tic Tac Toe board.
               0: empty, 1: AI (you), -1: Opponent.

    Returns:
        A tuple (row, col) indicating the chosen move.
    """
    best_move = (-1, -1)
    best_score = -math.inf

    # Iterate through all possible moves (empty cells)
    for r in range(3):
        for c in range(3):
            if board[r][c] == 0: # If the cell is empty
                # Try making this move
                board[r][c] = 1 
                # Evaluate this move using minimax_alpha_beta. 
                # The next turn will be the opponent's (minimizing player).
                score = minimax_alpha_beta(board, 0, False, -math.inf, math.inf)
                # Undo the move to restore the board state for evaluating other moves
                board[r][c] = 0

                # If this move leads to a better score, update best_score and best_move
                if score > best_score:
                    best_score = score
                    best_move = (r, c)
                
                # If we've found a winning move, we can return it immediately
                # as there's no need to search for a "better" winning move.
                if best_score == 1:
                    return best_move

    # This part handles a theoretical edge case where best_move is not updated
    # (e.g., if the board is full from the start, though policy shouldn't be called then,
    # or if no move makes sense, which is not true for Tic-Tac-Toe).
    # For a valid game state where AI has a turn, best_move will always be updated.
    # However, as a failsafe, return the first empty cell if no "best" move was found.
    # (This case should ideally never be hit in a standard game since Minimax always finds a move)
    if best_move == (-1, -1):
         for r in range(3):
             for c in range(3):
                 if board[r][c] == 0:
                     return (r, c) # Fallback: return the first empty spot

    return best_move
