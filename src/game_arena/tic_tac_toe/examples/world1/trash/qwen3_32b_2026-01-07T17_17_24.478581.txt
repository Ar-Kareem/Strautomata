def check_win(board, player):
    # Check rows
    for row in board:
        if all(cell == player for cell in row):
            return True
    # Check columns
    for col in range(3):
        if all(board[row][col] == player for row in range(3)):
            return True
    # Check diagonals
    if all(board[i][i] == player for i in range(3)):
        return True
    if all(board[i][2 - i] == player for i in range(3)):
        return True
    return False

def policy(board: list[list[int]]) -> tuple[int, int]:
    # Check for immediate win
    for i in range(3):
        for j in range(3):
            if board[i][j] == 0:
                board_copy = [row[:] for row in board]
                board_copy[i][j] = 1
                if check_win(board_copy, 1):
                    return (i, j)
    
    # Check for opponent win and block
    for i in range(3):
        for j in range(3):
            if board[i][j] == 0:
                board_copy = [row[:] for row in board]
                board_copy[i][j] = -1
                if check_win(board_copy, -1):
                    return (i, j)
    
    # Center priority
    if board[1][1] == 0:
        return (1, 1)
    
    # Check corners
    corners = [(0, 0), (0, 2), (2, 0), (2, 2)]
    for (i, j) in corners:
        if board[i][j] == 0:
            return (i, j)
    
    # Check edges
    edges = [(0, 1), (1, 0), (1, 2), (2, 1)]
    for (i, j) in edges:
        if board[i][j] == 0:
            return (i, j)
    
    # Fallback (should not be reached in game logic)
    return (0, 0)
