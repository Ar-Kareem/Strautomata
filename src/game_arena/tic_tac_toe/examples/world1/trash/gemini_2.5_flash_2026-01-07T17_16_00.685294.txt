def policy(board: list[list[int]]) -> tuple[int, int]:
    """
    Determines the next optimal move for a Tic Tac Toe game.

    The strategy is as follows:
    1.  Check if an immediate winning move is available for the AI (player 1).
    2.  If not, check if the opponent (player -1) has an immediate winning move
        and block it.
    3.  If no immediate wins or blocks are needed, take the center position (1,1) if available.
    4.  If the center is not available, take any available corner position.
    5.  If no corners are available, take any available side position.
    6.  As a fallback, return the first empty cell found (should rarely be reached).

    Args:
        board: A 3x3 list of lists representing the Tic Tac Toe board.
               0: empty, 1: AI (you), -1: Opponent.

    Returns:
        A tuple (row, col) representing the chosen move.
    """

    my_player = 1
    opponent_player = -1

    def get_winner(current_board: list[list[int]]) -> int | None:
        """
        Checks the given board for a winner.
        Returns 1 if my_player wins, -1 if opponent_player wins, otherwise None.
        """
        # Check rows
        for row in current_board:
            if all(cell == my_player for cell in row):
                return my_player
            if all(cell == opponent_player for cell in row):
                return opponent_player

        # Check columns
        for col in range(3):
            if all(current_board[row][col] == my_player for row in range(3)):
                return my_player
            if all(current_board[row][col] == opponent_player for row in range(3)):
                return opponent_player

        # Check diagonals
        if (current_board[0][0] == my_player and current_board[1][1] == my_player and current_board[2][2] == my_player) or \
           (current_board[0][2] == my_player and current_board[1][1] == my_player and current_board[2][0] == my_player):
            return my_player
        if (current_board[0][0] == opponent_player and current_board[1][1] == opponent_player and current_board[2][2] == opponent_player) or \
           (current_board[0][2] == opponent_player and current_board[1][1] == opponent_player and current_board[2][0] == opponent_player):
            return opponent_player

        return None  # No winner yet

    # 1. Check for immediate win
    for r in range(3):
        for c in range(3):
            if board[r][c] == 0:
                board[r][c] = my_player  # Simulate placing my piece
                if get_winner(board) == my_player:
                    board[r][c] = 0  # Undo the simulation
                    return (r, c)
                board[r][c] = 0  # Undo the simulation

    # 2. Check for immediate block
    for r in range(3):
        for c in range(3):
            if board[r][c] == 0:
                board[r][c] = opponent_player  # Simulate placing opponent's piece
                if get_winner(board) == opponent_player:
                    board[r][c] = 0  # Undo the simulation
                    return (r, c)  # Block the opponent's win
                board[r][c] = 0  # Undo the simulation

    # 3. Take the center if available
    if board[1][1] == 0:
        return (1, 1)

    # 4. Take an available corner
    corners = [(0, 0), (0, 2), (2, 0), (2, 2)]
    for r, c in corners:
        if board[r][c] == 0:
            return (r, c)

    # 5. Take an available side position
    sides = [(0, 1), (1, 0), (1, 2), (2, 1)]
    for r, c in sides:
        if board[r][c] == 0:
            return (r, c)

    # Fallback: This case should only be reached if the board is full and it's somehow
    # still the AI's turn without a winner, which is an invalid game state from a game engine perspective.
    # However, to guarantee a return, we provide a default (e.g., first empty spot).
    # Since all above steps prioritize moves, this implies no empty spots.
    # For safety (though unlikely to be hit in a valid game flow), we return (0,0)
    # if somehow no empty cells were found and no move was selected earlier.
    for r in range(3):
        for c in range(3):
            if board[r][c] == 0:
                return (r,c)
    
    return (0,0) # Should be unreachable in a logically sound game.
