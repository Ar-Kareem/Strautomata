def policy(board: list[list[int]]) -> tuple[int, int]:
    # Helper to check if a player has won on the CURRENT board state
    def check_win_state(current_board, player):
        # Check rows
        for r in range(3):
            if all(current_board[r][c] == player for c in range(3)):
                return True
        # Check columns
        for c in range(3):
            if all(current_board[r][c] == player for r in range(3)):
                return True
        # Check diagonals
        if all(current_board[i][i] == player for i in range(3)):
            return True
        if all(current_board[i][2-i] == player for i in range(3)):
            return True
        return False

    # --- Step 1: Check for winning move ---
    for r in range(3):
        for c in range(3):
            if board[r][c] == 0:
                # Try placing our mark (1)
                board[r][c] = 1
                if check_win_state(board, 1):
                    board[r][c] = 0  # Undo change for next potential checks
                    return (r, c)
                board[r][c] = 0  # Undo change

    # --- Step 2: Check for blocking move ---
    for r in range(3):
        for c in range(3):
            if board[r][c] == 0:
                # Try placing opponent's mark (-1)
                board[r][c] = -1
                if check_win_state(board, -1):
                    board[r][c] = 0  # Undo change
                    return (r, c)
                board[r][c] = 0  # Undo change

    # --- Step 3: Take the center ---
    if board[1][1] == 0:
        return (1, 1)

    # --- Step 4: Prioritize corners ---
    corners = [(0, 0), (0, 2), (2, 0), (2, 2)]
    for r, c in corners:
        if board[r][c] == 0:
            return (r, c)

    # --- Step 5: Prioritize sides ---
    sides = [(0, 1), (1, 0), (1, 2), (2, 1)]
    for r, c in sides:
        if board[r][c] == 0:
            return (r, c)

    # This case should ideally not be reached if the game is still ongoing.
    # If somehow all cells are filled and no winner, it implies a draw.
    # However, the problem asks for a move, so we might need a fallback.
    # In any valid game state, there should be at least one empty cell if no winner.
    # If we reach here, it means the board is full or some edge case.
    # For robustness, we can iterate and return the first available 0, though this shouldn't happen in a typical game flow.
    for r in range(3):
        for c in range(3):
            if board[r][c] == 0:
                return (r, c)

    # Should not be reached in a valid ongoing game
    return (-1, -1) # Placeholder, should not happen
