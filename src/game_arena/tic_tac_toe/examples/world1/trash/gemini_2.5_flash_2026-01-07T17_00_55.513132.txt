def minimax(board: list[list[int]], depth: int, is_maximizing_player: bool, alpha: float, beta: float) -> int:
    """
    Implements the minimax algorithm with alpha-beta pruning to evaluate board states.
    Args:
        board: The current 3x3 Tic Tac Toe board.
        depth: The current depth in the game tree (not strictly used for scoring in TicTacToe, but for recursion).
        is_maximizing_player: True if the current call is for the maximizing player (us, value 1), 
                              False if for the minimizing player (opponent, value -1).
        alpha: The alpha value for alpha-beta pruning.
        beta: The beta value for alpha-beta pruning.
    Returns:
        The evaluated score of the board state from the perspective of the maximizing player.
    """
    winner = check_winner(board)

    # Base cases: If game is over, return the score
    if winner is not None:
        return winner # 1 for our win, -1 for opponent win, 0 for draw

    if is_maximizing_player: # Our turn, we want to maximize our score
        best_score = -math.inf
        for r in range(3):
            for c in range(3):
                if board[r][c] == 0:
                    board[r][c] = 1 # Make the move for player 1
                    score = minimax(board, depth + 1, False, alpha, beta) # Recurse for opponent
                    board[r][c] = 0 # Undo the move
                    best_score = max(best_score, score)
                    alpha = max(alpha, best_score)
                    if beta <= alpha: # Alpha-beta pruning
                        break # Prune this branch
            if beta <= alpha: # Break outer loop too for row iteration
                break
        return best_score
    else: # Opponent's turn, they want to minimize our score
        best_score = math.inf
        for r in range(3):
            for c in range(3):
                if board[r][c] == 0:
                    board[r][c] = -1 # Make the move for player -1
                    score = minimax(board, depth + 1, True, alpha, beta) # Recurse for us
                    board[r][c] = 0 # Undo the move
                    best_score = min(best_score, score)
                    beta = min(beta, best_score)
                    if beta <= alpha: # Beta-alpha pruning
                        break # Prune this branch
            if beta <= alpha: # Break outer loop too for row iteration
                break
        return best_score

# Main policy function for the AI
def policy(board: list[list[int]]) -> tuple[int, int]:
    """
    Determines the best move for the AI (player 1) in a Tic Tac Toe game.
    Args:
        board: The current 3x3 Tic Tac Toe board.
    Returns:
        A tuple (row, col) indicating the chosen move.
    """
    best_score = -math.inf
    best_move = None

    empty_cells = []
    for r in range(3):
        for c in range(3):
            if board[r][c] == 0:
                empty_cells.append((r, c))

    # If the board is full, return a dummy move (this scenario should not happen in actual game flow)
    if not empty_cells:
        return (0, 0)
    
    # --- Priority 1: Check for an immediate winning move ---
    # If we can win in the next move, take it.
    for r, c in empty_cells:
        board[r][c] = 1 # Try placing our piece
        if check_winner(board) == 1:
            board[r][c] = 0 # Undo the move
            return (r, c)
        board[r][c] = 0 # Undo the move

    # --- Priority 2: Check for an immediate blocking move ---
    # If the opponent can win in their next move, block them.
    for r, c in empty_cells:
        board[r][c] = -1 # Try placing opponent's piece
        if check_winner(board) == -1:
            board[r][c] = 0 # Undo the move
            return (r, c) # Block the opponent by playing our piece here
        board[r][c] = 0 # Undo the move

    # --- Priority 3: Use Minimax to find the optimal move for other scenarios ---
    # If no immediate win or block, use minimax to find the best move.
    # The minimax function evaluates moves assuming optimal play from both sides.
    # We iterate through possible moves for 'us' and evaluate the resulting board state
    # by calling minimax for the opponent.
    for r, c in empty_cells:
        board[r][c] = 1 # Make a potential move for 'us'
        # Call minimax for the opponent (first call, depth 0, is_maximizing_player=False)
        # This evaluates the board assuming optimal play from the opponent immediately after our move.
        score = minimax(board, 0, False, -math.inf, math.inf)
        board[r][c] = 0 # Undo the move to restore the board state for the next iteration

        if score > best_score:
            best_score = score
            best_move = (r, c)
        # In case of ties (multiple moves lead to the same optimal minimax score, e.g., a guaranteed draw),
        # the first one encountered will be chosen. This is perfectly acceptable for optimal play.

    # If best_move is still None (e.g., if empty_cells was empty, though checked earlier),
    # or if some very unusual game state, return a safe default.
    # However, given empty_cells is checked, best_move will always be assigned.
    return best_move
