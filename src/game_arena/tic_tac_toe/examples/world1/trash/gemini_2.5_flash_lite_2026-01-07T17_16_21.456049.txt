import random # Not strictly needed for this deterministic strategy, but good to have if randomness was introduced.

def policy(board: list[list[int]]) -> tuple[int, int]:
    """
    Determines the next move for a 3x3 Tic Tac Toe game AI player (1).

    Args:
        board: A 3x3 list of lists representing the game board.
               0: empty, 1: AI player, -1: opponent player.

    Returns:
        A tuple (row, col) of the chosen move (0-2, 0-2).
    """
    EMPTY = 0
    PLAYER_ME = 1
    PLAYER_OPPONENT = -1

    def get_empty_cells(current_board):
        """Returns a list of (row, col) for empty cells."""
        cells = []
        for r in range(3):
            for c in range(3):
                if current_board[r][c] == EMPTY:
                    cells.append((r, c))
        return cells

    def check_win(current_board, player):
        """Checks if the given player has won on this board state."""
        # Check rows
        for r in range(3):
            if all(current_board[r][c] == player for c in range(3)):
                return True
        # Check columns
        for c in range(3):
            if all(current_board[r][c] == player for r in range(3)):
                return True
        # Check diagonals
        if all(current_board[i][i] == player for i in range(3)):
            return True
        if all(current_board[i][2 - i] == player for i in range(3)):
            return True
        return False

    empty_cells = get_empty_cells(board)

    # Strategy 1: Win
    # If a move exists that makes the AI win, take it.
    for r, c in empty_cells:
        # Make a hypothetical move
        board[r][c] = PLAYER_ME
        if check_win(board, PLAYER_ME):
            board[r][c] = EMPTY  # Undo hypothetical move
            return (r, c)
        board[r][c] = EMPTY  # Undo hypothetical move

    # Strategy 2: Block opponent
    # If the opponent can win on their next move, block it.
    for r, c in empty_cells:
        # Make a hypothetical move for opponent
        board[r][c] = PLAYER_OPPONENT
        if check_win(board, PLAYER_OPPONENT):
            board[r][c] = EMPTY  # Undo hypothetical move
            return (r, c)
        board[r][c] = EMPTY  # Undo hypothetical move

    # Strategy 3: Take the center
    # The center square is strategically important.
    if board[1][1] == EMPTY:
        return (1, 1)

    # Strategy 4: Take an opposite corner
    # If the opponent occupies a corner, take the opposite corner.
    corners = [(0, 0), (0, 2), (2, 0), (2, 2)]
    opposite_corner_map = {
        (0, 0): (2, 2), (2, 2): (0, 0),
        (0, 2): (2, 0), (2, 0): (0, 2)
    }
    for r_opp, c_opp in corners:
        if board[r_opp][c_opp] == PLAYER_OPPONENT:  # If opponent occupies this corner
            r_mine, c_mine = opposite_corner_map[(r_opp, c_opp)] # Get the opposite corner
            if board[r_mine][c_mine] == EMPTY:  # If the opposite corner is free
                return (r_mine, c_mine)

    # Strategy 5: Take an empty corner
    # If no winning or blocking moves, take any available corner.
    for r, c in corners:
        if board[r][c] == EMPTY:
            return (r, c)

    # Strategy 6: Take an empty side
    # If no corners are available, take any available side (edge middle).
    sides = [(0, 1), (1, 0), (1, 2), (2, 1)]
