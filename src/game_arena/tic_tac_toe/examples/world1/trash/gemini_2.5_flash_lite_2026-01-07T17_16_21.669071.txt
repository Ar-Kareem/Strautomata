```python
import typing

def policy(board: list[list[int]]) -> tuple[int, int]:
    """
    Determines the next optimal move for a 3x3 Tic Tac Toe game.

    The strategy follows the standard priority:
    1. Win: If a move can win the game, take it.
    2. Block: If the opponent can win in their next move, block them.
    3. Center: Take the center square if available.
    4. Corners: Take an available corner square.
    5. Edges: Take an available edge square.

    Args:
        board: A 3x3 list of lists representing the Tic Tac Toe board.
               0: Empty, 1: Your symbol, -1: Opponent's symbol.

    Returns:
        A tuple (row, col) indicating the chosen move (0-indexed).
        Returns (-1, -1) if the board is full (no moves available).
    """
    lines = [
        [(0, 0), (0, 1), (0, 2)],  # Row 0
        [(1, 0), (1, 1), (1, 2)],  # Row 1
        [(2, 0), (2, 1), (2, 2)],  # Row 2
        [(0, 0), (1, 0), (2, 0)],  # Col 0
        [(0, 1), (1, 1), (2, 1)],  # Col 1
        [(0, 2), (1, 2), (2, 2)],  # Col 2
        [(0, 0), (1, 1), (2, 2)],  # Main Diagonal
        [(0, 2), (1, 1), (2, 0)]   # Anti-Diagonal
    ]

    # 1. Check for a winning move for '1'
    for line in lines:
        my_count = 0
        empty_cell = None
        for r, c in line:
            if board[r][c] == 1:
                my_count += 1
            elif board[r][c] == 0:
                empty_cell = (r, c)
        # If two of my symbols are in a line and the third is empty, that's a win
        if my_count == 2 and empty_cell is not None:
            return empty_cell

    # 2. Check for a move to block the opponent ('-1')
    for line in lines:
        opponent_count = 0
        empty_cell = None
        for r, c in line:
            if board[r][c] == -1:
                opponent_count += 1
            elif board[r][c] == 0:
                empty_cell = (r, c)
        # If two of the opponent's symbols are in a line and the third is empty, block it
        if opponent_count == 2 and empty_cell is not None:
            return empty_cell

    # 3. Take the center square if it's empty
    if board[1][1] == 0:
        return (1, 1)

    # 4. Take any available corner square
    corners = [(0, 0), (0, 2), (2, 0), (2, 2)]
    for r, c in corners:
        if board[r][c] == 0:
            return (r, c)

    # 5. Take any available edge square
    edges = [(0, 1), (1, 0), (1, 2), (2, 1)]
    for r, c in edges:
        if board[r][c] == 0:
            return (r, c)

    # Fallback: If no strategic move found (e.g., board is full or game is in an unexpected state)
    # Find any empty cell. This block should theoretically not be reached
    # unless the board is full.
    for r in range(3):
        for c in range(3):
            if board[r][c] == 0:
                return (r, c)

    # If the board is completely full and no empty cells were found
```
