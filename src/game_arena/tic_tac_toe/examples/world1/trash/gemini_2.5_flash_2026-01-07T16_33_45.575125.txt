def check_winner(board):
    # Check rows
    for row in board:
        if all(cell == 1 for cell in row):
            return 1
        if all(cell == -1 for cell in row):
            return -1

    # Check columns
    for col_idx in range(3):
        if all(board[row_idx][col_idx] == 1 for row_idx in range(3)):
            return 1
        if all(board[row_idx][col_idx] == -1 for row_idx in range(3)):
            return -1

    if (board[0][0] == board[1][1] == board[2][2] == 1) or \
       (board[0][2] == board[1][1] == board[2][0] == 1):
        return 1
    if (board[0][0] == board[1][1] == board[2][2] == -1) or \
       (board[0][2] == board[1][1] == board[2][0] == -1):
        return -1

    # No winner yet, check for draw
    if all(cell != 0 for row in board for cell in row):

import math

# Helper function to check the current status of the game board
def check_winner(board: list[list[int]]) -> int | None:
    """
    Checks if there's a winner or a draw on the 3x3 Tic-Tac-Toe board.

    Args:
        board: A 3x3 list of lists representing the game board.
               0: empty, 1: AI's piece, -1: opponent's piece.

    Returns:
        1: If player 1 (AI) wins.
        -1: If player -1 (opponent) wins.
        0: If it's a draw (no winner and no empty cells).
        None: If the game is still in progress.
    """
    # Check rows
    for row in board:
        if all(cell == 1 for cell in row):
            return 1
        if all(cell == -1 for cell in row):
            return -1

    # Check columns
    for col_idx in range(3):
        if all(board[row_idx][col_idx] == 1 for row_idx in range(3)):
            return 1
        if all(board[row_idx][col_idx] == -1 for row_idx in range(3)):
            return -1

    # Check diagonals
    # Main diagonal
    if (board[0][0] == board[1][1] == board[2][2] == 1) or \
       (board[0][0] == board[1][1] == board[2][2] == -1):
        return board[0][0]
    # Anti-diagonal
    if (board[0][2] == board[1][1] == board[2][0] == 1) or \
       (board[0][2] == board[1][1] == board[2][0] == -1):
        return board[0][2]

    # Check for draw (no empty cells left)
    if all(cell != 0 for row in board for cell in row):
        return 0  # Draw

    # Game is still in progress
    return None

# Minimax algorithm to find the best score for the current player
def minimax(board: list[list[int]], is_maximizing_player: bool) -> int:
    """
    Implements the Minimax algorithm to determine the best possible score
    from the current board state assuming optimal play from both sides.

    Args:
        board: The current 3x3 Tic-Tac-Toe board.
        is_maximizing_player: True if it's the maximizing player's (player 1) turn,
                              False if it's the minimizing player's (player -1) turn.

    Returns:
        The best score achievable from this state given optimal play:
        1 for a win by player 1, -1 for a win by player -1, 0 for a draw.
    """

    # Base case: if the game is over, return the score
    if winner_status is not None:
        return winner_status

    # Get all available empty cells
    empty_cells = []
    for r in range(3):
        for c in range(3):
            if board[r][c] == 0:
                empty_cells.append((r, c))

    if is_maximizing_player:  # Player 1's turn (our AI)
        best_score = -math.inf
        for r, c in empty_cells:
            board[r][c] = 1  # Make a hypothetical move for player 1
            score = minimax(board, False)  # Recurse for the opponent (minimizing player)
            board[r][c] = 0  # Undo the move (backtrack)
            best_score = max(best_score, score)
        return best_score
    else:  # Player -1's turn (opponent, minimizing player)
        best_score = math.inf
        for r, c in empty_cells:
            board[r][c] = -1  # Make a hypothetical move for player -1
            score = minimax(board, True)  # Recurse for player 1 (maximizing player)
            board[r][c] = 0  # Undo the move (backtrack)
            best_score = min(best_score, score)
        return best_score
