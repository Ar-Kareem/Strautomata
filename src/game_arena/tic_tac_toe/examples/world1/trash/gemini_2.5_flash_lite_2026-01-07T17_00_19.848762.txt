import typing # Not strictly necessary for this code, but good practice for type hints

def policy(board: list[list[int]]) -> tuple[int, int]:
    """
    Determines the next move for a 3x3 Tic Tac Toe game based on an optimal strategy.

    Args:
        board: A 3x3 list of lists representing the game board.
               0: Empty cell
               1: Player's mark (you)
              -1: Opponent's mark

    Returns:
        A tuple (row, column) representing the chosen move (0-2, 0-2).
        Returns (-1, -1) if no valid move can be made (e.g., board is full).
    """

    def get_winning_move(current_board: list[list[int]], player: int) -> typing.Optional[tuple[int, int]]:
        """
        Checks if the player can win on their next move and returns the winning move coordinates.
        Also checks if the opponent can win and returns the blocking move coordinates.
        """
        lines = [
            [(0, 0), (0, 1), (0, 2)], [(1, 0), (1, 1), (1, 2)], [(2, 0), (2, 1), (2, 2)],  # Rows
            [(0, 0), (1, 0), (2, 0)], [(0, 1), (1, 1), (2, 1)], [(0, 2), (1, 2), (2, 2)],  # Columns
            [(0, 0), (1, 1), (2, 2)], [(0, 2), (1, 1), (2, 0)]   # Diagonals
        ]

        for line in lines:
            player_count = 0
            empty_coord = None
            for r, c in line:
                if current_board[r][c] == player:
                    player_count += 1
                elif current_board[r][c] == 0:
                    empty_coord = (r, c)

            # If player has two marks and there's an empty spot, this is a winning move.
            if player_count == 2 and empty_coord:
                return empty_coord
        return None

    # --- Strategy Implementation ---

    # 1. Check if I (player 1) can win on the next move.
    win_move = get_winning_move(board, 1)
    if win_move:
        return win_move

    # 2. Check if the opponent (-1) can win on their next move, and block them.
    block_move = get_winning_move(board, -1)
    if block_move:
        return block_move

    # 3. Take the center if it's empty.
    if board[1][1] == 0:
        return (1, 1)

    # 4. Take an opposite corner if the opponent is in a corner and the opposite is free.
    corner_pairs = [((0, 0), (2, 2)), ((0, 2), (2, 0))]
    for (c1_r, c1_c), (c2_r, c2_c) in corner_pairs:
        if board[c1_r][c1_c] == -1 and board[c2_r][c2_c] == 0:
            return (c2_r, c2_c)
        if board[c2_r][c2_c] == -1 and board[c1_r][c1_c] == 0:
            return (c1_r, c1_c)

    # 5. Take any empty corner if available.
    corners = [(0, 0), (0, 2), (2, 0), (2, 2)]
    for r, c in corners:
        if board[r][c] == 0:
            return (r, c)

    # 6. Take any empty side if available.
    sides = [(0, 1), (1, 0), (1, 2), (2, 1)]
    for r, c in sides:
        if board[r][c] == 0:
            return (r, c)

    # Fallback: If somehow no strategic move was found but cells are empty.
    # This should ideally not be reached with the above logic unless the board is full.
    for r in range(3):
        for c in range(3):
            if board[r][c] == 0:
                return (r, c)

    # If the board is full and no move was possible.
    return (-1, -1)
