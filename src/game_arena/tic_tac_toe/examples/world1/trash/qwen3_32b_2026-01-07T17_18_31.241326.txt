To solve the problem of choosing the optimal move in a 3x3 Tic Tac Toe game, a **Minimax algorithm with memoization** is implemented. This algorithm explores all possible future moves and outcomes, assuming optimal play from both the AI and the opponent. The core idea is to:

1. **Check if the game is over (a player has won or it's a draw).**  
   If so, return a terminal score (`+1` for AI win, `-1` for opponent win, `0` for draw).
2. **For each possible next move, recursively simulate the game.**  
   The AI aims to *maximize* its chances, while the opponent *minimizes* the AI's score.
3. **Use memoization to cache already evaluated board states.**  
   This significantly speeds up the algorithm and avoids redundant computation.

def policy(board: list[list[int]]) -> tuple[int, int]:
    def check_winner(bd):
        # Check rows
        for row in bd:
            if row[0] != 0 and row[0] == row[1] == row[2]:
                return row[0]
        # Check columns
        for col in range(3):
            if bd[0][col] != 0 and bd[0][col] == bd[1][col] == bd[2][col]:
                return bd[0][col]
        # Check diagonals
        if bd[0][0] != 0 and bd[0][0] == bd[1][1] == bd[2][2]:
            return bd[0][0]
        if bd[0][2] != 0 and bd[0][2] == bd[1][1] == bd[2][0]:
            return bd[0][2]
        return 0

    def board_to_key(bd):
        return tuple(tuple(row) for row in bd)

    memo = {}

    def minimax(bd):
        key = board_to_key(bd)
        if key in memo:
            return memo[key]

        winner = check_winner(bd)
        if winner == 1:
            memo[key] = (None, 1)
            return (None, 1)
        elif winner == -1:
            memo[key] = (None, -1)
            return (None, -1)

        is_full = all(cell != 0 for row in bd for cell in row)
        if is_full:
            memo[key] = (None, 0)
            return (None, 0)

        count_ones = sum(row.count(1) for row in bd)
        count_minus_ones = sum(row.count(-1) for row in bd)
        is_ai_turn = (count_ones == count_minus_ones)

        if is_ai_turn:
            best_score = -float('inf')
            best_move = None
            for i in range(3):
                for j in range(3):
                    if bd[i][j] == 0:
                        new_bd = [row[:] for row in bd]
                        new_bd[i][j] = 1
                        _, score = minimax(new_bd)
                        if score > best_score:
                            best_score = score
                            best_move = (i, j)
            memo[key] = (best_move, best_score)
            return (best_move, best_score)
        else:
            best_score = float('inf')
            best_move = None
            for i in range(3):
                for j in range(3):
                    if bd[i][j] == 0:
                        new_bd = [row[:] for row in bd]
                        new_bd[i][j] = -1
                        _, score = minimax(new_bd)
                        if score < best_score:
                            best_score = score
                            best_move = (i, j)
            memo[key] = (best_move, best_score)
            return (best_move, best_score)

    best_move, _ = minimax(board)
    return best_move
