        row_values = board[r]
        if row_values.count(player_mark) == 2 and 0 in row_values:

            return (r, c)

        return (idx, idx)

            c = row_values.index(0)
            return (r, c)

    # Check columns
    for c in range(3):
        col_values = [board[r][c] for r in range(3)]
        if col_values.count(player_mark) == 2 and 0 in col_values:
            r = col_values.index(0)
            return (r, c)

    # Check diagonals
    # Top-left to bottom-right
    diag1 = [board[i][i] for i in range(3)]
    if diag1.count(player_mark) == 2 and 0 in diag1:
        idx = diag1.index(0)
        return (idx, idx)

    # Top-right to bottom-left
    diag2 = [board[i][2 - i] for i in range(3)]
    if diag2.count(player_mark) == 2 and 0 in diag2:
        idx = diag2.index(0)
        return (idx, 2 - idx)

    return None

    """
    Determines the next move for a 3x3 Tic Tac Toe game.

    The strategy prioritizes moves in the following order:
    1. Win the game if possible.
    2. Block the opponent from winning.
    3. Take the center square.
    4. If the opponent occupies a corner, take the opposite corner if it's free.
    5. Take any available corner.
    6. Take any available side (edge) square.

    Args:
        board: A 3x3 list of lists representing the game board.
               0 for empty, 1 for self, -1 for opponent.

    Returns:
        A tuple (row, col) representing the chosen move (0-2, 0-2).
    """

    # 1. Check if I can win on this turn.
    win_move = _find_opportunity(board, 1)
    if win_move:
        return win_move

    # 2. Check if the opponent can win on their next turn and block them.
    block_move = _find_opportunity(board, -1)
    if block_move:
        return block_move

    # 3. Take the center square if it's available.
    if board[1][1] == 0:
        return (1, 1)

    # 4. If opponent occupies a corner, take the opposite corner if it's free.
    # This is a heuristic to counter a common attacking pattern and create opportunities.
    corners = [(0, 0), (0, 2), (2, 0), (2, 2)]
    opp_corner_map = {
        (0, 0): (2, 2), (2, 2): (0, 0),
        (0, 2): (2, 0), (2, 0): (0, 2)
    }

    for r_corner, c_corner in corners:
        if board[r_corner][c_corner] == -1:  # Opponent occupies this corner
            r_opposite, c_opposite = opp_corner_map[(r_corner, c_corner)]
            if board[r_opposite][c_opposite] == 0:  # Opposite corner is free
                return (r_opposite, c_opposite)

    # 5. Take any available corner.
    # This is done after checking for win, block, center, and specific opposite corner plays.
    for r, c in corners:
        if board[r][c] == 0:
            return (r, c)

    # 6. Take any available side (edge) square.
    sides = [(0, 1), (1, 0), (1, 2), (2, 1)]
    for r, c in sides:
        if board[r][c] == 0:
            return (r, c)
