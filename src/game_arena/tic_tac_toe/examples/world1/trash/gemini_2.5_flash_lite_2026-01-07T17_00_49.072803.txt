def policy(board: list[list[int]]) -> tuple[int, int]:
    """
    Determines the next move for a 3x3 Tic Tac Toe game based on a greedy
    heuristic strategy:
    1. Win if possible.
    2. Block opponent's win if possible.
    3. Take the center if available.
    4. Take any available corner.
    5. Take any available side.
    """

    def get_empty_cells(b: list[list[int]]) -> list[tuple[int, int]]:
        """Returns a list of (row, col) tuples for all empty cells (value 0)."""
        empty = []
        for r in range(3):
            for c in range(3):
                if b[r][c] == 0:
                    empty.append((r, c))
        return empty

    def find_special_move(b: list[list[int]], target_player: int) -> tuple[int, int] | None:
        """
        Searches for a move that would result in `target_player` winning or
        blocking an immediate win for `target_player`.
        It looks for a line (row, column, or diagonal) that contains two of
        `target_player`'s marks and one empty cell.

        Args:
            b: The current board state.
            target_player: The player (1 or -1) to check for a winning/blocking move.

        Returns:
            A tuple (row, col) of the move if found, otherwise None.
        """
        # Define all 8 winning lines (3 rows, 3 columns, 2 diagonals)
        lines = [
            # Rows
            [(0, 0), (0, 1), (0, 2)],
            [(1, 0), (1, 1), (1, 2)],
            [(2, 0), (2, 1), (2, 2)],
            # Columns
            [(0, 0), (1, 0), (2, 0)],
            [(0, 1), (1, 1), (2, 1)],
            [(0, 2), (1, 2), (2, 2)],
            # Diagonals
            [(0, 0), (1, 1), (2, 2)],
            [(0, 2), (1, 1), (2, 0)]
        ]

        for line in lines:
            player_count = 0
            empty_cell_in_line = None
            for r, c in line:
                if b[r][c] == target_player:
                    player_count += 1
                elif b[r][c] == 0:
                    empty_cell_in_line = (r, c)

            # If we found a line with exactly two of the target_player's marks
            # and one empty spot, then this empty spot is the winning/blocking move.
            if player_count == 2 and empty_cell_in_line is not None:
                return empty_cell_in_line
        return None

    # --- Core Policy Logic ---

    # 1. Check if player 1 (myself) can win in the next move
    win_move = find_special_move(board, 1)
    if win_move:
        return win_move

    # 2. Check if opponent (-1) can win in the next move and block them
    block_move = find_special_move(board, -1)
    if block_move:
        return block_move

    # 3. Take the center square if it's empty (generally the best strategic move)
    if board[1][1] == 0:
        return (1, 1)

    # 4. Take any available corner square
    corners = [(0, 0), (0, 2), (2, 0), (2, 2)]
    for r, c in corners:
        if board[r][c] == 0:
            return (r, c)

    # 5. Take any available side square (non-corner, non-center)
    sides = [(0, 1), (1, 0), (1, 2), (2, 1)]
    for r, c in sides:
        if board[r][c] == 0:
            return (r, c)

    # Fallback: If the board is completely full (should only happen at the end of a game)
    # or if the logic above somehow fails to find an empty cell (highly unlikely for a valid game state),
    # return the first available empty cell. This part should ideally not be reached
    # in a valid, non-full game where a move is expected.
    empty_cells = get_empty_cells(board)
    if empty_cells:
        return empty_cells[0]
    else:
        # This case should indicate a full board, meaning draw or win, and no move is possible.
        # Returning an invalid move to signify this if absolutely forced.
        return (-1, -1) # Should not occur if called mid-game.
