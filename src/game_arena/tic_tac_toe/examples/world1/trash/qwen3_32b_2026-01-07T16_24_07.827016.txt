def policy(board: list[list[int]]) -> tuple[int, int]:
    def is_terminal(current_board):
        # Check rows
        for row in current_board:
            if row[0] != 0 and row[0] == row[1] == row[2]:
                return (True, row[0])
        # Check columns
        for col in range(3):
            if current_board[0][col] != 0 and current_board[0][col] == current_board[1][col] == current_board[2][col]:
                return (True, current_board[0][col])
        # Check diagonals
        if current_board[0][0] != 0 and current_board[0][0] == current_board[1][1] == current_board[2][2]:
            return (True, current_board[0][0])
        if current_board[0][2] != 0 and current_board[0][2] == current_board[1][1] == current_board[2][0]:
            return (True, current_board[0][2])
        # Check if full
        full = all(all(cell != 0 for cell in row) for row in current_board)
        if full:
            return (True, 0)
        return (False, None)
    
    def minimax(current_board, is_maximizing):
        terminal, score = is_terminal(current_board)
        if terminal:
            return score
        if is_maximizing:
            best_score = -float('inf')
            for i in range(3):
                for j in range(3):
                    if current_board[i][j] == 0:
                        current_board[i][j] = 1
                        new_score = minimax(current_board, False)
                        current_board[i][j] = 0
                        best_score = max(best_score, new_score)
            return best_score
        else:
            best_score = float('inf')
            for i in range(3):
                for j in range(3):
                    if current_board[i][j] == 0:
                        current_board[i][j] = -1
                        new_score = minimax(current_board, True)
                        current_board[i][j] = 0
                        best_score = min(best_score, new_score)
            return best_score
    
    best_move = None
    best_score = -float('inf')
    for i in range(3):
        for j in range(3):
            if board[i][j] == 0:
                board[i][j] = 1
                score = minimax(board, False)
                board[i][j] = 0
                if score > best_score or best_move is None:
                    best_score = score
                    best_move = (i, j)
    return best_move
