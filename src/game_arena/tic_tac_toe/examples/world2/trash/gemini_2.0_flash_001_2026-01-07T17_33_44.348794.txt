    Chooses the next move for a 4x4 Tic Tac Toe game.

    Args:

                return True

            for j in range(4):
                if board[i][j] == 0:
                    moves.append((i, j))
        return moves

    # 1. Check for winning moves
    possible_moves = get_possible_moves(board)
    for row, col in possible_moves:
