def policy(board: list[list[int]]) -> tuple[int, int]:
    """
    Determines the next move for a 4x4 Tic Tac Toe game.

    Args:
        board: A 4x4 list of lists containing 0 (empty), 1 (you), and -1 (opponent).

    Returns:
        A tuple of two integers (0-3, 0-3) indicating the row and column you choose.
    """

    def check_win(board: list[list[int]], player: int) -> bool:
        """Checks if the given player has won the board."""
        # Check rows
        for row in board:
            if all(cell == player for cell in row):
                return True

        # Check columns
        for col in range(4):
            if all(board[row][col] == player for row in range(4)):
                return True

        # Check diagonals
        if all(board[i][i] == player for i in range(4)):
            return True
        if all(board[i][3 - i] == player for i in range(4)):
            return True

        return False

    # Check for winning moves
    for row in range(4):
        for col in range(4):
            if board[row][col] == 0:
                temp_board = [r[:] for r in board]  # Create a copy
                temp_board[row][col] = 1
                if check_win(temp_board, 1):
                    return (row, col)

    # Check for blocking moves
    for row in range(4):
        for col in range(4):
            if board[row][col] == 0:
                temp_board = [r[:] for r in board]  # Create a copy
                temp_board[row][col] = -1
                if check_win(temp_board, -1):
                    return (row, col)

    # Take the center
    center_moves = [(1, 1), (1, 2), (2, 1), (2, 2)]
    for move in center_moves:
        if board[move[0]][move[1]] == 0:
            return move

    # Take any available cell
    for row in range(4):
        for col in range(4):
            if board[row][col] == 0:
                return (row, col)

    # Should never happen in a valid game
    return (0, 0)
