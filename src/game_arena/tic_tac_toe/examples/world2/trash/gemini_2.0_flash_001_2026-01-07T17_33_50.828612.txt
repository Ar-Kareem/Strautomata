def policy(board: list[list[int]]) -> tuple[int, int]:
    """
    Determines the next move for a 4x4 Tic Tac Toe game.

    Args:
        board: A 4x4 list of lists containing 0 (empty), 1 (you), and -1 (opponent).

    Returns:
        A tuple of two integers (0-3, 0-3) indicating the row and column you choose.
    """

    def check_win(board_state: list[list[int]], player: int) -> bool:
        """Checks if the given player has won the board."""
        # Check rows
        for row in board_state:
            if all(cell == player for cell in row):
                return True

        # Check columns
        for col in range(4):
            if all(board_state[row][col] == player for row in range(4)):
                return True

        # Check diagonals
        if all(board_state[i][i] == player for i in range(4)):
            return True
        if all(board_state[i][3 - i] == player for i in range(4)):
            return True

        return False

    def is_valid_move(board_state: list[list[int]], row: int, col: int) -> bool:
        """Checks if the given move is valid (within bounds and cell is empty)."""
        if not (0 <= row < 4 and 0 <= col < 4):
            return False
        return board_state[row][col] == 0

    # 1. Check for winning moves
    for row in range(4):
        for col in range(4):
            if is_valid_move(board, row, col):
                temp_board = [row[:] for row in board]  # Create a copy
                temp_board[row][col] = 1
                if check_win(temp_board, 1):
                    return (row, col)

    # 2. Check for blocking moves
    for row in range(4):
        for col in range(4):
            if is_valid_move(board, row, col):
                temp_board = [row[:] for row in board]  # Create a copy
                temp_board[row][col] = -1
                if check_win(temp_board, -1):
                    return (row, col)

    # 3. Prioritize center cells
    center_cells = [(1, 1), (1, 2), (2, 1), (2, 2)]
    for row, col in center_cells:
        if is_valid_move(board, row, col):
            return (row, col)

    # 4. Choose the first available cell
    for row in range(4):
        for col in range(4):
            if is_valid_move(board, row, col):
                return (row, col)

    return (0, 0)  # Should not happen in a valid game
