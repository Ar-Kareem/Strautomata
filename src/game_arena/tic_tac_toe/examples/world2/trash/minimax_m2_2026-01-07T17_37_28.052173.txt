import time

WINNING_LINES = []
# Rows
for i in range(4):
    WINNING_LINES.append(tuple((i, j) for j in range(4)))
# Columns
for j in range(4):
    WINNING_LINES.append(tuple((i, j) for i in range(4)))
# Diagonals
WINNING_LINES.append(tuple((i, i) for i in range(4)))
WINNING_LINES.append(tuple((i, 3-i) for i in range(4)))

memo = {}

def check_win(board, player):
    for line in WINNING_LINES:
        if all(board[i][j] == player for (i, j) in line):
            return True
    return False

def heuristic_value(board):
    score = 0
    for line in WINNING_LINES:
        s1 = 0
        s2 = 0
        for (i, j) in line:
            val = board[i][j]
            if val == 1:
                s1 += 1
            elif val == -1:
                s2 += 1
        if s2 == 0:
            score += 10 ** s1
        if s1 == 0:
            score -= 10 ** s2
    return score

def minimax(board, depth, maximizing_player, alpha, beta):
    state = tuple(tuple(row) for row in board)
    key = (state, depth, maximizing_player)
    if key in memo:
        return memo[key]
    
    if depth == 0:
        result = heuristic_value(board)
        memo[key] = result
        return result
    if check_win(board, 1):
        result = 1000000
        memo[key] = result
        return result
    if check_win(board, -1):
        result = -1000000
        memo[key] = result
        return result
    if all(board[i][j] != 0 for i in range(4) for j in range(4)):
        result = 0
        memo[key] = result
        return result
    
    empty_cells = [(i, j) for i in range(4) for j in range(4) if board[i][j] == 0]
    
    if maximizing_player:
        max_val = -1000000
        for (i, j) in empty_cells:
            board[i][j] = 1
            val = minimax(board, depth-1, False, alpha, beta)
            board[i][j] = 0
            if val > max_val:
                max_val = val
            alpha = max(alpha, val)
            if beta <= alpha:
                break
        memo[key] = max_val
        return max_val
    else:
        min_val = 1000000
        for (i, j) in empty_cells:
            board[i][j] = -1
            val = minimax(board, depth-1, True, alpha, beta)
            board[i][j] = 0
            if val < min_val:
                min_val = val
            beta = min(beta, val)
            if beta <= alpha:
                break
        memo[key] = min_val
        return min_val

def policy(board):
    start_time = time.time()
    time_limit = 0.9
    depth = 1
    best_move = None
    max_depth = 5
    
    while time.time() - start_time < time_limit and depth <= max_depth:
        current_best_value = -1000000
        current_best_moves = []
        for i in range(4):
            for j in range(4):
                if board[i][j] == 0:
                    board[i][j] = 1
                    value = minimax(board, depth-1, False, -1000000, 1000000)
                    board[i][j] = 0
                    if value > current_best_value:
                        current_best_value = value
                        current_best_moves = [(i, j)]
                    elif value == current_best_value:
                        current_best_moves.append((i, j))
        if current_best_moves:
            best_move = current_best_moves[0]
        depth += 1
    
    return best_move
