The strategy for this policy is:

- Use a bitboard representation for speed.
- Always take an immediate win.
- Always block a single immediate opponent win if no direct win exists.
- On late-game positions (11 or fewer empty cells), switch to an exact memoized minimax solver to play perfectly to the end.
- On earlier positions, use iterative-deepening alpha-beta search with move ordering.
- The heuristic values:
  - immediate threats and double-threats very highly,
  - open lines with 1/2/3 marks,
  - strong squares (inner 2x2, then corners, then edges).
- The policy always keeps a legal fallback move, so it will still return a valid move even if search is interrupted.

import time
from functools import lru_cache

N = 4
FULL_MASK = (1 &lt;&lt; 16) - 1
INF = 10**9

EXACT = 0
LOWER = 1
UPPER = 2

# Precompute row/col/diag win masks for 4-in-a-row on a 4x4 board.
WIN_MASKS = []

# Rows
for r in range(4):
    m = 0
    for c in range(4):
        m |= 1 &lt;&lt; (4 * r + c)
    WIN_MASKS.append(m)

# Cols
for c in range(4):
    m = 0
    for r in range(4):
        m |= 1 &lt;&lt; (4 * r + c)
    WIN_MASKS.append(m)

# Main diagonal
m = 0
for i in range(4):
    m |= 1 &lt;&lt; (4 * i + i)
WIN_MASKS.append(m)

# Anti-diagonal
m = 0
for i in range(4):
    m |= 1 &lt;&lt; (4 * i + (3 - i))
WIN_MASKS.append(m)

IDX_TO_RC = [(i // 4, i % 4) for i in range(16)]
BIT_TO_IDX = {1 &lt;&lt; i: i for i in range(16)}

CELL_MASKS = [[] for _ in range(16)]
for mask in WIN_MASKS:
    bits = mask
    while bits:
        b = bits &amp; -bits
        bits -= b
        CELL_MASKS[BIT_TO_IDX[b]].append(mask)

# Piece-square values: prefer inner 2x2, then corners, then edges.
POS_SCORES = []
for i in range(16):
    r, c = IDX_TO_RC[i]
    if 0 &lt; r &lt; 3 and 0 &lt; c &lt; 3:
        POS_SCORES.append(7)   # inner 2x2
    elif r in (0, 3) and c in (0, 3):
        POS_SCORES.append(6)   # corners
    else:
        POS_SCORES.append(3)   # edges

LOCAL_ATTACK = (0, 6, 25, 120, 0)
LOCAL_BLOCK = (0, 5, 22, 110, 0)
LINE_VALUES = (0, 4, 18, 120, 0)

ROOT_EXACT_THRESHOLD = 11


class SearchTimeout(Exception):
    pass


def has_win(bits: int) -&gt; bool:
    for mask in WIN_MASKS:
        if bits &amp; mask == mask:
            return True
    return False


def winning_moves_bits(player: int, opponent: int) -&gt; int:
    occ = player | opponent
    result = 0
    for mask in WIN_MASKS:
        if opponent &amp; mask:
            continue
        if (player &amp; mask).bit_count() == 3:
            result |= mask &amp; ~occ
    return result


def board_to_bits(board: list[list[int]]) -&gt; tuple[int, int]:
    cur = 0
    opp = 0
    for r in range(4):
        row = board[r]
        for c in range(4):
            v = row[c]
            bit = 1 &lt;&lt; (4 * r + c)
            if v == 1:
                cur |= bit
            elif v == -1:
                opp |= bit
    return cur, opp


def first_empty(board: list[list[int]]) -&gt; tuple[int, int]:
    for r in range(4):
        for c in range(4):
            if board[r][c] == 0:
                return (r, c)
    return (0, 0)


def ordered_moves(cur: int, opp: int, hint: int = 0) -&gt; list[int]:
    occ = cur | opp
    empty = FULL_MASK ^ occ
    opp_wins_now = winning_moves_bits(opp, cur)
    moves = []

    bits = empty
    while bits:
        bit = bits &amp; -bits
        bits -= bit
        idx = BIT_TO_IDX[bit]
        score = POS_SCORES[idx]

        if bit == hint:
            score += 1_000_000

        my2 = cur | bit

        if has_win(my2):
            score += 500_000

        if opp_wins_now &amp; bit:
            score += 200_000

        # Reward moves that create one or more direct threats next turn.
        score += 30_000 * winning_moves_bits(my2, opp).bit_count()

        # Local line potential and blocking value.
        for mask in CELL_MASKS[idx]:
            myc = (cur &amp; mask).bit_count()
            opc = (opp &amp; mask).bit_count()
            if opc == 0:
                score += LOCAL_ATTACK[myc]
            elif myc == 0:
                score += LOCAL_BLOCK[opc]

        moves.append((score, bit))

    moves.sort(key=lambda x: x[0], reverse=True)
    return [bit for _, bit in moves]


@lru_cache(maxsize=300000)
def solve(cur: int, opp: int) -&gt; int:
    """
    Exact minimax score from the perspective of side-to-move = cur.
    Larger is better for cur.
    """
    occ = cur | opp
    rem = 16 - occ.bit_count()

    if has_win(opp):
        return -10000 - rem
    if occ == FULL_MASK:
        return 0

    wins = winning_moves_bits(cur, opp)
    if wins:
        return 10000 + rem - 1

    best = -20000
    for bit in ordered_moves(cur, opp):
        score = -solve(opp, cur | bit)
        if score &gt; best:
            best = score
    return best


def choose_exact_move(cur: int, opp: int) -&gt; int:
    moves = ordered_moves(cur, opp)
    best_bit = moves[0]
    best_score = -20000
    for bit in moves:
        score = -solve(opp, cur | bit)
        if score &gt; best_score:
            best_score = score
            best_bit = bit
    return best_bit


def evaluate(cur: int, opp: int) -&gt; int:
    occ = cur | opp
    rem = 16 - occ.bit_count()

    if has_win(cur):
        return 100000 + rem
    if has_win(opp):
        return -100000 - rem

    score = 0

    my_wins = winning_moves_bits(cur, opp).bit_count()
    opp_wins = winning_moves_bits(opp, cur).bit_count()

    score += 1200 * my_wins
    score -= 1300 * opp_wins

    if my_wins &gt;= 2:
        score += 3000
    if opp_wins &gt;= 2:
        score -= 3200

    for mask in WIN_MASKS:
        myc = (cur &amp; mask).bit_count()
        opc = (opp &amp; mask).bit_count()
        if myc and opc:
            continue
        if opc == 0:
            score += LINE_VALUES[myc]
        else:
            score -= LINE_VALUES[opc]

    bits = cur
    while bits:
        b = bits &amp; -bits
        bits -= b
        score += POS_SCORES[BIT_TO_IDX[b]]

    bits = opp
    while bits:
        b = bits &amp; -bits
        bits -= b
        score -= POS_SCORES[BIT_TO_IDX[b]]

    return score


def negamax(cur: int, opp: int, depth: int, alpha: int, beta: int,
            deadline: float, tt: dict) -&gt; int:
    if time.perf_counter() &gt;= deadline:
        raise SearchTimeout

    occ = cur | opp
    rem = 16 - occ.bit_count()

    if has_win(opp):
        return -100000 - rem
    if occ == FULL_MASK:
        return 0
    if depth == 0:
        return evaluate(cur, opp)

    key = (cur, opp, depth)
    alpha_orig = alpha
    beta_orig = beta
    hint = 0

    entry = tt.get(key)
    if entry is not None:
        val, flag, move = entry
        hint = move
        if flag == EXACT:
            return val
        elif flag == LOWER:
            if val &gt; alpha:
                alpha = val
        else:
            if val &lt; beta:
                beta = val
        if alpha &gt;= beta:
            return val

    best = -INF
    best_move = 0

    for bit in ordered_moves(cur, opp, hint):
        score = -negamax(opp, cur | bit, depth - 1, -beta, -alpha, deadline, tt)
        if score &gt; best:
            best = score
            best_move = bit
        if best &gt; alpha:
            alpha = best
        if alpha &gt;= beta:
            break

    if best &lt;= alpha_orig:
        flag = UPPER
    elif best &gt;= beta_orig:
        flag = LOWER
    else:
        flag = EXACT

    tt[key] = (best, flag, best_move)
    return best


def search_root(cur: int, opp: int, depth: int, deadline: float,
                tt: dict, best_hint: int = 0) -&gt; tuple[int, int]:
    alpha = -INF
    beta = INF
    best_score = -INF
    best_move = 0

    for bit in ordered_moves(cur, opp, best_hint):
        if time.perf_counter() &gt;= deadline:
            raise SearchTimeout
        score = -negamax(opp, cur | bit, depth - 1, -beta, -alpha, deadline, tt)
        if score &gt; best_score:
            best_score = score
            best_move = bit
        if score &gt; alpha:
            alpha = score

    return best_move, best_score


def policy(board: list[list[int]]) -&gt; tuple[int, int]:
    fallback = first_empty(board)

    try:
        cur, opp = board_to_bits(board)
        occ = cur | opp
        rem = 16 - occ.bit_count()

        if rem &lt;= 0:
            return fallback

        # Immediate win.
        my_wins = winning_moves_bits(cur, opp)
        if my_wins:
            for bit in ordered_moves(cur, opp):
                if my_wins &amp; bit:
                    move = IDX_TO_RC[BIT_TO_IDX[bit]]
                    if board[move[0]][move[1]] == 0:
                        return move
            bit = my_wins &amp; -my_wins
            move = IDX_TO_RC[BIT_TO_IDX[bit]]
            if board[move[0]][move[1]] == 0:
                return move

        # Mandatory block if opponent has exactly one immediate winning move.
        opp_wins = winning_moves_bits(opp, cur)
        if opp_wins and opp_wins.bit_count() == 1:
            bit = opp_wins &amp; -opp_wins
            move = IDX_TO_RC[BIT_TO_IDX[bit]]
            if board[move[0]][move[1]] == 0:
                return move

        # Opening book: choose a strong central square.
        if rem == 16:
            if board[1][1] == 0:
                return (1, 1)
            return fallback

        # Exact play in the late game.
        if rem &lt;= ROOT_EXACT_THRESHOLD:
            bit = choose_exact_move(cur, opp)
            move = IDX_TO_RC[BIT_TO_IDX[bit]]
            if board[move[0]][move[1]] == 0:
                return move
            return fallback

        # Earlier game: iterative-deepening alpha-beta.
        deadline = time.perf_counter() + 0.90
        tt = {}

        moves = ordered_moves(cur, opp)
        best_bit = moves[0]
        best_hint = best_bit

        max_depth = min(8, rem)

        for depth in range(1, max_depth + 1):
            try:
                bit, score = search_root(cur, opp, depth, deadline, tt, best_hint)
                best_bit = bit
                best_hint = bit
                if abs(score) &gt;= 90000:
                    break
            except SearchTimeout:
                break

        move = IDX_TO_RC[BIT_TO_IDX[best_bit]]
        if board[move[0]][move[1]] == 0:
            return move
        return fallback

    except Exception:
        return fallback
