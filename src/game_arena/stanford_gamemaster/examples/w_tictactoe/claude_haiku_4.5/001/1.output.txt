# Strategy

For Tic Tac Toe, I'll implement a minimax algorithm with alpha-beta pruning to find the optimal move. The strategy prioritizes:

1. **Win immediately** - If the current player can win in one move, take it
2. **Block opponent** - If the opponent can win in one move, block them
3. **Minimax with pruning** - For deeper analysis, use minimax to evaluate all possible future game states
4. **Center and corners** - As a heuristic, prefer center and corner positions when minimax scores are equal

This approach will beat most policies because:
- It never misses a winning move
- It never allows the opponent to win when preventable
- It plays optimally by looking ahead multiple moves
- It's efficient enough to complete within time constraints

def policy(state: list[list[object]], role: str) -> str:
    # Parse the board state
    board = [[None for _ in range(3)] for _ in range(3)]
    
    for prop in state:
        if prop[0] == "cell":
            row = int(prop[1]) - 1
            col = int(prop[2]) - 1
            cell_state = prop[3]
            if cell_state == "b":
                board[row][col] = None
            else:
                board[row][col] = cell_state
    
    # Get opponent role
    opponent = "o" if role == "x" else "x"
    
    # Check for winning move or blocking move
    legal_moves = []
    for r in range(3):
        for c in range(3):
            if board[r][c] is None:
                legal_moves.append((r, c))
    
    # Check if we can win
    for r, c in legal_moves:
        board[r][c] = role
        if check_winner(board) == role:
            board[r][c] = None
            return f"mark({r+1},{c+1})"
        board[r][c] = None
    
    # Check if we need to block opponent
    for r, c in legal_moves:
        board[r][c] = opponent
        if check_winner(board) == opponent:
            board[r][c] = None
            return f"mark({r+1},{c+1})"
        board[r][c] = None
    
    # Use minimax to find best move
    best_score = float('-inf')
    best_move = legal_moves[0]
    
    for r, c in legal_moves:
        board[r][c] = role
        score = minimax(board, 0, False, role, opponent, float('-inf'), float('inf'))
        board[r][c] = None
        
        if score > best_score:
            best_score = score
            best_move = (r, c)
    
    return f"mark({best_move[0]+1},{best_move[1]+1})"


def check_winner(board):
    # Check rows
    for r in range(3):
        if board[r][0] == board[r][1] == board[r][2] and board[r][0] is not None:
            return board[r][0]
    
    # Check columns
    for c in range(3):
        if board[0][c] == board[1][c] == board[2][c] and board[0][c] is not None:
            return board[0][c]
    
    # Check diagonals
    if board[0][0] == board[1][1] == board[2][2] and board[0][0] is not None:
        return board[0][0]
    if board[0][2] == board[1][1] == board[2][0] and board[0][2] is not None:
        return board[0][2]
    
    return None


def is_board_full(board):
    for r in range(3):
        for c in range(3):
            if board[r][c] is None:
                return False
    return True


def minimax(board, depth, is_maximizing, role, opponent, alpha, beta):
    winner = check_winner(board)
    
    if winner == role:
        return 10 - depth
    elif winner == opponent:
        return depth - 10
    elif is_board_full(board):
        return 0
    
    if is_maximizing:
        max_eval = float('-inf')
        for r in range(3):
            for c in range(3):
                if board[r][c] is None:
                    board[r][c] = role
                    eval_score = minimax(board, depth + 1, False, role, opponent, alpha, beta)
                    board[r][c] = None
                    max_eval = max(max_eval, eval_score)
                    alpha = max(alpha, eval_score)
                    if beta <= alpha:
                        return max_eval
        return max_eval
    else:
        min_eval = float('inf')
        for r in range(3):
            for c in range(3):
                if board[r][c] is None:
                    board[r][c] = opponent
                    eval_score = minimax(board, depth + 1, True, role, opponent, alpha, beta)
                    board[r][c] = None
                    min_eval = min(min_eval, eval_score)
                    beta = min(beta, eval_score)
                    if beta <= alpha:
                        return min_eval
        return min_eval
