Looking at this game, I need to understand the structure:

1. **Board**: Hexagonal board with letters a-i and numbers 1-9, with 6 unusable cells
2. **Lines**: Three types - letter lines (a-i), number lines (1-9), and diagonal lines (a5-a9, b9-e9)
3. **Winning**: Win majority of lines in a direction → win that direction → win majority of directions

From analyzing the examples, I discovered the diagonal mapping: a cell (letter, number) is on the diagonal where `letter_index + number` determines the diagonal name. For example:
- (a,5), (b,4), (c,3), (e,1) all have sum=5 → diagonal a5
- (a,9), (c,7), (e,5), (g,3), (i,1) all have sum=9 → diagonal a9

My strategy:
1. Prioritize moves that create winning positions (lines where we have 3+ stones and opponent has 0)
2. Block opponent's potential winning lines
3. Prefer improving lines where we're competitive
4. Consider all three line types for each move

```python
import random

def policy(state: list[list[object]], role: str) -> str:
    opponent = "blue" if role == "red" else "red"
    
    occupied = set()
    line_counts = {}
    
    for fact in state:
        if fact[0] == "cell":
            _, letter, number, color = fact
            occupied.add((letter, number))
        elif fact[0] == "line_count":
            _, line_id, color, count = fact
            if line_id not in line_counts:
                line_counts[line_id] = {}
            line_counts[line_id][color] = int(count)
    
    valid_ranges = {
        'a': [5, 6, 7, 8, 9],
        'b': [4, 5, 6, 7, 9],
        'c': [3, 4, 5, 6, 7, 8, 9],
        'd': [2, 3, 4, 5, 7, 8, 9],
        'e': [1, 3, 5, 6, 7, 8, 9],
        'f': [1, 2, 3, 4, 6, 7, 8],
        'g': [1, 2, 3, 4, 5, 6, 7],
        'h': [1, 2, 3, 4, 6],
        'i': [1, 2, 3, 4, 5]
    }
    
    legal_moves = []
    for letter, numbers in valid_ranges.items():
        for number in numbers:
            cell = (letter, str(number))
            if cell not in occupied:
                legal_moves.append(cell)
    
    if not legal_moves:
        return "place(a,5)"
    
    def get_diagonal(letter, number):
        letter_idx = ord(letter) - ord('a')
        num = int(number)
        total = letter_idx + num
        if 5 <= total <= 9:
            return f"a{total}"
        elif 10 <= total <= 13:
            return f"{chr(ord('a') + total - 9)}9"
        return None
    
    best_move = legal_moves[0]
    best_score = float('-inf')
    
    for cell in legal_moves:
        letter, number = cell
        diag = get_diagonal(letter, number)
        
        score = 0
        for line_id in [letter, number] + ([diag] if diag else []):
            counts = line_counts.get(line_id, {})
            my_count = counts.get(role, 0)
            opp_count = counts.get(opponent, 0)
            
            diff = my_count - opp_count
            score += diff * 25
            
            if my_count >= 3 and opp_count == 0:
                score += 250
            elif my_count >= 2 and opp_count == 0:
                score += 100
            
            if opp_count >= 3 and my_count == 0:
                score += 200
            elif opp_count >= 2 and my_count == 0:
                score += 75
            
            if my_count >= 1:
                score += 20
        
        score += random.random() * 5
        
        if score > best_score:
            best_score = score
            best_move = cell
    
    letter, number = best_move
    return f"place({letter},{number})"
```
