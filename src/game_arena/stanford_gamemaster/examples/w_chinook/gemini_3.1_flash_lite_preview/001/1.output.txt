The strategy for this policy is to perform a deterministic search for all possible legal moves by simulating the game's rules: forward movement and diagonal jumps. Given that the Chinook board is a 4x8 grid with periodic (wrap-around) boundaries, the policy identifies all pieces belonging to the current `role` and checks for empty target squares (for regular moves) or valid jump sequences (for jumping over pieces of the same color). To prioritize winning, the policy evaluates potential moves based on the distance of the pieces toward the target rows. Since the game length is limited to 41 steps, speed-to-target is the primary heuristic for selection.

def policy(state: list[list[object]], role: str) -> str:
    cells = {}
    for fact in state:
        if fact[0] == "cell":
            cells[(fact[1], int(fact[2]))] = fact[3]

    rows = ['a', 'b', 'c', 'd']
    row_idx = {r: i for i, r in enumerate(rows)}
    
    def get_cell(c, r):
        # Wrap around: columns a-d (4), rows 1-8
        c_mapped = rows[row_idx[c] % 4]
        r_mapped = ((r - 1) % 8) + 1
        return cells.get((c_mapped, r_mapped), 'b')

    legal_moves = []
    
    # Directions: Red (up: +1 row index), Blue (down: -1 row index)
    direction = 1 if role == "red" else -1
    
    for (c, r), দখলে in cells.items():
        if দখলে == role:
            # Check moves
            for dc in [-1, 1]:
                dr = direction
                nc, nr = rows[(row_idx[c] + dc) % 4], ((r + dr - 1) % 8) + 1
                if get_cell(nc, nr) == 'b':
                    legal_moves.append(f"move({c},{r},{nc},{nr})")
                
                # Check jumps
                nnr = ((nr + dr - 1) % 8) + 1
                nnc = rows[(row_idx[nc] + dc) % 4]
                if get_cell(nc, nr) == role and get_cell(nnc, nnr) == 'b':
                    legal_moves.append(f"jump({c},{r},{nc},{nr},{nnc},{nnr})")

    # Heuristic: Prefer jumps over moves, prioritize forward movement
    if not legal_moves:
        return "noop"
        
    # Simple strategy: prioritize jumps, then moves that advance rows
    jumps = [m for m in legal_moves if m.startswith('jump')]
    if jumps:
        return jumps[0]
        
    return legal_moves[0]
