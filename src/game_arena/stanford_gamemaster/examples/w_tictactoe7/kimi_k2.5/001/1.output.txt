The strategy for this policy is to implement perfect play for Tic-Tac-Toe using the minimax algorithm with alpha-beta pruning. The policy:

1. Parses the state into a 3x3 board representation
2. Checks for immediate winning moves for the current player and plays them
3. Checks for immediate winning threats from the opponent and blocks them
4. If no critical moves exist, uses minimax to evaluate all legal moves and selects the one with the optimal score (preferring wins, then draws, avoiding losses)
5. Prefers winning quickly (higher score for earlier wins) and delaying losses (less negative score for later losses)

This ensures the policy never loses with perfect play (guarantees at least a draw) and capitalizes on opponent mistakes to win.

def policy(state: list[list[object]], role: str) -> str:
    # Initialize board with all cells blank
    board = {(r, c): "b" for r in range(1, 4) for c in range(1, 4)}
    
    # Parse state
    for fact in state:
        if fact[0] == "cell":
            r, c, mark = int(fact[1]), int(fact[2]), fact[3]
            board[(r, c)] = mark
    
    opponent = "o" if role == "x" else "x"
    
    # Win lines (rows, cols, diagonals)
    lines = [
        [(1,1),(1,2),(1,3)], [(2,1),(2,2),(2,3)], [(3,1),(3,2),(3,3)],
        [(1,1),(2,1),(3,1)], [(1,2),(2,2),(3,2)], [(1,3),(2,3),(3,3)],
        [(1,1),(2,2),(3,3)], [(1,3),(2,2),(3,1)]
    ]
    
    def check_winner(b):
        for line in lines:
            v1, v2, v3 = b[line[0]], b[line[1]], b[line[2]]
            if v1 == v2 == v3 and v1 != "b":
                return v1
        return None
    
    def is_full(b):
        return all(b[(r,c)] != "b" for r in range(1,4) for c in range(1,4))
    
    def minimax(b, depth, is_maximizing, alpha, beta):
        winner = check_winner(b)
        if winner == role:
            return 10 - depth
        if winner == opponent:
            return depth - 10
        if is_full(b):
            return 0
        
        if is_maximizing:
            max_eval = -float('inf')
            for r in range(1, 4):
                for c in range(1, 4):
                    if b[(r,c)] == "b":
                        b[(r,c)] = role
                        eval_score = minimax(b, depth + 1, False, alpha, beta)
                        b[(r,c)] = "b"
                        max_eval = max(max_eval, eval_score)
                        alpha = max(alpha, eval_score)
                        if beta <= alpha:
                            break
                if beta <= alpha:
                    break
            return max_eval
        else:
            min_eval = float('inf')
            for r in range(1, 4):
                for c in range(1, 4):
                    if b[(r,c)] == "b":
                        b[(r,c)] = opponent
                        eval_score = minimax(b, depth + 1, True, alpha, beta)
                        b[(r,c)] = "b"
                        min_eval = min(min_eval, eval_score)
                        beta = min(beta, eval_score)
                        if beta <= alpha:
                            break
                if beta <= alpha:
                    break
            return min_eval
    
    # Check for immediate win
    for r in range(1, 4):
        for c in range(1, 4):
            if board[(r,c)] == "b":
                board[(r,c)] = role
                if check_winner(board) == role:
                    return f"mark({r},{c})"
                board[(r,c)] = "b"
    
    # Check for immediate block (opponent win)
    for r in range(1, 4):
        for c in range(1, 4):
            if board[(r,c)] == "b":
                board[(r,c)] = opponent
                if check_winner(board) == opponent:
                    board[(r,c)] = "b"
                    return f"mark({r},{c})"
                board[(r,c)] = "b"
    
    # Find best move using minimax
    best_score = -float('inf')
    best_move = None
    
    for r in range(1, 4):
        for c in range(1, 4):
            if board[(r,c)] == "b":
                board[(r,c)] = role
                score = minimax(board, 0, False, -float('inf'), float('inf'))
                board[(r,c)] = "b"
                if score > best_score:
                    best_score = score
                    best_move = (r, c)
    
    if best_move:
        return f"mark({best_move[0]},{best_move[1]})"
    
    # Fallback: return first available move (should never reach here)
    for r in range(1, 4):
        for c in range(1, 4):
            if board[(r,c)] == "b":
                return f"mark({r},{c})"
    
    return "mark(1,1)"
